6//! RFC 9106: Argon2id memory-hard password-based key derivation
7//!
8//! Argon2id derives a key from a password and a salt while forcing every
9//! guess to fill and revisit a configurable amount of memory. It is the
10//! variant RFC 9106 §4 recommends when the deployment does not know which
11//! Argon2 type to pick.
12//!
13//! This module wraps the RustCrypto `argon2` crate (raw-hash API only, no PHC
14//! string format) with:
15//! - Parameter validation against RFC 9106 §3.1 plus library floors
16//!   ([`Argon2Params::validate`])
17//! - A memory cap enforced through
18//!   [`ResourceLimits::max_argon2_memory_kib`](crate::primitives::resource_limits::ResourceLimits)
19//!   before any allocation
20//! - Zeroized working memory and output ([`Argon2idResult`])
21//! - Machine-local cost calibration ([`Argon2Params::calibrate`])
22//!
23//! # Threat model
24//!
25//! - **Offline guessing.** An attacker who obtains the salt, the parameters
26//!   and anything derived from the key (a ciphertext, a verifier) can test
27//!   password guesses offline. Argon2id does not stop this; it makes each
28//!   guess cost `memory_kib` KiB of memory for the full duration of
29//!   `iterations` passes. Low-entropy passwords remain guessable; the
30//!   parameters set the price per guess.
31//! - **Memory hardness.** Fast hashes such as PBKDF2 parallelise cheaply on
32//!   GPUs and ASICs because each guess needs almost no memory. Argon2id
33//!   binds cost to memory bandwidth and capacity, where attacker hardware has
34//!   far less advantage. RFC 9106 §7.2 bounds the best known time-memory
35//!   trade-off against multi-pass Argon2id to a factor of about 1.33.
36//! - **Salt length.** The salt makes precomputed tables useless and forces
37//!   the attacker to attack each password record separately. RFC 9106 §3.1
38//!   recommends 16 bytes; §4 step 7 allows 64 bits (8 bytes) where space is
39//!   short. This module rejects salts shorter than 8 bytes and rejects
40//!   all-zero salts, which indicate uninitialised buffers.
41//! - **Side channels.** Argon2d indexes memory by password-dependent values,
42//!   which leaks through cache timing to a co-located attacker. Argon2i uses
43//!   data-independent indexing but is weaker against trade-off attacks.
44//!   Argon2id runs the first half of the first pass data-independently and
45//!   the rest data-dependently, so a cache-timing observer learns less than
46//!   with Argon2d while trade-off resistance stays close to Argon2d. It is
47//!   not fully constant-time: an attacker who can observe memory access
48//!   patterns during the data-dependent phase still learns something.
49//! - **Denial of service.** Argon2 parameters often come from stored or
50//!   deserialized records. The memory cap, the iteration cap and the output
51//!   length cap bound the work any single call can be made to do.
52//!
53//! # Not exposed
54//!
55//! RFC 9106 also defines an optional secret value `K` and associated data
56//! `X`. The public API exposes neither: Argon2id takes no domain-separation
57//! context in this library. The crate-internal path that feeds them is used
58//! only to replay the RFC 9106 §5.3 test vector. The underlying crate caps
59//! associated data at 32 bytes, while the RFC allows up to `2^32 - 1`.
60//!
61//! # FIPS 140-3
62//!
63//! Argon2id is not a NIST-approved KDF. Like BLAKE2, it stays compiled under
64//! the `fips` feature so callers can reach it explicitly, but no policy-engine
65//! path selects it. Deployments that must stay in an approved mode should use
66//! [`pbkdf2`](super::pbkdf2::pbkdf2).
67//!
68//! # Lanes
69//!
70//! This build does not enable the `argon2` crate's `parallel` feature, so
71//! lanes (`parallelism`) are computed one after another on a single thread.
72//! `parallelism` still changes the output, as RFC 9106 requires.
73

85/// Argon2id cost parameters.
86///
87/// The parameters are public protocol data (they are stored next to the
88/// salt), so this type derives `Debug`, `Clone` and `PartialEq`.
89///
90/// Fields are public for wire-format parsers; [`argon2id`] validates them on
91/// every call, so a struct built by hand cannot bypass the checks.
92#[derive(Debug, Clone, Copy, PartialEq, Eq)]
93pub struct Argon2Params {
94    /// Memory cost `m` in KiB. Must be at least `8 * parallelism` (RFC 9106
95    /// §3.1) and at most the [`max_argon2_memory_kib`][cap] resource limit.
96    ///
97    /// Consumer: argon2id()
98    ///
99    /// [cap]: crate::primitives::resource_limits::ResourceLimits::max_argon2_memory_kib
100    pub memory_kib: u32,
101    /// Number of passes `t` over memory. Between
102    /// [`Argon2Params::MIN_ITERATIONS`] and [`Argon2Params::MAX_ITERATIONS`].
103    ///
104    /// Consumer: argon2id()
105    pub iterations: u32,
106    /// Degree of parallelism `p` (number of lanes). Between
107    /// [`Argon2Params::MIN_PARALLELISM`] and [`Argon2Params::MAX_PARALLELISM`].
108    ///
109    /// Consumer: argon2id()
110    pub parallelism: u32,
111    /// Derived key (tag) length in bytes. Between
112    /// [`Argon2Params::MIN_OUTPUT_LEN`] and [`Argon2Params::MAX_OUTPUT_LEN`].
113    ///
114    /// Consumer: argon2id()
115    pub output_len: usize,
116}
117
118impl Argon2Params {
119    /// RFC 9106 §3.1 minimum memory per lane, in KiB (`m >= 8 * p`).
120    pub const MIN_MEMORY_KIB_PER_LANE: u32 = 8;
121
122    /// RFC 9106 §3.1 minimum number of passes.
123    pub const MIN_ITERATIONS: u32 = 1;
124
125    /// Library cap on the number of passes.
126    ///
127    /// RFC 9106 allows up to `2^32 - 1`; its §4 profiles use one or three
128    /// passes. The cap stops a stored parameter block from pinning a CPU for
129    /// hours.
130    pub const MAX_ITERATIONS: u32 = 1024;
131
132    /// RFC 9106 §3.1 minimum degree of parallelism.
133    pub const MIN_PARALLELISM: u32 = 1;
134
135    /// RFC 9106 §3.1 maximum degree of parallelism (`2^24 - 1`).
136    pub const MAX_PARALLELISM: u32 = 0x00FF_FFFF;
137
138    /// Library floor on the derived key length: 16 bytes (128 bits).
139    ///
140    /// RFC 9106 §3.1 permits tags as short as 4 bytes, but §4 step 8 names
141    /// 128 bits as the length that suffices for key derivation. Shorter keys
142    /// are rejected.
143    pub const MIN_OUTPUT_LEN: usize = 16;
144
145    /// Library cap on the derived key length: 1024 bytes.
146    ///
147    /// Bounds the output allocation for parameters read from untrusted
148    /// records. Callers needing more key material should expand the output
149    /// with HKDF.
150    pub const MAX_OUTPUT_LEN: usize = 1024;
151
152    /// Minimum salt length in bytes (RFC 9106 §4 step 7: 64 bits).
153    pub const MIN_SALT_LEN: usize = 8;
154
155    /// Recommended salt length in bytes (RFC 9106 §3.1: 16 bytes).
156    pub const RECOMMENDED_SALT_LEN: usize = 16;
157
158    /// Upper bound on the calibration target, in milliseconds (60 s).
159    pub const MAX_CALIBRATION_TARGET_MS: u64 = 60_000;
160
161    /// Memory cost of the first calibration run, in KiB (1 MiB), or the
162    /// memory cap if that is smaller.
163    pub const CALIBRATION_START_MEMORY_KIB: u32 = 1024;
164
165    /// Create validated Argon2id parameters.
166    ///
167    /// # Errors
168    /// Returns [`LatticeArcError::InvalidParameter`] if any value is outside
169    /// the bounds listed in [`Self::validate`].
170    pub fn new(
171        memory_kib: u32,
172        iterations: u32,
173        parallelism: u32,
174        output_len: usize,
175    ) -> Result<Self> {
176        let params = Self { memory_kib, iterations, parallelism, output_len };
177        params.validate()?;
178        Ok(params)
179    }
180
181    /// Check the parameters against RFC 9106 §3.1 and the library bounds.
182    ///
183    /// Checks, in order:
184    /// - `parallelism` in [`Self::MIN_PARALLELISM`]..=[`Self::MAX_PARALLELISM`]
185    /// - `memory_kib >= 8 * parallelism`
186    /// - `iterations` in [`Self::MIN_ITERATIONS`]..=[`Self::MAX_ITERATIONS`]
187    /// - `output_len` in [`Self::MIN_OUTPUT_LEN`]..=[`Self::MAX_OUTPUT_LEN`]
188    ///
189    /// The memory cap from resource limits is checked separately by
190    /// [`argon2id`], because it can change at runtime.
191    ///
192    /// # Errors
193    /// Returns [`LatticeArcError::InvalidParameter`] naming the first failed
194    /// check.
195    pub fn validate(&self) -> Result<()> {
196        let Self { memory_kib, iterations, parallelism, output_len } = *self;
197        if !(Self::MIN_PARALLELISM..=Self::MAX_PARALLELISM).contains(&parallelism) {
198            return Err(LatticeArcError::InvalidParameter(format!(
199                "Argon2id parallelism {parallelism} outside RFC 9106 range {}..={}",
200                Self::MIN_PARALLELISM,
201                Self::MAX_PARALLELISM
202            )));
203        }
204        // `parallelism <= 2^24 - 1`, so `8 * parallelism` fits in u32; the
205        // saturating form keeps the arithmetic lint-clean.
206        let min_memory = parallelism.saturating_mul(Self::MIN_MEMORY_KIB_PER_LANE);
207        if memory_kib < min_memory {
208            return Err(LatticeArcError::InvalidParameter(format!(
209                "Argon2id memory {memory_kib} KiB below RFC 9106 minimum of 8 * parallelism \
210                 = {min_memory} KiB"
211            )));
212        }
213        if !(Self::MIN_ITERATIONS..=Self::MAX_ITERATIONS).contains(&iterations) {
214            return Err(LatticeArcError::InvalidParameter(format!(
215                "Argon2id iterations {iterations} outside allowed range {}..={}",
216                Self::MIN_ITERATIONS,
217                Self::MAX_ITERATIONS
218            )));
219        }
220        if !(Self::MIN_OUTPUT_LEN..=Self::MAX_OUTPUT_LEN).contains(&output_len) {
221            return Err(LatticeArcError::InvalidParameter(format!(
222                "Argon2id output length {output_len} outside allowed range {}..={} bytes",
223                Self::MIN_OUTPUT_LEN,
224                Self::MAX_OUTPUT_LEN
225            )));
226        }
227        Ok(())
228    }
229
230    /// Measure Argon2id on this machine and pick parameters that take about
231    /// `target_ms` milliseconds per derivation.
232    ///
233    /// Follows the procedure in RFC 9106 §4: memory first, then passes.
234    /// With `parallelism = 1` and a 32-byte output, it starts at
235    /// [`Self::CALIBRATION_START_MEMORY_KIB`] and one pass, doubles the
236    /// memory until it reaches the cap or a run exceeds the target, and then
237    /// sets the pass count from the measured one-pass time, rounding down so
238    /// the estimate stays within the target.
239    ///
240    /// The memory cap is the smaller of `max_memory_kib` and the
241    /// [`max_argon2_memory_kib`][cap] resource limit, so the result is always
242    /// accepted by [`argon2id`] under the current limits.
243    ///
244    /// The result is machine- and load-dependent. Calibrate once at
245    /// provisioning time and store the parameters with the salt; do not
246    /// recalibrate when verifying, or the derived key changes. If even the
247    /// first run exceeds the target, the first run's parameters are returned.
248    ///
249    /// Memory doubles geometrically and the run that overshoots costs at most
250    /// about twice the target, so calibration takes roughly three times
251    /// `target_ms` plus the first run.
252    ///
253    /// # Errors
254    /// Returns [`LatticeArcError::InvalidParameter`] if `target_ms` is zero or
255    /// above [`Self::MAX_CALIBRATION_TARGET_MS`], or if the effective memory
256    /// cap is below [`Self::MIN_MEMORY_KIB_PER_LANE`]. Propagates any error
257    /// from the measurement runs.
258    ///
259    /// [cap]: crate::primitives::resource_limits::ResourceLimits::max_argon2_memory_kib
260    #[instrument(level = "debug")]
261    pub fn calibrate(target_ms: u64, max_memory_kib: u32) -> Result<Self> {
262        calibrate_with_limits(target_ms, max_memory_kib, get_global_resource_limits())
263    }
264}

266impl Default for Argon2Params {
267    /// Mobile-friendly profile: 64 MiB, 3 passes, 1 lane, 32-byte key.
268    ///
269    /// Memory and passes match the RFC 9106 §4 SECOND RECOMMENDED option
270    /// (`t = 3`, `m = 2^16` KiB), which the RFC gives for systems with much
271    /// less memory than the 2 GiB first option; 64 MiB fits phones and small
272    /// containers. The RFC option uses `p = 4`; this profile uses `p = 1`
273    /// because lanes run sequentially in this build (see the
274    /// [module docs](self)), so extra lanes add no defender throughput. The
275    /// 32-byte output is a 256-bit key. Use [`Argon2Params::calibrate`] to
276    /// tune for a specific machine.
277    fn default() -> Self {
278        Self { memory_kib: 64 * 1024, iterations: 3, parallelism: 1, output_len: 32 }
279    }
280}

331/// Derive a key from a password with Argon2id (RFC 9106, version 0x13).
332///
333/// # Arguments
334/// * `password` - Password bytes. Must not be empty.
335/// * `salt` - Unique random salt, at least [`Argon2Params::MIN_SALT_LEN`]
336///   (8) bytes; [`Argon2Params::RECOMMENDED_SALT_LEN`] (16) is recommended.
337/// * `params` - Cost parameters; see [`Argon2Params`].
338///
339/// # Input policy
340///
341/// RFC 9106 allows an empty password. This function rejects it, matching
342/// `hkdf_extract`'s rejection of empty input keying material: an empty
343/// password protects nothing and usually signals a caller bug. All-zero
344/// salts are rejected as a guard against uninitialised buffers, matching
345/// [`pbkdf2`](super::pbkdf2::pbkdf2).
346///
347/// # Errors
348/// - [`LatticeArcError::InvalidParameter`] for an empty password, a salt
349///   shorter than 8 bytes or all zero, or parameters that fail
350///   [`Argon2Params::validate`].
351/// - [`LatticeArcError::ResourceExhausted`] if `params.memory_kib` exceeds
352///   the `max_argon2_memory_kib` resource limit, or the working memory
353///   cannot be allocated.
354/// - [`LatticeArcError::InvalidConfiguration`] if the resource limits cannot
355///   be read.
356#[instrument(
357    level = "debug",
358    skip(password, salt),
359    fields(password_len = password.len(), salt_len = salt.len())
360)]
361pub fn argon2id(password: &[u8], salt: &[u8], params: &Argon2Params) -> Result<Argon2idResult> {
362    argon2id_inner(password, salt, &[], &[], params, get_global_resource_limits())
363}

365/// Shared Argon2id derivation path.
366///
367/// `secret` and `associated_data` are the optional RFC 9106 inputs `K` and
368/// `X`. The public [`argon2id`] passes empty slices, which RFC 9106 §3.2
369/// encodes identically to omitting them. Only the RFC 9106 §5.3 test-vector
370/// replay passes non-empty values.
371///
372/// # Errors
373/// See [`argon2id`].
374pub(crate) fn argon2id_inner(
375    password: &[u8],
376    salt: &[u8],
377    secret: &[u8],
378    associated_data: &[u8],
379    params: &Argon2Params,
380    limits: &ResourceLimitsManager,
381) -> Result<Argon2idResult> {
382    params.validate()?;
383    let Argon2Params { memory_kib, iterations, parallelism, output_len } = *params;
384
385    if password.is_empty() {
386        return Err(LatticeArcError::InvalidParameter(
387            "Argon2id password must not be empty".to_string(),
388        ));
389    }
390    if salt.len() < Argon2Params::MIN_SALT_LEN {
391        return Err(LatticeArcError::InvalidParameter(format!(
392            "Argon2id salt length {} below RFC 9106 minimum of {} bytes",
393            salt.len(),
394            Argon2Params::MIN_SALT_LEN
395        )));
396    }
397    if crate::primitives::ct::is_all_zero_bytes(salt) {
398        return Err(LatticeArcError::InvalidParameter(
399            "Argon2id salt must not be all zeros - use a cryptographically random salt".to_string(),
400        ));
401    }
402
403    // Memory cap: checked before any allocation.
404    limits.validate_argon2_memory_kib(memory_kib).map_err(map_resource_error)?;
405
406    let data = argon2::AssociatedData::new(associated_data).map_err(map_argon2_error)?;
407    let argon2_params = argon2::ParamsBuilder::new()
408        .m_cost(memory_kib)
409        .t_cost(iterations)
410        .p_cost(parallelism)
411        .output_len(output_len)
412        .data(data)
413        .build()
414        .map_err(map_argon2_error)?;
415    let block_count = argon2_params.block_count();
416    let context = argon2::Argon2::new_with_secret(
417        secret,
418        argon2::Algorithm::Argon2id,
419        argon2::Version::V0x13,
420        argon2_params,
421    )
422    .map_err(map_argon2_error)?;
423
424    // Allocate the working memory ourselves so it is zeroized on drop; the
425    // crate's own `hash_password_into` frees its blocks without wiping them.
426    // `try_reserve_exact` turns allocation failure into an error instead of
427    // an abort.
428    let mut blocks: Zeroizing<Vec<argon2::Block>> = Zeroizing::new(Vec::new());
429    blocks.try_reserve_exact(block_count).map_err(|e| {
430        tracing::debug!(error = %e, block_count, "Argon2id working memory allocation failed");
431        LatticeArcError::ResourceExhausted
432    })?;
433    blocks.resize(block_count, argon2::Block::new());
434
435    let mut output: Zeroizing<Vec<u8>> = Zeroizing::new(vec![0u8; output_len]);
436    context
437        .hash_password_into_with_memory(password, salt, &mut output, blocks.as_mut_slice())
438        .map_err(map_argon2_error)?;
439
440    // `SecretVec::new` copies into an exact-size buffer and wipes the moved
441    // vector; `Zeroizing` then drops the empty husk.
442    let key = SecretVec::new(std::mem::take(&mut *output));
443    Ok(Argon2idResult { key, params: *params })
444}

446/// Map a resource-limit failure to the crate error type.
447fn map_resource_error(err: ResourceError) -> LatticeArcError {
448    match err {
449        ResourceError::Argon2MemoryLimitExceeded { requested_kib, limit_kib } => {
450            tracing::debug!(requested_kib, limit_kib, "Argon2id memory cap exceeded");
451            LatticeArcError::ResourceExhausted
452        }
453        other => LatticeArcError::InvalidConfiguration(other.to_string()),
454    }
455}

462/// Calibration against an explicit resource-limit manager.
463///
464/// [`Argon2Params::calibrate`] passes the global manager; tests pass their
465/// own so they can vary the cap without touching global state.
466///
467/// # Errors
468/// See [`Argon2Params::calibrate`].
469pub(crate) fn calibrate_with_limits(
470    target_ms: u64,
471    max_memory_kib: u32,
472    limits: &ResourceLimitsManager,
473) -> Result<Argon2Params> {
474    if target_ms == 0 || target_ms > Argon2Params::MAX_CALIBRATION_TARGET_MS {
475        return Err(LatticeArcError::InvalidParameter(format!(
476            "Argon2id calibration target {target_ms} ms outside 1..={} ms",
477            Argon2Params::MAX_CALIBRATION_TARGET_MS
478        )));
479    }
480    let limit_kib = limits.get_limits().map_err(map_resource_error)?.max_argon2_memory_kib;
481    let cap_kib = max_memory_kib.min(limit_kib);
482    if cap_kib < Argon2Params::MIN_MEMORY_KIB_PER_LANE {
483        return Err(LatticeArcError::InvalidParameter(format!(
484            "Argon2id calibration memory cap {cap_kib} KiB below RFC 9106 minimum of {} KiB",
485            Argon2Params::MIN_MEMORY_KIB_PER_LANE
486        )));
487    }
488    let target = Duration::from_millis(target_ms);
489
490    // Fixed inputs: Argon2id running time does not depend on input values.
491    let password = [0x5a_u8; 32];
492    let salt = [0xa5_u8; Argon2Params::RECOMMENDED_SALT_LEN];
493    let measure = |params: &Argon2Params| -> Result<Duration> {
494        let start = Instant::now();
495        argon2id_inner(&password, &salt, &[], &[], params, limits)?;
496        Ok(start.elapsed())
497    };
498
499    // Phase 1: grow memory at one pass until the cap or the target.
500    let mut best = Argon2Params {
501        memory_kib: Argon2Params::CALIBRATION_START_MEMORY_KIB.min(cap_kib),
502        iterations: 1,
503        parallelism: 1,
504        output_len: 32,
505    };
506    let mut elapsed = measure(&best)?;
507    while elapsed < target && best.memory_kib < cap_kib {
508        let candidate =
509            Argon2Params { memory_kib: best.memory_kib.saturating_mul(2).min(cap_kib), ..best };
510        let candidate_elapsed = measure(&candidate)?;
511        if candidate_elapsed > target {
512            return Ok(best);
513        }
514        best = candidate;
515        elapsed = candidate_elapsed;
516    }
517    if elapsed >= target {
518        return Ok(best);
519    }
520
521    // Phase 2: memory is at the cap; scale passes linearly from the
522    // one-pass time, rounding down so the estimate stays within target.
523    let per_pass_ns = elapsed.as_nanos().max(1);
524    // `per_pass_ns >= 1`, so `checked_div` cannot return `None`.
525    let passes = target.as_nanos().checked_div(per_pass_ns).unwrap_or(1);
526    best.iterations = u32::try_from(passes)
527        .unwrap_or(Argon2Params::MAX_ITERATIONS)
528        .clamp(Argon2Params::MIN_ITERATIONS, Argon2Params::MAX_ITERATIONS);
529    best.validate()?;
530    Ok(best)
531}
