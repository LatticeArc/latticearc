29/// Result of a self-test operation
30#[non_exhaustive]
31#[derive(Debug, Clone, PartialEq, Eq)]
32pub enum SelfTestResult {
33    /// All tests passed successfully
34    Pass,
35    /// One or more tests failed with the given error message
36    Fail(String),
37}
38
39impl SelfTestResult {
40    /// Returns true if the self-test passed
41    #[must_use]
42    pub fn is_pass(&self) -> bool {
43        matches!(self, SelfTestResult::Pass)
44    }
45
46    /// Returns true if the self-test failed
47    #[must_use]
48    pub fn is_fail(&self) -> bool {
49        matches!(self, SelfTestResult::Fail(_))
50    }
51
52    /// Converts the result to a standard Result type
53    ///
54    /// # Errors
55    /// Returns `LatticeArcError::ValidationError` if the self-test failed
56    pub fn to_result(&self) -> Result<()> {
57        match self {
58            SelfTestResult::Pass => Ok(()),
59            SelfTestResult::Fail(msg) => Err(LatticeArcError::ValidationError {
60                message: format!("FIPS 140-3 self-test failed: {msg}"),
61            }),
62        }
63    }
64}

92/// Run all FIPS 140-3 power-up self-tests
93///
94/// This function runs Known Answer Tests (KATs) for all approved algorithms.
95/// According to FIPS 140-3, these tests must pass before any cryptographic
96/// operation can be performed.
97///
98/// # Returns
99///
100/// - `SelfTestResult::Pass` if all tests pass
101/// - `SelfTestResult::Fail(message)` if any test fails
102///
103/// # Example
104///
105/// ```no_run
106/// use latticearc::primitives::self_test::run_power_up_tests;
107///
108/// let result = run_power_up_tests();
109/// if result.is_fail() {
110///     // Enter error state - no crypto operations allowed
111///     eprintln!("CRITICAL: FIPS self-tests failed!");
112/// }
113/// ```
114#[must_use]
115pub fn run_power_up_tests() -> SelfTestResult {
116    // Run each test in sequence - any failure stops further tests
117    // Order follows FIPS 140-3 requirements: integrity first, then KATs
118
119    // 0. Software/Firmware Integrity Test (FIPS 140-3 Section 9.2.2)
120    // MUST be performed before any cryptographic operations
121    if let Err(e) = integrity_test() {
122        return SelfTestResult::Fail(format!("Module Integrity Test failed: {e}"));
123    }
124
125    // 1. SHA-256 KAT (foundational - other tests depend on hash)
126    if let Err(e) = kat_sha256() {
127        return SelfTestResult::Fail(format!("SHA-256 KAT failed: {e}"));
128    }
129
130    // 2. HKDF-SHA256 KAT (depends on HMAC-SHA256)
131    if let Err(e) = kat_hkdf_sha256() {
132        return SelfTestResult::Fail(format!("HKDF-SHA256 KAT failed: {e}"));
133    }
134
135    // 3. AES-256-GCM KAT
136    if let Err(e) = kat_aes_256_gcm() {
137        return SelfTestResult::Fail(format!("AES-256-GCM KAT failed: {e}"));
138    }
139
140    // 4. SHA3-256 KAT
141    if let Err(e) = kat_sha3_256() {
142        return SelfTestResult::Fail(format!("SHA3-256 KAT failed: {e}"));
143    }
144
145    // 5. HMAC-SHA256 KAT
146    if let Err(e) = kat_hmac_sha256() {
147        return SelfTestResult::Fail(format!("HMAC-SHA256 KAT failed: {e}"));
148    }
149
150    // 6-9. PQ algorithm self-consistency tests.
151    //
152    // These functions are named `roundtrip_*` because they are NOT
153    // Known Answer Tests in the FIPS 140-3 §10.3.1 / CAVP sense —
154    // they generate a fresh keypair (random seed) and verify a
155    // roundtrip rather than comparing against precomputed test
156    // vectors. A bug producing self-consistent-but-wrong output would
157    // pass indefinitely.
158    //
159    // Real KAT validation against NIST test vectors runs in the
160    // `latticearc-tests` crate at `tests/tests/nist_kat/{ml_kem,
161    // ml_dsa, slh_dsa}_vectors.rs` and `tests/tests/fips_cavp.rs`.
162    // Those tests are required for CAVP submission; the roundtrip
163    // self-tests below provide power-on smoke coverage that the
164    // algorithm chains are wired correctly.
165    //
166    // Promoting the power-on path to true KATs requires a
167    // deterministic-keygen API that the wrapper layers don't yet
168    // expose; tracked as TRK-007 in docs/TRACKING.md.
169
170    // 6. ML-KEM-768 self-consistency (encap/decap roundtrip).
171    if let Err(e) = roundtrip_ml_kem_768() {
172        return SelfTestResult::Fail(format!("ML-KEM-768 roundtrip failed: {e}"));
173    }
174
175    // 7. ML-DSA-44 KAT (ACVP keygen vector + sign/verify roundtrip).
176    if let Err(e) = kat_ml_dsa() {
177        return SelfTestResult::Fail(format!("ML-DSA-44 KAT failed: {e}"));
178    }
179
180    // 8. SLH-DSA-SHAKE-192s KAT (ACVP keygen vector + sign/verify roundtrip).
181    if let Err(e) = kat_slh_dsa() {
182        return SelfTestResult::Fail(format!("SLH-DSA-SHAKE-192s KAT failed: {e}"));
183    }
184
185    // 9. FN-DSA self-consistency (runs in separate thread with 32 MB stack).
186    if let Err(e) = kat_fn_dsa() {
187        return SelfTestResult::Fail(format!("FN-DSA roundtrip failed: {e}"));
188    }
189
190    // The doc example advertises this function as a standalone entry
191    // point, so it must set `SELF_TEST_PASSED` itself — otherwise
192    // `is_module_operational()` returns `false` after a clean Pass for
193    // callers that don't go through `initialize_and_test`. `SeqCst`
194    // pairs with the `SeqCst` loads on the reader side and gives a
195    // single total order across `SELF_TEST_PASSED` and
196    // `MODULE_ERROR_CODE` — needed because the FIPS module-state
197    // gate reads both atomics non-atomically.
198    SELF_TEST_PASSED.store(true, Ordering::SeqCst);
199    SelfTestResult::Pass
200}

202/// Run power-up tests with detailed reporting.
203///
204/// Same sequence as [`run_power_up_tests`] — integrity test first per
205/// FIPS 140-3 §9.2.2, then the KATs / self-consistency tests — but
206/// returns a [`SelfTestReport`] capturing individual results and timings
207/// for diagnostics. On Pass this sets `SELF_TEST_PASSED` so the module
208/// becomes operational; on Fail it calls [`set_module_error`] so
209/// subsequent crypto-op gates reject calls.
210///
211/// Unlike [`initialize_and_test`], this function does **not** abort the
212/// process on failure — FIPS 140-3 §9.1's "immediate abort" requirement
213/// is the caller's responsibility for this entry point. Inspect the
214/// returned report and abort if your deployment needs strict §9.1
215/// behaviour, or call [`initialize_and_test`] instead.
216///
217/// # Returns
218///
219/// A `SelfTestReport` containing individual test results and timing information.
220#[must_use]
221pub fn run_power_up_tests_with_report() -> SelfTestReport {
222    use std::time::Instant;
223
224    /// Convert duration to u64 microseconds with saturation
225    fn duration_to_us(duration: std::time::Duration) -> u64 {
226        // Saturate at u64::MAX if duration exceeds ~584,942 years
227        u64::try_from(duration.as_micros()).unwrap_or(u64::MAX)
228    }
229
230    let start = Instant::now();
231    let mut tests = Vec::new();
232    let mut overall_pass = true;
233
234    // Module integrity test (FIPS 140-3 §9.2.2): MUST run first, before
235    // any cryptographic operation. Same ordering as `run_power_up_tests`.
236    // If the integrity test fails, §9.2.2 requires the module to inhibit
237    // all subsequent crypto — we record a skip for each downstream KAT and
238    // return early. The sibling `run_power_up_tests` already short-circuits
239    // on integrity failure; the reporting variant previously continued
240    // through eight more KATs on a binary that just failed tamper
241    // detection, which is exactly the threat scenario this test exists for.
242    let integrity_start = Instant::now();
243    let integrity_outcome = integrity_test();
244    let integrity_result = match &integrity_outcome {
245        Ok(()) => SelfTestResult::Pass,
246        Err(e) => {
247            overall_pass = false;
248            SelfTestResult::Fail(e.to_string())
249        }
250    };
251    tests.push(IndividualTestResult {
252        algorithm: "Module-Integrity".to_string(),
253        result: integrity_result,
254        duration_us: Some(duration_to_us(integrity_start.elapsed())),
255    });
256    if integrity_outcome.is_err() {
257        // Set the FIPS module-error state so downstream `verify_operational`
258        // calls reject too; mirror `run_power_up_tests`'s effect even though
259        // this variant returns a Report rather than aborting.
260        set_module_error(ModuleErrorCode::IntegrityFailure);
261        // Record every downstream KAT as "not run" so report consumers see
262        // exactly which tests were inhibited by the §9.2.2 halt. We don't
263        // execute the underlying KAT functions — running tamper-suspect
264        // crypto code is precisely what §9.2.2 forbids.
265        // Names MUST match the success-path KAT identifiers byte-for-byte —
266        // a downstream report consumer that filters or correlates by the
267        // `algorithm` field must see the same identifier whether the test
268        // ran or was inhibited. The success branches below populate
269        // `algorithm: "<NAME>".to_string()`; this list mirrors them.
270        const SKIPPED_TESTS: &[&str] = &[
271            "SHA-256",
272            "HKDF-SHA256",
273            "AES-256-GCM",
274            "SHA3-256",
275            "HMAC-SHA256",
276            "ML-KEM-768",
277            "ML-DSA-44",
278            "SLH-DSA-SHAKE-192s",
279            "FN-DSA-512",
280        ];
281        for name in SKIPPED_TESTS {
282            tests.push(IndividualTestResult {
283                algorithm: (*name).to_string(),
284                result: SelfTestResult::Fail(
285                    "Not run: inhibited by integrity-test failure (FIPS 140-3 §9.2.2)".to_string(),
286                ),
287                duration_us: None,
288            });
289        }
290        return SelfTestReport {
291            overall_result: SelfTestResult::Fail(
292                "Module-Integrity Test failed; downstream KATs inhibited per FIPS 140-3 §9.2.2"
293                    .to_string(),
294            ),
295            tests,
296            total_duration_us: duration_to_us(start.elapsed()),
297        };
298    }
299
300    // SHA-256 KAT
301    let sha_start = Instant::now();
302    let sha_result = match kat_sha256() {
303        Ok(()) => SelfTestResult::Pass,
304        Err(e) => {
305            overall_pass = false;
306            SelfTestResult::Fail(e.to_string())
307        }
308    };
309    tests.push(IndividualTestResult {
310        algorithm: "SHA-256".to_string(),
311        result: sha_result,
312        duration_us: Some(duration_to_us(sha_start.elapsed())),
313    });
314
315    // HKDF-SHA256 KAT
316    let hkdf_start = Instant::now();
317    let hkdf_result = match kat_hkdf_sha256() {
318        Ok(()) => SelfTestResult::Pass,
319        Err(e) => {
320            overall_pass = false;
321            SelfTestResult::Fail(e.to_string())
322        }
323    };
324    tests.push(IndividualTestResult {
325        algorithm: "HKDF-SHA256".to_string(),
326        result: hkdf_result,
327        duration_us: Some(duration_to_us(hkdf_start.elapsed())),
328    });
329
330    // AES-256-GCM KAT
331    let aes_start = Instant::now();
332    let aes_result = match kat_aes_256_gcm() {
333        Ok(()) => SelfTestResult::Pass,
334        Err(e) => {
335            overall_pass = false;
336            SelfTestResult::Fail(e.to_string())
337        }
338    };
339    tests.push(IndividualTestResult {
340        algorithm: "AES-256-GCM".to_string(),
341        result: aes_result,
342        duration_us: Some(duration_to_us(aes_start.elapsed())),
343    });
344
345    // SHA3-256 KAT
346    let sha3_start = Instant::now();
347    let sha3_result = match kat_sha3_256() {
348        Ok(()) => SelfTestResult::Pass,
349        Err(e) => {
350            overall_pass = false;
351            SelfTestResult::Fail(e.to_string())
352        }
353    };
354    tests.push(IndividualTestResult {
355        algorithm: "SHA3-256".to_string(),
356        result: sha3_result,
357        duration_us: Some(duration_to_us(sha3_start.elapsed())),
358    });
359
360    // HMAC-SHA256 KAT
361    let hmac_start = Instant::now();
362    let hmac_result = match kat_hmac_sha256() {
363        Ok(()) => SelfTestResult::Pass,
364        Err(e) => {
365            overall_pass = false;
366            SelfTestResult::Fail(e.to_string())
367        }
368    };
369    tests.push(IndividualTestResult {
370        algorithm: "HMAC-SHA256".to_string(),
371        result: hmac_result,
372        duration_us: Some(duration_to_us(hmac_start.elapsed())),
373    });
374
375    // ML-KEM-768 roundtrip
376    let kem_start = Instant::now();
377    let kem_result = match roundtrip_ml_kem_768() {
378        Ok(()) => SelfTestResult::Pass,
379        Err(e) => {
380            overall_pass = false;
381            SelfTestResult::Fail(e.to_string())
382        }
383    };
384    tests.push(IndividualTestResult {
385        algorithm: "ML-KEM-768".to_string(),
386        result: kem_result,
387        duration_us: Some(duration_to_us(kem_start.elapsed())),
388    });
389
390    // ML-DSA-44 KAT (ACVP keygen + roundtrip)
391    let mldsa_start = Instant::now();
392    let mldsa_result = match kat_ml_dsa() {
393        Ok(()) => SelfTestResult::Pass,
394        Err(e) => {
395            overall_pass = false;
396            SelfTestResult::Fail(e.to_string())
397        }
398    };
399    tests.push(IndividualTestResult {
400        algorithm: "ML-DSA-44".to_string(),
401        result: mldsa_result,
402        duration_us: Some(duration_to_us(mldsa_start.elapsed())),
403    });
404
405    // SLH-DSA-SHAKE-192s KAT (ACVP keygen) + SHAKE-128s roundtrip
406    let slhdsa_start = Instant::now();
407    let slhdsa_result = match kat_slh_dsa() {
408        Ok(()) => SelfTestResult::Pass,
409        Err(e) => {
410            overall_pass = false;
411            SelfTestResult::Fail(e.to_string())
412        }
413    };
414    tests.push(IndividualTestResult {
415        // The underlying KAT exercises `fips205::slh_dsa_shake_192s`
416        // (ACVP keygen vector + sign/verify roundtrip). The audit-record
417        // label must name the parameter set that actually ran, not the
418        // smaller `128s` variant — that mislabel previously misrepresented
419        // which FIPS 205 parameter set was validated on power-up.
420        algorithm: "SLH-DSA-SHAKE-192s".to_string(),
421        result: slhdsa_result,
422        duration_us: Some(duration_to_us(slhdsa_start.elapsed())),
423    });
424
425    // FN-DSA KAT
426    let fndsa_start = Instant::now();
427    let fndsa_result = match kat_fn_dsa() {
428        Ok(()) => SelfTestResult::Pass,
429        Err(e) => {
430            overall_pass = false;
431            SelfTestResult::Fail(e.to_string())
432        }
433    };
434    tests.push(IndividualTestResult {
435        algorithm: "FN-DSA-512".to_string(),
436        result: fndsa_result,
437        duration_us: Some(duration_to_us(fndsa_start.elapsed())),
438    });
439
440    let overall_result = if overall_pass {
441        // Mirror `run_power_up_tests`: on Pass, mark the module operational so
442        // `is_module_operational()` returns true for callers that routed
443        // through `_with_report` instead of `initialize_and_test`. SeqCst
444        // pairs with the SeqCst loads on the reader side.
445        SELF_TEST_PASSED.store(true, Ordering::SeqCst);
446        SelfTestResult::Pass
447    } else {
448        // Mirror `initialize_and_test`: on Fail, set the FIPS module-error
449        // state so subsequent crypto-op gates reject calls. We do NOT
450        // `process::abort()` here — the contract of this entry point is to
451        // return a diagnostic report. Callers in FIPS deployments who need
452        // §9.1's immediate-abort behaviour must inspect the returned report
453        // (or call `initialize_and_test`).
454        set_module_error(ModuleErrorCode::SelfTestFailure);
455        let failed: Vec<_> =
456            tests.iter().filter(|t| t.result.is_fail()).map(|t| t.algorithm.clone()).collect();
457        SelfTestResult::Fail(format!("Failed tests: {}", failed.join(", ")))
458    };
459
460    SelfTestReport { overall_result, tests, total_duration_us: duration_to_us(start.elapsed()) }
461}

467/// Run power-up tests and set the module state
468///
469/// This function runs all power-up tests and updates the module state
470/// accordingly. It should be called once during module initialization.
471/// On failure, the module enters an error state and no cryptographic
472/// services will be provided.
473///
474/// # Returns
475///
476/// The result of the self-tests
477#[must_use]
478pub fn initialize_and_test() -> SelfTestResult {
479    let result = run_power_up_tests();
480    if result.is_pass() {
481        SELF_TEST_PASSED.store(true, Ordering::SeqCst);
482    } else {
483        // FIPS 140-3 §9.1: Self-test failure requires module abort.
484        // Set error state first so any concurrent readers see a definitive error.
485        set_module_error(ModuleErrorCode::SelfTestFailure);
486        // FIPS 140-3 §9.1 requires immediate abort on self-test failure.
487        // No logging after this point — abort is non-recoverable.
488        std::process::abort();
489    }
490    result
491}

493/// Verify module is operational before performing cryptographic operations
494///
495/// This function checks if the module has passed self-tests and is ready
496/// to perform cryptographic operations. It also verifies that no error
497/// state has been set.
498///
499/// According to FIPS 140-3, a cryptographic module must not provide any
500/// cryptographic services when it is in an error state.
501///
502/// # Errors
503///
504/// Returns `LatticeArcError::ValidationError` if:
505/// - Self-tests have not passed
506/// - The module is in an error state
507pub fn verify_operational() -> Result<()> {
508    // Check for error state first
509    let error_state = get_module_error_state();
510    if error_state.is_error() {
511        return Err(LatticeArcError::ValidationError {
512            message: format!(
513                "FIPS module not operational: {} (error set at timestamp {})",
514                error_state.error_code.description(),
515                error_state.timestamp
516            ),
517        });
518    }
519
520    // Check self-test status
521    if !self_tests_passed() {
522        return Err(LatticeArcError::ValidationError {
523            message: "FIPS module not operational: self-tests have not passed".to_string(),
524        });
525    }
526
527    // Strict-integrity gate. Under the `fips-strict-integrity` feature the
528    // module must not enter operational state when the pre-operational
529    // integrity test had no configured HMAC to verify against. This is the
530    // right layer for that check — `integrity_test` itself returns a Result
531    // so it cannot abort, and `initialize_and_test`'s §9.1 abort is reserved
532    // for real tamper or KAT failure. Returning Err here lets the caller
533    // handle the deployment-config error without taking the process down.
534    //
535    // The `not(any(test, feature = "test-utils"))` clause keeps the gate
536    // out of unit-test and integration-test builds. Inside this crate's
537    // own `cargo test`, `cfg(test)` is true. The integration crate
538    // `latticearc-tests` enables the `test-utils` feature on its dep
539    // line, so its build of `latticearc` lib is also covered. Downstream
540    // consumers (and `latticearc-cli` builds, which deliberately do NOT
541    // enable `test-utils`) get the gate intact. Without this scoping, any
542    // `--all-features --release` CI test run would enable the gate AND
543    // ship no `PRODUCTION_HMAC.txt` — every test that asserts
544    // `verify_operational().is_ok()` would then break in lockstep.
545    #[cfg(all(feature = "fips-strict-integrity", not(any(test, feature = "test-utils"))))]
546    {
547        if !INTEGRITY_TEST_CONFIGURED.load(Ordering::SeqCst) {
548            return Err(LatticeArcError::ValidationError {
549                message: "FIPS module not operational: fips-strict-integrity is enabled but \
550                          no PRODUCTION_HMAC.txt was provisioned, so the pre-operational \
551                          integrity test (FIPS 140-3 §9.2) could not establish module \
552                          authenticity."
553                    .to_string(),
554            });
555        }
556    }
557
558    Ok(())
559}
