15/// Heuristic: does the file at `path` look like a LatticeArc artifact?
16///
17/// `current_exe()` returns the path to the running executable; when
18/// LatticeArc is loaded as a dynamic library by a host (Python, Node,
19/// JVM, etc.) that path is the host interpreter, not the library. This
20/// helper recognises three kinds of legitimate LatticeArc-on-disk
21/// artifacts:
22///   * the platform-specific shared library (`liblatticearc.so` /
23///     `liblatticearc.dylib` / `latticearc.dll`)
24///   * the LatticeArc CLI binary (`latticearc-cli` / `latticearc-cli.exe`)
25///   * any cargo-generated `<crate-name>-<16-hex>` binary under a
26///     `target` directory. The crate-name prefix is intentionally
27///     unconstrained: integration tests across the workspace produce
28///     binaries named after the test file, not after `latticearc`, and
29///     they all statically link this crate.
30///
31/// Anything else is treated as an unverifiable host process and the
32/// integrity test refuses to HMAC it.
33fn path_looks_like_latticearc_module(path: &std::path::Path) -> bool {
34    let Some(file_name) = path.file_name().and_then(|n| n.to_str()) else {
35        return false;
36    };
37    let lower = file_name.to_ascii_lowercase();
38    // tightened from a permissive
39    // `starts_with("latticearc")` plus a blanket `target/deps/` accept
40    // to (a) an exact list of names produced by this build, plus
41    // (b) cargo-test binary recognition via the `<crate-name>-<16-hex>`
42    // suffix shape inside `target/{debug,release}/deps/`. Previously
43    // a binary named `latticearc-evil-host` would have been HMAC-
44    // checked against the EXPECTED_HMAC, surfacing as "module
45    // integrity OK" for a non-LatticeArc image.
46    //
47    // Note: the cargo-deps shape check intentionally does NOT require
48    // a `latticearc` crate-name prefix — integration tests under
49    // `latticearc/tests/`, `tests/tests/`, and other workspace
50    // members produce binaries whose crate name reflects the test
51    // file (e.g. `primitives_self_test_conditional_kats-<hex>`),
52    // none of which start with `latticearc`. They all statically
53    // link the latticearc crate and are legitimate consumers, so
54    // the integrity test must accept them. The 16-hex-suffix shape
55    // is what excludes adversary binaries: a hand-crafted binary
56    // dropped into `deps/` to fake the integrity check would either
57    // (i) lack the 16-hex suffix (cargo always generates one) or
58    // (ii) be a real cargo-test binary, in which case the question
59    // reduces to "does the developer trust their own `target/`",
60    // which the workspace already assumes.
61    let exact_names = [
62        "liblatticearc.so",
63        "liblatticearc.dylib",
64        "latticearc.dll",
65        // Downstream binary name. The package is `latticearc-cli` but its
66        // `[[bin]] name = "latticearc"`, so `current_exe()` reports the
67        // short form — both spellings must be accepted to keep FIPS POST
68        // (which calls `path_looks_like_latticearc_module`) from rejecting
69        // legitimate cli invocations and SIGABRT-ing during keygen.
70        "latticearc",
71        "latticearc.exe",
72        "latticearc-cli",
73        "latticearc-cli.exe",
74    ];
75    if exact_names.iter().any(|n| lower == *n) {
76        return true;
77    }
78    // Accept any cargo-generated test binary that sits under a `target`
79    // ancestor. Deliberately does NOT pin the directory that immediately
80    // contains the binary.
81    //
82    // Cargo documents the build-dir layout as "internal to Cargo, and
83    // subject to change", and it does change: build-dir layout v2
84    // (rust-lang/cargo#17258, default on nightly from 2026-07-24, and
85    // stabilized for 1.99.0 by rust-lang/cargo#16807) moved unit-test
86    // binaries out of
87    //     target/<triple>/<profile>/deps/<crate>-<hash>
88    // and into
89    //     target/<triple>/<profile>/build/<pkg>/<hash>/out/<crate>-<hash>
90    // A check that required the parent to be `deps` rejected every test
91    // binary the moment that landed, which failed the integrity test,
92    // aborted the FIPS POST, and SIGABRT'd the whole test runner.
93    //
94    // So the shape we match on is the one cargo does NOT reserve the
95    // right to reshuffle: the `<crate-name>-<16-hex>` file name (checked
96    // below) plus containment under `target`. Pinning any intermediate
97    // directory name re-creates the same outage on the next layout
98    // revision.
99    //
100    // Security note: this is a "did we find our own artifact, or the host
101    // interpreter that dlopen'd us" check, NOT the tamper gate — an
102    // adversary-injected binary is rejected by HMAC mismatch, not by
103    // path. Which subdirectory of `target/` the build system chose is
104    // not a security property: an attacker who can write into
105    // `target/<anything>/` can write into `target/release/deps/` too.
106    //
107    // Hop bound: the deepest known layout is a nested tool target dir
108    // plus an explicit `--target <triple>` under layout v2 —
109    // `target/llvm-cov-target/<triple>/<profile>/build/<pkg>/<hash>/out/`
110    // — 8 hops from the binary's parent. Bounded at 12 rather than 8:
111    // the bound exists only to stop the walk wandering to the filesystem
112    // root, it is not a trust boundary, and sizing it flush against the
113    // deepest layout known today is exactly the brittleness that caused
114    // the layout-v2 outage. The margin costs nothing.
115    const MAX_TARGET_ANCESTOR_HOPS: usize = 12;
116    let under_target = path.parent().is_some_and(|p| {
117        p.ancestors()
118            .take(MAX_TARGET_ANCESTOR_HOPS)
119            .any(|a| a.file_name().and_then(|n| n.to_str()) == Some("target"))
120    });
121    if under_target {
122        // Strip `.exe` if present, then split on the LAST `-` to get
123        // crate-name vs hex-suffix. Accept any 16-hex-suffix file under
124        // `target/` — see the comment above for why neither the
125        // crate-name prefix nor the containing directory is constrained.
126        let stem = lower.strip_suffix(".exe").unwrap_or(&lower);
127        if let Some((_crate_name, suffix)) = stem.rsplit_once('-')
128            && suffix.len() == 16
129            && suffix.chars().all(|c| c.is_ascii_hexdigit())
130        {
131            return true;
132        }
133    }
134    false
135}

137/// Software/Firmware Integrity Test
138///
139/// FIPS 140-3 Software/Firmware Load Test (Section 9.2.2).
140///
141/// Verifies the integrity of the cryptographic module at power-up by
142/// computing an HMAC-SHA256 digest of the on-disk module artifact and
143/// comparing it against the build-time-recorded expected value.
144///
145/// # Module location
146///
147/// The module path is resolved via `std::env::current_exe()` and then
148/// cross-checked with [`path_looks_like_latticearc_module`]. If the
149/// resolved path does not look like a LatticeArc shared library or
150/// LatticeArc-bearing binary (e.g. when the library is loaded by a
151/// host interpreter and `current_exe()` returns the interpreter
152/// itself), the test returns an explicit "cannot locate" error rather
153/// than HMACing the wrong file. Without `unsafe`, which the workspace
154/// `unsafe_code` lint forbids, there is no portable way to call the
155/// platform dynamic-loader APIs (`dladdr`, `dl_iterate_phdr`,
156/// `GetModuleFileName`) that would recover the library's path
157/// directly; the dynamic-load case must be handled by the deployment
158/// (e.g. by also shipping a static-link CLI that runs the integrity
159/// test out-of-band).
160///
161/// # Errors
162///
163/// Returns error if:
164/// - `current_exe()` is unavailable
165/// - The resolved path does not look like a LatticeArc artifact
166/// - The artifact cannot be read
167/// - HMAC computation fails
168/// - The computed HMAC does not match the build-time expected value
169pub fn integrity_test() -> Result<()> {
170    // FIPS requires using a cryptographic key for HMAC
171    // For a self-contained integrity test, we use a deterministic key derived
172    // from the module identity. In production FIPS, this would come from HSM/TPM.
173    const INTEGRITY_KEY: &[u8] = crate::types::domains::MODULE_INTEGRITY_HMAC_KEY;
174
175    // Locate the latticearc module binary on disk.
176    //
177    // `std::env::current_exe()` returns the path to the *host* binary,
178    // which is correct only when latticearc is statically linked into
179    // that binary. When latticearc is loaded as a `.so`/`.dylib`/`.dll`
180    // (e.g. from a Python or Node.js extension), `current_exe()` points
181    // at the host interpreter, and HMACing it would silently verify
182    // the wrong file.
183    //
184    // Without `unsafe` (forbidden crate-wide) we cannot call
185    // platform dynamic-loader APIs (`dladdr`, `dl_iterate_phdr`,
186    // `GetModuleFileName`) to recover the library's own path. Instead
187    // we read `current_exe()`, then check whether the resolved file
188    // name matches one of the LatticeArc artifact names compiled
189    // into this build (`liblatticearc.so`, `liblatticearc.dylib`,
190    // `latticearc.dll`, or any binary that links them statically).
191    // If the path looks like a host-process executable rather than the
192    // LatticeArc library, return an explicit "cannot locate" error so
193    // FIPS callers see the integrity gap rather than a silent
194    // false-positive verification of the wrong file.
195    let module_path = std::env::current_exe().map_err(|e| LatticeArcError::ValidationError {
196        message: format!("Integrity test: cannot locate module binary: {e}"),
197    })?;
198
199    if !path_looks_like_latticearc_module(&module_path) {
200        return Err(LatticeArcError::ValidationError {
201            message: format!(
202                "Integrity test: current_exe() = {:?} does not appear to be a \
203                 LatticeArc library or a binary that statically links it. \
204                 This build cannot verify dynamic-library integrity without \
205                 platform dynamic-loader APIs (forbidden by the workspace \
206                 `unsafe_code` lint). Run the integrity test from a binary \
207                 that statically links latticearc, or supply an external \
208                 library path via the FIPS deployment manifest.",
209                module_path,
210            ),
211        });
212    }
213
214    // Read the module binary with an explicit upper bound. The path is
215    // `current_exe()` (not adversary-controlled), but a runaway binary
216    // size or a /dev/* substitution could still OOM the process. 512 MB
217    // is well above any realistic statically-linked LatticeArc binary
218    // and well below the smallest deployment target's RAM ceiling.
219    const MAX_MODULE_SIZE: u64 = 512 * 1024 * 1024;
220    use std::io::Read;
221    let f = std::fs::File::open(&module_path).map_err(|e| LatticeArcError::ValidationError {
222        message: format!("Integrity test: cannot open module binary: {e}"),
223    })?;
224    let mut module_bytes = Vec::new();
225    f.take(MAX_MODULE_SIZE + 1).read_to_end(&mut module_bytes).map_err(|e| {
226        LatticeArcError::ValidationError {
227            message: format!("Integrity test: cannot read module binary: {e}"),
228        }
229    })?;
230    if module_bytes.len() as u64 > MAX_MODULE_SIZE {
231        return Err(LatticeArcError::ValidationError {
232            message: format!("Integrity test: module binary exceeds {MAX_MODULE_SIZE}-byte cap"),
233        });
234    }
235
236    // Compute HMAC-SHA256 over the module binary via the primitives wrapper
237    // (FIPS-validated aws-lc-rs backend).
238    let computed_hmac = crate::primitives::mac::hmac::hmac_sha256(INTEGRITY_KEY, &module_bytes)
239        .map_err(|e| LatticeArcError::ValidationError {
240            message: format!("Integrity test: HMAC computation failed: {e}"),
241        })?;
242
243    // In a production FIPS module, the expected HMAC would be:
244    // 1. Computed in a secure build environment
245    // 2. Stored in tamper-evident storage (HSM, TPM, or signed manifest)
246    // 3. Verified against the runtime-computed value
247    //
248    // For this implementation, we use a reference HMAC that represents the
249    // "known-good" state. This demonstrates the verification mechanism.
250    //
251    // NOTE: The expected HMAC must be updated whenever the module is recompiled.
252    // For production FIPS certification, implement automated HMAC generation
253    // in the build pipeline.
254
255    // Expected HMAC generated by build script
256    // The build script creates a file defining EXPECTED_HMAC in OUT_DIR
257    // We include it here to get the constant value
258    mod generated {
259        include!(concat!(env!("OUT_DIR"), "/integrity_hmac.rs"));
260    }
261    let expected_hmac = generated::EXPECTED_HMAC;
262
263    // "No HMAC configured" is a deployment-configuration condition, NOT a
264    // tamper detection. The integrity test couldn't run — that's different
265    // from "the integrity test ran and detected tamper". We always return
266    // Ok here (with a loud warning) so `initialize_and_test`'s FIPS 140-3
267    // §9.1 abort path fires ONLY on real tamper (HMAC mismatch) or KAT
268    // failures, never on a missing config file. Under the
269    // `fips-strict-integrity` feature, `verify_operational` separately
270    // rejects operational entry when `INTEGRITY_TEST_CONFIGURED` is still
271    // false at gate time — that's the right layer to enforce "config
272    // must be present", because the caller can handle a Result whereas
273    // initialize_and_test can only abort.
274    let Some(expected_hmac) = expected_hmac else {
275        tracing::warn!(
276            hmac_computed = ?computed_hmac.as_slice(),
277            "FIPS integrity test SKIPPED — no PRODUCTION_HMAC.txt configured. \
278             Module-integrity verification did NOT run. Provision \
279             PRODUCTION_HMAC.txt for FIPS 140-3 §9 compliance; under the \
280             fips-strict-integrity feature, verify_operational will refuse \
281             to enter operational state until this is configured."
282        );
283        #[expect(
284            clippy::print_stderr,
285            reason = "Operator-facing diagnostic — must surface even when tracing is unconfigured"
286        )]
287        {
288            eprintln!("WARNING: FIPS Integrity Test SKIPPED (no PRODUCTION_HMAC.txt)");
289            eprintln!("   Module-integrity verification did NOT run.");
290            eprintln!("   Computed HMAC: {:02x?}", computed_hmac.as_slice());
291            eprintln!(
292                "   Build with --features fips (or --features fips-strict-integrity) \
293                 and provision PRODUCTION_HMAC.txt for FIPS 140-3 §9 compliance."
294            );
295        }
296        // INTEGRITY_TEST_CONFIGURED stays false — strict-integrity gate
297        // in `verify_operational` reads this and refuses Ok.
298        return Ok(());
299    };
300    // From here on, an HMAC IS configured and a mismatch indicates real
301    // tamper — record the fact so the strict-integrity gate knows a real
302    // check ran. Store before the comparison so a panicking compare (which
303    // cannot happen with `ct_eq` but is belt-and-suspenders) doesn't leave
304    // the flag false on a path where tamper detection actually executed.
305    INTEGRITY_TEST_CONFIGURED.store(true, Ordering::SeqCst);
306
307    // Constant-time comparison using subtle crate
308    use subtle::ConstantTimeEq;
309    let hmac_match = computed_hmac.ct_eq(expected_hmac);
310
311    if hmac_match.into() {
312        Ok(())
313    } else {
314        // Integrity violation detected
315        Err(LatticeArcError::ValidationError {
316            message: "FIPS Integrity Test FAILED: Module binary has been modified or corrupted. \
317                     This is a critical security violation."
318                .to_string(),
319        })
320    }
321}

331    #[test]
332    fn test_path_looks_like_latticearc_module_accepts_known_shapes() {
333        use std::path::PathBuf;
334
335        // Exact-name binaries (CLI / dynamic library names).
336        for name in [
337            "liblatticearc.so",
338            "liblatticearc.dylib",
339            "latticearc.dll",
340            "latticearc",
341            "latticearc.exe",
342            "latticearc-cli",
343            "latticearc-cli.exe",
344        ] {
345            assert!(
346                path_looks_like_latticearc_module(&PathBuf::from(name)),
347                "exact-name binary should be accepted: {name}"
348            );
349        }
350
351        // Cargo-generated `<crate>-<16-hex>` binaries under `target`.
352        // The containing directory is deliberately not pinned: cargo
353        // treats the build-dir layout as internal and has already moved
354        // these binaries once (layout v1 `deps/` -> layout v2
355        // `build/<pkg>/<hash>/out/`).
356        for path in [
357            // Layout v1: standard `target/<profile>/deps/`.
358            "/work/repo/target/release/deps/latticearc-0123456789abcdef",
359            "/work/repo/target/debug/deps/audit_regression_signatures-fedcba9876543210",
360            // Layout v1 with an explicit `--target <triple>`.
361            "/work/repo/target/x86_64-unknown-linux-gnu/debug/deps/latticearc-53a673efabc83b87",
362            // Custom profile (e.g. `[profile.valgrind]`).
363            "/work/repo/target/valgrind/deps/latticearc-0123456789abcdef",
364            // Directly under the profile dir, with no intervening
365            // artifact directory at all. Accepted because the trust
366            // scope is `target/` plus the cargo file-name shape, not
367            // any particular subdirectory — the same reason layout v2
368            // is accepted below.
369            "/work/repo/target/release/latticearc-0123456789abcdef",
370            // `cargo llvm-cov` nested target dir.
371            "/work/repo/target/llvm-cov-target/release/deps/latticearc-0123456789abcdef",
372            // Layout v2 (`build/<pkg>/<hash>/out/`), as produced by the
373            // sanitizer jobs' `cargo +nightly test -Z build-std
374            // --target x86_64-unknown-linux-gnu`. This exact path shape
375            // aborted the FIPS POST on every sanitizer run once cargo
376            // enabled layout v2 by default on nightly.
377            "/work/repo/target/x86_64-unknown-linux-gnu/debug/build/latticearc/\
378             2dfe6b1649723639/out/latticearc-2dfe6b1649723639",
379            // Layout v2 without an explicit target triple.
380            "/work/repo/target/debug/build/latticearc/2dfe6b1649723639/out/\
381             latticearc-2dfe6b1649723639",
382        ] {
383            assert!(
384                path_looks_like_latticearc_module(&PathBuf::from(path)),
385                "cargo-generated binary under target/ should be accepted: {path}"
386            );
387        }
388    }

390    #[test]
391    fn test_path_looks_like_latticearc_module_rejects_adversarial_shapes() {
392        use std::path::PathBuf;
393
394        for path in [
395            // Wrong hex-suffix length (15 vs required 16).
396            "/work/repo/target/release/deps/latticearc-0123456789abcde",
397            // Suffix is not hex.
398            "/work/repo/target/release/deps/latticearc-evil-not-hex-here",
399            // Not under any `target` ancestor — host interpreter case.
400            // This is the case the helper actually exists to catch: a
401            // Python/Node process that dlopen'd liblatticearc.
402            "/usr/bin/python3.12",
403            "/opt/node/bin/node",
404            // Cargo-shaped file name, but no `target` ancestor: an
405            // installed artifact is not a build-tree artifact.
406            "/usr/lib/python3/dist-packages/latticearc-0123456789abcdef",
407            // `target` exists but is further up than the ancestor walk
408            // goes, so the walk stops before reaching it rather than
409            // climbing to the filesystem root. Deeper than any real
410            // cargo/llvm-cov layout — this pins the walk's termination,
411            // not a trust boundary.
412            "/builds/foo/target/a/b/c/d/e/f/g/h/i/j/k/deps/latticearc-0123456789abcdef",
413        ] {
414            assert!(
415                !path_looks_like_latticearc_module(&PathBuf::from(path)),
416                "non-LatticeArc path should be rejected: {path}"
417            );
418        }
419    }
