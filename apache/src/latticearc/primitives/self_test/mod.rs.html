1//! FIPS 140-3 Self-Test Module
2//!
3//! This module provides power-up and conditional self-tests for FIPS 140-3 compliance.
4//! According to FIPS 140-3 IG 10.3.A, cryptographic modules must perform Known Answer
5//! Tests (KATs) at power-up before any cryptographic operation can be performed.
6//!
7//! This is the **canonical** FIPS 140-3 self-test module for the LatticeArc
8//! cryptographic module. The `latticearc-tests` workspace crate provides test/validation
9//! utilities for development; this module contains the production self-tests.
10//!
11//! ## Power-Up Self-Tests
12//!
13//! The following algorithms are tested at power-up:
14//! - SHA-256: Cryptographic hash function (FIPS 180-4)
15//! - SHA3-256: Cryptographic hash function (FIPS 202)
16//! - HMAC-SHA256: Message authentication code (FIPS 198-1)
17//! - HKDF-SHA256: Key derivation function (NIST SP 800-56C)
18//! - AES-256-GCM: Authenticated encryption (NIST SP 800-38D)
19//! - ML-KEM-768: Key encapsulation mechanism (FIPS 203)
20//! - ML-DSA-44: Digital signatures (FIPS 204)
21//! - SLH-DSA-SHAKE-128s: Hash-based signatures (FIPS 205)
22//! - FN-DSA-512: Lattice-based signatures (draft FIPS 206)
23//!
24//! ## Usage
25//!
26//! ```no_run
27//! use latticearc::primitives::self_test::run_power_up_tests;
28//!
29//! let result = run_power_up_tests();
30//! assert!(result.is_pass(), "FIPS 140-3 power-up self-tests must pass");
31//! ```
32//!
33//! ## FIPS 140-3 Compliance Notes
34//!
35//! - All KATs use NIST-approved test vectors where available
36//! - Test vectors are hardcoded to ensure deterministic verification
37//! - Any self-test failure should result in the module entering an error state
38//! - No cryptographic services should be provided after a self-test failure
39

76// =============================================================================
77// Module State Management
78// =============================================================================
79//
80// These two flags are genuinely shared across the split: `kat` and
81// `integrity` don't touch them, but `post` (power-up orchestration) and
82// `error_state` (the FIPS §9.6 error-state machine) both read and write
83// `SELF_TEST_PASSED`, and both `integrity` (sets it) and `post` (reads it
84// in the strict-integrity gate) touch `INTEGRITY_TEST_CONFIGURED`.
85// Declared here, at the shared parent, rather than arbitrarily homed in
86// one submodule: a plain private item declared in this file is visible to
87// `self_test` and all of its descendant modules (including each
88// submodule's own `tests`), which is exactly the access shape needed
89// without exceeding the `pub(super)` promotion ceiling used elsewhere in
90// this split.
91
92static SELF_TEST_PASSED: AtomicBool = AtomicBool::new(false);
93
94/// Records whether the pre-operational integrity test had a real expected
95/// HMAC to compare against. Set to `true` by `integrity_test` only after a
96/// configured `EXPECTED_HMAC` was consumed; stays `false` if the test
97/// short-circuited because no `PRODUCTION_HMAC.txt` was provisioned. The
98/// `fips-strict-integrity` branch in `verify_operational` reads this flag
99/// to refuse operational entry on a deployment-misconfiguration that
100/// previously would have silently passed.
101static INTEGRITY_TEST_CONFIGURED: AtomicBool = AtomicBool::new(false);

103// =============================================================================
104// Shared test support
105// =============================================================================
106//
107// `FipsStateGuard` is used by tests in both `error_state` (which owns
108// `restore_operational_state`) and `post` (which owns
109// `verify_operational`/`initialize_and_test`). Declared here for the same
110// reason as the statics above: a private item here is visible to every
111// submodule's `tests` without needing anything above `pub(super)`.
112
113#[cfg(test)]
114/// RAII guard that restores `SELF_TEST_PASSED = true` and clears
115/// the module error state when dropped. Tests that deliberately
116/// set the FIPS module to an error state to exercise blocking
117/// behaviour MUST instantiate this at the top of the test body
118/// so that — even on panic — the global state is restored before
119/// the next test (which may run in parallel and depend on
120/// `is_module_operational()` returning true) starts. Without
121/// this, a parallel runner schedules a state-reading test
122/// during the brief window where SELF_TEST_PASSED is false and
123/// it (correctly) reports the module as non-operational, then
124/// fails its own assertion.
125///
126/// `serial_test`-style serialisation is strictly worse here: it
127/// widens the false-state window from "during this test only" to
128/// "during this test AND every concurrent reader the queue blocks
129/// on", cascading the failure into hundreds of unrelated tests.
130/// The guard pattern keeps the window scoped to the test body.
131///
132/// `_not_send_or_sync: PhantomData<*mut ()>` makes the guard
133/// neither `Send` nor `Sync` at compile time (raw pointers are
134/// neither). Without it, a zero-size struct is implicitly
135/// `Send + Sync`, so a test that accidentally moves the guard
136/// into a `thread::spawn(..)` closure would restore the FIPS
137/// state on the spawned thread's drop — racing with the main
138/// test body and re-opening exactly the false-state-leak we are
139/// guarding against. The marker raises that to a compile error.
140struct FipsStateGuard {
141    _not_send_or_sync: core::marker::PhantomData<*mut ()>,
142}
143#[cfg(test)]
144impl FipsStateGuard {
145    const fn new() -> Self {
146        Self { _not_send_or_sync: core::marker::PhantomData }
147    }
148}
149#[cfg(test)]
150impl Drop for FipsStateGuard {
151    fn drop(&mut self) {
152        // `restore_operational_state` is the canonical "bring
153        // the module back to operational" entry point used by
154        // the `clear_error_state` doc-tests. It sets
155        // SELF_TEST_PASSED = true (SeqCst) and clears the error
156        // code + timestamp.
157        restore_operational_state();
158    }
159}
