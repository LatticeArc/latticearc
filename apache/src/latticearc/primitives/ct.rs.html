1//! Constant-time helpers usable across primitive modules.
2//!
3//! Centralises CT predicates that several primitives share (AEAD weak-key
4//! check, KDF salt validation, tag/MAC comparison, ...) so each call site cannot accidentally
5//! reach for the early-exiting `iter().all()` form. All functions here use
6//! [`subtle::ConstantTimeEq`] under the hood.
7
8use subtle::ConstantTimeEq;
9
10/// Constant-time equality of two byte slices.
11///
12/// Use this for comparing secret or attacker-influenced values (MAC tags,
13/// derived keys, hashes of secrets) instead of `==`, which exits at the
14/// first differing byte.
15///
16/// Slices of different lengths compare unequal and return `false`
17/// immediately: lengths are treated as public, so this early return is
18/// not a timing leak for fixed-size values. For slices of equal length the
19/// running time is independent of the contents and of the position of any
20/// difference. Two empty slices are equal.
21///
22/// When the length is known at compile time, prefer [`ct_eq_array`].
23#[inline]
24#[must_use]
25pub fn ct_eq(a: &[u8], b: &[u8]) -> bool {
26    if a.len() != b.len() {
27        return false;
28    }
29    bool::from(a.ct_eq(b))
30}
31
32/// Constant-time equality of two fixed-length byte arrays.
33///
34/// Like [`ct_eq`], but the shared length `N` is enforced by the type
35/// system, so a length mismatch cannot occur and no length branch is taken.
36/// Two zero-length arrays are equal.
37#[inline]
38#[must_use]
39pub fn ct_eq_array<const N: usize>(a: &[u8; N], b: &[u8; N]) -> bool {
40    bool::from(a.as_slice().ct_eq(b.as_slice()))
41}
42
43/// Constant-time all-zero check for fixed-length byte arrays.
44///
45/// Prefer this over [`is_all_zero_bytes`] whenever the length is known
46/// at compile time: the loop iteration count is `ceil(N / 32)` — a
47/// constant — so the caller cannot accidentally pass a runtime-derived
48/// length that would leak through timing. AEAD key validation (16-,
49/// 24-, 32-byte keys) is the canonical call site.
50///
51/// Returns `false` when `N == 0` so a zero-length array cannot trigger
52/// a false-positive weak-input rejection.
53#[inline]
54#[must_use]
55pub fn is_all_zero<const N: usize>(bytes: &[u8; N]) -> bool {
56    if N == 0 {
57        return false;
58    }
59    is_all_zero_bytes(bytes.as_slice())
60}
61
62/// Returns `true` iff every byte of `bytes` is zero, in constant time
63/// **with respect to byte contents**.
64///
65/// Empty slices return `false` so that a caller passing a zero-length
66/// secret cannot trigger a false-positive weak-input rejection.
67///
68/// # Constant-time scope and a length-leak caveat
69///
70/// Runtime is independent of *which byte* is non-zero, so an attacker
71/// who can time this function cannot learn the position of the first
72/// difference. The loop, however, iterates `ceil(bytes.len() / 32)`
73/// times — the iteration count itself is **not constant-time over the
74/// input length**. For variable-length secret inputs (KDF output of
75/// caller-chosen size, secret-length nonces, etc.) the timing leaks
76/// the length.
77///
78/// **Production callers must validate `bytes.len()` before calling this
79/// function** (which the current sole caller `aead::is_all_zero_key`
80/// does, by way of the AEAD constructors' fixed-size key requirement).
81/// For new fixed-length use sites, prefer the const-generic
82/// [`is_all_zero`] above — its type signature makes the length-non-
83/// secrecy precondition compile-time enforced.
84#[inline]
85#[must_use]
86pub fn is_all_zero_bytes(bytes: &[u8]) -> bool {
87    if bytes.is_empty() {
88        return false;
89    }
90    const CHUNK: usize = 32;
91    let zero = [0u8; CHUNK];
92    let mut acc = subtle::Choice::from(1u8);
93    let mut i = 0usize;
94    while i < bytes.len() {
95        let end = i.saturating_add(CHUNK).min(bytes.len());
96        let len = end.saturating_sub(i);
97        let chunk = bytes.get(i..end).unwrap_or(&[]);
98        let zero_slice = zero.get(..len).unwrap_or(&zero);
99        acc &= chunk.ct_eq(zero_slice);
100        i = end;
101    }
102    bool::from(acc)
103}

105#[cfg(test)]
106mod tests {
107    use super::{ct_eq, ct_eq_array, is_all_zero_bytes};
108
109    #[test]
110    fn test_all_zero_returns_true() {
111        assert!(is_all_zero_bytes(&[0u8; 16]));
112        assert!(is_all_zero_bytes(&[0u8; 32]));
113        assert!(is_all_zero_bytes(&[0u8; 33]));
114        assert!(is_all_zero_bytes(&[0u8; 64]));
115        assert!(is_all_zero_bytes(&[0u8; 100]));
116    }
117
118    #[test]
119    fn test_any_nonzero_returns_false() {
120        let mut bytes = [0u8; 32];
121        bytes[0] = 1;
122        assert!(!is_all_zero_bytes(&bytes));
123        let mut bytes = [0u8; 32];
124        bytes[31] = 1;
125        assert!(!is_all_zero_bytes(&bytes));
126        let mut bytes = [0u8; 64];
127        bytes[63] = 1;
128        assert!(!is_all_zero_bytes(&bytes));
129    }
130
131    #[test]
132    fn test_empty_returns_false() {
133        // Empty inputs are not "weak" — they're just unsupported. Returning
134        // `false` prevents downstream callers from rejecting an empty key
135        // as weak-zero when the actual issue is length validation.
136        assert!(!is_all_zero_bytes(&[]));
137    }
138
139    #[test]
140    fn test_oversize_chunked_path() {
141        // Larger than CHUNK (32) and not a multiple, exercises the partial-
142        // last-chunk branch. PBKDF2 salts in the wild are 16-64 bytes.
143        assert!(is_all_zero_bytes(&[0u8; 65]));
144        let mut bytes = [0u8; 65];
145        bytes[64] = 1;
146        assert!(!is_all_zero_bytes(&bytes));
147    }
148
149    #[test]
150    fn test_ct_eq_equal_slices() {
151        assert!(ct_eq(b"abcdef", b"abcdef"));
152        assert!(ct_eq(&[0u8; 100], &[0u8; 100]));
153    }
154
155    #[test]
156    fn test_ct_eq_empty_slices_are_equal() {
157        assert!(ct_eq(&[], &[]));
158    }
159
160    #[test]
161    fn test_ct_eq_detects_difference_at_every_position() {
162        let base = [0x5au8; 65];
163        for i in 0..base.len() {
164            let mut other = base;
165            if let Some(byte) = other.get_mut(i) {
166                *byte ^= 0x01;
167            }
168            assert!(!ct_eq(&base, &other), "difference at {i} not detected");
169        }
170    }
171
172    #[test]
173    fn test_ct_eq_length_mismatch_is_false() {
174        assert!(!ct_eq(b"abc", b"abcd"));
175        assert!(!ct_eq(b"abcd", b"abc"));
176        assert!(!ct_eq(&[], &[0u8]));
177        // A shared prefix must not make a shorter slice compare equal.
178        assert!(!ct_eq(&[0u8; 32], &[0u8; 33]));
179    }
180
181    #[test]
182    fn test_ct_eq_array_equal_and_different() {
183        assert!(ct_eq_array(&[7u8; 32], &[7u8; 32]));
184        let mut other = [7u8; 32];
185        other[31] = 8;
186        assert!(!ct_eq_array(&[7u8; 32], &other));
187        other = [7u8; 32];
188        other[0] = 0;
189        assert!(!ct_eq_array(&[7u8; 32], &other));
190    }
191
192    #[test]
193    fn test_ct_eq_array_zero_length_is_equal() {
194        assert!(ct_eq_array(&[], &[]));
195    }
