6//! Hybrid Digital Signatures Module
7//!
8//! This module provides hybrid digital signatures that combine post-quantum
9//! (ML-DSA) and classical (Ed25519) signature algorithms for enhanced security
10//! during the quantum transition period.
11//!
12//! # Architecture
13//!
14//! ```text
15//! ┌─────────────────────────────────────────────────────────────────────────┐
16//! │                    HYBRID SIGNATURE: Signing Flow                       │
17//! ├─────────────────────────────────────────────────────────────────────────┤
18//! │                                                                         │
19//! │  ┌─────────────┐                                                        │
20//! │  │   Message   │                                                        │
21//! │  │     M       │                                                        │
22//! │  └──────┬──────┘                                                        │
23//! │         │                                                               │
24//! │         ├────────────────────────────────────────┐                      │
25//! │         │                                        │                      │
26//! │         ▼                                        ▼                      │
27//! │  ┌──────────────────────┐              ┌──────────────────────┐         │
28//! │  │    ML-DSA-65 Sign    │              │   Ed25519 Sign       │         │
29//! │  │                      │              │                      │         │
30//! │  │  SK_pq + M ──► σ_pq  │              │  SK_ed + M ──► σ_ed  │         │
31//! │  │    (3309 bytes)      │              │    (64 bytes)        │         │
32//! │  └──────────┬───────────┘              └──────────┬───────────┘         │
33//! │             │                                     │                     │
34//! │             └────────────────┬────────────────────┘                     │
35//! │                              │                                          │
36//! │                              ▼                                          │
37//! │                  ┌───────────────────────┐                              │
38//! │                  │  Hybrid Signature     │                              │
39//! │                  │  σ = σ_pq ║ σ_ed      │                              │
40//! │                  │  (3309 + 64 = 3373 B) │                              │
41//! │                  └───────────────────────┘                              │
42//! │                                                                         │
43//! └─────────────────────────────────────────────────────────────────────────┘
44//!
45//! ┌─────────────────────────────────────────────────────────────────────────┐
46//! │                    HYBRID SIGNATURE: Verification Flow                  │
47//! ├─────────────────────────────────────────────────────────────────────────┤
48//! │                                                                         │
49//! │  ┌─────────────┐         ┌───────────────────────┐                      │
50//! │  │   Message   │         │  Hybrid Signature     │                      │
51//! │  │     M       │         │  σ = σ_pq ║ σ_ed      │                      │
52//! │  └──────┬──────┘         └───────────┬───────────┘                      │
53//! │         │                            │                                  │
54//! │         │  ┌─────────────────────────┴─────────────────────────┐        │
55//! │         │  │                                                   │        │
56//! │         │  │ Parse: first 3309 bytes = σ_pq, last 64 = σ_ed   │        │
57//! │         │  │                                                   │        │
58//! │         │  └──────────────────┬────────────────────────────────┘        │
59//! │         │                     │                                         │
60//! │         ├────────────────────┬┴───────────────────┐                     │
61//! │         │                    │                    │                     │
62//! │         ▼                    ▼                    ▼                     │
63//! │  ┌────────────────┐   ┌────────────────┐   ┌────────────────┐           │
64//! │  │ ML-DSA Verify  │   │                │   │ Ed25519 Verify │           │
65//! │  │                │   │                │   │                │           │
66//! │  │ PK_pq, M, σ_pq │   │     AND        │   │ PK_ed, M, σ_ed │           │
67//! │  │       │        │   │                │   │       │        │           │
68//! │  └───────┼────────┘   └───────┬────────┘   └───────┼────────┘           │
69//! │          │                    │                    │                    │
70//! │          ▼                    ▼                    ▼                    │
71//! │       ┌─────┐             ┌──────┐             ┌─────┐                  │
72//! │       │ OK? │─────────────┤ BOTH ├─────────────│ OK? │                  │
73//! │       └─────┘             └──┬───┘             └─────┘                  │
74//! │                              │                                          │
75//! │                              ▼                                          │
76//! │                     ┌────────────────┐                                  │
77//! │                     │  Valid = true  │                                  │
78//! │                     │  iff BOTH pass │                                  │
79//! │                     └────────────────┘                                  │
80//! │                                                                         │
81//! └─────────────────────────────────────────────────────────────────────────┘
82//! ```
83//!
84//! # Key Sizes Summary
85//!
86//! | Component   | Public Key | Secret Key | Signature  |
87//! |-------------|------------|------------|------------|
88//! | ML-DSA-65   | 1952 B     | 4032 B     | 3309 B     |
89//! | Ed25519     | 32 B       | 32 B       | 64 B       |
90//! | **Hybrid**  | **1984 B** | **4064 B** | **3373 B** |
91//!
92//! # Security Properties
93//!
94//! - **EUF-CMA** (Existential Unforgeability under Chosen Message Attack) security
95//! - **AND-composition**: Requires breaking BOTH ML-DSA AND Ed25519 to forge
96//! - Automatic memory zeroization for secret keys via [`ZeroizeOnDrop`]
97//!
98//! # Transcript modes
99//!
100//! Every hybrid signature binds a per-scheme label into both legs (the
101//! "H1 / M1" transcript fix). 0.13 adds two further modes, each with its own
102//! label family so no signature from one mode verifies under another, even
103//! if one leg is broken:
104//!
105//! | Mode | Entry points | ML-DSA leg | Ed25519 leg |
106//! |------|--------------|------------|-------------|
107//! | plain | [`sign`] / [`verify`] | `ML-DSA.Sign(M, ctx = LABEL)` | `Ed25519(SHA-512(LABEL ‖ 0x00 ‖ M))` |
108//! | context | [`sign_with_context`] / [`verify_with_context`] | `ML-DSA.Sign(M, ctx = LABEL-ctx ‖ 0x00 ‖ app_ctx)` | `Ed25519(SHA-512(LABEL-ctx ‖ 0x00 ‖ ‖app_ctx‖ ‖ app_ctx ‖ M))` |
109//! | pre-hashed | [`sign_prehashed`] / [`verify_prehashed`] | `HashML-DSA.Sign(SHA-512(M), ctx = LABEL-ph ‖ 0x00 ‖ app_ctx)` | `Ed25519ph(SHA-512(M), ctx = LABEL-ph ‖ 0x00 ‖ app_ctx)` |
110//!
111//! The context and pre-hashed modes require a non-empty application context
112//! of at most [`MAX_HYBRID_SIG_CONTEXT_LEN`] bytes. The pre-hashed mode takes
113//! a caller-computed SHA-512 digest, so content of any size can be signed
114//! without the 64 KiB message cap that bounds the other two modes.
115//! [`generate_keypair_from_seed`] derives both legs from one 32-byte seed.
116//!
117//! # Example
118//!
119//! ```rust,no_run
120//! # fn main() -> Result<(), Box<dyn std::error::Error>> {
121//! use latticearc::hybrid::sig_hybrid::{generate_keypair, sign, verify};
122//!
123//! // Generate hybrid keypair
124//! let (pk, sk) = generate_keypair()?;
125//!
126//! // Sign a message (deterministic - no RNG needed)
127//! let message = b"Hello, hybrid signatures!";
128//! let signature = sign(&sk, message)?;
129//!
130//! // Verify the signature
131//! let is_valid = verify(&pk, message, &signature)?;
132//! assert!(is_valid);
133//! # Ok(())
134//! # }
135//! ```
136//!
137//! [`ZeroizeOnDrop`]: zeroize::ZeroizeOnDrop

175/// Error types for hybrid signature operations.
176///
177/// This enum captures all possible error conditions that can occur during
178/// hybrid signature generation and verification.
179// PartialEq is intentionally not derived: crypto error types should be
180// inspected by variant (`matches!`) rather than value-compared. The
181// `String`-carrying variants would otherwise compare upstream error
182// messages, which is too brittle.
183#[non_exhaustive]
184#[derive(Debug, Clone, Error)]
185pub enum HybridSignatureError {
186    /// Error during ML-DSA signature operations.
187    ///
188    /// Retained for keygen-side reporting where the failure is not
189    /// adversary-reachable. The sign path now folds component failures
190    /// into the opaque [`SigningFailed`](Self::SigningFailed) variant
191    /// per Pattern 6 (DP-M3).
192    #[error("ML-DSA error: {0}")]
193    MlDsaError(String),
194    /// Error during Ed25519 signature operations.
195    ///
196    /// Retained for keygen-side reporting where the failure is not
197    /// adversary-reachable. The sign path now folds component failures
198    /// into the opaque [`SigningFailed`](Self::SigningFailed) variant
199    /// per Pattern 6 (DP-M3).
200    #[error("Ed25519 error: {0}")]
201    Ed25519Error(String),
202    /// Hybrid sign failed (component-opaque).
203    ///
204    /// DP-M3: the sign path returns a single variant regardless of
205    /// which leg rejected. Pattern 6 forbids per-component variant
206    /// distinguishability on adversary-reachable paths — a remote
207    /// caller able to influence sign inputs could otherwise match on
208    /// `MlDsaError` vs `Ed25519Error` to fingerprint the failure
209    /// stage. The actual cause is routed to `tracing::debug!` for
210    /// operator-side correlation.
211    #[error("Hybrid signing failed: {0}")]
212    SigningFailed(String),
213    /// Signature verification failed for one or both components.
214    #[error("Signature verification failed: {0}")]
215    VerificationFailed(String),
216    /// Invalid key material provided (wrong length, format, etc.).
217    #[error("Invalid key material: {0}")]
218    InvalidKeyMaterial(String),
219    /// General cryptographic operation failure.
220    #[error("Cryptographic operation failed: {0}")]
221    CryptoError(String),
222    /// The application context passed to a context-taking entry point is
223    /// empty or longer than [`MAX_HYBRID_SIG_CONTEXT_LEN`] bytes. Caller-side
224    /// configuration error, not adversary-reachable.
225    #[error("Invalid signature context: {0}")]
226    InvalidContext(String),
227    /// The pre-hash digest passed to [`sign_prehashed`] / [`verify_prehashed`]
228    /// is not the 64 bytes of a SHA-512 digest. Caller-side error.
229    #[error("Invalid pre-hash digest length: expected {expected}, got {actual}")]
230    InvalidDigestLength {
231        /// Required digest length (64 for SHA-512).
232        expected: usize,
233        /// Length the caller supplied.
234        actual: usize,
235    },
236}

238/// Longest application context (bytes) accepted by [`sign_with_context`],
239/// [`sign_prehashed`] and their verifiers.
240///
241/// Both legs carry `LABEL || 0x00 || context` in a context field whose
242/// length is encoded in one byte (FIPS 204 §5.2 for ML-DSA, RFC 8032 §5.1
243/// `dom2` for Ed25519ph), so the budget is `255 - longest_label - 1`.
244pub const MAX_HYBRID_SIG_CONTEXT_LEN: usize = 208;

313/// Reject an application context that is empty or over budget.
314fn check_context(context: &[u8]) -> Result<(), HybridSignatureError> {
315    if context.is_empty() {
316        return Err(HybridSignatureError::InvalidContext(
317            "context must not be empty (domain separation is required)".to_string(),
318        ));
319    }
320    if context.len() > MAX_HYBRID_SIG_CONTEXT_LEN {
321        return Err(HybridSignatureError::InvalidContext(format!(
322            "context is {} bytes; maximum is {MAX_HYBRID_SIG_CONTEXT_LEN}",
323            context.len()
324        )));
325    }
326    Ok(())
327}

329/// Build the per-leg context string `LABEL || 0x00 || context` for the
330/// context and pre-hashed modes. The label is NUL-free (checked in
331/// `types::domains` tests), so the first `0x00` always delimits it.
332fn build_mode_ctx(label: &'static [u8], context: &[u8]) -> Result<Vec<u8>, HybridSignatureError> {
333    check_context(context)?;
334    let mut ctx = Vec::with_capacity(label.len().saturating_add(1).saturating_add(context.len()));
335    ctx.extend_from_slice(label);
336    ctx.push(0x00);
337    ctx.extend_from_slice(context);
338    Ok(ctx)
339}

343/// Carries the [`MlDsaParameterSet`] that the ML-DSA half was generated

361impl HybridSigPublicKey {
362    /// Construct a `HybridSigPublicKey` from its components, validating
363    /// that each PK matches its parameter set's expected length.
364    ///
365    /// # Errors
366    /// Returns `LatticeArcError::InvalidParameter` if `ml_dsa_pk.len()`
367    /// does not match `parameter_set.public_key_size()`, or if
368    /// `ed25519_pk.len() != 32`.
369    pub fn new(
370        parameter_set: MlDsaParameterSet,
371        ml_dsa_pk: Vec<u8>,
372        ed25519_pk: Vec<u8>,
373    ) -> Result<Self, crate::prelude::LatticeArcError> {
374        let expected_ml_dsa = parameter_set.public_key_size();
375        if ml_dsa_pk.len() != expected_ml_dsa {
376            return Err(crate::prelude::LatticeArcError::InvalidParameter(format!(
377                "ML-DSA public key for {:?}: expected {} bytes, got {}",
378                parameter_set,
379                expected_ml_dsa,
380                ml_dsa_pk.len()
381            )));
382        }
383        if ed25519_pk.len() != 32 {
384            return Err(crate::prelude::LatticeArcError::InvalidParameter(format!(
385                "Ed25519 public key: expected 32 bytes, got {}",
386                ed25519_pk.len()
387            )));
388        }
389        Ok(Self { parameter_set, ml_dsa_pk, ed25519_pk })
390    }

396    }

411/// Hybrid secret key combining ML-DSA and Ed25519
412///
413/// # Security Guarantees
414///
415/// This struct implements automatic memory zeroization via the [`ZeroizeOnDrop`] derive.
416/// When a `HybridSigSecretKey` is dropped (goes out of scope), all secret
417/// key material is immediately overwritten with zeros using constant-time volatile writes.
418/// This prevents secret material from remaining in memory after use.
419///
420/// # Zeroization Implementation
421///
422/// The [`ZeroizeOnDrop`] derive automatically calls [`Zeroize::zeroize()`]
423/// on all fields when the struct is dropped. This happens using volatile
424/// operations that prevent compiler optimization and ensure constant-time execution.
425///
426/// # Cloning
427///
428/// **Important**: This type does NOT implement [`Clone`] to prevent accidental
429/// copying of secret keys. If you need to clone, you must implement it
430/// explicitly with proper security considerations, including zeroizing the copy.
431///
432/// # Memory Safety
433///
434/// - All secret fields are wrapped in `Zeroizing<Vec<u8>>` for explicit zeroization
435/// - Drop implementation ensures zeroization even on panic
436/// - Constant-time operations prevent timing side-channels
437///
438/// # Example
439///
440/// ```rust,no_run
441/// use latticearc::hybrid::sig_hybrid::generate_keypair;
442///
443/// // Generate keypair
444/// let (pk, sk) = generate_keypair().expect("keypair generation failed");
445///
446/// //.. use sk for cryptographic operations..
447///
448/// // Drop secret key - automatically zeroized
449/// drop(sk);  // Secret material automatically zeroized
450/// ```
451#[derive(Zeroize, ZeroizeOnDrop)]
452pub struct HybridSigSecretKey {
453    /// ML-DSA parameter set this SK was generated at. Read by [`sign`]
454    /// to construct the right-sized `MlDsaSecretKey`. `MlDsaParameterSet`
455    /// is `Copy` and contains no secret material, so `#[zeroize(skip)]`.
456    #[zeroize(skip)]
457    parameter_set: MlDsaParameterSet,
458    /// ML-DSA secret key bytes (size depends on parameter set), automatically zeroized on drop.
459    ml_dsa_sk: Zeroizing<Vec<u8>>,
460    /// Ed25519 secret key bytes (32 bytes), automatically zeroized on drop.
461    ed25519_sk: Zeroizing<Vec<u8>>,
462}

478impl ConstantTimeEq for HybridSigSecretKey {
479    fn ct_eq(&self, other: &Self) -> subtle::Choice {
480        // Parameter-set match is a precondition for equality. ML-DSA-44
481        // SK bytes have a different length from ML-DSA-87 SK bytes, so a
482        // length-mismatch slice ct_eq would already return Choice(0) —
483        // but checking the parameter set explicitly fails earlier and
484        // makes the contract loud.
485        //
486        // Use the canonical `(self as u8).ct_eq(...)` pattern (per
487        // DESIGN_PATTERNS.md:793, Constant-Time Equality reference)
488        // rather than `==`. The parameter set is a public parameter,
489        // so the timing isn't security-sensitive HERE — but the
490        // canonical pattern keeps the call shape uniform with the
491        // secret-bearing legs below, so a future developer copying
492        // this shape into a secret context starts from a CT-safe
493        // primitive. `#[repr(u8)]` on MlDsaParameterSet makes the
494        // cast well-defined.
495        let param_eq = (self.parameter_set as u8).ct_eq(&(other.parameter_set as u8));
496        param_eq
497            & self.ml_dsa_sk.as_slice().ct_eq(other.ml_dsa_sk.as_slice())
498            & self.ed25519_sk.as_slice().ct_eq(other.ed25519_sk.as_slice())
499    }
500}

615/// [`generate_keypair_with_parameter_set`].
616///
617/// # Errors
618///
619/// Returns an error if ML-DSA keypair generation fails or Ed25519 keypair
620/// generation (including its pairwise consistency test) fails.
621///
622/// # Entropy source
623/// ML-DSA and Ed25519 key generation route through the primitives layer,
624/// which uses the OS CSPRNG internally — callers cannot supply an external RNG.

626pub fn generate_keypair() -> Result<(HybridSigPublicKey, HybridSigSecretKey), HybridSignatureError>

631/// Generate a hybrid keypair at a specified ML-DSA parameter set.
632///
633/// `parameter_set` selects the ML-DSA strength of the post-quantum half of
634/// the hybrid keypair. The Ed25519 half is unchanged across parameter sets
635/// (Ed25519 has a single fixed strength).
636///
637/// | `parameter_set` | NIST level | Use case |
638/// |-----------------|------------|----------|
639/// | `MlDsa44`       | 1 (~128-bit) | size-constrained, lowest security tier |
640/// | `MlDsa65`       | 3 (~192-bit) | default, balanced |
641/// | `MlDsa87`       | 5 (~256-bit) | CNSA 2.0, highest security |
642///
643/// # Errors
644///
645/// Returns an error if ML-DSA keypair generation fails for the requested
646/// parameter set, or if Ed25519 keypair generation (including its pairwise
647/// consistency test) fails.
648#[must_use = "generated keypair must be stored or used"]
649pub fn generate_keypair_with_parameter_set(
650    parameter_set: MlDsaParameterSet,
651) -> Result<(HybridSigPublicKey, HybridSigSecretKey), HybridSignatureError> {
652    // DP-H1 fix: opaque keygen errors. Pattern 6 SHOULD-be-opaque scope
653    // covers sign-side keygen as well as the adversary-reachable verify
654    // path — the upstream `fips204` and `ed25519-dalek` Display strings
655    // are not contractually stable and could leak implementation detail
656    // (KEM-fault category, RNG-failure reason, etc.) into the typed
657    // error a remote caller can match on. Route the actual cause to
658    // `tracing::debug!` for operator-side correlation.
659    let (ml_dsa_pk, ml_dsa_sk) = ml_dsa_generate_keypair(parameter_set).map_err(|e| {
660        tracing::debug!(error = %e, "hybrid sig keygen: ML-DSA keypair generation failed");
661        HybridSignatureError::MlDsaError("keypair generation failed".to_string())
662    })?;
663
664    // Generate Ed25519 keypair through the primitives wrapper so all
665    // Ed25519 operations go through a single entry point. The wrapper
666    // performs a pairwise consistency test before returning.
667    let ed25519_kp = Ed25519KeyPair::generate().map_err(|e| {
668        tracing::debug!(error = %e, "hybrid sig keygen: Ed25519 keypair generation failed");
669        HybridSignatureError::Ed25519Error("keypair generation failed".to_string())
670    })?;
671
672    let ed25519_pk = ed25519_kp.public_key_bytes();
673    // secret_key_bytes() returns Zeroizing<Vec<u8>>; move the inner buffer
674    // into the HybridSigSecretKey so the wrapping Zeroizing is preserved.
675    let ed25519_sk_zeroizing = ed25519_kp.secret_key_bytes();
676
677    assemble_keypair(parameter_set, &ml_dsa_pk, &ml_dsa_sk, ed25519_pk, ed25519_sk_zeroizing)
678}

680/// Build the hybrid `(pk, sk)` pair from freshly generated or derived legs.
681///
682/// Routes construction through the validating `HybridSigPublicKey::new`
683/// instead of building the struct literal directly: the production keygen
684/// is the most common construction path and must not be the only one that
685/// skips the length validator. The bytes come from real keygen output so the
686/// validator never rejects in practice; the error branch documents the
687/// invariant.
688fn assemble_keypair(
689    parameter_set: MlDsaParameterSet,
690    ml_dsa_pk: &MlDsaPublicKey,
691    ml_dsa_sk: &MlDsaSecretKey,
692    ed25519_pk: Vec<u8>,
693    ed25519_sk: Zeroizing<Vec<u8>>,
694) -> Result<(HybridSigPublicKey, HybridSigSecretKey), HybridSignatureError> {
695    let pk = HybridSigPublicKey::new(parameter_set, ml_dsa_pk.as_bytes().to_vec(), ed25519_pk)
696        .map_err(|e| {
697            HybridSignatureError::InvalidKeyMaterial(format!(
698                "internal invariant violated: keygen produced a PK that failed length validation: {e}"
699            ))
700        })?;
701
702    let sk = HybridSigSecretKey {
703        parameter_set,
704        ml_dsa_sk: Zeroizing::new(ml_dsa_sk.expose_secret().to_vec()),
705        ed25519_sk,
706    };
707
708    Ok((pk, sk))
709}

711/// The two per-leg seeds derived from one master seed: `(ξ, ed25519_seed)`.
712type HybridSubSeeds = (Zeroizing<[u8; 32]>, Zeroizing<[u8; 32]>);
713
714/// Expand a 32-byte master seed into the two per-leg seeds with HKDF-SHA256:
715/// `ξ = HKDF(seed, info = "latticearc/v1/ml-dsa-seed")` and
716/// `ed25519_seed = HKDF(seed, info = "latticearc/v1/ed25519-seed")`.
717///
718/// HKDF's salt is the RFC 5869 default (`None`): the master seed is required
719/// to be uniformly random (a 256-bit recovery secret), so Extract performs no
720/// entropy conditioning and the two labels alone separate the legs. Each
721/// sub-seed is a PRF output under a distinct label, so holding one leg's seed
722/// reveals nothing about the other's.
723fn derive_hybrid_sub_seeds(seed: &[u8; 32]) -> Result<HybridSubSeeds, HybridSignatureError> {
724    let expand = |info: &[u8]| -> Result<Zeroizing<[u8; 32]>, HybridSignatureError> {
725        let okm = hkdf(seed, None, Some(info), 32).map_err(|e| {
726            tracing::debug!(error = %e, "hybrid sig seed keygen: HKDF sub-seed derivation failed");
727            HybridSignatureError::CryptoError("seed derivation failed".to_string())
728        })?;
729        let mut out = Zeroizing::new([0u8; 32]);
730        let okm_bytes = okm.expose_secret().get(..32).ok_or_else(|| {
731            HybridSignatureError::CryptoError("seed derivation output length".to_string())
732        })?;
733        out.copy_from_slice(okm_bytes);
734        Ok(out)
735    };
736    Ok((expand(HYBRID_SIG_SEED_ML_DSA_INFO)?, expand(HYBRID_SIG_SEED_ED25519_INFO)?))
737}

739/// Derive a hybrid signing keypair deterministically from a 32-byte seed.
740///
741/// One seed (for example a 256-bit secret held in a recovery phrase) yields
742/// the same `(pk, sk)` on every platform and build: the seed is expanded by
743/// HKDF-SHA256 under two fixed labels (see
744/// [`HYBRID_SIG_SEED_ML_DSA_INFO`] / [`HYBRID_SIG_SEED_ED25519_INFO`]) into
745/// the ML-DSA keygen seed `ξ` — fed to FIPS 204 Algorithm 6 via
746/// [`generate_keypair_from_seed`](crate::primitives::sig::ml_dsa::generate_keypair_from_seed)
747/// — and the Ed25519 private key (RFC 8032 §5.1.5). Both legs pass their
748/// pairwise consistency tests before the pair is returned. The determinism
749/// vectors in `tests/src/validation/nist_kat/hybrid_sig_seed_kat.rs` pin the
750/// derivation for all three parameter sets.
751///
752/// Signing keys only: ML-KEM keygen has no seed entry point in the
753/// FIPS-validated backend, so hybrid KEM keys stay random.
754///
755/// # Security
756/// The seed is the whole secret. Callers must keep their copy in a
757/// zeroizing container and treat it like the private key it reproduces.
758///
759/// # Errors
760/// Returns `InvalidKeyMaterial` for an all-zero seed, `CryptoError` if the
761/// HKDF expansion fails, and the component keygen errors if either leg's
762/// pairwise consistency test fails.
763#[must_use = "generated keypair must be stored or used"]
764pub fn generate_keypair_from_seed(
765    seed: &[u8; 32],
766    parameter_set: MlDsaParameterSet,
767) -> Result<(HybridSigPublicKey, HybridSigSecretKey), HybridSignatureError> {
768    if seed.iter().all(|&b| b == 0) {
769        return Err(HybridSignatureError::InvalidKeyMaterial(
770            "seed must not be all zero".to_string(),
771        ));
772    }
773    let (xi, ed25519_seed) = derive_hybrid_sub_seeds(seed)?;
774
775    let (ml_dsa_pk, ml_dsa_sk) =
776        ml_dsa_generate_keypair_from_seed(parameter_set, &xi).map_err(|e| {
777            tracing::debug!(error = %e, "hybrid sig seed keygen: ML-DSA keypair derivation failed");
778            HybridSignatureError::MlDsaError("keypair generation failed".to_string())
779        })?;
780
781    let ed25519_kp = Ed25519KeyPair::from_secret_key(ed25519_seed.as_slice()).map_err(|e| {
782        tracing::debug!(error = %e, "hybrid sig seed keygen: Ed25519 keypair derivation failed");
783        HybridSignatureError::Ed25519Error("keypair generation failed".to_string())
784    })?;
785
786    let ed25519_pk = ed25519_kp.public_key_bytes();
787    let ed25519_sk = ed25519_kp.secret_key_bytes();
788    assemble_keypair(parameter_set, &ml_dsa_pk, &ml_dsa_sk, ed25519_pk, ed25519_sk)
789}

791/// Sign using hybrid signature scheme
792///
793/// Both ML-DSA and Ed25519 signing are deterministic, so no RNG is required.
794///
795/// Planned deprecation (0.14): prefer [`sign_with_context`], which binds a
796/// required application context into both legs. The attribute is not applied
797/// in 0.13 because the crate's own callers (unified API) still route here.
798///
799/// # Errors
800///
801/// Returns an error if:
802/// - The Ed25519 secret key is not exactly 32 bytes.
803/// - ML-DSA secret key construction or signing fails.
804/// - The Ed25519 secret key format is invalid for conversion.
805pub fn sign(
806    sk: &HybridSigSecretKey,
807    message: &[u8],
808) -> Result<HybridSignature, HybridSignatureError> {
809    // H1 / M1 fix: bind the signed transcript to this specific hybrid scheme.
810    // ML-DSA accepts a FIPS 204 §5.2 context parameter natively; the
811    // previous empty `&[]` left the ML-DSA leg byte-identical to a standalone
812    // ML-DSA signature, enabling the hybrid → PQ-only downgrade attack
813    // (auditor H1). Ed25519 (RFC 8032) has no native context, so its leg
814    // signs `SHA-512(scheme_ctx || 0x00 || message)` — the fixed 64-byte
815    // digest also stays under the signature-size cap regardless of message
816    // size (see `tests/hybrid_sig.rs::test_sign_verify_large_message_roundtrip`).
817    let scheme_ctx = sig_context(hybrid_scheme_label(sk.parameter_set));
818    sign_legs(
819        sk,
820        |ml_dsa_sk| ml_dsa_sk.sign(message, scheme_ctx),
821        |ed25519_kp| ed25519_kp.sign(&hash_with_context(scheme_ctx, message)),
822    )
823}

825/// Sign with a required application context bound into both legs.
826///
827/// `context` (1..=[`MAX_HYBRID_SIG_CONTEXT_LEN`] bytes, e.g.
828/// `b"vault/v1/root-sig"`) separates signatures made for different purposes
829/// under one key: the ML-DSA leg receives `LABEL-ctx || 0x00 || context` as
830/// its FIPS 204 context string and the Ed25519 leg signs
831/// `SHA-512(LABEL-ctx || 0x00 || |context| || context || message)`. A
832/// signature verifies only under [`verify_with_context`] with the same
833/// context — never under [`verify`] or [`verify_prehashed`].
834///
835/// The message is bounded by the configured signature resource limit
836/// (64 KiB by default); use [`sign_prehashed`] for larger content.
837///
838/// # Errors
839/// Returns `InvalidContext` if `context` is empty or over budget, otherwise
840/// the same errors as [`sign`].
841pub fn sign_with_context(
842    sk: &HybridSigSecretKey,
843    message: &[u8],
844    context: &[u8],
845) -> Result<HybridSignature, HybridSignatureError> {
846    let label = hybrid_mode_label(sk.parameter_set, HybridTranscriptMode::Context);
847    let ml_dsa_ctx = build_mode_ctx(label, context)?;
848    let ed25519_input = hash_with_context_and_app_ctx(label, context, message)
849        .ok_or_else(|| HybridSignatureError::InvalidContext("context over budget".to_string()))?;
850    sign_legs(
851        sk,
852        |ml_dsa_sk| ml_dsa_sk.sign(message, &ml_dsa_ctx),
853        |ed25519_kp| ed25519_kp.sign(&ed25519_input),
854    )
855}

857/// Sign a caller-computed SHA-512 digest (pre-hashed mode).
858///
859/// `digest` is `SHA-512(M)` over content of any size, computed by the caller
860/// (streaming); the signer never sees `M`, so the message resource limit
861/// does not apply. `context` is required (1..=[`MAX_HYBRID_SIG_CONTEXT_LEN`]
862/// bytes). Both legs use their standard pre-hash constructions with
863/// `LABEL-ph || 0x00 || context` as the context string: HashML-DSA with the
864/// SHA-512 OID (FIPS 204 Algorithm 4) and Ed25519ph (RFC 8032 §5.1). SHA-512
865/// is the only pre-hash because Ed25519ph is defined over SHA-512 and SHA-512
866/// meets the FIPS 204 §5.4 strength floor at every ML-DSA parameter set.
867///
868/// A pre-hashed signature verifies only under [`verify_prehashed`] with the
869/// same context — never under [`verify`] or [`verify_with_context`].
870///
871/// # Errors
872/// Returns `InvalidDigestLength` if `digest.len() != 64`, `InvalidContext`
873/// if `context` is empty or over budget, otherwise the same errors as
874/// [`sign`].
875pub fn sign_prehashed(
876    sk: &HybridSigSecretKey,
877    digest: &[u8],
878    context: &[u8],
879) -> Result<HybridSignature, HybridSignatureError> {
880    let digest: &[u8; 64] = digest.try_into().map_err(|_e| {
881        HybridSignatureError::InvalidDigestLength { expected: 64, actual: digest.len() }
882    })?;
883    let label = hybrid_mode_label(sk.parameter_set, HybridTranscriptMode::Prehashed);
884    let mode_ctx = build_mode_ctx(label, context)?;
885    sign_legs(
886        sk,
887        |ml_dsa_sk| ml_dsa_sk.sign_prehashed(digest, PrehashAlgorithm::Sha512, &mode_ctx),
888        |ed25519_kp| ed25519_kp.sign_prehashed(digest, &mode_ctx),
889    )
890}

892/// Shared sign body: materialise both leg keys, run the mode-specific leg
893/// closures, assemble the hybrid signature.
894///
895/// Sign-side opacity (defense-in-depth per Pattern 6): every component
896/// failure folds into `SigningFailed` and the stage is routed to the
897/// operation log.
898fn sign_legs<M, E>(
899    sk: &HybridSigSecretKey,
900    ml_dsa_sign: M,
901    ed25519_sign: E,
902) -> Result<HybridSignature, HybridSignatureError>
903where
904    M: FnOnce(&MlDsaSecretKey) -> Result<MlDsaSignature, MlDsaError>,
905    E: FnOnce(&Ed25519KeyPair) -> crate::prelude::error::Result<DalekSignature>,
906{
907    // Borrow for the length check instead of `ed25519_sk_bytes()` — the
908    // latter allocates and zeroizes a fresh `Zeroizing<Vec<u8>>` on every
909    // call just to read `.len()`, and this is the hot sign() path.
910    if sk.expose_ed25519_secret().len() != 32 {
911        return Err(HybridSignatureError::InvalidKeyMaterial(
912            "Ed25519 secret key must be 32 bytes".to_string(),
913        ));
914    }
915
916    // `MlDsaSecretKey::new()` wraps its `Vec<u8>` argument in `Zeroizing`
917    // on entry, so this single bare clone is consumed-and-wiped on both the
918    // success and the error path.
919    let ml_dsa_sk_bytes = sk.ml_dsa_sk_bytes();
920    let ml_dsa_sk_struct = MlDsaSecretKey::new(sk.parameter_set, (*ml_dsa_sk_bytes).clone())
921        .map_err(|_e| {
922            log_crypto_operation_error!(op::HYBRID_SIGN, "ML-DSA SK init failed");
923            HybridSignatureError::SigningFailed("signing failed".to_string())
924        })?;
925    let ml_dsa_sig = ml_dsa_sign(&ml_dsa_sk_struct)
926        .map_err(|_e| {
927            log_crypto_operation_error!(op::HYBRID_SIGN, "ML-DSA sign failed");
928            HybridSignatureError::SigningFailed("signing failed".to_string())
929        })?
930        .as_bytes()
931        .to_vec();
932
933    let ed25519_sk_zeroizing = sk.ed25519_sk_bytes();
934    let ed25519_keypair = Ed25519KeyPair::from_secret_key(ed25519_sk_zeroizing.as_slice())
935        .map_err(|_e| {
936            log_crypto_operation_error!(op::HYBRID_SIGN, "Ed25519 SK init failed");
937            HybridSignatureError::SigningFailed("signing failed".to_string())
938        })?;
939    let ed25519_signature = ed25519_sign(&ed25519_keypair).map_err(|_e| {
940        log_crypto_operation_error!(op::HYBRID_SIGN, "Ed25519 sign rejected");
941        HybridSignatureError::SigningFailed("signing failed".to_string())
942    })?;
943    let ed25519_sig = Ed25519SignatureOps::signature_bytes(&ed25519_signature);
944
945    Ok(HybridSignature { ml_dsa_sig, ed25519_sig })
946}

948/// Verify using hybrid signature scheme
949///
950/// Planned deprecation (0.14): prefer [`verify_with_context`]. See [`sign`].
951///
952/// # Errors
953///
954/// Returns an error if:
955/// - The Ed25519 public key is not exactly 32 bytes.
956/// - The Ed25519 signature is not exactly 64 bytes.
957/// - ML-DSA public key or signature construction fails.
958/// - ML-DSA signature verification fails.
959/// - The Ed25519 public key is invalid or signature verification fails.
960pub fn verify(
961    pk: &HybridSigPublicKey,
962    message: &[u8],
963    sig: &HybridSignature,
964) -> Result<bool, HybridSignatureError> {
965    // H1 / M1 fix: bind the verified transcript to this scheme. Must mirror
966    // the sign-side context exactly — any mismatch produces a verify-false
967    // for legitimate signatures. Old (pre-fix) hybrid signatures were
968    // Ed25519 signatures over the raw message and will not verify here —
969    // the deliberate wire-format break documented in CHANGELOG.
970    let scheme_ctx = sig_context(hybrid_scheme_label(pk.parameter_set));
971    let ed25519_input = hash_with_context(scheme_ctx, message);
972    verify_legs(
973        pk,
974        sig,
975        |ml_dsa_pk, ml_dsa_sig| ml_dsa_pk.verify(message, ml_dsa_sig, scheme_ctx),
976        |ed25519_pk, ed25519_sig| {
977            Ed25519SignatureOps::verify(ed25519_pk, &ed25519_input, ed25519_sig)
978        },
979    )
980}

1018    digest: &[u8],
1019    sig: &HybridSignature,
1020    context: &[u8],
1021) -> Result<bool, HybridSignatureError> {
1022    let digest: &[u8; 64] = digest.try_into().map_err(|_e| {
1023        HybridSignatureError::InvalidDigestLength { expected: 64, actual: digest.len() }
1024    })?;
1025    let label = hybrid_mode_label(pk.parameter_set, HybridTranscriptMode::Prehashed);
1026    let mode_ctx = build_mode_ctx(label, context)?;
1027    verify_legs(
1028        pk,
1029        sig,
1030        |ml_dsa_pk, ml_dsa_sig| {
1031            ml_dsa_pk.verify_prehashed(digest, PrehashAlgorithm::Sha512, ml_dsa_sig, &mode_ctx)
1032        },
1033        |ed25519_pk, ed25519_sig| {
1034            Ed25519SignatureOps::verify_prehashed(ed25519_pk, digest, &mode_ctx, ed25519_sig)
1035        },
1036    )
1037}

1039/// Shared verify body for every transcript mode.
1040///
1041/// SECURITY: Verify BOTH components unconditionally with bitwise AND combination.
1042/// This preserves AND-security: a partial break of one component must not leak which
1043/// one failed, because distinct error paths (or early exit) turn AND into OR.
1044///
1045/// Rules:
1046/// 1. No early return on a single component failure.
1047/// 2. No `&&` (short-circuit) — use `&` (bitwise) so both leg verifies always execute.
1048/// 3. A SINGLE opaque error string regardless of which component failed.
1049/// 4. Structural parse failures collapse to bit=0 (same as verify-fail),
1050///    matching FIPS 204 §5.3 Verify-returns-bool shape. This removes the
1051///    early-exit timing / variant oracle that would otherwise distinguish
1052///    "malformed ML-DSA" from "Ed25519 failed".
1053///
1054/// No length pre-checks are performed here. Earlier revisions had four
1055/// distinguishable `InvalidKeyMaterial` returns for the four buffer
1056/// lengths; that surface let an adversary varying wire-supplied PK/sig
1057/// bytes enumerate which length got rejected. The legs collapse parse
1058/// failures (which include length mismatches) to bit = 0, which is
1059/// indistinguishable from a verify failure — exactly what Pattern 6
1060/// §"adversary-reachable paths" requires.
1061///
1062/// Per-stage `tracing` was previously emitted under four distinct
1063/// sub-stage tags; that made the Pattern 6 returned-error opacity
1064/// reconstructable from the debug log. Operators still get a
1065/// single "verify failure occurred" event for alerting; the
1066/// granular sub-stage detail is intentionally dropped (see
1067/// `docs/DESIGN_PATTERNS.md` Pattern 6).
1068fn verify_legs<M, E>(
1069    pk: &HybridSigPublicKey,
1070    sig: &HybridSignature,
1071    ml_dsa_verify: M,
1072    ed25519_verify: E,
1073) -> Result<bool, HybridSignatureError>
1074where
1075    M: Fn(&MlDsaPublicKey, &MlDsaSignature) -> Result<bool, MlDsaError>,
1076    E: Fn(&[u8], &DalekSignature) -> crate::prelude::error::Result<()>,
1077{
1078    let ml_dsa_valid = verify_ml_dsa_leg(pk, sig, &ml_dsa_verify);
1079    let ed25519_valid = verify_ed25519_leg(pk, sig, &ed25519_verify);
1080
1081    // One generic event on aggregate failure — preserves the "something
1082    // went wrong with hybrid verify" alerting signal without leaking
1083    // which component(s) failed.
1084    if (ml_dsa_valid & ed25519_valid) != 1 {
1085        log_crypto_operation_error!(op::HYBRID_VERIFY, "hybrid signature verification failed");
1086    }
1087
1088    // The two component bits are combined via bitwise AND (not short-circuit
1089    // `&&`) so verification work is identical regardless of which component
1090    // failed first. The branch below on `both_valid` does not leak anything
1091    // the return value doesn't already carry — pass/fail is observable from
1092    // the caller either way.
1093    let both_valid: u8 = ml_dsa_valid & ed25519_valid;
1094    if both_valid != 1 {
1095        return Err(HybridSignatureError::VerificationFailed(
1096            "hybrid signature verification failed".to_string(),
1097        ));
1098    }
1099
1100    // Both signatures verified successfully
1101    Ok(true)
1102}

1104/// ML-DSA leg of hybrid verify. Returns `1` iff the caller's bytes passed
1105/// shape-check AND both `from_bytes` parses AND the real verify
1106/// (`real_verify`, which closes over the mode-specific message / context).
1107///
1108/// Timing equalizer — see `crate::hybrid::verify_equalizer` for the full
1109/// rationale. Briefly: select shape-correct bytes (real or zero-byte dummy)
1110/// and run `from_bytes` + `verify` once on the selection so the wall-clock
1111/// cost is identical for shape-fail and verify-fail.
1112///
1113/// The dummy equalizer paths intentionally verify with `&[]` context because
1114/// they verify pre-generated material against fixed internal messages — that
1115/// material was produced with empty ctx, and its sole purpose is wall-clock
1116/// parity (results are discarded). The FIPS 204 §5.2 ctx parsing cost is
1117/// negligible relative to the polynomial arithmetic that dominates verify,
1118/// so equal cost holds; HashML-DSA verify runs the same `Verify_internal`
1119/// and so has the same cost profile as the pure dummy verify.
1120fn verify_ml_dsa_leg(
1121    pk: &HybridSigPublicKey,
1122    sig: &HybridSignature,
1123    real_verify: &dyn Fn(&MlDsaPublicKey, &MlDsaSignature) -> Result<bool, MlDsaError>,
1124) -> u8 {
1125    let dummy = crate::hybrid::verify_equalizer::hybrid_verify_dummy_material(pk.parameter_set);
1126
1127    let pq_pk_len = pk.parameter_set.public_key_size();
1128    let pq_sig_len = pk.parameter_set.signature_size();
1129    let pq_shape_ok = pk.ml_dsa_pk.len() == pq_pk_len && sig.ml_dsa_sig.len() == pq_sig_len;
1130    let (pq_pk_bytes, pq_sig_bytes): (&[u8], &[u8]) = if pq_shape_ok {
1131        (&pk.ml_dsa_pk, &sig.ml_dsa_sig)
1132    } else {
1133        (dummy.pq_pk.as_slice(), dummy.pq_sig.as_slice())
1134    };
1135    // Match-on-parse, never `?`-propagate: if `from_bytes` ever fails
1136    // (today unreachable for our zero-byte dummies of correct length,
1137    // but a future `fips204` release adding content validation could
1138    // change that), we still want the verify pipeline to run so the
1139    // wall-clock cost stays equal between shape-fail and verify-fail.
1140    //
1141    // When parse fails on the substituted bytes, we DO still run a
1142    // real verify against the equalizer's pre-parsed valid material
1143    // (`dummy.parsed`). This guarantees verify-pipeline execution
1144    // regardless of whether `from_bytes` adds content validation
1145    // downstream. The pre-parsed PK + sig verify against an internal
1146    // test message — content-dependent verify timing is
1147    // content-independent of the caller, which is exactly the property
1148    // we want from the equalizer.
1149    //
1150    // If `dummy.parsed` is `None` (init keygen+sign failed at module
1151    // load — extremely rare RNG/PCT path), we fall back to the
1152    // legacy `Ok(false)` skip. The bit is then computed via the
1153    // shape-check AND parse-check AND verify-check, so a degraded
1154    // equalizer never affects correctness — only the strength of the
1155    // timing-oracle countermeasure for the degraded parameter set.
1156    let parse_ok =
1157        MlDsaPublicKey::from_bytes(pq_pk_bytes, pk.parameter_set).and_then(|parsed_pk| {
1158            MlDsaSignature::from_bytes(pq_sig_bytes, pk.parameter_set)
1159                .map(|parsed_sig| (parsed_pk, parsed_sig))
1160        });
1161    // `MlDsaPublicKey::verify` internally calls
1162    // `ml_dsa_NN::PublicKey::try_from_bytes`, which short-circuits with
1163    // `Err(VerificationError)` on structurally-invalid PKs (e.g.
1164    // all-zero) before any actual ML-DSA verify runs. So a shape-pass-
1165    // but-inner-parse-fail input would pay only the cheap parse-fail cost
1166    // while a shape-pass-and-inner-parse-pass input pays the full
1167    // ML-DSA verify cost — a measurable timing oracle on the difference.
1168    // ANY path that doesn't reach a successful real verify falls
1169    // through to the dummy verify so the wall-clock cost is equal
1170    // across all reject reasons.
1171    //
1172    // Pre-parsed material goes through `parsed_or_init()`, which retries
1173    // init when prior attempts produced None. The result is a clone
1174    // (cheap — public bytes only); we own it locally.
1175    let dummy_parsed = dummy.parsed_or_init();
1176    let ml_dsa_verify_result = match &parse_ok {
1177        Ok((parsed_pk, parsed_sig)) => {
1178            let inner = real_verify(parsed_pk, parsed_sig);
1179            match &inner {
1180                Ok(_) => inner,
1181                // Inner parse failed (structurally-invalid PK reached
1182                // the short-circuit in `try_from_bytes`). Run the
1183                // equalizer dummy verify to spend verify-time budget;
1184                // discard its result.
1185                Err(_) => match &dummy_parsed {
1186                    Some(parsed) => {
1187                        let _ = parsed.pq_pk.verify(&parsed.pq_test_message, &parsed.pq_sig, &[]);
1188                        // Bubble the original Err so `matches!(., Ok(true))`
1189                        // below still rejects the substituted input.
1190                        inner
1191                    }
1192                    None => inner,
1193                },
1194            }
1195        }
1196        Err(_) => match &dummy_parsed {
1197            // Equalizer fallback: run verify against pre-validated
1198            // material so the wall-clock cost stays equal between
1199            // shape-fail-then-parse-fail and shape-good-then-verify.
1200            // The result is discarded by the AND with `parse_ok.is_ok()`
1201            // below — its only purpose is to consume verify-time
1202            // wall-clock budget.
1203            Some(parsed) => parsed.pq_pk.verify(&parsed.pq_test_message, &parsed.pq_sig, &[]),
1204            // Init keygen failed (extremely rare; retries
1205            // on every call). Equalizer degraded; legacy fast-fail
1206            // behavior. Not a correctness regression because the bit
1207            // is computed below via AND with multiple guards.
1208            None => Ok(false),
1209        },
1210    };
1211    // Bit is 1 only when the real input passed shape-check, both
1212    // `from_bytes` calls succeeded, AND the verify returned Ok(true).
1213    // When shape failed, the verify still ran (against the dummy or
1214    // against pre-parsed material) for timing equalization; its
1215    // result is discarded by the bitwise AND below.
1216    //
1217    // DP-H3 fix: compute the bit via bitwise `&` (Anti-Pattern 3) so the
1218    // three predicates are always evaluated in identical order rather
1219    // than short-circuiting. The wall-clock parity is already preserved
1220    // by the timing equalizer (the dummy verify runs on the failure
1221    // path), but the explicit `u8 & u8 & u8` composition documents the
1222    // CT-discipline intent and prevents a future refactor from sliding
1223    // `&&` back in.
1224    let shape_bit: u8 = u8::from(pq_shape_ok);
1225    let parse_bit: u8 = u8::from(parse_ok.is_ok());
1226    let verify_bit: u8 = u8::from(matches!(ml_dsa_verify_result, Ok(true)));
1227    shape_bit & parse_bit & verify_bit
1228}

1230/// Ed25519 leg of hybrid verify. Returns `1` iff the caller's signature
1231/// parsed AND the real verify (`real_verify`, which closes over the
1232/// mode-specific digest / context) passed.
1233///
1234/// The Ed25519 leg has a real verify-time equalizer: parse is a 64-byte
1235/// length check while verify is one EC scalar mul (~3 orders of magnitude
1236/// apart), so on parse-fail we run verify against pre-parsed cached material
1237/// so the wall-clock cost matches a real verify, mirroring the ML-DSA
1238/// equalizer.
1239fn verify_ed25519_leg(
1240    pk: &HybridSigPublicKey,
1241    sig: &HybridSignature,
1242    real_verify: &dyn Fn(&[u8], &DalekSignature) -> crate::prelude::error::Result<()>,
1243) -> u8 {
1244    let ed_dummy = crate::hybrid::verify_equalizer::ed25519_verify_dummy_material();
1245    let ed_dummy_parsed = ed_dummy.parsed_or_init();
1246    if let Ok(ed25519_signature) =
1247        Ed25519SignatureOps::signature_from_bytes(sig.ed25519_sig.as_slice())
1248    {
1249        // Wrong-length PK short-circuits inside the real verify before any
1250        // scalar mul runs, distinguishing "32-byte PK, bad sig" from
1251        // "wrong-length PK" by wall-clock. Run the dummy verify against
1252        // parsed dummy material on that path so the wall-clock matches the
1253        // valid-PK code path.
1254        let pk_bytes = pk.ed25519_pk.as_slice();
1255        let result = real_verify(pk_bytes, &ed25519_signature);
1256        if pk_bytes.len() != 32
1257            && let Some(parsed) = &ed_dummy_parsed
1258            && let Ok(parsed_sig) =
1259                Ed25519SignatureOps::signature_from_bytes(parsed.ed_sig.as_slice())
1260        {
1261            // Discarded; runs the EC scalar mul.
1262            let _ = Ed25519SignatureOps::verify(
1263                parsed.ed_pk.as_slice(),
1264                parsed.ed_test_message.as_slice(),
1265                &parsed_sig,
1266            );
1267        }
1268        match result {
1269            Ok(()) => 1u8,
1270            _ => 0u8,
1271        }
1272    } else {
1273        // Parse failure on caller-supplied bytes: still spend a
1274        // verify-cost worth of cycles against the dummy material so
1275        // the wall-clock between parse-fail and verify-fail is equal.
1276        // Discard the result — `0u8` is the verdict because the
1277        // caller's input was structurally wrong.
1278        if let Some(parsed) = &ed_dummy_parsed
1279            && let Ok(parsed_sig) =
1280                Ed25519SignatureOps::signature_from_bytes(parsed.ed_sig.as_slice())
1281        {
1282            // Discarded; runs the EC scalar mul.
1283            let _ = Ed25519SignatureOps::verify(
1284                parsed.ed_pk.as_slice(),
1285                parsed.ed_test_message.as_slice(),
1286                &parsed_sig,
1287            );
1288        }
1289        // If `ed_dummy_parsed` is None (RNG/PCT init failure —
1290        // retries on every call), fall through to the
1291        // legacy fast-fail. Bit is computed by the caller via AND with
1292        // the ML-DSA leg, so a degraded equalizer cannot affect
1293        // correctness.
1294        0u8
1295    }
1296}
