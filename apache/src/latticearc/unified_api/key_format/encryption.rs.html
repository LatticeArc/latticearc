18    /// Returns `true` if the key material is passphrase-encrypted.
19    ///
20    /// Encrypted keys must be unwrapped via [`Self::decrypt_with_passphrase`]
21    /// before their raw bytes can be extracted via [`KeyData::decode_raw`] or
22    /// [`KeyData::decode_composite`].
23    #[must_use]
24    pub fn is_encrypted(&self) -> bool {
25        matches!(self.key_data, KeyData::Encrypted { .. })
26    }
27
28    /// Encrypt the key material in place using a passphrase.
29    ///
30    /// Derives a 32-byte AES key via PBKDF2-HMAC-SHA256 (600,000 iterations,
31    /// 16-byte random salt) and encrypts the JSON-serialized `KeyData` under
32    /// AES-256-GCM with a fresh 12-byte random nonce. The full envelope
33    /// (version, algorithm, key_type, KDF name, iteration count, salt, AEAD
34    /// name) is mixed into the AEAD AAD, so tampering with any metadata
35    /// field on disk causes decryption to fail at the tag check. See
36    /// [`Self::encryption_aad`] for the exact byte layout.
37    ///
38    /// # Errors
39    ///
40    /// Returns an error if the key is already encrypted, if the passphrase is
41    /// empty, or if the KDF / AEAD operation fails.
42    pub fn encrypt_with_passphrase(&mut self, passphrase: &[u8]) -> Result<()> {
43        if self.is_encrypted() {
44            return Err(CoreError::InvalidKey("Key is already passphrase-encrypted".to_string()));
45        }
46        if passphrase.is_empty() {
47            return Err(CoreError::InvalidKey("Passphrase must not be empty".to_string()));
48        }
49
50        // 1. Serialize the current plaintext KeyData to its own JSON. We
51        //    serialize only the KeyData (not the whole PortableKey) so the
52        //    ciphertext is self-contained and round-trips cleanly.
53        let plaintext_json = serde_json::to_vec(&self.key_data).map_err(|e| {
54            CoreError::SerializationError(format!(
55                "Failed to serialize key data for encryption: {e}"
56            ))
57        })?;
58        let plaintext = zeroize::Zeroizing::new(plaintext_json);
59
60        // 2. Generate a fresh random salt.
61        let salt = crate::primitives::rand::csprng::random_bytes(PBKDF2_SALT_LEN);
62
63        // 3. Derive the AES key via PBKDF2-HMAC-SHA256. Salt-length
64        // validation lives at the `pbkdf2(...)` call below; `with_salt`
65        // is intentionally infallible so wire-format parsers can also
66        // round-trip pre-0.8.0 short-salt envelopes for inspection.
67        let kdf_params = crate::primitives::kdf::pbkdf2::Pbkdf2Params::with_salt(&salt)
68            .iterations(PBKDF2_DEFAULT_ITERATIONS)
69            .key_length(AES_256_KEY_LEN);
70        let derived = crate::primitives::kdf::pbkdf2::pbkdf2(passphrase, &kdf_params)
71            .map_err(|e| CoreError::InvalidKey(format!("PBKDF2 derivation failed: {e}")))?;
72
73        // 4. Encrypt via AES-256-GCM with a fresh random nonce. Bind the
74        //    full envelope (version, algorithm, key_type, KDF name,
75        //    iterations, salt, AEAD name) to the ciphertext via AAD so an
76        //    attacker who modifies any of these fields on disk breaks the
77        //    AEAD tag.
78        use crate::primitives::aead::AeadCipher;
79        let cipher = crate::primitives::aead::aes_gcm::AesGcm256::new(derived.expose_secret())
80            .map_err(|e| CoreError::InvalidKey(format!("Failed to initialize AES-256-GCM: {e}")))?;
81        let aad = Self::encryption_aad(
82            ENCRYPTED_ENVELOPE_VERSION,
83            self.algorithm,
84            self.key_type,
85            PBKDF2_KDF_ID,
86            PBKDF2_DEFAULT_ITERATIONS,
87            &salt,
88            AES_GCM_AEAD_ID,
89            &self.metadata,
90        )?;
91        let (nonce, mut ct, tag) = cipher
92            .seal(&plaintext, Some(&aad))
93            .map_err(|e| CoreError::InvalidKey(format!("AES-256-GCM sealing failed: {e}")))?;
94
95        // 5. Pack ciphertext || tag for on-wire storage.
96        ct.extend_from_slice(&tag);
97
98        // 6. Replace the plaintext KeyData with the encrypted envelope.
99        self.key_data = KeyData::Encrypted {
100            enc: ENCRYPTED_ENVELOPE_VERSION,
101            kdf: PBKDF2_KDF_ID.to_string(),
102            kdf_iterations: PBKDF2_DEFAULT_ITERATIONS,
103            kdf_salt: BASE64_ENGINE.encode(&salt),
104            aead: AES_GCM_AEAD_ID.to_string(),
105            nonce: BASE64_ENGINE.encode(nonce),
106            ciphertext: BASE64_ENGINE.encode(&ct),
107        };
108
109        Ok(())
110    }

112    /// Decrypt the key material in place using a passphrase.
113    ///
114    /// Reverses [`Self::encrypt_with_passphrase`]. On success the `key_data`
115    /// field is replaced with the underlying `Single` or `Composite` variant.
116    ///
117    /// # Errors
118    ///
119    /// Returns an error if the key is not encrypted, the envelope is
120    /// malformed, the passphrase is wrong (AEAD authentication fails), or
121    /// the decrypted plaintext is not a valid `KeyData` serialization.
122    pub fn decrypt_with_passphrase(&mut self, passphrase: &[u8]) -> Result<()> {
123        if passphrase.is_empty() {
124            return Err(CoreError::InvalidKey("Passphrase must not be empty".to_string()));
125        }
126
127        // Borrow the envelope to validate shape and decode its fields into
128        // owned bytes. After validation the `kdf` and `aead` string fields
129        // are known to equal the envelope's fixed constants, so we don't
130        // bother cloning them into `Decoded` — we pass the constants to
131        // `encryption_aad` at the call site instead.
132        struct Decoded {
133            enc: u32,
134            kdf_iterations: u32,
135            salt: Vec<u8>,
136            nonce: [u8; AES_GCM_NONCE_LEN],
137            ct_and_tag: Vec<u8>,
138        }
139        let decoded = match &self.key_data {
140            KeyData::Encrypted { enc, kdf, kdf_iterations, kdf_salt, aead, nonce, ciphertext } => {
141                Self::validate_encrypted_envelope_fields(
142                    *enc,
143                    kdf,
144                    *kdf_iterations,
145                    kdf_salt,
146                    aead,
147                    nonce,
148                    ciphertext,
149                )?;
150                let salt = decode_b64_opaque(kdf_salt, "decrypt.encrypted.kdf_salt")?;
151                let nonce_bytes = decode_b64_opaque(nonce, "decrypt.encrypted.nonce")?;
152                let mut nonce_array = [0u8; AES_GCM_NONCE_LEN];
153                nonce_array.copy_from_slice(&nonce_bytes);
154                let ct_and_tag = decode_b64_opaque(ciphertext, "decrypt.encrypted.ciphertext")?;
155                Decoded {
156                    enc: *enc,
157                    kdf_iterations: *kdf_iterations,
158                    salt,
159                    nonce: nonce_array,
160                    ct_and_tag,
161                }
162            }
163            _ => {
164                return Err(CoreError::InvalidKey("Key is not passphrase-encrypted".to_string()));
165            }
166        };
167        let Decoded { enc, kdf_iterations, salt, nonce: nonce_array, ct_and_tag } = decoded;
168
169        let tag_offset = ct_and_tag
170            .len()
171            .checked_sub(AES_GCM_TAG_LEN)
172            .ok_or_else(|| CoreError::InvalidKey("Ciphertext shorter than tag".to_string()))?;
173        let (ct_bytes, tag_bytes) = ct_and_tag.split_at(tag_offset);
174        let mut tag_array = [0u8; AES_GCM_TAG_LEN];
175        tag_array.copy_from_slice(tag_bytes);
176
177        // Derive the AES key from the passphrase + salt.
178        // `with_salt` is the right tool here: `salt` was
179        // deserialized from an externally-supplied envelope and may be
180        // a pre-0.8.0 short salt that `with_salt`'s NIST SP 800-132 §5.1
181        // floor would reject. The validating min-length check happens at
182        // the `pbkdf2(...)` call below instead, so the unchecked
183        // construction cannot smuggle a short salt past actual
184        // derivation. This is the documented wire-format-parser use
185        // case for `with_salt`.
186        let kdf_params = crate::primitives::kdf::pbkdf2::Pbkdf2Params::with_salt(&salt)
187            .iterations(kdf_iterations)
188            .key_length(AES_256_KEY_LEN);
189        // Use the legacy load-side floor (PBKDF2_MIN_ITERATIONS = 100k)
190        // rather than the public OWASP 2023 floor: the iteration count
191        // here is integrity-protected by the envelope AAD, so a count
192        // below 600k means the legitimate keyholder wrote it under
193        // OWASP 2018 guidance. Refusing to load it would strand
194        // historical keys; the caller already saw a re-protect warning
195        // in `validate_encrypted_envelope_fields` above.
196        let derived = crate::primitives::kdf::pbkdf2::pbkdf2_with_floor(
197            passphrase,
198            &kdf_params,
199            PBKDF2_MIN_ITERATIONS,
200        )
201        .map_err(|e| CoreError::InvalidKey(format!("PBKDF2 derivation failed: {e}")))?;
202
203        // Decrypt. A wrong passphrase produces a wrong AES key, which causes
204        // AEAD authentication to fail with an opaque error — we do NOT leak
205        // whether the passphrase was wrong vs. the envelope was corrupted.
206        // The AAD binds the full envelope so any tampered metadata field
207        // also breaks the tag.
208        use crate::primitives::aead::AeadCipher;
209        let cipher = crate::primitives::aead::aes_gcm::AesGcm256::new(derived.expose_secret())
210            .map_err(|e| CoreError::InvalidKey(format!("Failed to initialize AES-256-GCM: {e}")))?;
211        // `kdf` and `aead` are the constants: `validate_encrypted_envelope_fields`
212        // rejected any other value, so we can pass the literals directly
213        // instead of cloning them out of the borrowed envelope.
214        let aad = Self::encryption_aad(
215            enc,
216            self.algorithm,
217            self.key_type,
218            PBKDF2_KDF_ID,
219            kdf_iterations,
220            &salt,
221            AES_GCM_AEAD_ID,
222            &self.metadata,
223        )?;
224        let plaintext = cipher
225            .decrypt(&nonce_array, ct_bytes, &tag_array, Some(&aad))
226            .map_err(|_e| {
227                CoreError::InvalidKey(
228                    "Passphrase-protected key unwrap failed (wrong passphrase or corrupted envelope)"
229                        .to_string(),
230                )
231            })?;
232
233        // Deserialize the plaintext bytes back into a KeyData variant.
234        // Defense-in-depth: AEAD authentication has already passed by
235        // this point, so an attacker cannot reach this branch without
236        // the passphrase — but a corrupted between-sessions disk write
237        // could surface attacker-controlled token detail through
238        // serde_json's error display. Route through the opaque helper.
239        let plaintext_str = std::str::from_utf8(&plaintext).map_err(|e| {
240            tracing::debug!(error = %e, "decrypted keyfile: non-UTF8 plaintext");
241            CoreError::SerializationError("decrypted plaintext is not UTF-8".to_string())
242        })?;
243        let new_key_data: KeyData = decode_json_opaque(plaintext_str, "decrypt.plaintext")?;
244        // Reject nested encryption (prevents re-wrap confusion).
245        if matches!(new_key_data, KeyData::Encrypted { .. }) {
246            return Err(CoreError::InvalidKey(
247                "Decrypted payload was itself an encrypted envelope".to_string(),
248            ));
249        }
250
251        // Stage the new KeyData and run the same coherence check that
252        // `from_json`/`load_from_file` apply on entry. An attacker who
253        // crafted an envelope where the AEAD-bound metadata declares one
254        // algorithm but the decrypted KeyData payload describes another
255        // would otherwise install a structurally-incoherent PortableKey
256        // that the rest of the API would only catch much later (or, in
257        // some paths, not at all). Run validation before mutating self
258        // so a failed decrypt leaves the receiver untouched.
259        let original = std::mem::replace(&mut self.key_data, new_key_data);
260        if let Err(e) = self.validate() {
261            // Restore the original encrypted envelope on validation
262            // failure so the caller can retry, inspect, or re-wrap.
263            self.key_data = original;
264            return Err(e);
265        }
266        Ok(())
267    }

269    /// Build the AAD bound to an encrypted key envelope.
270    ///
271    /// AEAD AAD is authenticated (not encrypted), so fields folded in here
272    /// are protected against tampering but not against disclosure. Binding
273    /// all envelope parameters — version, algorithm, key type, KDF name,
274    /// iteration count, salt, and AEAD name — ensures that any attacker
275    /// modification of a stored key file's metadata or KDF parameters
276    /// causes `cipher.decrypt` to fail with an opaque error.
277    ///
278    /// Uses stable kebab-case / lowercase names (`ml-kem-768`, `secret`)
279    /// via `canonical_name` accessors on the enums. These are load-bearing:
280    /// changing the returned strings breaks every existing encrypted key
281    /// file. A pinned byte-level test in the `tests` module guards this.
282    ///
283    /// # Byte layout
284    ///
285    /// ```text
286    /// "latticearc-lpk-v1-enc" || 0x00
287    /// || enc (u32 BE)
288    /// || algorithm_name || 0x00
289    /// || key_type_name || 0x00
290    /// || kdf_name || 0x00
291    /// || kdf_iterations (u32 BE)
292    /// || kdf_salt_len (u32 BE) || kdf_salt_raw_bytes
293    /// || aead_name
294    /// ```
295    ///
296    /// Length prefixes and null separators prevent ambiguity between
297    /// adjacent variable-length fields. The salt is included as its raw
298    /// (base64-decoded) bytes, not the base64 string, so an attacker
299    /// cannot use base64 non-canonical encodings to get past the check.
300    ///
301    /// `pub(super)`: exercised directly by `key_format::tests_a`'s
302    /// AAD byte-layout-stability regression tests.
303    #[expect(
304        clippy::too_many_arguments,
305        reason = "8 fields are inherent to the AAD layout; bundling into a struct adds indirection without simplifying the call sites."
306    )]
307    pub(super) fn encryption_aad(
308        enc: u32,
309        algorithm: KeyAlgorithm,
310        key_type: KeyType,
311        kdf: &str,
312        kdf_iterations: u32,
313        kdf_salt: &[u8],
314        aead: &str,
315        metadata: &BTreeMap<String, serde_json::Value>,
316    ) -> Result<Vec<u8>> {
317        let algorithm_name = algorithm.canonical_name();
318        let key_type_name = key_type.canonical_name();
319
320        // Canonical metadata serialization: BTreeMap iterates in
321        // lexicographic key order, so `serde_json::to_vec` produces
322        // deterministic bytes for flat string→string maps (the actual
323        // shape used by this format — `ml_kem_pk` and friends are
324        // base64 strings). Length-prefix the result so a metadata
325        // append/truncate cannot collide with an adjacent AAD field.
326        let metadata_bytes = serde_json::to_vec(metadata).map_err(|e| {
327            CoreError::SerializationError(format!(
328                "Failed to canonicalize key-format metadata for AEAD AAD: {e}"
329            ))
330        })?;
331        let metadata_len_u32 = u32::try_from(metadata_bytes.len()).map_err(|_e| {
332            CoreError::SerializationError(
333                "Encrypted-envelope metadata exceeds u32::MAX bytes".to_string(),
334            )
335        })?;
336
337        let mut aad = Vec::with_capacity(
338            b"latticearc-lpk-v3-enc"
339                .len()
340                .saturating_add(1) // null after magic
341                .saturating_add(4) // enc
342                .saturating_add(algorithm_name.len())
343                .saturating_add(1) // null after algorithm_name
344                .saturating_add(key_type_name.len())
345                .saturating_add(1) // null after key_type_name
346                .saturating_add(kdf.len())
347                .saturating_add(1) // null after kdf
348                .saturating_add(4) // kdf_iterations
349                .saturating_add(4) // salt len
350                .saturating_add(kdf_salt.len())
351                .saturating_add(aead.len())
352                .saturating_add(1) // null after aead (parallel to other string fields)
353                .saturating_add(4) // metadata len
354                .saturating_add(metadata_bytes.len()),
355        );
356        // AAD magic `lpk-v3-enc`: bumped from `lpk-v2-enc` for the
357        // canonicalization fix below. The previous v2 layout omitted the
358        // null terminator after the `aead` string field while every other
359        // string field (label, algorithm_name, key_type_name, kdf) had
360        // one — the moment a second AEAD name was added (e.g.
361        // ChaCha20-Poly1305), the (aead || metadata_len) concatenation
362        // could collide with a different (aead' || metadata_len')
363        // prefix-shifted by the length difference. Adding the terminator
364        // closes the collision class. The envelope's `enc` wire field is
365        // bumped to `ENCRYPTED_ENVELOPE_VERSION` (3) in lockstep, so a v2
366        // envelope is caught by the version check at load time and
367        // reported with a distinct "re-protect" error rather than failing
368        // opaquely as a wrong passphrase.
369        aad.extend_from_slice(b"latticearc-lpk-v3-enc");
370        aad.push(0);
371        aad.extend_from_slice(&enc.to_be_bytes());
372        aad.extend_from_slice(algorithm_name.as_bytes());
373        aad.push(0);
374        aad.extend_from_slice(key_type_name.as_bytes());
375        aad.push(0);
376        aad.extend_from_slice(kdf.as_bytes());
377        aad.push(0);
378        aad.extend_from_slice(&kdf_iterations.to_be_bytes());
379        // The `unwrap_or(u32::MAX)` saturation that would have lived
380        // here was collapsed at four other length-prefix sites for
381        // good reason: saturation collapses every `>4 GiB` salt onto
382        // the same length prefix, leading to an AAD canonicalization
383        // collision. Refuse instead, with the same shape as the
384        // metadata-len overflow check above.
385        let salt_len_u32 = u32::try_from(kdf_salt.len()).map_err(|_e| {
386            CoreError::SerializationError(
387                "Encrypted-envelope kdf_salt exceeds u32::MAX bytes".to_string(),
388            )
389        })?;
390        aad.extend_from_slice(&salt_len_u32.to_be_bytes());
391        aad.extend_from_slice(kdf_salt);
392        aad.extend_from_slice(aead.as_bytes());
393        // null terminator after `aead` to match every
394        // other string field. Without it, a hypothetical
395        // (aead="ChaCha20Poly1305", metadata_len=N) pair could
396        // canonicalize identically to (aead="ChaCha20", metadata_len=
397        // 'P'<<24|...) for cleverly-crafted suffixes. Today aead is
398        // hard-coded to AES-256-GCM so unexploitable; the fix closes
399        // the collision class before a second AEAD ships.
400        aad.push(0);
401        aad.extend_from_slice(&metadata_len_u32.to_be_bytes());
402        aad.extend_from_slice(&metadata_bytes);
403        Ok(aad)
404    }
