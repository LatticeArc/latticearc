14    /// Wrap a hybrid KEM keypair (from `generate_hybrid_keypair()`) into
15    /// a pair of `PortableKey`s (public + secret).
16    ///
17    /// Algorithm is auto-derived from the keypair's `security_level`.
18    ///
19    /// # Errors
20    /// Returns an error if secret key export fails.
21    pub fn from_hybrid_kem_keypair(
22        use_case: crate::types::types::UseCase,
23        pk: &crate::hybrid::kem_hybrid::HybridKemPublicKey,
24        sk: &crate::hybrid::kem_hybrid::HybridKemSecretKey,
25    ) -> Result<(Self, Self)> {
26        let algorithm: KeyAlgorithm = pk.security_level().into();
27
28        let pub_key = Self {
29            version: Self::CURRENT_VERSION,
30            use_case: Some(use_case),
31            security_level: None,
32            algorithm,
33            key_type: KeyType::Public,
34            key_data: KeyData::from_composite(pk.ml_kem_pk(), pk.ecdh_pk()),
35            created: Utc::now(),
36            metadata: BTreeMap::new(),
37            not_after: None,
38        };
39
40        let ml_kem_sk = sk
41            .ml_kem_sk_bytes()
42            .map_err(|e| CoreError::InvalidKey(format!("ML-KEM SK export: {e}")))?;
43        let ecdh_seed = sk
44            .ecdh_seed_bytes()
45            .map_err(|e| CoreError::InvalidKey(format!("ECDH seed export: {e}")))?;
46
47        // Store ML-KEM public key in metadata so the secret key file is
48        // self-contained for decryption (no separate public key file needed).
49        // This follows the PKCS#12 pattern of bundling public + private material.
50        // Note: ECDH public key is not stored — it's derived from the seed at
51        // reconstruction time via X25519StaticKeyPair::from_seed_bytes().
52        let mut sk_metadata = BTreeMap::new();
53        sk_metadata.insert(
54            ML_KEM_PK_METADATA_KEY.to_string(),
55            serde_json::Value::String(BASE64_ENGINE.encode(pk.ml_kem_pk())),
56        );
57
58        let sec_key = Self {
59            version: Self::CURRENT_VERSION,
60            use_case: Some(use_case),
61            security_level: None,
62            algorithm,
63            key_type: KeyType::Secret,
64            key_data: KeyData::from_composite(&ml_kem_sk, &*ecdh_seed),
65            created: Utc::now(),
66            metadata: sk_metadata,
67            not_after: None,
68        };
69
70        Ok((pub_key, sec_key))
71    }

73    /// Extract a `HybridPublicKey` from a portable key.
74    ///
75    /// # Errors
76    /// Returns an error if the algorithm is not a hybrid KEM, the key file is
77    /// not marked [`KeyType::Public`], or key data is invalid.
78    ///
79    /// # M2 fix
80    ///
81    /// The companion [`to_hybrid_secret_key`](Self::to_hybrid_secret_key)
82    /// already required `key_type == Secret`; this extractor previously
83    /// accepted any `key_type` and would silently treat the embedded
84    /// composite of a [`KeyType::Secret`] file as a public key. Most real
85    /// callers were protected by downstream length validation inside
86    /// `HybridKemPublicKey::new`, but the asymmetry was a defense-in-depth
87    /// hole. The guard makes the contract structural.
88    pub fn to_hybrid_public_key(&self) -> Result<crate::hybrid::kem_hybrid::HybridKemPublicKey> {
89        // M-B: expiry gate must run before extraction. A key file with
90        // `not_after` in the past loads cleanly through validate() (by
91        // design — inspection / migration paths need it), but emitting
92        // the typed HybridKemPublicKey lets a downstream encrypt path
93        // use expired key material. Route through the M-B helper.
94        self.validate_with_expiry_now()?;
95        if self.key_type != KeyType::Public {
96            return Err(CoreError::InvalidKey(
97                "Cannot extract hybrid public key from a non-public key file".to_string(),
98            ));
99        }
100
101        let level =
102            crate::primitives::kem::MlKemSecurityLevel::try_from(self.algorithm).map_err(|()| {
103                CoreError::InvalidKey(format!(
104                    "Not a hybrid KEM algorithm: {alg:?}",
105                    alg = self.algorithm
106                ))
107            })?;
108
109        let (pq_bytes, classical_bytes) = self.key_data.decode_composite()?;
110
111        Ok(crate::hybrid::kem_hybrid::HybridKemPublicKey::new(pq_bytes, classical_bytes, level))
112    }

114    /// Reconstruct a `HybridSecretKey` from a portable key pair.
115    ///
116    /// Requires the corresponding public key PortableKey because ML-KEM
117    /// key reconstruction needs both the secret key bytes and public key bytes.
118    ///
119    /// # Arguments
120    /// * `public_key` - The corresponding `PortableKey` with `KeyType::Public`
121    ///
122    /// # Errors
123    /// Returns an error if the algorithm is not hybrid KEM, key data is invalid,
124    /// or key reconstruction fails.
125    ///
126    /// The ML-KEM public key is extracted from the secret key file's metadata
127    /// (stored at keygen time), making the secret key file fully self-contained.
128    /// No separate public key file is needed for decryption.
129    pub fn to_hybrid_secret_key(&self) -> Result<crate::hybrid::kem_hybrid::HybridKemSecretKey> {
130        // M-B: see `to_hybrid_public_key`. Expired secret keys must not be
131        // turned into a typed secret-key value that a downstream decrypt
132        // / agree path would use.
133        self.validate_with_expiry_now()?;
134        let level =
135            crate::primitives::kem::MlKemSecurityLevel::try_from(self.algorithm).map_err(|()| {
136                CoreError::InvalidKey(format!(
137                    "Not a hybrid KEM algorithm: {alg:?}",
138                    alg = self.algorithm
139                ))
140            })?;
141
142        if self.key_type != KeyType::Secret {
143            return Err(CoreError::InvalidKey(
144                "Cannot reconstruct secret key from a public key".to_string(),
145            ));
146        }
147
148        // Use the zeroizing decoder so the ML-KEM secret-key bytes and
149        // the X25519 seed are wiped from heap memory when this scope
150        // ends, regardless of which downstream branch we take.
151        let (ml_kem_sk, ecdh_seed_vec) = self.key_data.decode_composite_zeroized()?;
152
153        // derive the ML-KEM public key from the SK
154        // bytes' embedded layout (FIPS 203 §6.1) rather than trusting
155        // the unauthenticated `ml_kem_pk` metadata field. The PK
156        // metadata, when present, was used as load-time input to
157        // HKDF info — but the metadata BTreeMap is only included in
158        // AAD when the SK file is passphrase-encrypted (see test at
159        // line 2980). Plaintext-stored SKs had no integrity binding
160        // for that field, so a file-write attacker could swap it.
161        // Pattern-6: plaintext-keyfile path is reachable without AEAD,
162        // so the upstream `MlKemSecretKey::new` / `embedded_public_key_bytes`
163        // errors are attacker-touchable. Collapse to a fixed string and
164        // route the raw error to tracing::debug! for operator visibility.
165        let parsed_sk =
166            crate::primitives::kem::ml_kem::MlKemSecretKey::new(level, ml_kem_sk.to_vec())
167                .map_err(|e| {
168                    tracing::debug!(error = %e, "plaintext keyfile: ML-KEM SK parse rejected");
169                    CoreError::InvalidKey("Invalid ML-KEM secret key".to_string())
170                })?;
171        let ml_kem_pk_slice = parsed_sk.embedded_public_key_bytes().map_err(|e| {
172            tracing::debug!(error = %e, "plaintext keyfile: ML-KEM SK pk-extract rejected");
173            CoreError::InvalidKey("Invalid ML-KEM secret key".to_string())
174        })?;
175        let ml_kem_pk: Vec<u8> = ml_kem_pk_slice.to_vec();
176
177        if ecdh_seed_vec.len() != 32 {
178            return Err(CoreError::InvalidKey(format!(
179                "X25519 seed must be 32 bytes, got {}",
180                ecdh_seed_vec.len()
181            )));
182        }
183
184        let mut ecdh_seed = zeroize::Zeroizing::new([0u8; 32]);
185        ecdh_seed.copy_from_slice(&ecdh_seed_vec);
186
187        crate::hybrid::kem_hybrid::HybridKemSecretKey::from_serialized(
188            level, &ml_kem_sk, &ml_kem_pk, &ecdh_seed,
189        )
190        .map_err(|e| {
191            tracing::debug!(error = %e, "plaintext keyfile: hybrid SK reconstruction rejected");
192            CoreError::InvalidKey("Hybrid secret key reconstruction failed".to_string())
193        })
194    }

198    /// Wrap a hybrid signature keypair (from `generate_hybrid_signing_keypair()`)
199    /// into a pair of `PortableKey`s (public + secret).
200    ///
201    /// The ML-DSA parameter set is auto-detected from the public key byte length:
202    /// - 1,312 bytes → ML-DSA-44 (`HybridMlDsa44Ed25519`)
203    /// - 1,952 bytes → ML-DSA-65 (`HybridMlDsa65Ed25519`)
204    /// - 2,592 bytes → ML-DSA-87 (`HybridMlDsa87Ed25519`)
205    ///
206    /// # Arguments
207    /// * `use_case` - The use case this key was generated for
208    /// * `pk` - Hybrid signature public key (ML-DSA + Ed25519)
209    /// * `sk` - Hybrid signature secret key (ML-DSA + Ed25519)
210    ///
211    /// # Errors
212    /// Returns an error if `pk.parameter_set()` does not map to a known
213    /// hybrid-signature `KeyAlgorithm` variant. (As of 0.8.0 all three
214    /// ML-DSA parameter sets are mapped, so this is currently
215    /// unreachable; the `Result` shape is retained for forward
216    /// compatibility with future ML-DSA variants.)
217    pub fn from_hybrid_sig_keypair(
218        use_case: crate::types::types::UseCase,
219        pk: &crate::hybrid::sig_hybrid::HybridSigPublicKey,
220        sk: &crate::hybrid::sig_hybrid::HybridSigSecretKey,
221    ) -> Result<(Self, Self)> {
222        // Read the parameter set off the typed handle introduced in
223        // 0.8.0 — see HybridSigPublicKey::parameter_set. Earlier
224        // revisions sniffed it from the PK byte length (1312/1952/2592);
225        // that worked because the FIPS 204 sets have unique lengths,
226        // but it was inconsistent with the KEM side (which used the
227        // typed level()) and would silently break if any future ML-DSA
228        // variant collided on length.
229        let algorithm: KeyAlgorithm = pk.parameter_set().into();
230
231        let pub_key = Self {
232            version: Self::CURRENT_VERSION,
233            use_case: Some(use_case),
234            security_level: None,
235            algorithm,
236            key_type: KeyType::Public,
237            key_data: KeyData::from_composite(pk.ml_dsa_pk(), pk.ed25519_pk()),
238            created: Utc::now(),
239            metadata: BTreeMap::new(),
240            not_after: None,
241        };
242
243        let sec_key = Self {
244            version: Self::CURRENT_VERSION,
245            use_case: Some(use_case),
246            security_level: None,
247            algorithm,
248            key_type: KeyType::Secret,
249            key_data: KeyData::from_composite(
250                sk.expose_ml_dsa_secret(),
251                sk.expose_ed25519_secret(),
252            ),
253            created: Utc::now(),
254            metadata: BTreeMap::new(),
255            not_after: None,
256        };
257
258        Ok((pub_key, sec_key))
259    }

261    /// Extract a hybrid signature `HybridPublicKey` from a portable key.
262    ///
263    /// # Errors
264    /// Returns an error if the algorithm is not a hybrid signature, the key
265    /// file is not marked [`KeyType::Public`], or key data is invalid.
266    ///
267    /// # M2 fix
268    ///
269    /// Mirrors [`to_hybrid_public_key`](Self::to_hybrid_public_key). See its
270    /// docs for the asymmetry rationale.
271    pub fn to_hybrid_sig_public_key(
272        &self,
273    ) -> Result<crate::hybrid::sig_hybrid::HybridSigPublicKey> {
274        // M-B: see `to_hybrid_public_key`.
275        self.validate_with_expiry_now()?;
276        if self.key_type != KeyType::Public {
277            return Err(CoreError::InvalidKey(
278                "Cannot extract hybrid signature public key from a non-public key file".to_string(),
279            ));
280        }
281
282        let parameter_set = crate::primitives::sig::ml_dsa::MlDsaParameterSet::try_from(
283            self.algorithm,
284        )
285        .map_err(|()| {
286            CoreError::InvalidKey(format!(
287                "Not a hybrid signature algorithm: {alg:?}",
288                alg = self.algorithm
289            ))
290        })?;
291
292        let (pq_bytes, classical_bytes) = self.key_data.decode_composite()?;
293
294        crate::hybrid::sig_hybrid::HybridSigPublicKey::new(parameter_set, pq_bytes, classical_bytes)
295            .map_err(|e| CoreError::InvalidKey(format!("hybrid sig public key: {e}")))
296    }

298    /// Extract a hybrid signature `HybridSecretKey` from a portable key.
299    ///
300    /// # Errors
301    /// Returns an error if the algorithm is not hybrid signature or key data is invalid.
302    pub fn to_hybrid_sig_secret_key(
303        &self,
304    ) -> Result<crate::hybrid::sig_hybrid::HybridSigSecretKey> {
305        // M-B: see `to_hybrid_public_key`.
306        self.validate_with_expiry_now()?;
307        let parameter_set = crate::primitives::sig::ml_dsa::MlDsaParameterSet::try_from(
308            self.algorithm,
309        )
310        .map_err(|()| {
311            CoreError::InvalidKey(format!(
312                "Not a hybrid signature algorithm: {alg:?}",
313                alg = self.algorithm
314            ))
315        })?;
316
317        if self.key_type != KeyType::Secret {
318            return Err(CoreError::InvalidKey(
319                "Cannot reconstruct secret key from a public key".to_string(),
320            ));
321        }
322
323        // Use the zeroizing decoder so the ML-DSA secret-key bytes and
324        // Ed25519 seed are never held as bare `Vec<u8>` between decode
325        // and `Zeroizing` wrapping.
326        let (pq_bytes, classical_bytes) = self.key_data.decode_composite_zeroized()?;
327
328        Ok(crate::hybrid::sig_hybrid::HybridSigSecretKey::new(
329            parameter_set,
330            pq_bytes,
331            classical_bytes,
332        ))
333    }

490    /// Extract ML-KEM public (encapsulation) key.
491    ///
492    /// # Errors
493    /// Returns an error if the algorithm is not a standalone ML-KEM variant,
494    /// key type is not Public, or the key data is malformed.
495    pub fn to_ml_kem_public_key(&self) -> Result<crate::primitives::kem::ml_kem::MlKemPublicKey> {
496        let level = match self.algorithm {
497            KeyAlgorithm::MlKem512 => crate::primitives::kem::MlKemSecurityLevel::MlKem512,
498            KeyAlgorithm::MlKem768 => crate::primitives::kem::MlKemSecurityLevel::MlKem768,
499            KeyAlgorithm::MlKem1024 => crate::primitives::kem::MlKemSecurityLevel::MlKem1024,
500            other => {
501                return Err(CoreError::InvalidKey(format!(
502                    "Not a standalone ML-KEM algorithm: {other:?}"
503                )));
504            }
505        };
506        if self.key_type != KeyType::Public {
507            return Err(CoreError::InvalidKey(
508                "ML-KEM public key requires Public key type".to_string(),
509            ));
510        }
511        let bytes = self.key_data.decode_raw()?;
512        crate::primitives::kem::ml_kem::MlKemPublicKey::new(level, bytes)
513            .map_err(|e| CoreError::InvalidKey(format!("ML-KEM public key: {e}")))
514    }

516    /// Extract ML-KEM secret (decapsulation) key.
517    ///
518    /// # Errors
519    /// Returns an error if the algorithm is not a standalone ML-KEM variant,
520    /// key type is not Secret, or the key data is malformed.
521    pub fn to_ml_kem_secret_key(&self) -> Result<crate::primitives::kem::ml_kem::MlKemSecretKey> {
522        let level = match self.algorithm {
523            KeyAlgorithm::MlKem512 => crate::primitives::kem::MlKemSecurityLevel::MlKem512,
524            KeyAlgorithm::MlKem768 => crate::primitives::kem::MlKemSecurityLevel::MlKem768,
525            KeyAlgorithm::MlKem1024 => crate::primitives::kem::MlKemSecurityLevel::MlKem1024,
526            other => {
527                return Err(CoreError::InvalidKey(format!(
528                    "Not a standalone ML-KEM algorithm: {other:?}"
529                )));
530            }
531        };
532        if self.key_type != KeyType::Secret {
533            return Err(CoreError::InvalidKey(
534                "ML-KEM secret key requires Secret key type".to_string(),
535            ));
536        }

539        // error path where `MlKemSecretKey::new` rejects malformed
540        // material before consuming the bytes.
541        let bytes = self.key_data.decode_raw_zeroized()?;
542        crate::primitives::kem::ml_kem::MlKemSecretKey::new(level, bytes.to_vec())
543            .map_err(|e| CoreError::InvalidKey(format!("ML-KEM secret key: {e}")))
544    }

548    /// Extract ML-DSA verifying (public) key bytes.
549    ///
550    /// # Errors
551    /// Returns an error if the algorithm is not a standalone ML-DSA variant
552    /// or key type is not Public.
553    pub fn to_ml_dsa_verifying_key_bytes(&self) -> Result<Vec<u8>> {
554        if !matches!(
555            self.algorithm,
556            KeyAlgorithm::MlDsa44 | KeyAlgorithm::MlDsa65 | KeyAlgorithm::MlDsa87
557        ) {
558            return Err(CoreError::InvalidKey(format!(
559                "Not a standalone ML-DSA algorithm: {:?}",
560                self.algorithm
561            )));
562        }
563        if self.key_type != KeyType::Public {
564            return Err(CoreError::InvalidKey(
565                "ML-DSA verifying key requires Public key type".to_string(),
566            ));
567        }
568        self.key_data.decode_raw()
569    }
570
571    /// Extract ML-DSA signing (secret) key bytes (zeroized on drop).
572    ///
573    /// # Errors
574    /// Returns an error if the algorithm is not a standalone ML-DSA variant
575    /// or key type is not Secret.
576    pub fn to_ml_dsa_signing_key_bytes(&self) -> Result<zeroize::Zeroizing<Vec<u8>>> {
577        if !matches!(
578            self.algorithm,
579            KeyAlgorithm::MlDsa44 | KeyAlgorithm::MlDsa65 | KeyAlgorithm::MlDsa87
580        ) {
581            return Err(CoreError::InvalidKey(format!(
582                "Not a standalone ML-DSA algorithm: {:?}",
583                self.algorithm
584            )));
585        }
586        if self.key_type != KeyType::Secret {
587            return Err(CoreError::InvalidKey(
588                "ML-DSA signing key requires Secret key type".to_string(),
589            ));
590        }
591        self.key_data.decode_raw_zeroized()
592    }
593
594    // --- Bridge: Symmetric key ---
595
596    /// Create a `PortableKey` from raw symmetric key bytes.
597    ///
598    /// now requires `security_level: SecurityLevel`. The
599    /// previous signature produced a `PortableKey` with both
600    /// `use_case = None` AND `security_level = None`, which the
601    /// invariant check at deserialization (`from_json` / `from_cbor`)
602    /// rejects. Symmetric keys round-tripped through JSON/CBOR were
603    /// silently broken on reload. The added parameter makes the
604    /// invariant satisfiable at construction time and is the same
605    /// shape callers already use for non-symmetric constructors.
606    /// **Breaking change**: callers must add a `security_level`
607    /// argument.
608    ///
609    /// # Errors
610    /// Returns an error if the algorithm is not symmetric (AES-256 or ChaCha20).
611    pub fn from_symmetric_key(
612        algorithm: KeyAlgorithm,
613        security_level: crate::types::SecurityLevel,
614        key: &[u8],
615    ) -> Result<Self> {
616        if !algorithm.is_symmetric() {
617            return Err(CoreError::InvalidKey(format!(
618                "{algorithm:?} is not a symmetric algorithm"
619            )));
620        }
621        Ok(Self {
622            version: Self::CURRENT_VERSION,
623            use_case: None,
624            security_level: Some(security_level),
625            algorithm,
626            key_type: KeyType::Symmetric,
627            key_data: KeyData::from_raw(key),
628            created: Utc::now(),
629            metadata: BTreeMap::new(),
630            not_after: None,
631        })
632    }
633}
