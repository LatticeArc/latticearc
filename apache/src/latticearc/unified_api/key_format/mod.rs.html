5//! LatticeArc Portable Key Format (LPK v1)
6//!
7//! Schema-first dual-format key serialization supporting JSON (human-readable)
8//! and CBOR (compact binary, RFC 8949) for all LatticeArc key types.
9//!
10//! # Design Principles
11//!
12//! - **Schema-first**: One Rust struct, two wire formats (JSON + CBOR)
13//! - **CBOR primary**: Wire protocol, database storage, enterprise containers
14//! - **JSON secondary**: CLI display, REST APIs, debugging, key export
15//! - **Standards-aligned**: Algorithm IDs from FIPS 203-206 / RFC 7748 / RFC 8032;
16//!   composite key layout follows `draft-ietf-lamps-pq-composite-kem` (concatenation);
17//!   JWK-compatible structure (per `draft-ietf-jose-pqc-kem`)
18//!
19//! # Key Identity
20//!
21//! Keys are identified by **use case** or **security level** — mirroring
22//! how the library's API works. The algorithm is auto-derived and stored
23//! internally for version-stability.
24//!
25//! # JSON Format
26//!
27//! ```json
28//! {
29//!   "version": 1,
30//!   "use_case": "file-storage",
31//!   "algorithm": "hybrid-ml-kem-1024-x25519",
32//!   "key_type": "public",
33//!   "key_data": { "raw": "Base64..." },
34//!   "created": "2026-03-19T..."
35//! }
36//! ```
37//!
38//! # CBOR Format
39//!
40//! Same logical schema. Key material stored as CBOR byte strings (`bstr`) —
41//! no base64 encoding, no string overhead. See [`crate::unified_api::key_format::PortableKey::to_cbor`].
42//!
43//! # Enterprise Extension Model
44//!
45//! The `metadata` field is an open `BTreeMap<String, serde_json::Value>`.
46//! Enterprise crates add dimensions, key expiry,
47//! hardware binding, etc. via extension traits — the base library preserves
48//! unknown metadata keys during roundtrips without modification:
49//!
50//! ```text
51//! // Enterprise crate — typed accessors over the metadata map (pseudocode)
52//! impl EnterpriseKeyExt for PortableKey {
53//!     fn dimensions(&self) -> Option<Vec<String>> { ... }
54//!     fn key_expiry(&self) -> Option<DateTime<Utc>> { ... }
55//!     fn hsm_binding(&self) -> Option<HsmSlot> { ... }
56//! }
57//! ```

63// --- Passphrase-encryption envelope constants (LPK v1 encrypted variant) ---
64
65/// Current version of the encrypted-key envelope schema.
66///
67/// Each bump marks an AEAD AAD-layout change. An envelope written by an
68/// older version has the same wire shape but a different AAD, so a newer
69/// verifier's AAD will not match and AEAD authentication fails. Without a
70/// version field that failure is indistinguishable from a wrong
71/// passphrase. Bumping the field on every AAD change lets the loader emit
72/// a distinct "old envelope; re-protect" error so users decrypt and
73/// re-encrypt rather than chase a passphrase that was already correct.
74///
75/// History:
76/// - `1` → `2`: metadata-binding fix; AAD magic `lpk-v1-enc` → `lpk-v2-enc`.
77/// - `2` → `3`: AAD canonicalization fix — a null terminator was added
78///   after the `aead` string field; AAD magic `lpk-v2-enc` → `lpk-v3-enc`.
79///   See `encryption_aad` for the layout.
80const ENCRYPTED_ENVELOPE_VERSION: u32 = 3;
81/// KDF identifier for the encrypted-key envelope.
82const PBKDF2_KDF_ID: &str = "PBKDF2-HMAC-SHA256";
83/// AEAD identifier for the encrypted-key envelope.
84const AES_GCM_AEAD_ID: &str = "AES-256-GCM";
85/// PBKDF2 default iteration count (OWASP 2023 recommendation for HMAC-SHA256).
86const PBKDF2_DEFAULT_ITERATIONS: u32 = 600_000;
87/// PBKDF2 minimum iteration count accepted when loading an encrypted key.
88const PBKDF2_MIN_ITERATIONS: u32 = 100_000;
89/// PBKDF2 maximum iteration count accepted when loading an encrypted key.
90///
91/// an attacker-supplied envelope with `kdf_iterations =
92/// u32::MAX` (4.3 billion) would force tens of minutes of HMAC-SHA256 work
93/// per decrypt attempt, denying service to the legitimate keyholder. This
94/// upper bound is two orders of magnitude above the OWASP-2023
95/// recommendation (600 k) and well above any reasonable interactive-login
96/// budget, so it does not constrain legitimate callers.
97const PBKDF2_MAX_ITERATIONS: u32 = 10_000_000;
98/// PBKDF2 salt length in bytes (SP 800-132 recommends ≥ 16).
99const PBKDF2_SALT_LEN: usize = 16;
100/// PBKDF2 minimum salt length accepted when loading an encrypted key.
101/// Re-exported from the canonical NIST SP 800-132 §5.1 constant on
102/// [`Pbkdf2Params::MIN_SALT_LEN`](crate::primitives::kdf::pbkdf2::Pbkdf2Params::MIN_SALT_LEN)
103/// so the load-side check and the construction-side check cannot drift.
104const PBKDF2_MIN_SALT_LEN: usize = crate::primitives::kdf::pbkdf2::Pbkdf2Params::MIN_SALT_LEN;
105/// AES-256-GCM nonce length in bytes (NIST SP 800-38D).
106const AES_GCM_NONCE_LEN: usize = 12;
107/// AES-256-GCM tag length in bytes.
108const AES_GCM_TAG_LEN: usize = 16;
109/// AES-256 key length in bytes.
110const AES_256_KEY_LEN: usize = 32;
