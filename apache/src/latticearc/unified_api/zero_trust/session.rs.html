1//! `ZeroTrustSession`: manual challenge/proof/verify authentication flow,
2//! convertible into a [`super::VerifiedSession`].
3
4use super::{Challenge, VerifiedSession, ZeroKnowledgeProof, ZeroTrustAuth};
5use crate::log_zero_trust_session_verification_failed;
6use crate::types::traits::ZeroTrustAuthenticable;
7use crate::unified_api::error::{CoreError, Result};
8use chrono::{DateTime, Utc};
9
10/// A zero-trust session managing the authentication flow.
11pub struct ZeroTrustSession {
12    /// The underlying authentication handler.
13    pub(crate) auth: ZeroTrustAuth,
14    /// Current active challenge, if any.
15    challenge: Option<Challenge>,
16    /// Whether the session has been verified.
17    verified: bool,
18    /// When the session started.
19    session_start: DateTime<Utc>,
20}
21
22impl ZeroTrustSession {
23    /// Creates a new zero-trust session.
24    #[must_use]
25    pub fn new(auth: ZeroTrustAuth) -> Self {
26        Self { auth, challenge: None, verified: false, session_start: Utc::now() }
27    }
28
29    /// Initiates the authentication flow by generating a new challenge.
30    ///
31    /// # Errors
32    ///
33    /// Returns `CoreError::EntropyDepleted` if the system cannot generate random bytes
34    /// for the challenge.
35    pub fn initiate_authentication(&mut self) -> Result<Challenge> {
36        let challenge = self.auth.generate_challenge()?;
37        self.challenge = Some(challenge.clone());
38        Ok(challenge)
39    }
40
41    /// Verifies a proof response against the active challenge.
42    ///
43    /// # Errors
44    ///
45    /// Returns `CoreError::AuthenticationFailed` if:
46    /// - No challenge has been initiated (no active challenge).
47    /// - The challenge has expired.
48    /// - The proof format is invalid for Medium/High complexity proofs.
49    ///
50    /// Returns `CoreError::InvalidInput` if the public key or signature format is invalid
51    /// during Ed25519 verification.
52    ///
53    /// Returns `CoreError::InvalidKeyLength` if the public key has incorrect length.
54    pub fn verify_response(&mut self, proof: &ZeroKnowledgeProof) -> Result<bool> {
55        // Pattern-6: collapse the "no active challenge" and "challenge
56        // expired" cases to a single fixed string in the typed error
57        // AND in the SIEM audit log. Both are caller-protocol mistakes
58        // (not adversary-controllable crypto material), but distinct
59        // strings let an attacker mapping the protocol state machine
60        // fingerprint which check tripped. The discriminator goes to
61        // `tracing::debug!` for operator visibility, mirroring
62        // `verify_pop`'s posture.
63        const VERIFY_RESPONSE_FAILED: &str = "challenge verification failed";
64
65        let challenge = self.challenge.as_ref().ok_or_else(|| {
66            log_zero_trust_session_verification_failed!("pending", VERIFY_RESPONSE_FAILED);
67            tracing::debug!("verify_response rejected: no active challenge");
68            CoreError::AuthenticationFailed(VERIFY_RESPONSE_FAILED.to_string())
69        })?;
70
71        if challenge.is_expired() {
72            log_zero_trust_session_verification_failed!("pending", VERIFY_RESPONSE_FAILED);
73            tracing::debug!("verify_response rejected: challenge expired");
74            return Err(CoreError::AuthenticationFailed(VERIFY_RESPONSE_FAILED.to_string()));
75        }
76
77        // Snapshot challenge data, then immediately consume the
78        // challenge regardless of verification outcome. Without this,
79        // a captured (challenge, proof) pair on the wire is bit-for-bit
80        // replayable against the SAME ZeroTrustSession instance within
81        // the `challenge_timeout_ms` window (Ed25519 is deterministic
82        // per RFC 8032; the proof bytes don't change). Mirrors the
83        // posture of `verify_pop` (which uses `pop_replay_cache` for
84        // the same attack class). Off-machine replay defense still
85        // requires transport-binding (TLS); see SECURITY.md.
86        let challenge_data = challenge.data().to_vec();
87        self.challenge = None;
88
89        let verified = self.auth.verify_proof(proof, &challenge_data)?;
90        self.verified = verified;
91
92        if !verified {
93            log_zero_trust_session_verification_failed!("pending", "Proof verification failed");
94        }
95
96        Ok(self.verified)
97    }
98
99    /// Generates a zero-knowledge proof for the given challenge.
100    ///
101    /// This is used in the manual authentication flow: after calling
102    /// [`initiate_authentication`](Self::initiate_authentication) to get a challenge,
103    /// generate a proof and then pass it to [`verify_response`](Self::verify_response).
104    ///
105    /// # Errors
106    ///
107    /// Returns `CoreError::AuthenticationFailed` if the challenge data is empty.
108    ///
109    /// Returns `CoreError::InvalidKeyLength` if the private key has incorrect length.
110    ///
111    /// Returns `CoreError::InvalidInput` if the private key format is invalid.
112    pub fn generate_proof(&self, challenge: &Challenge) -> Result<ZeroKnowledgeProof> {
113        self.auth.generate_proof(challenge.data())
114    }
115
116    /// Returns `true` if the session has been successfully authenticated.
117    #[must_use]
118    pub fn is_authenticated(&self) -> bool {
119        self.verified
120    }
121
122    /// Returns the age of the session in milliseconds since creation.
123    ///
124    /// # Errors
125    ///
126    /// This function does not currently return errors, but returns `Result` for
127    /// API consistency and future extensibility.
128    pub fn session_age_ms(&self) -> Result<u64> {
129        let elapsed = Utc::now().signed_duration_since(self.session_start);
130        // Convert safely: negative elapsed times treated as 0 (just started)
131        let elapsed_u64 = u64::try_from(elapsed.num_milliseconds()).unwrap_or(0);
132        Ok(elapsed_u64)
133    }
134
135    /// Convert this session into a `VerifiedSession` after successful authentication.
136    ///
137    /// This consumes the `ZeroTrustSession` and returns a `VerifiedSession` that
138    /// can be used to authorize cryptographic operations.
139    ///
140    /// # Errors
141    ///
142    /// Returns `CoreError::AuthenticationRequired` if the session has not been
143    /// successfully authenticated via `verify_response()`.
144    ///
145    /// Returns `CoreError::EntropyDepleted` if session ID generation fails.
146    pub fn into_verified(self) -> Result<VerifiedSession> {
147        VerifiedSession::from_authenticated(&self)
148    }
149}
