6//! ECDH (Elliptic Curve Diffie-Hellman) Key Exchange
7//!
8//! This module provides X25519 (RFC 7748, 128-bit security) key agreement via
9//! aws-lc-rs for FIPS 140-3 compliance and optimized performance (AVX2, AES-NI).
10//! X25519 is the classical leg of the crate's hybrid KEM; no other ECDH curves
11//! are exposed.
12//!
13//! # Performance
14//!
15//! aws-lc-rs provides ~4x speedup over pure-Rust implementations:
16//! - Key generation: ~6µs (vs ~24µs for pure-Rust x25519)
17//! - Key agreement: ~6µs (vs ~20µs for pure-Rust x25519)
18//!
19//! # Example
20//!
21//! ```no_run
22//! use latticearc::primitives::kem::ecdh::X25519KeyPair;
23//!
24//! let alice = X25519KeyPair::generate()?;
25//! let bob = X25519KeyPair::generate()?;
26//!
27//! let alice_pk = alice.public_key_bytes().to_vec();
28//! let bob_pk = bob.public_key_bytes().to_vec();
29//!
30//! let alice_secret = alice.agree(&bob_pk)?;
31//! let bob_secret = bob.agree(&alice_pk)?;
32//!
33//! assert_eq!(alice_secret, bob_secret);
34//! # Ok::<(), latticearc::primitives::kem::ecdh::EcdhError>(())
35//! ```

53
54    /// Shared secret derivation failed
55    #[error("ECDH shared secret derivation failed")]
56    SharedSecretDerivationFailed,
57
58    /// Invalid key size
59    #[error("Invalid key size: expected {expected}, got {actual}")]
60    InvalidKeySize {
61        /// Expected key size in bytes

63        /// Actual key size in bytes

65    },
66

68    #[error("ECDH key agreement failed")]
69    AgreementFailed,

113/// L5 fix: constant-time comparison for `X25519PublicKey`.
114///
115/// The derived `PartialEq` above is varying-time and is preserved because the
116/// 32-byte X25519 public key is not secret material — callers using it as a
117/// `HashMap` key or in `assert_eq!` test fixtures rely on `==`. Adding a
118/// `ConstantTimeEq` impl is defense-in-depth: callers that DO need timing-
119/// independent comparison (e.g. CT-discipline trust-anchor pins) have it
120/// without an obvious foot-gun if a future refactor secretizes the wrapped
121/// bytes.
122impl subtle::ConstantTimeEq for X25519PublicKey {
123    fn ct_eq(&self, other: &Self) -> subtle::Choice {
124        self.bytes.ct_eq(&other.bytes)
125    }

128/// Low-order points on Curve25519 that must be rejected per RFC 7748 §6.1.
129///
130/// These are the small-order subgroup elements (orders 1, 2, 4, 8) on the
131/// Montgomery curve and its twist. Using any of them as a peer public key
132/// forces the shared secret to be predictable (typically all-zeros), so a
133/// malicious peer can learn the private scalar modulo the small subgroup.
134///
135/// The exact byte values mirror the `has_small_order()` blacklist in
136/// libsodium's `crypto_scalarmult/curve25519/ref10/x25519_ref10.c`, which is
137/// the canonical reference implementation and is itself derived from RFC 7748
138/// §6.1. libsodium ships seven entries; we use the same seven, unchanged.
139///
140/// # Defense-in-depth rationale
141///
142/// aws-lc-rs (the backing implementation) may or may not reject these inputs
143/// internally before invoking the X25519 scalar-mult ladder. Rather than rely
144/// on that assumption, every call that accepts peer-supplied X25519 public
145/// key bytes routes through `X25519PublicKey::from_bytes`, which checks this
146/// list in constant time. This closes the hybrid KEM bypass where raw peer
147/// bytes were previously passed directly to `agreement::agree_ephemeral` /
148/// `agreement::agree` without going through the wrapper.
149///
150/// # Prior list note
151///
152/// Earlier revisions of this module carried twelve entries — the seven
153/// canonical points plus five "high-bit-set" variants. Audit review found
154/// that two of those variants did not match any canonical reference, and the
155/// RFC 7748 §5 clamping step already masks bit 255 before the ladder runs,
156/// making the extras either redundant or potentially incorrect. The list was
157/// trimmed to the seven canonical points to eliminate ambiguity.
158const X25519_LOW_ORDER_POINTS: [[u8; X25519_KEY_SIZE]; 7] = [
159    // 0 (order 4 — identity representation).
160    [0x00; 32],
161    // 1 (order 1 — neutral element image).
162    [
163        0x01, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00,
164        0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00,
165        0x00, 0x00,
166    ],
167    // Order 8 — Curve25519 small-order point
168    // 325606250916557431795983626356110631294008115727848805560023387167927233504.
169    [
170        0xe0, 0xeb, 0x7a, 0x7c, 0x3b, 0x41, 0xb8, 0xae, 0x16, 0x56, 0xe3, 0xfa, 0xf1, 0x9f, 0xc4,
171        0x6a, 0xda, 0x09, 0x8d, 0xeb, 0x9c, 0x32, 0xb1, 0xfd, 0x86, 0x62, 0x05, 0x16, 0x5f, 0x49,
172        0xb8, 0x00,
173    ],
174    // Order 8 — Curve25519 small-order point
175    // 39382357235489614581723060781553021112529911719440698176882885853963445705823.
176    [
177        0x5f, 0x9c, 0x95, 0xbc, 0xa3, 0x50, 0x8c, 0x24, 0xb1, 0xd0, 0xb1, 0x55, 0x9c, 0x83, 0xef,
178        0x5b, 0x04, 0x44, 0x5c, 0xc4, 0x58, 0x1c, 0x8e, 0x86, 0xd8, 0x22, 0x4e, 0xdd, 0xd0, 0x9f,
179        0x11, 0x57,
180    ],
181    // p - 1 (order 2).
182    [
183        0xec, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff,
184        0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff,
185        0xff, 0x7f,
186    ],
187    // p (order 4 — zero after reduction).
188    [
189        0xed, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff,
190        0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff,
191        0xff, 0x7f,
192    ],
193    // p + 1 (order 1 — one after reduction).
194    [
195        0xee, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff,
196        0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff,
197        0xff, 0x7f,
198    ],
199];

201impl X25519PublicKey {
202    /// Create a new X25519 public key from bytes
203    ///
204    /// # Errors
205    /// Returns an error if the provided bytes are not exactly 32 bytes
206    /// or if the key is a known low-order point on Curve25519 (RFC 7748 §6.1).
207    pub fn from_bytes(bytes: &[u8]) -> Result<Self, EcdhError> {
208        use subtle::ConstantTimeEq;
209
210        if bytes.len() != X25519_KEY_SIZE {
211            return Err(EcdhError::InvalidKeySize {
212                expected: X25519_KEY_SIZE,
213                actual: bytes.len(),
214            });
215        }
216        // Reject the 7 canonical low-order points on Curve25519 (RFC 7748 §6.1).
217        // We compare in constant time using subtle::Choice so a peer cannot
218        // distinguish "all-zero" from "ordered-4 point" from timing.
219        let mut is_low_order = subtle::Choice::from(0u8);
220        for point in &X25519_LOW_ORDER_POINTS {
221            is_low_order |= bytes.ct_eq(point);
222        }
223        if bool::from(is_low_order) {
224            return Err(EcdhError::InvalidKeyMaterial(
225                "X25519 public key is a low-order point (RFC 7748 §6.1)".to_string(),
226            ));
227        }
228        let mut key_bytes = [0u8; X25519_KEY_SIZE];
229        key_bytes.copy_from_slice(bytes);
230        Ok(Self { bytes: key_bytes })
231    }
232
233    /// Get the public key as bytes
234    #[must_use]
235    pub fn as_bytes(&self) -> &[u8; X25519_KEY_SIZE] {
236        &self.bytes
237    }
238
239    /// Convert to `Vec<u8>`
240    #[must_use]
241    pub fn to_vec(&self) -> Vec<u8> {
242        self.bytes.to_vec()
243    }
244}

299/// X25519 key pair containing both public and secret keys
300///
301/// This struct holds an ephemeral private key from aws-lc-rs along with
302/// the computed public key bytes for transmission.
303///
304/// # Zeroization
305///
306/// The inner `EphemeralPrivateKey` is managed by aws-lc-rs. Zeroization of
307/// the private key material on drop is delegated to the aws-lc-rs
308/// (BoringSSL) allocator, which zeros memory on free. See SECURITY.md
309/// ("aws-lc-rs-Wrapped Secret Types").
310///
311/// # Constant-Time Comparison
312///
313/// `ConstantTimeEq` is not implemented because the inner aws-lc-rs type
314/// does not expose raw key bytes. `PartialEq` is also not implemented,
315/// and a compile-time barrier in
316/// `latticearc/tests/no_partial_eq_on_secret_types.rs` prevents one from
317/// being added without removing the explicit assertion. See SECURITY.md
318/// ("aws-lc-rs-Wrapped Secret Types").
319pub struct X25519KeyPair {
320    private: EphemeralPrivateKey,
321    public_bytes: [u8; X25519_KEY_SIZE],
322}
323
324impl X25519KeyPair {
325    /// Generate a new X25519 key pair
326    ///
327    /// # Errors
328    /// Returns an error if key generation fails.
329    pub fn generate() -> Result<Self, EcdhError> {
330        let rng = aws_lc_rs::rand::SystemRandom::new();
331        let private = EphemeralPrivateKey::generate(&X25519, &rng)
332            .map_err(|_e| EcdhError::KeyGenerationFailed)?;
333        let public = private.compute_public_key().map_err(|_e| EcdhError::KeyGenerationFailed)?;
334
335        let mut public_bytes = [0u8; X25519_KEY_SIZE];
336        public_bytes.copy_from_slice(public.as_ref());
337
338        Ok(Self { private, public_bytes })
339    }
340
341    /// Get public key bytes for transmission
342    #[must_use]
343    pub fn public_key_bytes(&self) -> &[u8; X25519_KEY_SIZE] {
344        &self.public_bytes
345    }
346
347    /// Get the public key
348    #[must_use]
349    pub fn public_key(&self) -> X25519PublicKey {
350        X25519PublicKey { bytes: self.public_bytes }
351    }
352
353    /// Perform X25519 key agreement with a peer's public key
354    ///
355    /// Consumes the private key to ensure single-use (ephemeral) semantics.
356    ///
357    /// Validates the peer key against the RFC 7748 §6.1 low-order point
358    /// blacklist before performing the Diffie-Hellman exchange. This is
359    /// defense-in-depth on top of aws-lc-rs's own validation; it ensures
360    /// every caller of `agree()` gets the low-order rejection, including
361    /// hybrid KEM paths that receive peer keys as raw bytes.
362    ///
363    /// # Errors
364    /// Returns an error if the peer key is a low-order point (per RFC 7748
365    /// §6.1) or key agreement fails.
366    pub fn agree(
367        self,
368        peer_public_bytes: &[u8],
369    ) -> Result<Zeroizing<[u8; X25519_KEY_SIZE]>, EcdhError> {
370        // Reject low-order peer keys before reaching aws-lc-rs.
371        // `from_bytes` enforces the 7-point blacklist (RFC 7748 §6.1).
372        // the validated bytes are explicitly
373        // documented as the source the aws-lc-rs primitive consumes —
374        // `from_bytes` does not transform the bytes (X25519 PKs are
375        // their own canonical form modulo the low-order check) so
376        // passing the original `peer_public_bytes` to aws-lc-rs is
377        // semantically identical to passing the validated copy.
378        // Documenting the invariant here so a future refactor that
379        // changes `from_bytes` to canonicalize cannot silently let
380        // the unvalidated bytes through.
381        let _validated_peer = X25519PublicKey::from_bytes(peer_public_bytes)?;
382        let peer_public = UnparsedPublicKey::new(&X25519, peer_public_bytes);
383
384        agreement::agree_ephemeral(
385            self.private,
386            peer_public,
387            EcdhError::AgreementFailed,
388            |shared_secret| {
389                // Allocate the destination as `Zeroizing<[u8; 32]>`
390                // first so the stack slot we copy into is owned by a
391                // type that scrubs on drop. A `let mut result =
392                // [0u8; 32]; ...; Ok(Zeroizing::new(result))` shape
393                // leaves the unwrapped stack slot holding the secret
394                // bytes after the move into Zeroizing.
395                let mut result: Zeroizing<[u8; X25519_KEY_SIZE]> =
396                    Zeroizing::new([0u8; X25519_KEY_SIZE]);
397                result.copy_from_slice(shared_secret);
398                // L-B: defense-in-depth all-zero shared-secret check,
399                // mirroring the P-256 / P-384 / P-521 closures above.
400                // The from_bytes(peer_public_bytes) call earlier already
401                // rejects the RFC 7748 §6.1 low-order set (including the
402                // all-zero key), so this is redundant under aws-lc-rs's
403                // current behaviour. The check still guards against a
404                // future backend swap that lets a degenerate output
405                // through and keeps the CT discipline uniform across
406                // ECDH variants. is_all_zero_bytes is constant-time over
407                // the fixed-length slice.
408                if crate::primitives::ct::is_all_zero_bytes(result.as_slice()) {
409                    return Err(EcdhError::AgreementFailed);
410                }
411                Ok(result)
412            },
413        )
414    }
415}

426/// X25519 static key pair for reusable Diffie-Hellman key agreement.
427///
428/// Unlike [`X25519KeyPair`] which uses ephemeral keys (consumed on use),
429/// this type wraps [`PrivateKey`] which supports multiple `agree()` calls
430/// without consuming the key. This is essential for hybrid KEM where the
431/// recipient's X25519 key must persist for decapsulation.
432///
433/// # Zeroization
434///
435/// Zeroization of the inner `PrivateKey` is delegated to aws-lc-rs
436/// (BoringSSL), which zeros key material on free. See SECURITY.md
437/// ("aws-lc-rs-Wrapped Secret Types").
438///
439/// # Constant-Time Comparison
440///
441/// `ConstantTimeEq` is not implemented because this type is not compared in
442/// any production code path. If byte-level comparison is required, extract
443/// the seed via aws-lc-rs `AsBigEndian<Curve25519SeedBin>` and compare those
444/// bytes in constant time — materializing secret bytes on every compare is
445/// avoided by default.
446///
447/// # Limitations
448///
449/// aws-lc-rs 1.16+ supports X25519 raw-bytes import via `from_private_key`
450/// and export via `AsBigEndian<Curve25519SeedBin>`. DER encoding of X25519
451/// private keys is still unsupported upstream. This wrapper currently exposes
452/// only the generate/agree path; add serialization helpers if persistence
453/// is needed.
454///
455/// # Example
456///
457/// ```no_run
458/// # fn main() -> Result<(), Box<dyn std::error::Error>> {
459/// use latticearc::primitives::kem::ecdh::X25519StaticKeyPair;
460///
461/// let alice = X25519StaticKeyPair::generate()?;
462/// let bob = X25519StaticKeyPair::generate()?;
463///
464/// // Commutativity: alice.agree(bob_pk) == bob.agree(alice_pk)
465/// let ss1 = alice.agree(bob.public_key_bytes())?;
466/// let ss2 = bob.agree(alice.public_key_bytes())?;
467/// assert_eq!(ss1, ss2);
468///
469/// // Reusable: alice can agree with multiple peers
470/// let carol = X25519StaticKeyPair::generate()?;
471/// let ss3 = alice.agree(carol.public_key_bytes())?;
472/// # Ok(())
473/// # }
474/// ```
475pub struct X25519StaticKeyPair {
476    private: PrivateKey,
477    public_bytes: [u8; X25519_KEY_SIZE],
478}

487        let public = private.compute_public_key().map_err(|_e| EcdhError::KeyGenerationFailed)?;

507    /// Export the private key seed bytes (32 bytes).

522    /// Reconstruct a key pair from previously exported seed bytes.
523    ///
524    /// # Errors

527        let private =
528            PrivateKey::from_private_key(&X25519, seed).map_err(|_e| EcdhError::InvalidKeyData)?;
529        let public = private.compute_public_key().map_err(|_e| EcdhError::KeyGenerationFailed)?;
530
531        let mut public_bytes = [0u8; X25519_KEY_SIZE];
532        public_bytes.copy_from_slice(public.as_ref());
533
534        Ok(Self { private, public_bytes })
535    }

537    /// Perform X25519 key agreement with a peer's public key.
538    ///
539    /// Unlike [`X25519KeyPair::agree`], this does **not** consume the key,
540    /// allowing multiple agreements with different peers.
541    ///
542    /// Validates the peer key against the RFC 7748 §6.1 low-order point
543    /// blacklist before performing the Diffie-Hellman exchange. This closes
544    /// the hybrid KEM bypass where `HybridKemSecretKey::ecdh_agree` passes
545    /// raw peer bytes to the low-level `agree()` without validation.
546    ///
547    /// # Errors
548    /// Returns an error if the peer key is a low-order point (per RFC 7748
549    /// §6.1) or key agreement fails.
550    pub fn agree(
551        &self,
552        peer_public_bytes: &[u8],
553    ) -> Result<Zeroizing<[u8; X25519_KEY_SIZE]>, EcdhError> {
554        // Reject low-order peer keys before reaching aws-lc-rs.
555        let _validated_peer = X25519PublicKey::from_bytes(peer_public_bytes)?;
556        let peer_public = UnparsedPublicKey::new(&X25519, peer_public_bytes);
557
558        agreement::agree(&self.private, peer_public, EcdhError::AgreementFailed, |shared_secret| {
559            // See X25519KeyPair::agree above for the Zeroizing-first
560            // rationale.
561            let mut result: Zeroizing<[u8; X25519_KEY_SIZE]> =
562                Zeroizing::new([0u8; X25519_KEY_SIZE]);
563            result.copy_from_slice(shared_secret);
564            // L-B: defense-in-depth all-zero shared-secret check. See
565            // X25519KeyPair::agree above for the rationale.
566            if crate::primitives::ct::is_all_zero_bytes(result.as_slice()) {
567                return Err(EcdhError::AgreementFailed);
568            }
569            Ok(result)
570        })
571    }
572}

583/// Generate a new X25519 keypair
584///
585/// Returns the public key and secret key bytes. The secret key is stored
586/// in a zeroizing container for security.
587///
588/// Validate a public key has correct size.
589///
590/// # Errors
591/// Returns an error if the public key size does not match the expected X25519 key size.
592pub fn validate_public_key(public_key: &X25519PublicKey) -> Result<(), EcdhError> {
593    if public_key.as_bytes().len() != X25519_KEY_SIZE {
594        return Err(EcdhError::InvalidKeySize {
595            expected: X25519_KEY_SIZE,
596            actual: public_key.as_bytes().len(),
597        });
598    }
599    Ok(())
600}
601
602/// Validate a secret key has correct size.
603///
604/// # Errors
605/// Returns an error if the secret key size does not match the expected X25519 key size.
606pub fn validate_secret_key(secret_key: &X25519SecretKey) -> Result<(), EcdhError> {
607    if secret_key.expose_secret().len() != X25519_KEY_SIZE {
608        return Err(EcdhError::InvalidKeySize {
609            expected: X25519_KEY_SIZE,
610            actual: secret_key.expose_secret().len(),
611        });
612    }
613    Ok(())
614}
615
616/// Perform X25519 key agreement
617///
618/// This creates an ephemeral key pair and performs Diffie-Hellman with the peer's
619/// public key. For static-ephemeral or static-static DH, use `X25519KeyPair`.
620///
621/// # Errors
622/// Returns an error if key agreement fails.
623pub fn agree_ephemeral(
624    peer_public_bytes: &[u8],
625) -> Result<(Zeroizing<[u8; X25519_KEY_SIZE]>, [u8; X25519_KEY_SIZE]), EcdhError> {
626    let keypair = X25519KeyPair::generate()?;
627    let our_public = *keypair.public_key_bytes();
628    let shared_secret = keypair.agree(peer_public_bytes)?;
629    Ok((shared_secret, our_public))
630}
