6//! Shared ECDSA verification core for the NIST prime curves.
7//!
8//! [`super::p256`] and [`super::p384`] are thin per-curve front ends over the
9//! single implementation here, which routes every verification through
10//! aws-lc-rs (`signature::ParsedPublicKey`). Keeping one code path means the
11//! input checks (key encoding, signature length bounds, message resource
12//! limit) and the result convention are identical for both curves.
13//!
14//! Result convention, shared by every public function built on this module:
15//!
16//! - `Err` for inputs that are wrong *before* any signature arithmetic runs:
17//!   a public key that is not uncompressed SEC1 of the expected length or is
18//!   not a point on the curve, a signature whose length is impossible for the
19//!   encoding, or a message above the signature resource limit.
20//! - `Ok(false)` for a well-formed request whose signature does not verify,
21//!   including a DER signature whose contents do not parse.
22//! - `Ok(true)` only for a valid signature.
23
24use aws_lc_rs::digest;
25use aws_lc_rs::signature::{EcdsaVerificationAlgorithm, ParsedPublicKey};

27use crate::prelude::error::{LatticeArcError, Result};

30/// SEC1 tag byte for an uncompressed point (`0x04 || X || Y`).
31const SEC1_UNCOMPRESSED_TAG: u8 = 0x04;
32
33/// Per-curve parameters for the shared verifier.
34pub(super) struct CurveParams {
35    /// Curve name used in debug events and error strings.
36    pub(super) name: &'static str,
37    /// Uncompressed SEC1 public key length (`1 + 2 * field_len`).
38    pub(super) public_key_len: usize,
39    /// Fixed-width `r || s` signature length (`2 * scalar_len`).
40    pub(super) fixed_signature_len: usize,
41    /// Upper bound on a DER `SEQUENCE { INTEGER r, INTEGER s }` encoding.
42    pub(super) max_der_signature_len: usize,
43    /// aws-lc-rs algorithm for ASN.1 DER signatures.
44    pub(super) asn1: &'static EcdsaVerificationAlgorithm,
45    /// aws-lc-rs algorithm for fixed-width `r || s` signatures.
46    pub(super) fixed: &'static EcdsaVerificationAlgorithm,
47    /// Digest algorithm the curve is paired with.
48    pub(super) digest: &'static digest::Algorithm,
49}
50
51/// Signature encoding accepted by a verification call.
52#[derive(Clone, Copy)]
53pub(super) enum SigEncoding {
54    /// ASN.1 DER `Ecdsa-Sig-Value` (RFC 3279 §2.2.3), as used by X.509.
55    Der,
56    /// Fixed-width big-endian `r || s` (IEEE P1363), as used by JWS and COSE.
57    Fixed,
58}
59
60/// What the signature covers.
61pub(super) enum Signed<'a> {
62    /// The message itself; it is hashed with the curve's digest.
63    Message(&'a [u8]),
64    /// A digest computed by the caller with the curve's digest algorithm.
65    Digest(&'a [u8]),
66}

68fn parse_public_key(
69    curve: &CurveParams,
70    alg: &'static EcdsaVerificationAlgorithm,
71    public_key_sec1: &[u8],
72) -> Result<ParsedPublicKey> {
73    if public_key_sec1.len() != curve.public_key_len {
74        tracing::debug!(
75            curve = curve.name,
76            pk_len = public_key_sec1.len(),
77            "ECDSA verify rejected: public key length"
78        );
79        return Err(LatticeArcError::InvalidKeyLength {
80            expected: curve.public_key_len,
81            actual: public_key_sec1.len(),
82        });
83    }
84    // Only the uncompressed form is accepted: aws-lc-rs would otherwise also
85    // take compressed points and SubjectPublicKeyInfo DER, which gives one
86    // key several byte identities.
87    if public_key_sec1.first() != Some(&SEC1_UNCOMPRESSED_TAG) {
88        tracing::debug!(curve = curve.name, "ECDSA verify rejected: SEC1 tag is not uncompressed");
89        return Err(LatticeArcError::InvalidKey(format!(
90            "{} public key must be uncompressed SEC1",
91            curve.name
92        )));
93    }
94    ParsedPublicKey::new(alg, public_key_sec1).map_err(|e| {
95        tracing::debug!(curve = curve.name, error = %e, "ECDSA verify rejected: point validation");
96        LatticeArcError::InvalidKey(format!("invalid {} public key", curve.name))
97    })
98}

100fn check_signature_len(curve: &CurveParams, encoding: SigEncoding, len: usize) -> Result<()> {
101    let ok = match encoding {
102        SigEncoding::Fixed => len == curve.fixed_signature_len,
103        SigEncoding::Der => len > 0 && len <= curve.max_der_signature_len,
104    };
105    if ok {
106        return Ok(());
107    }
108    tracing::debug!(curve = curve.name, sig_len = len, "ECDSA verify rejected: signature length");
109    let expected = match encoding {
110        SigEncoding::Fixed => curve.fixed_signature_len,
111        SigEncoding::Der => curve.max_der_signature_len,
112    };
113    Err(LatticeArcError::InvalidSignatureLength { expected, got: len })
114}

116/// Verify one ECDSA signature for `curve`. See the module docs for the result
117/// convention.
118pub(super) fn verify(
119    curve: &CurveParams,
120    encoding: SigEncoding,
121    public_key_sec1: &[u8],
122    signed: &Signed<'_>,
123    signature: &[u8],
124) -> Result<bool> {
125    let alg = match encoding {
126        SigEncoding::Der => curve.asn1,
127        SigEncoding::Fixed => curve.fixed,
128    };
129    let key = parse_public_key(curve, alg, public_key_sec1)?;
130    check_signature_len(curve, encoding, signature.len())?;
131
132    let verified = match signed {
133        Signed::Message(message) => {
134            if let Err(e) = validate_signature_size(message.len()) {
135                tracing::debug!(
136                    curve = curve.name,
137                    error = %e,
138                    msg_len = message.len(),
139                    "ECDSA verify rejected: message exceeds resource limit"
140                );
141                return Err(LatticeArcError::MessageTooLong);
142            }
143            key.verify_sig(message, signature).is_ok()
144        }
145        Signed::Digest(bytes) => {
146            let digest = digest::Digest::import_less_safe(bytes, curve.digest).map_err(|_e| {
147                // Unreachable through the public API (callers pass a
148                // fixed-size array), kept as an error rather than a panic.
149                LatticeArcError::InvalidInput(format!("{} digest length", curve.name))
150            })?;
151            key.verify_digest_sig(&digest, signature).is_ok()
152        }
153    };
154    if !verified {
155        tracing::debug!(curve = curve.name, "ECDSA signature did not verify");
156    }
157    Ok(verified)
158}
