6//! # Hash-to-curve for NIST P-256 (RFC 9380)
7//!
8//! [`p256_xmd_sha256_sswu_ro`] implements the RFC 9380 suite
9//! `P256_XMD:SHA-256_SSWU_RO_` (§8.2): `expand_message_xmd` with SHA-256,
10//! `hash_to_field` producing two field elements, the simplified SWU map, and
11//! point addition (the random-oracle construction; P-256 has cofactor 1).
12//! The output is the 65-byte uncompressed SEC1 encoding of the point.
13//!
14//! aws-lc-rs exposes no hash-to-curve API, so this module uses the RustCrypto
15//! `p256` crate (`hash2curve` feature, pure Rust, `#![forbid(unsafe_code)]`).
16//! Correctness is pinned by the five RFC 9380 Appendix J.1.1 vectors in the
17//! tests below. The `p256` crate's arithmetic has not been independently
18//! audited (its README says so); it aims to be constant time but that is
19//! unverified, so treat hashing *secret* inputs to the curve accordingly.
20//!
21//! ## Inputs
22//!
23//! - `dst`, the domain separation tag, must be 1 to 255 bytes (RFC 9380
24//!   §3.1). Longer tags are refused rather than hashed down with the
25//!   `H2C-OVERSIZE-DST-` rule; callers with such a tag can apply that rule
26//!   themselves.
27//! - `msg` is bounded by the signature-message resource limit
28//!   (`max_signature_size_bytes`, 64 KiB by default).
29//!
30//! ## FIPS
31//!
32//! Hash-to-curve is not a FIPS-approved algorithm and the `p256` crate is
33//! outside the aws-lc-rs module boundary; see
34//! `docs/FIPS_SECURITY_POLICY.md`.
35
36use p256::NistP256;
37use p256::elliptic_curve::sec1::ToSec1Point;
38use p256::hash2curve::{ExpandMsgXmd, hash_from_bytes};
39use sha2::Sha256;
40
41use crate::prelude::error::{LatticeArcError, Result};
42use crate::primitives::resource_limits::validate_signature_size;
43
44/// Length of the uncompressed SEC1 point returned by
45/// [`p256_xmd_sha256_sswu_ro`].
46pub const P256_H2C_OUTPUT_LEN: usize = 65;
47
48/// Largest domain separation tag accepted (RFC 9380 §3.1).
49pub const H2C_MAX_DST_LEN: usize = 255;
50
51/// Hash `msg` to a point on NIST P-256 with the RFC 9380 suite
52/// `P256_XMD:SHA-256_SSWU_RO_` and return it as 65-byte uncompressed SEC1
53/// (`0x04 || X || Y`).
54///
55/// # Errors
56///
57/// Returns `InvalidInput` if `dst` is empty or longer than
58/// [`H2C_MAX_DST_LEN`] bytes, and `MessageTooLong` if `msg` exceeds the
59/// signature-message resource limit. Returns `InvalidData` in the
60/// probability-2^-256 case that the result is the point at infinity, which
61/// has no uncompressed encoding.
62#[must_use = "the hashed point must be used"]
63pub fn p256_xmd_sha256_sswu_ro(msg: &[u8], dst: &[u8]) -> Result<[u8; P256_H2C_OUTPUT_LEN]> {
64    if dst.is_empty() || dst.len() > H2C_MAX_DST_LEN {
65        tracing::debug!(dst_len = dst.len(), "hash-to-curve rejected: DST length");
66        return Err(LatticeArcError::InvalidInput(
67            "hash-to-curve DST must be 1 to 255 bytes".to_string(),
68        ));
69    }
70    if let Err(e) = validate_signature_size(msg.len()) {
71        tracing::debug!(error = %e, msg_len = msg.len(), "hash-to-curve rejected: message exceeds resource limit");
72        return Err(LatticeArcError::MessageTooLong);
73    }
74
75    let point = hash_from_bytes::<NistP256, ExpandMsgXmd<Sha256>>(&[msg], &[dst]).map_err(|e| {
76        tracing::debug!(error = %e, "hash-to-curve expand_message_xmd failed");
77        LatticeArcError::InvalidInput("hash-to-curve input rejected".to_string())
78    })?;
79
80    let encoded = point.to_sec1_point(false);
81    <[u8; P256_H2C_OUTPUT_LEN]>::try_from(encoded.as_bytes()).map_err(|_e| {
82        tracing::debug!("hash-to-curve produced the identity point");
83        LatticeArcError::InvalidData("hash-to-curve produced the identity point".to_string())
84    })
85}

94    /// DST of the RFC 9380 Appendix J.1.1 vectors.
95    const DST: &[u8] = b"QUUX-V01-CS02-with-P256_XMD:SHA-256_SSWU_RO_";
96
97    enum Msg {
98        Literal(&'static [u8]),
99        /// `prefix` followed by `count` copies of `byte` (the q128 / a512
100        /// inputs).
101        Repeat(&'static [u8], u8, usize),
102    }
103
104    impl Msg {
105        fn bytes(&self) -> Vec<u8> {
106            match self {
107                Msg::Literal(b) => b.to_vec(),
108                Msg::Repeat(prefix, byte, count) => {
109                    let mut v = prefix.to_vec();
110                    v.resize(prefix.len() + count, *byte);
111                    v
112                }
113            }
114        }
115    }
116
117    struct Vector {
118        msg: Msg,
119        x: &'static str,
120        y: &'static str,
121    }
122
123    /// RFC 9380 Appendix J.1.1 (`P256_XMD:SHA-256_SSWU_RO_`), the `P.x` and
124    /// `P.y` values for each of the five messages, copied from
125    /// <https://www.rfc-editor.org/rfc/rfc9380.txt>.
126    const RFC9380_J_1_1: &[Vector] = &[

154    #[test]
155    fn rfc9380_appendix_j_1_1_vectors() {
156        let lens: Vec<usize> = RFC9380_J_1_1.iter().map(|v| v.msg.bytes().len()).collect();
157        assert_eq!(lens, vec![0, 3, 16, 133, 517]);
158        for v in RFC9380_J_1_1 {
159            let msg = v.msg.bytes();
160            let point = p256_xmd_sha256_sswu_ro(&msg, DST).unwrap();
161            assert_eq!(point[0], 0x04);
162            assert_eq!(hex::encode(&point[1..33]), v.x, "P.x for msg of {} bytes", msg.len());
163            assert_eq!(hex::encode(&point[33..]), v.y, "P.y for msg of {} bytes", msg.len());
164        }
165    }

167    #[test]
168    fn output_is_a_valid_p256_public_point() {
169        let point = p256_xmd_sha256_sswu_ro(b"abc", DST).unwrap();
170        assert!(p256::PublicKey::from_sec1_bytes(&point).is_ok());
171    }

173    #[test]
174    fn dst_and_message_separate_outputs() {
175        let a = p256_xmd_sha256_sswu_ro(b"abc", DST).unwrap();
176        let b = p256_xmd_sha256_sswu_ro(b"abd", DST).unwrap();
177        let c = p256_xmd_sha256_sswu_ro(b"abc", b"another-dst").unwrap();
178        assert_ne!(a, b);
179        assert_ne!(a, c);
180    }

182    #[test]
183    fn dst_length_bounds() {
184        assert!(matches!(
185            p256_xmd_sha256_sswu_ro(b"m", b""),
186            Err(LatticeArcError::InvalidInput(_))
187        ));
188        assert!(p256_xmd_sha256_sswu_ro(b"m", &[0x41; H2C_MAX_DST_LEN]).is_ok());
189        assert!(matches!(
190            p256_xmd_sha256_sswu_ro(b"m", &[0x41; H2C_MAX_DST_LEN + 1]),
191            Err(LatticeArcError::InvalidInput(_))
192        ));
193    }

195    #[test]
196    fn oversized_message_is_rejected() {
197        let limit = crate::primitives::resource_limits::get_global_resource_limits()
198            .get_limits()
199            .unwrap()
200            .max_signature_size_bytes;
201        let big = vec![0u8; limit + 1];
202        assert!(matches!(p256_xmd_sha256_sswu_ro(&big, DST), Err(LatticeArcError::MessageTooLong)));
203        assert!(p256_xmd_sha256_sswu_ro(&big[..limit], DST).is_ok());
204    }
