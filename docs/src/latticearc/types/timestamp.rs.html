1//! Wall-clock timestamps: `std::time::SystemTime` and its RFC 3339 text form.
2//!
3//! The public API carries every timestamp as [`SystemTime`], so no
4//! third-party date/time type is part of the crate's API surface. This
5//! module holds the conversions the crate needs internally: Unix-epoch
6//! arithmetic, signed differences, and the RFC 3339 encoding used by serde,
7//! by the audit-chain hash transcript and by audit filenames.
8//!
9//! The encoder reproduces byte-for-byte the text the crate emitted when it
10//! used `chrono` (fractional seconds omitted when zero, otherwise 3, 6 or 9
11//! digits; UTC written as `Z` in serde and as `+00:00` in the audit hash
12//! transcript; years outside 0000–9999 written with a sign and at least
13//! four digits, e.g. `+10000`). That is a compatibility requirement, not a
14//! style choice: audit integrity hashes cover the encoded timestamp, so a
15//! different encoding would make every audit chain written by an earlier
16//! version fail verification.
17//!
18//! The parser accepts everything the encoder (and chrono before it) wrote,
19//! plus the RFC 3339 variants a conforming producer may emit: `t` or a space
20//! as the date/time separator, `z`, any `±HH:MM` offset, more than nine
21//! fractional digits (truncated) and second `60`. It rejects the non-RFC
22//! forms chrono's parser also tolerated (a `UTC` suffix, `+HHMM` offsets
23//! without a colon, single-digit fields, embedded whitespace): nothing in
24//! this crate ever wrote them.
25

31const NANOS_PER_SEC: i128 = 1_000_000_000;
32const NANOS_PER_MILLI: i128 = 1_000_000;
33const NANOS_PER_MICRO: i128 = 1_000;
34const NANOS_PER_DAY: i128 = 86_400 * NANOS_PER_SEC;
35/// Julian day number of 1970-01-01.
36const UNIX_EPOCH_JULIAN_DAY: i128 = 2_440_588;
37
38/// How the UTC offset is written after the time of day.
39#[derive(Debug, Clone, Copy, PartialEq, Eq)]
40pub(crate) enum UtcSuffix {
41    /// `Z` — the serde encoding of every timestamp field.
42    Z,
43    /// `+00:00` — the audit-chain hash transcript encoding.
44    PlusZero,
45}

47/// Signed nanoseconds since the Unix epoch.
48#[must_use]
49pub(crate) fn unix_nanos(t: SystemTime) -> i128 {
50    match t.duration_since(UNIX_EPOCH) {
51        Ok(after) => duration_nanos(after),
52        Err(before) => duration_nanos(before.duration()).saturating_neg(),
53    }
54}
55
56/// `Duration::as_nanos` is at most `u64::MAX * 10^9 + 999_999_999`, far
57/// below `i128::MAX`, so the fallback is never taken.
58fn duration_nanos(d: Duration) -> i128 {
59    i128::try_from(d.as_nanos()).unwrap_or(i128::MAX)
60}
61
62/// The `SystemTime` that lies `nanos` nanoseconds from the Unix epoch, or
63/// `None` when the platform's `SystemTime` cannot represent it.
64#[must_use]
65pub(crate) fn from_unix_nanos(nanos: i128) -> Option<SystemTime> {
66    let secs = nanos.checked_div_euclid(NANOS_PER_SEC)?;
67    let subsec = u32::try_from(nanos.checked_rem_euclid(NANOS_PER_SEC)?).ok()?;
68    if let Ok(secs) = u64::try_from(secs) {
69        UNIX_EPOCH.checked_add(Duration::new(secs, subsec))
70    } else {
71        let back = Duration::from_secs(u64::try_from(secs.checked_neg()?).ok()?);
72        UNIX_EPOCH.checked_sub(back)?.checked_add(Duration::from_nanos(u64::from(subsec)))
73    }
74}

76/// Floor of `t` measured in `unit` nanoseconds since the epoch, saturated to
77/// the `i64` range.
78fn unix_floor(t: SystemTime, unit: i128) -> i64 {
79    let floored = unix_nanos(t).checked_div_euclid(unit).unwrap_or(0);
80    i64::try_from(floored).unwrap_or(if floored < 0 { i64::MIN } else { i64::MAX })
81}

83/// Whole seconds since the Unix epoch, rounded toward negative infinity.
84#[must_use]
85pub(crate) fn unix_seconds(t: SystemTime) -> i64 {
86    unix_floor(t, NANOS_PER_SEC)
87}

101/// `later - earlier` in whole `unit`s, truncated toward zero and saturated
102/// to the `i64` range. Negative when `later` precedes `earlier`.
103fn signed_between(earlier: SystemTime, later: SystemTime, unit: i128) -> i64 {
104    let diff = unix_nanos(later).saturating_sub(unix_nanos(earlier));
105    let units = diff.checked_div(unit).unwrap_or(0);
106    i64::try_from(units).unwrap_or(if units < 0 { i64::MIN } else { i64::MAX })
107}

142/// RFC 3339 text for `t` in UTC, or `None` outside years ±999,999.
143#[must_use]
144pub(crate) fn to_rfc3339(t: SystemTime, suffix: UtcSuffix) -> Option<String> {
145    let dt = utc_fields(t)?;
146    let mut out = String::with_capacity(35);
147    let year = dt.year();
148    if (0..=9999).contains(&year) {
149        write!(out, "{year:04}").ok()?;
150    } else {
151        // chrono's extended form: explicit sign, at least four digits.
152        write!(out, "{year:+05}").ok()?;
153    }
154    write!(
155        out,
156        "-{:02}-{:02}T{:02}:{:02}:{:02}",
157        u8::from(dt.month()),
158        dt.day(),
159        dt.hour(),
160        dt.minute(),
161        dt.second()
162    )
163    .ok()?;
164    // Fraction omitted when zero, else the shortest of 3, 6 or 9 digits
165    // that is exact (chrono's `SecondsFormat::AutoSi`).
166    let nanos = dt.nanosecond();
167    if nanos != 0 {
168        if nanos % 1_000_000 == 0 {
169            write!(out, ".{:03}", nanos / 1_000_000).ok()?;
170        } else if nanos % 1_000 == 0 {
171            write!(out, ".{:06}", nanos / 1_000).ok()?;
172        } else {
173            write!(out, ".{nanos:09}").ok()?;
174        }
175    }
176    out.push_str(match suffix {
177        UtcSuffix::Z => "Z",
178        UtcSuffix::PlusZero => "+00:00",
179    });
180    Some(out)
181}

183/// Human-readable RFC 3339 rendering for log and error messages. Falls
184/// back to the `Debug` form for instants outside years ±999,999.
185#[must_use]
186pub(crate) fn display(t: SystemTime) -> String {
187    to_rfc3339(t, UtcSuffix::Z).unwrap_or_else(|| format!("{t:?}"))
188}

190/// Byte cursor for [`parse_rfc3339`].
191struct Cursor<'a> {
192    bytes: &'a [u8],
193    pos: usize,
194}
195
196impl Cursor<'_> {
197    fn peek(&self) -> Option<u8> {
198        self.bytes.get(self.pos).copied()
199    }
200
201    fn bump(&mut self) -> Option<u8> {
202        let b = self.peek()?;
203        self.pos = self.pos.checked_add(1)?;
204        Some(b)
205    }
206
207    fn expect(&mut self, want: u8) -> Option<()> {
208        (self.bump()? == want).then_some(())
209    }
210
211    fn digit(&mut self) -> Option<u32> {
212        let b = self.peek().filter(u8::is_ascii_digit)?;
213        self.pos = self.pos.checked_add(1)?;
214        Some(u32::from(b.checked_sub(b'0')?))
215    }
216
217    /// Exactly `n` ASCII digits as a number.
218    fn fixed(&mut self, n: usize) -> Option<u32> {
219        (0..n).try_fold(0u32, |acc, _| acc.checked_mul(10)?.checked_add(self.digit()?))
220    }
221
222    fn at_end(&self) -> bool {
223        self.pos == self.bytes.len()
224    }
225}

227/// Parse an RFC 3339 timestamp (see the module docs for the accepted
228/// forms), or `None` if it is malformed, names an impossible date or time,
229/// or is not representable as a `SystemTime`.
230#[must_use]
231pub(crate) fn parse_rfc3339(text: &str) -> Option<SystemTime> {
232    let mut c = Cursor { bytes: text.as_bytes(), pos: 0 };
233
234    // Year: four digits, or chrono's extended form — a sign and 4–6 digits.
235    let year = match c.peek()? {
236        sign @ (b'+' | b'-') => {
237            c.bump()?;
238            let mut value = c.fixed(4)?;
239            for _ in 0..2 {
240                match c.digit() {
241                    Some(d) => value = value.checked_mul(10)?.checked_add(d)?,
242                    None => break,
243                }
244            }
245            let value = i32::try_from(value).ok()?;
246            if sign == b'-' { value.checked_neg()? } else { value }
247        }
248        _ => i32::try_from(c.fixed(4)?).ok()?,
249    };
250    c.expect(b'-')?;
251    let month = c.fixed(2)?;
252    c.expect(b'-')?;
253    let day = c.fixed(2)?;
254    if !matches!(c.bump()?, b'T' | b't' | b' ') {
255        return None;
256    }
257    let hour = c.fixed(2)?;
258    c.expect(b':')?;
259    let minute = c.fixed(2)?;
260    c.expect(b':')?;
261    let second = c.fixed(2)?;
262    // RFC 3339 allows second 60 (leap second); like chrono, it denotes the
263    // instant one second after :59, i.e. the start of the next minute.
264    if hour > 23 || minute > 59 || second > 60 {
265        return None;
266    }
267
268    // Fraction: one or more digits; the first nine are significant.
269    let mut nanos: u32 = 0;
270    if c.peek() == Some(b'.') {
271        c.bump()?;
272        let mut scale: u32 = 100_000_000;
273        nanos = c.digit()?.checked_mul(scale)?;
274        while let Some(d) = c.digit() {
275            scale /= 10;
276            nanos = nanos.checked_add(d.checked_mul(scale)?)?;
277        }
278    }
279
280    // Offset: `Z`/`z` or `±HH:MM`.
281    let offset_secs: i128 = match c.bump()? {
282        b'Z' | b'z' => 0,
283        sign @ (b'+' | b'-') => {
284            let oh = c.fixed(2)?;
285            c.expect(b':')?;
286            let om = c.fixed(2)?;
287            if oh > 23 || om > 59 {
288                return None;
289            }
290            let secs = i128::from(oh.checked_mul(3_600)?.checked_add(om.checked_mul(60)?)?);
291            if sign == b'-' { secs.checked_neg()? } else { secs }
292        }
293        _ => return None,
294    };
295    if !c.at_end() {
296        return None;
297    }
298
299    let date = Date::from_calendar_date(
300        year,
301        Month::try_from(u8::try_from(month).ok()?).ok()?,
302        u8::try_from(day).ok()?,
303    )
304    .ok()?;
305    let days = i128::from(date.to_julian_day()).checked_sub(UNIX_EPOCH_JULIAN_DAY)?;
306    let secs = days
307        .checked_mul(86_400)?
308        .checked_add(i128::from(hour.checked_mul(3_600)?))?
309        .checked_add(i128::from(minute.checked_mul(60)?))?
310        .checked_add(i128::from(second))?
311        .checked_sub(offset_secs)?;
312    from_unix_nanos(secs.checked_mul(NANOS_PER_SEC)?.checked_add(i128::from(nanos))?)
313}

315/// Serde adapter: a `SystemTime` field encoded as an RFC 3339 string.
316pub(crate) mod serde_rfc3339 {
317    use std::time::SystemTime;
318
319    use serde::{Deserialize, Deserializer, Serializer, de, ser};
320
321    /// Serialize as RFC 3339 text with a `Z` suffix.
322    ///
323    /// # Errors
324    /// Fails when `t` lies outside years ±999,999.
325    pub(crate) fn serialize<S: Serializer>(t: &SystemTime, s: S) -> Result<S::Ok, S::Error> {
326        let text = super::to_rfc3339(*t, super::UtcSuffix::Z).ok_or_else(|| {
327            <S::Error as ser::Error>::custom("timestamp outside the encodable year range")
328        })?;
329        s.serialize_str(&text)
330    }
331
332    /// Deserialize from RFC 3339 text (see the module docs for accepted forms).
333    ///
334    /// # Errors
335    /// Fails when the input is not a valid RFC 3339 timestamp.
336    pub(crate) fn deserialize<'de, D: Deserializer<'de>>(d: D) -> Result<SystemTime, D::Error> {
337        let text = String::deserialize(d)?;
338        super::parse_rfc3339(&text)
339            .ok_or_else(|| <D::Error as de::Error>::custom("invalid RFC 3339 timestamp"))
340    }
341}

343/// Serde adapter: an `Option<SystemTime>` field encoded as an RFC 3339
344/// string or `null`. Pair it with `#[serde(default)]` so a missing field
345/// still deserializes as `None`.
346pub(crate) mod serde_rfc3339_option {
347    use std::time::SystemTime;
348
349    use serde::{Deserialize, Deserializer, Serializer, de, ser};
350
351    /// Serialize `Some` as RFC 3339 text with a `Z` suffix and `None` as `null`.
352    ///
353    /// # Errors
354    /// Fails when the timestamp lies outside years ±999,999.
355    pub(crate) fn serialize<S: Serializer>(
356        t: &Option<SystemTime>,
357        s: S,
358    ) -> Result<S::Ok, S::Error> {
359        match t {
360            Some(t) => {
361                let text = super::to_rfc3339(*t, super::UtcSuffix::Z).ok_or_else(|| {
362                    <S::Error as ser::Error>::custom("timestamp outside the encodable year range")
363                })?;
364                s.serialize_some(&text)
365            }
366            None => s.serialize_none(),
367        }
368    }
369
370    /// Deserialize `null` as `None` and RFC 3339 text as `Some`.
371    ///
372    /// # Errors
373    /// Fails when the input is neither `null` nor a valid RFC 3339 timestamp.
374    pub(crate) fn deserialize<'de, D: Deserializer<'de>>(
375        d: D,
376    ) -> Result<Option<SystemTime>, D::Error> {
377        Option::<String>::deserialize(d)?
378            .map(|text| {
379                super::parse_rfc3339(&text)
380                    .ok_or_else(|| <D::Error as de::Error>::custom("invalid RFC 3339 timestamp"))
381            })
382            .transpose()
383    }
384}

540    fn test_extended_years_match_chrono_golden_succeeds() {
541        for &(secs, nanos, plus_zero, z) in CHRONO_EXTENDED_YEARS {
542            let t = at(secs, nanos);
543            assert_eq!(to_rfc3339(t, UtcSuffix::PlusZero).as_deref(), Some(plus_zero));
544            assert_eq!(to_rfc3339(t, UtcSuffix::Z).as_deref(), Some(z));
545            assert_eq!(parse_rfc3339(plus_zero), Some(t));
546            assert_eq!(parse_rfc3339(z), Some(t));
547        }
548        // Four-digit boundary years keep the unsigned form.
549        assert_eq!(
550            to_rfc3339(at(253_402_300_799, 999_999_999), UtcSuffix::Z).as_deref(),
551            Some("9999-12-31T23:59:59.999999999Z")

554            to_rfc3339(at(-62_167_219_200, 0), UtcSuffix::Z).as_deref(),

556        );

592    #[test]
593    fn test_serde_adapters_roundtrip_and_reject_garbage_succeeds() {
594        #[derive(serde::Serialize, serde::Deserialize, PartialEq, Debug)]
595        struct Probe {
596            #[serde(with = "serde_rfc3339")]
597            at: SystemTime,
598            #[serde(default, with = "serde_rfc3339_option")]
599            maybe: Option<SystemTime>,
600        }
601        let probe = Probe { at: at(1_700_000_000, 123_456_000), maybe: None };
602        let json = serde_json::to_string(&probe).unwrap();
603        assert_eq!(json, r#"{"at":"2023-11-14T22:13:20.123456Z","maybe":null}"#);
604        assert_eq!(serde_json::from_str::<Probe>(&json).unwrap(), probe);
605        // A missing optional field still deserializes as `None`.
606        let missing: Probe =
607            serde_json::from_str(r#"{"at":"2023-11-14T22:13:20.123456Z"}"#).unwrap();
608        assert_eq!(missing, probe);
609        let some = Probe { at: at(0, 0), maybe: Some(at(1, 1_000)) };
610        let json = serde_json::to_string(&some).unwrap();
611        assert_eq!(json, r#"{"at":"1970-01-01T00:00:00Z","maybe":"1970-01-01T00:00:01.000001Z"}"#);
612        assert_eq!(serde_json::from_str::<Probe>(&json).unwrap(), some);
613        assert!(serde_json::from_str::<Probe>(r#"{"at":"yesterday"}"#).is_err());
614        assert!(serde_json::from_str::<Probe>(r#"{"at":1700000000}"#).is_err());
615        assert!(
616            serde_json::from_str::<Probe>(r#"{"at":"1970-01-01T00:00:00Z","maybe":"x"}"#).is_err()
617        );
618        let far = Probe { at: at(40_000_000_000_000, 0), maybe: None };
619        assert!(serde_json::to_string(&far).is_err());
620    }
