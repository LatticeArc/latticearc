1//! [`Timestamp`]: the crate's platform-independent UTC instant.
2//!
3//! Every timestamp in the public API is a [`Timestamp`] — nanoseconds since
4//! the Unix epoch, held identically on every platform. `std::time::SystemTime`
5//! is not used for stored values because its resolution and range are
6//! platform-specific (100 ns and years 1601–60056 on Windows): a nanosecond
7//! timestamp written on Linux would lose digits when loaded on Windows. No
8//! third-party date/time type is part of the API either. Convert to and from
9//! `SystemTime` with `TryFrom`.
10//!
11//! Text form: RFC 3339 in UTC. The encoder reproduces byte-for-byte what the
12//! crate wrote when it used `chrono` (fractional seconds omitted when zero,
13//! otherwise 3, 6 or 9 digits; `Z` in serde and [`Display`], `+00:00` in the
14//! version-1 audit-chain hash transcript; years outside 0000–9999 with a sign
15//! and at least four digits, e.g. `+10000`), so files written by earlier
16//! versions load and re-encode identically.
17//!
18//! The parser accepts everything the encoder (and chrono before it) wrote,
19//! plus the RFC 3339 variants a conforming producer may emit: `t` or a space
20//! as the date/time separator, `z`, any `±HH:MM` offset, more than nine
21//! fractional digits (truncated) and second `60`. It rejects the non-RFC
22//! forms chrono's parser also tolerated (a `UTC` suffix, `+HHMM` offsets
23//! without a colon, single-digit fields, embedded whitespace): nothing in
24//! this crate ever wrote them.
25//!
26//! [`Display`]: std::fmt::Display
27

36const NANOS_PER_SEC: i128 = 1_000_000_000;
37const NANOS_PER_MILLI: i128 = 1_000_000;
38const NANOS_PER_MICRO: i128 = 1_000;
39const NANOS_PER_DAY: i128 = 86_400 * NANOS_PER_SEC;
40/// Julian day number of 1970-01-01.
41const UNIX_EPOCH_JULIAN_DAY: i128 = 2_440_588;
42/// `-999999-01-01T00:00:00Z` in Unix seconds.
43const MIN_UNIX_SECS: i128 = -31_619_087_596_800;
44/// `+999999-12-31T23:59:59Z` in Unix seconds.
45const MAX_UNIX_SECS: i128 = 31_494_784_780_799;
46const MIN_NANOS: i128 = MIN_UNIX_SECS * NANOS_PER_SEC;
47const MAX_NANOS: i128 = MAX_UNIX_SECS * NANOS_PER_SEC + (NANOS_PER_SEC - 1);

49/// A UTC instant with nanosecond precision, independent of the platform.
50///
51/// Range: years −999,999 through +999,999 (a superset of every instant
52/// `chrono` could represent), so every `Timestamp` has an RFC 3339 encoding.
53/// Leap seconds are not represented: like Unix time, a parsed `:60` second
54/// denotes the first instant of the following minute.
55///
56/// Serializes as an RFC 3339 string (`"2026-11-18T09:30:00.123456Z"`), and
57/// [`Display`](fmt::Display) / [`FromStr`] use the same form.
58#[derive(Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
59pub struct Timestamp {
60    /// Nanoseconds since 1970-01-01T00:00:00Z; always within
61    /// `MIN_NANOS..=MAX_NANOS`.
62    nanos: i128,
63}
64
65/// Why a [`Timestamp`] could not be built, parsed or converted.
66#[derive(Debug, Clone, Copy, PartialEq, Eq, thiserror::Error)]
67#[non_exhaustive]
68pub enum TimestampError {
69    /// The instant lies outside the representable range — years ±999,999
70    /// for a `Timestamp`, or the platform's range for a `SystemTime`.
71    #[error("timestamp out of range")]
72    OutOfRange,
73    /// The text is not a valid RFC 3339 timestamp.
74    #[error("invalid RFC 3339 timestamp")]
75    InvalidFormat,
76}
77
78impl Timestamp {
79    /// 1970-01-01T00:00:00Z.
80    pub const UNIX_EPOCH: Self = Self { nanos: 0 };
81
82    /// The earliest representable instant, `-999999-01-01T00:00:00Z`.
83    pub const MIN: Self = Self { nanos: MIN_NANOS };
84
85    /// The latest representable instant, `+999999-12-31T23:59:59.999999999Z`
86    /// — e.g. a "never expires" `not_after`.
87    pub const MAX: Self = Self { nanos: MAX_NANOS };
88
89    /// The current wall-clock time.
90    ///
91    /// A system clock outside years ±999,999 (which no real clock reports)
92    /// saturates to the nearest bound rather than failing.
93    #[must_use]
94    pub fn now() -> Self {
95        Self { nanos: system_time_nanos(SystemTime::now()).clamp(MIN_NANOS, MAX_NANOS) }
96    }
97
98    /// The instant `secs` seconds plus `nanos` nanoseconds after the Unix
99    /// epoch (`secs` may be negative).
100    ///
101    /// # Errors
102    /// [`TimestampError::OutOfRange`] when `nanos >= 1_000_000_000` or the
103    /// instant lies outside years ±999,999.
104    pub fn from_unix(secs: i64, nanos: u32) -> Result<Self, TimestampError> {
105        if nanos >= 1_000_000_000 {
106            return Err(TimestampError::OutOfRange);
107        }
108        i128::from(secs)
109            .checked_mul(NANOS_PER_SEC)
110            .and_then(|n| n.checked_add(i128::from(nanos)))
111            .and_then(Self::from_nanos)
112            .ok_or(TimestampError::OutOfRange)
113    }
114
115    /// The instant `nanos` nanoseconds after the Unix epoch (negative
116    /// before it). Lossless inverse of [`unix_nanos`](Self::unix_nanos).
117    ///
118    /// # Errors
119    /// [`TimestampError::OutOfRange`] outside [`MIN`](Self::MIN)..=[`MAX`](Self::MAX).
120    pub fn from_unix_nanos(nanos: i128) -> Result<Self, TimestampError> {
121        Self::from_nanos(nanos).ok_or(TimestampError::OutOfRange)
122    }
123
124    /// Nanoseconds since the Unix epoch (negative before it), losslessly.
125    #[must_use]
126    pub fn unix_nanos(self) -> i128 {
127        self.nanos
128    }
129
130    /// Whole seconds since the Unix epoch, rounded toward negative infinity.
131    #[must_use]
132    pub fn unix_seconds(self) -> i64 {
133        self.unix_floor(NANOS_PER_SEC)
134    }
135
136    /// Nanoseconds past [`unix_seconds`](Self::unix_seconds), in `0..1_000_000_000`.
137    #[must_use]
138    pub fn subsec_nanos(self) -> u32 {
139        u32::try_from(self.nanos.rem_euclid(NANOS_PER_SEC)).unwrap_or(0)
140    }
141
142    /// `self + duration`, or `None` if the result leaves the range.
143    #[must_use]
144    pub fn checked_add(self, duration: Duration) -> Option<Self> {
145        Self::from_nanos(self.nanos.checked_add(duration_nanos(duration))?)
146    }
147
148    /// `self - duration`, or `None` if the result leaves the range.
149    #[must_use]
150    pub fn checked_sub(self, duration: Duration) -> Option<Self> {
151        Self::from_nanos(self.nanos.checked_sub(duration_nanos(duration))?)
152    }
153
154    /// The time elapsed from `earlier` to `self`, or `None` if `earlier`
155    /// is later than `self`.
156    #[must_use]
157    pub fn duration_since(self, earlier: Self) -> Option<Duration> {
158        let diff = self.nanos.checked_sub(earlier.nanos)?;
159        let secs = u64::try_from(diff.div_euclid(NANOS_PER_SEC)).ok()?;
160        let subsec = u32::try_from(diff.rem_euclid(NANOS_PER_SEC)).ok()?;
161        Some(Duration::new(secs, subsec))
162    }
163
164    fn from_nanos(nanos: i128) -> Option<Self> {
165        (MIN_NANOS..=MAX_NANOS).contains(&nanos).then_some(Self { nanos })
166    }
167
168    /// Floor of `self` in `unit` nanoseconds since the epoch. The range
169    /// bound keeps every unit used here (s, ms, µs) inside `i64`.
170    fn unix_floor(self, unit: i128) -> i64 {
171        let floored = self.nanos.div_euclid(unit);
172        i64::try_from(floored).unwrap_or(if floored < 0 { i64::MIN } else { i64::MAX })
173    }
174
175    /// Whole milliseconds since the Unix epoch, rounded toward negative infinity.
176    #[must_use]
177    pub(crate) fn unix_millis(self) -> i64 {
178        self.unix_floor(NANOS_PER_MILLI)
179    }
180
181    /// Whole microseconds since the Unix epoch, rounded toward negative infinity.
182    #[must_use]
183    pub(crate) fn unix_micros(self) -> i64 {
184        self.unix_floor(NANOS_PER_MICRO)
185    }
186
187    /// UTC calendar fields. Always `Some` for an in-range `Timestamp` (the
188    /// `time` crate's `large-dates` range is the same ±999,999 years).
189    #[must_use]
190    pub(crate) fn utc_fields(self) -> Option<OffsetDateTime> {
191        OffsetDateTime::from_unix_timestamp_nanos(self.nanos).ok()
192    }
193
194    /// RFC 3339 text in UTC with the given offset designator.
195    #[must_use]
196    pub(crate) fn to_rfc3339(self, suffix: UtcSuffix) -> Option<String> {
197        let dt = self.utc_fields()?;
198        let mut out = String::with_capacity(35);
199        let year = dt.year();
200        if (0..=9999).contains(&year) {
201            write!(out, "{year:04}").ok()?;
202        } else {
203            // chrono's extended form: explicit sign, at least four digits.
204            write!(out, "{year:+05}").ok()?;
205        }
206        write!(
207            out,
208            "-{:02}-{:02}T{:02}:{:02}:{:02}",
209            u8::from(dt.month()),
210            dt.day(),
211            dt.hour(),
212            dt.minute(),
213            dt.second()
214        )
215        .ok()?;
216        // Fraction omitted when zero, else the shortest of 3, 6 or 9 digits
217        // that is exact (chrono's `SecondsFormat::AutoSi`).
218        let nanos = dt.nanosecond();
219        if nanos != 0 {
220            if nanos % 1_000_000 == 0 {
221                write!(out, ".{:03}", nanos / 1_000_000).ok()?;
222            } else if nanos % 1_000 == 0 {
223                write!(out, ".{:06}", nanos / 1_000).ok()?;
224            } else {
225                write!(out, ".{nanos:09}").ok()?;
226            }
227        }
228        out.push_str(match suffix {
229            UtcSuffix::Z => "Z",
230            UtcSuffix::PlusZero => "+00:00",
231        });
232        Some(out)
233    }
234}

236/// Signed nanoseconds of `t` relative to the Unix epoch.
237fn system_time_nanos(t: SystemTime) -> i128 {
238    match t.duration_since(UNIX_EPOCH) {
239        Ok(after) => duration_nanos(after),
240        Err(before) => duration_nanos(before.duration()).saturating_neg(),
241    }
242}
243
244/// `Duration::as_nanos` is at most `u64::MAX * 10^9 + 999_999_999`, far
245/// below `i128::MAX`, so the fallback is never taken.
246fn duration_nanos(d: Duration) -> i128 {
247    i128::try_from(d.as_nanos()).unwrap_or(i128::MAX)
248}

250impl TryFrom<SystemTime> for Timestamp {
251    type Error = TimestampError;
252
253    /// # Errors
254    /// [`TimestampError::OutOfRange`] outside years ±999,999.
255    fn try_from(t: SystemTime) -> Result<Self, Self::Error> {
256        Self::from_nanos(system_time_nanos(t)).ok_or(TimestampError::OutOfRange)
257    }
258}
259
260impl TryFrom<Timestamp> for SystemTime {
261    type Error = TimestampError;
262
263    /// # Errors
264    /// [`TimestampError::OutOfRange`] when the platform's `SystemTime`
265    /// cannot represent the instant. On Windows (100 ns resolution) the
266    /// sub-100 ns digits are dropped.
267    fn try_from(t: Timestamp) -> Result<Self, Self::Error> {
268        let secs = t.nanos.div_euclid(NANOS_PER_SEC);
269        let subsec = u32::try_from(t.nanos.rem_euclid(NANOS_PER_SEC))
270            .map_err(|_e| TimestampError::OutOfRange)?;
271        let converted = if let Ok(secs) = u64::try_from(secs) {
272            UNIX_EPOCH.checked_add(Duration::new(secs, subsec))
273        } else {
274            u64::try_from(secs.unsigned_abs())
275                .ok()
276                .and_then(|back| UNIX_EPOCH.checked_sub(Duration::from_secs(back)))
277                .and_then(|base| base.checked_add(Duration::from_nanos(u64::from(subsec))))
278        };
279        converted.ok_or(TimestampError::OutOfRange)
280    }
281}

295impl FromStr for Timestamp {
296    type Err = TimestampError;
297
298    /// Parse RFC 3339 text (see the module docs for the accepted forms).
299    ///
300    /// # Errors
301    /// [`TimestampError::InvalidFormat`] for malformed text or an impossible
302    /// date/time; [`TimestampError::OutOfRange`] beyond years ±999,999.
303    fn from_str(text: &str) -> Result<Self, Self::Err> {
304        let nanos = parse_rfc3339_nanos(text).ok_or(TimestampError::InvalidFormat)?;
305        Self::from_nanos(nanos).ok_or(TimestampError::OutOfRange)
306    }
307}

309impl Serialize for Timestamp {
310    fn serialize<S: Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
311        let text = self
312            .to_rfc3339(UtcSuffix::Z)
313            .ok_or_else(|| serde::ser::Error::custom(TimestampError::OutOfRange))?;
314        serializer.serialize_str(&text)
315    }

318impl<'de> Deserialize<'de> for Timestamp {
319    fn deserialize<D: Deserializer<'de>>(deserializer: D) -> Result<Self, D::Error> {
320        struct Rfc3339Visitor;
321
322        impl Visitor<'_> for Rfc3339Visitor {
323            type Value = Timestamp;
324
325            fn expecting(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
326                f.write_str("an RFC 3339 timestamp string")
327            }
328
329            fn visit_str<E: de::Error>(self, v: &str) -> Result<Timestamp, E> {
330                v.parse().map_err(E::custom)
331            }
332
333            // Formats that carry strings as raw bytes (e.g. some binary
334            // encodings) deliver UTF-8 text here.
335            fn visit_bytes<E: de::Error>(self, v: &[u8]) -> Result<Timestamp, E> {
336                std::str::from_utf8(v)
337                    .map_err(|_e| E::custom(TimestampError::InvalidFormat))?
338                    .parse()
339                    .map_err(E::custom)
340            }
341        }
342
343        deserializer.deserialize_str(Rfc3339Visitor)
344    }
345}

356/// `later - earlier` in whole `unit`s, truncated toward zero. The range
357/// bound keeps every unit used here inside `i64`.
358fn signed_between(earlier: Timestamp, later: Timestamp, unit: i128) -> i64 {
359    let units = later.nanos.saturating_sub(earlier.nanos).checked_div(unit).unwrap_or(0);
360    i64::try_from(units).unwrap_or(if units < 0 { i64::MIN } else { i64::MAX })
361}

388/// Byte cursor for [`parse_rfc3339_nanos`].
389struct Cursor<'a> {
390    bytes: &'a [u8],
391    pos: usize,
392}
393
394impl Cursor<'_> {
395    fn peek(&self) -> Option<u8> {
396        self.bytes.get(self.pos).copied()
397    }
398
399    fn bump(&mut self) -> Option<u8> {
400        let b = self.peek()?;
401        self.pos = self.pos.checked_add(1)?;
402        Some(b)
403    }
404
405    fn expect(&mut self, want: u8) -> Option<()> {
406        (self.bump()? == want).then_some(())
407    }
408
409    fn digit(&mut self) -> Option<u32> {
410        let b = self.peek().filter(u8::is_ascii_digit)?;
411        self.pos = self.pos.checked_add(1)?;
412        Some(u32::from(b.checked_sub(b'0')?))
413    }
414
415    /// Exactly `n` ASCII digits as a number.
416    fn fixed(&mut self, n: usize) -> Option<u32> {
417        (0..n).try_fold(0u32, |acc, _| acc.checked_mul(10)?.checked_add(self.digit()?))
418    }
419
420    fn at_end(&self) -> bool {
421        self.pos == self.bytes.len()
422    }
423}

425/// Nanoseconds since the Unix epoch denoted by RFC 3339 `text`, or `None`
426/// if it is malformed or names an impossible date or time.
427fn parse_rfc3339_nanos(text: &str) -> Option<i128> {
428    let mut c = Cursor { bytes: text.as_bytes(), pos: 0 };
429
430    // Year: four digits, or chrono's extended form — a sign and 4–6 digits.
431    let year = match c.peek()? {
432        sign @ (b'+' | b'-') => {
433            c.bump()?;
434            let mut value = c.fixed(4)?;
435            for _ in 0..2 {
436                match c.digit() {
437                    Some(d) => value = value.checked_mul(10)?.checked_add(d)?,
438                    None => break,
439                }
440            }
441            let value = i32::try_from(value).ok()?;
442            if sign == b'-' { value.checked_neg()? } else { value }
443        }
444        _ => i32::try_from(c.fixed(4)?).ok()?,
445    };
446    c.expect(b'-')?;
447    let month = c.fixed(2)?;
448    c.expect(b'-')?;
449    let day = c.fixed(2)?;
450    if !matches!(c.bump()?, b'T' | b't' | b' ') {
451        return None;
452    }
453    let hour = c.fixed(2)?;
454    c.expect(b':')?;
455    let minute = c.fixed(2)?;
456    c.expect(b':')?;
457    let second = c.fixed(2)?;
458    // RFC 3339 allows second 60 (leap second); like chrono, it denotes the
459    // instant one second after :59, i.e. the start of the next minute.
460    if hour > 23 || minute > 59 || second > 60 {
461        return None;
462    }
463
464    // Fraction: one or more digits; the first nine are significant.
465    let mut nanos: u32 = 0;
466    if c.peek() == Some(b'.') {
467        c.bump()?;
468        let mut scale: u32 = 100_000_000;
469        nanos = c.digit()?.checked_mul(scale)?;
470        while let Some(d) = c.digit() {
471            scale /= 10;
472            nanos = nanos.checked_add(d.checked_mul(scale)?)?;
473        }
474    }
475
476    // Offset: `Z`/`z` or `±HH:MM`.
477    let offset_secs: i128 = match c.bump()? {
478        b'Z' | b'z' => 0,
479        sign @ (b'+' | b'-') => {
480            let oh = c.fixed(2)?;
481            c.expect(b':')?;
482            let om = c.fixed(2)?;
483            if oh > 23 || om > 59 {
484                return None;
485            }
486            let secs = i128::from(oh.checked_mul(3_600)?.checked_add(om.checked_mul(60)?)?);
487            if sign == b'-' { secs.checked_neg()? } else { secs }
488        }
489        _ => return None,
490    };
491    if !c.at_end() {
492        return None;
493    }
494
495    let month = Month::try_from(u8::try_from(month).ok()?).ok()?;
496    let date = Date::from_calendar_date(year, month, u8::try_from(day).ok()?).ok()?;
497    let days = i128::from(date.to_julian_day()).checked_sub(UNIX_EPOCH_JULIAN_DAY)?;
498    let secs = days
499        .checked_mul(86_400)?
500        .checked_add(i128::from(hour.checked_mul(3_600)?))?
501        .checked_add(i128::from(minute.checked_mul(60)?))?
502        .checked_add(i128::from(second))?
503        .checked_sub(offset_secs)?;
504    secs.checked_mul(NANOS_PER_SEC)?.checked_add(i128::from(nanos))
505}

744    #[test]
745    fn test_serde_roundtrip_and_reject_garbage_succeeds() {
746        #[derive(Serialize, Deserialize, PartialEq, Debug)]
747        struct Probe {
748            at: Timestamp,
749            maybe: Option<Timestamp>,
750        }
751        let probe = Probe { at: at(1_700_000_000, 123_456_000), maybe: None };
752        let json = serde_json::to_string(&probe).unwrap();
753        assert_eq!(json, r#"{"at":"2023-11-14T22:13:20.123456Z","maybe":null}"#);
754        assert_eq!(serde_json::from_str::<Probe>(&json).unwrap(), probe);
755        // A missing optional field still deserializes as `None`.
756        let missing: Probe =
757            serde_json::from_str(r#"{"at":"2023-11-14T22:13:20.123456Z"}"#).unwrap();
758        assert_eq!(missing, probe);
759        let some = Probe { at: at(0, 0), maybe: Some(at(1, 1_000)) };
760        let json = serde_json::to_string(&some).unwrap();
761        assert_eq!(json, r#"{"at":"1970-01-01T00:00:00Z","maybe":"1970-01-01T00:00:01.000001Z"}"#);
762        assert_eq!(serde_json::from_str::<Probe>(&json).unwrap(), some);
763        assert!(serde_json::from_str::<Probe>(r#"{"at":"yesterday"}"#).is_err());
764        assert!(serde_json::from_str::<Probe>(r#"{"at":1700000000}"#).is_err());
765        assert!(
766            serde_json::from_str::<Probe>(r#"{"at":"1970-01-01T00:00:00Z","maybe":"x"}"#).is_err()
767        );
768        assert_eq!(format!("{:?}", at(0, 0)), "Timestamp(1970-01-01T00:00:00Z)");
769    }
