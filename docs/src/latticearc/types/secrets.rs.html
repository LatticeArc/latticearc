1//! # Secret byte containers
2//!
3//! Provides the two primitive secret types used throughout LatticeArc:
4//!
5//! - [`SecretBytes<N>`]: fixed-size secret bytes, stack-allocated `[u8; N]` backing.
6//!   Preferred whenever the length is compile-time known.
7//! - [`SecretVec`]: variable-length secret bytes, heap-allocated `Vec<u8>` backing.
8//!   Used only when the length genuinely varies at runtime.
9//!
10//! Both types uphold the full Secret Type Invariant set — see
11//! [`docs/SECRET_TYPE_INVARIANTS.md`](../../../docs/SECRET_TYPE_INVARIANTS.md)
12//! for the canonical, normative specification.
13//!
14//! ## Quick reference
15//!
16//! | Invariant | Mechanism |
17//! |---|---|
18//! | Wipe on drop | `#[derive(Zeroize, ZeroizeOnDrop)]` — volatile writes |
19//! | Redacted `Debug` | Manual `fmt::Debug` impl emits `"[REDACTED]"` |
20//! | Timing-safe equality | `impl ConstantTimeEq` |
21//! | No `PartialEq` / `Eq` | Enforced at `tests/no_partial_eq_on_secret_types.rs` |
22//! | No `Clone` | Explicit `clone_for_transmission(&self) -> Self` |
23//! | Sealed accessor | `expose_secret()` only — no `AsRef`, no `Deref` |
24//! | Swap protection (opt-in, `SecretVec` only) | `secret-mlock` feature; outcome via [`SecretVec::is_locked`] and [`memory_lock_status`] |
25//!
26//! ## When to use which
27//!
28//! ```text
29//! // 32-byte HKDF-Extract output: fixed size → SecretBytes<32>
30//! fn hkdf_extract(...) -> SecretBytes<32>;
31//!
32//! // 64-byte hybrid shared secret: fixed size → SecretBytes<64>
33//! struct EncapsulatedKey { shared_secret: SecretBytes<64>, ... }
34//!
35//! // AEAD plaintext: length varies with input → SecretVec
36//! fn decrypt_aes_gcm(...) -> SecretVec;
37//!
38//! // PBKDF2 output with user-specified key_length → SecretVec
39//! fn pbkdf2(..., key_length: usize) -> SecretVec;
40//! ```

47// ============================================================================
48// SecretBytes<N> — fixed-size, stack-allocated
49// ============================================================================
50
51/// Fixed-size secret byte container, stack-allocated.
52///
53/// The `[u8; N]` backing store never reallocates, never migrates across the heap,
54/// and is wiped by volatile writes on drop. Prefer this type over [`SecretVec`]
55/// whenever `N` is compile-time known.
56///
57/// # Security invariants
58///
59/// - **Wipe on drop**: `#[derive(Zeroize, ZeroizeOnDrop)]` emits volatile writes
60///   that the compiler cannot optimize away.
61/// - **Redacted `Debug`**: the bytes are never printed; `"[REDACTED]"` is emitted.
62/// - **Timing-safe equality**: [`ConstantTimeEq`] compares all `N` bytes in time
63///   independent of their contents; `PartialEq`/`Eq` are not implemented.
64/// - **No `Clone`**: duplication must go through [`clone_for_transmission`] so
65///   every copy is a grep-able audit checkpoint.
66/// - **Sealed access**: the bytes are exposed only through [`expose_secret`]. No
67///   `AsRef<[u8]>`, no `Deref`, no public field.
68///
69/// [`clone_for_transmission`]: Self::clone_for_transmission
70/// [`expose_secret`]: Self::expose_secret
71///
72/// # Why stack-allocated matters
73///
74/// - Heap allocators retain size metadata after `free`, which is a side-channel
75///   leak about secret sizes in adversary-controlled environments.
76/// - A `Vec<u8>` with a single `.push()` past capacity silently reallocates and
77///   frees the old buffer **without zeroization**. `[u8; N]` has no such escape.
78/// - Stack memory is reclaimed in deterministic scope order, not at allocator
79///   whim.
80///
81/// # Example
82///
83/// ```
84/// use latticearc::types::SecretBytes;
85///
86/// // Construct from raw bytes (e.g., after filling a buffer with KDF output)
87/// let key: SecretBytes<32> = SecretBytes::new([0x42u8; 32]);
88/// assert_eq!(key.len(), 32);
89/// assert_eq!(key.expose_secret()[0], 0x42);
90///
91/// // Debug is redacted
92/// let debug = format!("{:?}", key);
93/// assert!(debug.contains("[REDACTED]"));
94/// assert!(!debug.contains("42"));
95/// ```
96#[derive(Zeroize, ZeroizeOnDrop)]
97pub struct SecretBytes<const N: usize> {
98    bytes: [u8; N],
99}
100
101impl<const N: usize> SecretBytes<N> {
102    /// Construct from raw bytes.
103    ///
104    /// The passed array is moved in; no copy is made beyond the move itself.
105    #[must_use]
106    #[inline]
107    pub const fn new(bytes: [u8; N]) -> Self {
108        Self { bytes }
109    }
110
111    /// Construct an all-zero buffer.
112    ///
113    /// Build the final byte pattern in a plain `[u8; N]` on the stack, then
114    /// hand the array to [`Self::new`]; use [`Self::zero`] when the caller
115    /// wants an all-zero placeholder (e.g., a struct field that is overwritten
116    /// by a downstream operation).
117    #[must_use]
118    #[inline]
119    pub const fn zero() -> Self {
120        Self { bytes: [0u8; N] }
121    }
122
123    /// Length in bytes. Always equals the const generic `N`.
124    #[must_use]
125    #[inline]
126    pub const fn len(&self) -> usize {
127        N
128    }
129
130    /// Returns `true` if `N == 0`.
131    #[must_use]
132    #[inline]
133    pub const fn is_empty(&self) -> bool {
134        N == 0
135    }
136
137    /// Expose the inner bytes.
138    ///
139    /// This is the **only** public read accessor. Every call site is a grep-able
140    /// audit checkpoint: `rg '\.expose_secret\(\)'` enumerates every point at
141    /// which secret bytes are read.
142    ///
143    /// Consumers should hold the returned slice only as long as needed and
144    /// should never copy it into a non-secret container (a subsequent `.to_vec()`
145    /// or struct field of type `Vec<u8>` would evade zeroization).
146    #[must_use]
147    #[inline]
148    pub const fn expose_secret(&self) -> &[u8; N] {
149        &self.bytes
150    }
151
152    /// Create an independent copy.
153    ///
154    /// `Clone` is intentionally not derived. Use this method when duplication is
155    /// semantically required — for example, sending a shared secret across a
156    /// channel boundary — so every duplication is grep-able.
157    ///
158    /// # Note on memory locking
159    ///
160    /// `SecretBytes<N>` is stack-allocated, so `mlock(2)`-style page
161    /// pinning does not apply: each clone is a fresh stack location
162    /// (no heap pages to lock). If a caller specifically needs OS-level
163    /// memory-lock semantics for a duplicated secret, use
164    /// [`SecretVec`] (heap-backed) instead.
165    #[must_use]
166    #[inline]
167    pub fn clone_for_transmission(&self) -> Self {
168        Self { bytes: self.bytes }
169    }
170}

172impl<const N: usize> fmt::Debug for SecretBytes<N> {
173    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
174        f.debug_struct("SecretBytes").field("len", &N).field("bytes", &"[REDACTED]").finish()
175    }
176}

178impl<const N: usize> ConstantTimeEq for SecretBytes<N> {
179    fn ct_eq(&self, other: &Self) -> Choice {
180        self.bytes.ct_eq(&other.bytes)
181    }
182}

184// ============================================================================
185// SecretVec — variable-length, heap-allocated
186// ============================================================================
187
188/// Variable-length secret byte container, heap-allocated.
189///
190/// Use only when the length genuinely varies at runtime (e.g., user-specified
191/// PBKDF2 output length, AEAD plaintext of arbitrary input). For any
192/// compile-time-known length, prefer [`SecretBytes<N>`].
193///
194/// # Security invariants
195///
196/// Same as [`SecretBytes`]. See type-level docs for the full list.
197///
198/// # Reallocation safety
199///
200/// The internal `Vec<u8>` is never reallocated by this type's public API: there
201/// is no `push`, no `extend`, no `reserve`. All construction paths allocate the
202/// final size up front. If a caller needs to build a secret incrementally, they
203/// must build the full `Vec<u8>` first (outside this wrapper), then pass it to
204/// [`Self::new`] — at which point reallocation is no longer a concern because
205/// subsequent mutation is not exposed.
206///
207/// # OS-level memory locking (`secret-mlock` feature)
208///
209/// With the `secret-mlock` feature enabled on Linux, macOS, Android and other
210/// Unix targets, each non-empty `SecretVec` asks the kernel to keep the pages
211/// backing its buffer resident in RAM via `mlock(2)`, so the bytes are not
212/// written to swap. This implements invariant I-10. `mlock` does not exclude
213/// the pages from core dumps; disable core dumps at the process level if that
214/// matters for your deployment.
215///
216/// Locking is best-effort and its outcome is reported, not hidden:
217///
218/// - [`SecretVec::is_locked`] says whether this buffer's pages are locked.
219/// - [`memory_lock_status`] returns a process-wide [`MemoryLockStatus`] with
220///   the number of currently locked buffers, the number of failed attempts and
221///   the most recent OS error.
222///
223/// If locking fails (for example `RLIMIT_MEMLOCK` exceeded — Android
224/// typically allows only 64 KiB per process), the `SecretVec` is still
225/// returned with an unlocked backing. The bytes remain zeroized on drop; only
226/// the swap protection is missing. Applications should surface
227/// [`memory_lock_status`] to the user or operator instead of assuming success.
228///
229/// Page locks do not nest in the kernel: unlocking a page unlocks it for every
230/// buffer on it. The crate therefore reference-counts locked pages process-wide
231/// and only unlocks a page when the last `SecretVec` using it is dropped, so
232/// [`SecretVec::is_locked`] stays accurate while neighbouring buffers come and
233/// go. Each newly locked page costs one `mlock` call.
234///
235/// On Windows the feature is a no-op (see [`MemoryLockStatus::supported`]):
236/// `is_locked` is always `false`. Without the feature, `mlock` is never called
237/// and the crate does not depend on `region`.
238///
239/// `SecretBytes<N>` is deliberately NOT covered by this feature: its `[u8; N]`
240/// backing is stack-allocated, so mlocking it would require per-instance
241/// page-aligned locking of arbitrary stack addresses — impractical and
242/// incompatible with Rust's move semantics. Callers who need OS-level
243/// protection for small fixed-size secrets should wrap them in `SecretVec` at
244/// a boundary where the sensitive bytes leave the stack.
245///
246/// # Example
247///
248/// ```
249/// use latticearc::types::{SecretVec, memory_lock_status};
250///
251/// let key = SecretVec::new(vec![0x42u8; 48]);
252/// assert_eq!(key.len(), 48);
253/// assert_eq!(key.expose_secret()[0], 0x42);
254///
255/// // Without the `secret-mlock` feature this is always `false`.
256/// let _locked: bool = key.is_locked();
257/// let status = memory_lock_status();
258/// assert_eq!(status.feature_enabled, cfg!(feature = "secret-mlock"));
259///
260/// let debug = format!("{:?}", key);
261/// assert!(debug.contains("[REDACTED]"));
262/// ```
263#[derive(Zeroize, ZeroizeOnDrop)]
264pub struct SecretVec {
265    // Drop order when `secret-mlock` is enabled:
266    //   1. `ZeroizeOnDrop`'s generated `Drop` runs first, calling `.zeroize()`
267    //      on `bytes` while the pages are still locked.
268    //   2. Fields drop in declaration order: `lock` first (releases this
269    //      buffer's page references via `MlockGuard`'s Drop), then `bytes`
270    //      (returns the zeroed buffer to the allocator).
271    // `#[zeroize(skip)]` is required because the lock holder does not
272    // implement `Zeroize` and there's nothing secret inside it — it only
273    // holds page addresses.
274    #[cfg(all(feature = "secret-mlock", not(target_os = "windows")))]
275    #[zeroize(skip)]
276    lock: Option<mlock::MlockGuard>,
277    bytes: Vec<u8>,
278}

280/// Process-wide snapshot of OS memory locking for [`SecretVec`] buffers.
281///
282/// Returned by [`memory_lock_status`]. Intended for display: a vault app can
283/// show "secrets pinned in RAM" or "memory locking unavailable (limit
284/// reached)" instead of silently running without swap protection.
285///
286/// On Android, `RLIMIT_MEMLOCK` is typically 64 KiB per process, so
287/// `failed_buffers > 0` is an expected state there once enough secrets are
288/// alive at the same time; show it rather than hide it.
289#[derive(Debug, Clone, PartialEq, Eq)]
290#[non_exhaustive]
291pub struct MemoryLockStatus {
292    /// `true` when this build attempts to lock `SecretVec` buffers: the
293    /// `secret-mlock` feature is enabled and the target is not Windows.
294    pub supported: bool,
295    /// `true` when the crate was built with the `secret-mlock` feature
296    /// (including Windows builds, where locking is still not attempted).
297    pub feature_enabled: bool,
298    /// Number of currently alive `SecretVec` buffers whose pages are locked.
299    /// Increases on each successful lock and decreases when such a buffer is
300    /// dropped.
301    pub locked_buffers: usize,
302    /// Number of lock attempts that failed since process start. Never
303    /// decreases; a non-zero value means at least one `SecretVec` ran
304    /// without swap protection.
305    pub failed_buffers: usize,
306    /// Operating-system error text of the most recent failed lock attempt,
307    /// or `None` if no attempt has failed.
308    pub last_error: Option<String>,
309}
310
311/// Return the process-wide [`MemoryLockStatus`].
312///
313/// Cheap: two atomic loads and one short mutex acquisition. Without the
314/// `secret-mlock` feature, or on Windows, every counter is zero and
315/// `supported` is `false`.
316#[must_use]
317pub fn memory_lock_status() -> MemoryLockStatus {
318    #[cfg(all(feature = "secret-mlock", not(target_os = "windows")))]
319    {
320        mlock::status()
321    }
322    #[cfg(not(all(feature = "secret-mlock", not(target_os = "windows"))))]
323    {
324        MemoryLockStatus {
325            supported: false,
326            feature_enabled: cfg!(feature = "secret-mlock"),
327            locked_buffers: 0,
328            failed_buffers: 0,
329            last_error: None,
330        }
331    }
332}

334/// Reference-counted page locking for [`SecretVec`] (Unix only).
335///
336/// # Why reference counting
337///
338/// `mlock` / `munlock` operate on whole pages and do not nest: two buffers on
339/// the same page share one lock, and `munlock` from the first buffer's drop
340/// would silently unlock the page under the second. The registry below keeps
341/// one [`region::LockGuard`] per locked page plus a count of the `SecretVec`
342/// buffers using it, and only drops the guard (unlocking the page) when the
343/// count reaches zero.
344///
345/// # Why Windows is excluded
346///
347/// `VirtualLock` is documented as best-effort: pages can be implicitly
348/// unlocked when the working set is trimmed, and a later `VirtualUnlock`
349/// returns `ERROR_NOT_LOCKED`. `region` asserts on that error in its `Drop`,
350/// so a routine working-set trim would turn into a Drop-time panic. The bytes
351/// are still zeroized at Drop via `ZeroizeOnDrop` on Windows.
352///
353/// # Drop-time panics
354///
355/// `region::LockGuard::drop` contains a `debug_assert!` on the unlock result.
356/// Each page guard is dropped inside [`std::panic::catch_unwind`] so a failed
357/// `munlock` in a debug build cannot escape a `SecretVec` Drop chain (which
358/// may run while already unwinding). `catch_unwind` is a no-op under
359/// `panic = "abort"`; release builds compile the `debug_assert!` out.
360#[cfg(all(feature = "secret-mlock", not(target_os = "windows")))]
361mod mlock {
362    use std::collections::BTreeMap;
363    use std::sync::atomic::{AtomicUsize, Ordering};
364
365    use parking_lot::Mutex;
366
367    use super::MemoryLockStatus;
368
369    /// One locked page and the number of live buffers that use it.
370    struct PageLock {
371        refs: usize,
372        guard: region::LockGuard,
373    }
374
375    /// Locked pages keyed by page base address.
376    static PAGES: Mutex<BTreeMap<usize, PageLock>> = Mutex::new(BTreeMap::new());
377    /// Live `SecretVec` buffers whose pages are all locked.
378    static LOCKED_BUFFERS: AtomicUsize = AtomicUsize::new(0);
379    /// Failed lock attempts since process start.
380    static FAILED_BUFFERS: AtomicUsize = AtomicUsize::new(0);
381    /// OS error text of the most recent failure.
382    static LAST_ERROR: Mutex<Option<String>> = Mutex::new(None);
383
384    /// Proof that one buffer holds a reference on every page in
385    /// `first_page .. first_page + page_count * page_size`.
386    pub(super) struct MlockGuard {
387        first_page: usize,
388        page_count: usize,
389        page_size: usize,
390    }
391
392    impl Drop for MlockGuard {
393        fn drop(&mut self) {
394            release_pages(self.first_page, self.page_count, self.page_size);
395            // Checked decrement: the counter was incremented when this guard
396            // was created, so it cannot be zero here; `try_update` leaves it
397            // at zero instead of wrapping if that invariant were ever broken.
398            let _ = LOCKED_BUFFERS
399                .try_update(Ordering::AcqRel, Ordering::Acquire, |n| n.checked_sub(1));
400        }
401    }
402
403    /// Lock every page overlapping `bytes`. Returns `None` for an empty
404    /// buffer (no attempt is made) or when the OS refuses the lock; the
405    /// failure case is recorded in the process-wide status.
406    pub(super) fn lock(bytes: &[u8]) -> Option<MlockGuard> {
407        if bytes.is_empty() {
408            return None;
409        }
410        match lock_pages(bytes) {
411            Ok(guard) => {
412                LOCKED_BUFFERS.fetch_add(1, Ordering::AcqRel);
413                Some(guard)
414            }
415            Err(message) => {
416                record_lock_failure(message);
417                None
418            }
419        }
420    }
421
422    /// Record a failed lock attempt. Crate-private so the bookkeeping can be
423    /// unit-tested without lowering `RLIMIT_MEMLOCK` in the test process.
424    pub(super) fn record_lock_failure(message: String) {
425        FAILED_BUFFERS.fetch_add(1, Ordering::AcqRel);
426        *LAST_ERROR.lock() = Some(message);
427    }
428
429    /// Snapshot of the counters.
430    pub(super) fn status() -> MemoryLockStatus {
431        MemoryLockStatus {
432            supported: true,
433            feature_enabled: true,
434            locked_buffers: LOCKED_BUFFERS.load(Ordering::Acquire),
435            failed_buffers: FAILED_BUFFERS.load(Ordering::Acquire),
436            last_error: LAST_ERROR.lock().clone(),
437        }
438    }
439
440    /// Page range `(first_page, page_count, page_size)` covering `bytes`.
441    fn page_range(bytes: &[u8]) -> Option<(usize, usize, usize)> {
442        let page_size = region::page::size();
443        let start = bytes.as_ptr().addr();
444        let last = start.checked_add(bytes.len())?.checked_sub(1)?;
445        let first_page = start.checked_sub(start.checked_rem(page_size)?)?;
446        let last_page = last.checked_sub(last.checked_rem(page_size)?)?;
447        let page_count =
448            last_page.checked_sub(first_page)?.checked_div(page_size)?.checked_add(1)?;
449        Some((first_page, page_count, page_size))
450    }
451
452    fn lock_pages(bytes: &[u8]) -> Result<MlockGuard, String> {
453        let (first_page, page_count, page_size) =
454            page_range(bytes).ok_or_else(|| "buffer address range overflows".to_string())?;
455        // Pointer to the first page, derived from the buffer pointer so the
456        // address handed to the OS keeps the allocation's provenance.
457        let offset = bytes.as_ptr().addr().saturating_sub(first_page);
458        let base = bytes.as_ptr().wrapping_sub(offset);
459
460        let mut pages = PAGES.lock();
461        for index in 0..page_count {
462            let delta = index.checked_mul(page_size);
463            let page = delta.and_then(|d| first_page.checked_add(d));
464            let (Some(delta), Some(page)) = (delta, page) else {
465                drop(pages);
466                release_pages(first_page, index, page_size);
467                return Err("buffer address range overflows".to_string());
468            };
469            if let Some(entry) = pages.get_mut(&page) {
470                entry.refs = entry.refs.saturating_add(1);
471                continue;
472            }
473            match region::lock(base.wrapping_add(delta), page_size) {
474                Ok(guard) => {
475                    pages.insert(page, PageLock { refs: 1, guard });
476                }
477                Err(err) => {
478                    drop(pages);
479                    // Roll back the references this call already took.
480                    release_pages(first_page, index, page_size);
481                    return Err(err.to_string());
482                }
483            }
484        }
485        Ok(MlockGuard { first_page, page_count, page_size })
486    }
487
488    /// Drop one reference on each of `page_count` pages starting at
489    /// `first_page`, unlocking pages whose count reaches zero.
490    fn release_pages(first_page: usize, page_count: usize, page_size: usize) {
491        let mut unlocked: Vec<region::LockGuard> = Vec::new();
492        {
493            let mut pages = PAGES.lock();
494            for index in 0..page_count {
495                let Some(page) =
496                    index.checked_mul(page_size).and_then(|d| first_page.checked_add(d))
497                else {
498                    break;
499                };
500                let remove = match pages.get_mut(&page) {
501                    Some(entry) => {
502                        entry.refs = entry.refs.saturating_sub(1);
503                        entry.refs == 0
504                    }
505                    None => false,
506                };
507                if remove && let Some(entry) = pages.remove(&page) {
508                    unlocked.push(entry.guard);
509                }
510            }
511        }
512        // `munlock` runs outside the registry mutex.
513        for guard in unlocked {
514            let _ = std::panic::catch_unwind(std::panic::AssertUnwindSafe(move || {
515                drop(guard);
516            }));
517        }
518    }
519
520    /// Reference count of the page containing `addr`, for tests.
521    #[cfg(test)]
522    pub(super) fn page_refs(addr: usize) -> Option<usize> {
523        let page_size = region::page::size();
524        let page = addr.checked_sub(addr.checked_rem(page_size)?)?;
525        PAGES.lock().get(&page).map(|entry| entry.refs)
526    }
527}

529impl SecretVec {
530    /// Construct from a `Vec<u8>`.
531    ///
532    /// The vector is moved in; its capacity is taken as-is. For best security
533    /// hygiene, construct the passed `Vec` at its final length (e.g.,
534    /// `vec![0u8; len]`) rather than growing it, so no unzeroized intermediate
535    /// buffer is ever freed.
536    #[must_use]
537    #[inline]
538    pub fn new(bytes: Vec<u8>) -> Self {
539        Self::from_bytes(bytes)
540    }
541
542    /// Construct an all-zero buffer of the given length.
543    #[must_use]
544    #[inline]
545    pub fn zero(len: usize) -> Self {
546        Self::from_bytes(vec![0u8; len])
547    }
548
549    /// Single construction path.
550    ///
551    /// Allocates a fresh `Vec<u8>` of exactly `bytes.len()` capacity, copies
552    /// the secret in, then zeroizes the caller-supplied buffer in place. This
553    /// is deliberately NOT `bytes.shrink_to_fit()`: `shrink_to_fit` may
554    /// internally call the allocator's `realloc`, which copies the secret
555    /// into a smaller allocation and frees the original WITHOUT zeroizing.
556    /// The freed buffer is then immediately available for allocator reuse
557    /// — a classic "secret in freed memory" leak that affects any caller
558    /// passing in an over-allocated `Vec` (e.g. `BASE64_ENGINE.decode(...)?`,
559    /// which pre-sizes with slack).
560    ///
561    /// The exact-size copy + caller-side wipe pattern guarantees:
562    /// 1. Backing capacity == length, so the owned `Vec` has no slack
563    ///    that could outlive a future shrink/realloc. (`zeroize 1.8`'s
564    ///    `Vec<Z>::zeroize` does wipe the full capacity — `iter_mut +
565    ///    clear + spare_capacity_mut().zeroize()` — but the exact-size
566    ///    invariant keeps that property robust against future moves
567    ///    that change the underlying allocation.)
568    /// 2. The original caller-supplied buffer is wiped *in place* before
569    ///    its `Vec` Drop hands the memory back to the allocator, so no
570    ///    copy of the secret persists in any allocator free-list. This
571    ///    is the property the new exact-size `owned` cannot provide on
572    ///    its own, because it inherits none of the original's slack
573    ///    capacity.
574    #[inline]
575    fn from_bytes(mut bytes: Vec<u8>) -> Self {
576        let len = bytes.len();
577        let mut owned: Vec<u8> = Vec::with_capacity(len);
578        owned.extend_from_slice(&bytes);
579        // Wipe the caller-supplied buffer in place: this covers any slack
580        // capacity in the ORIGINAL allocation, which the new `owned` does
581        // not inherit. Without this, an over-sized incoming `Vec` (e.g.
582        // base64 decode output) would leave secret material in the freed
583        // tail when `bytes` is dropped.
584        use zeroize::Zeroize;
585        bytes.zeroize();
586        drop(bytes);
587
588        #[cfg(all(feature = "secret-mlock", not(target_os = "windows")))]
589        let lock = mlock::lock(&owned);
590        Self {
591            #[cfg(all(feature = "secret-mlock", not(target_os = "windows")))]
592            lock,
593            bytes: owned,
594        }
595    }
596
597    /// Returns `true` if this buffer's pages are locked into RAM.
598    ///
599    /// `true` only when the `secret-mlock` feature is enabled, the target is
600    /// not Windows, the buffer is non-empty and `mlock` succeeded for every
601    /// page it occupies. Locked pages stay locked until this `SecretVec` is
602    /// dropped (see the reference-counting note on the type). A `false`
603    /// result means the bytes may be written to swap; they are still
604    /// zeroized on drop. Use [`memory_lock_status`] for the process-wide
605    /// picture.
606    #[must_use]
607    #[inline]
608    pub fn is_locked(&self) -> bool {
609        #[cfg(all(feature = "secret-mlock", not(target_os = "windows")))]
610        {
611            self.lock.is_some()
612        }
613        #[cfg(not(all(feature = "secret-mlock", not(target_os = "windows"))))]
614        {
615            false
616        }
617    }
618
619    /// Length in bytes.
620    #[must_use]
621    #[inline]
622    pub fn len(&self) -> usize {
623        self.bytes.len()
624    }
625
626    /// Returns `true` if the backing buffer is empty.
627    #[must_use]
628    #[inline]
629    pub fn is_empty(&self) -> bool {
630        self.bytes.is_empty()
631    }

633    /// Expose the inner bytes. The only public read accessor.
634    #[must_use]
635    #[inline]
636    pub fn expose_secret(&self) -> &[u8] {
637        &self.bytes
638    }
639
640    /// Create an independent copy. See [`SecretBytes::clone_for_transmission`].
641    ///
642    /// With the `secret-mlock` feature, the cloned buffer is independently
643    /// locked into RAM (the original's `LockGuard` does not transfer).
644    #[must_use]
645    pub fn clone_for_transmission(&self) -> Self {
646        Self::from_bytes(self.bytes.clone())
647    }

917    fn secrets_shared_pages_are_reference_counted() {
918        // Three pages of our own: the middle page cannot be shared with any
919        // other allocation, so its reference count is ours alone.
920        let page = region::page::size();
921        let buffer = vec![0u8; page.saturating_mul(3)];
922        let middle = buffer.as_ptr().addr().saturating_add(page.saturating_add(page / 2));
923        let Some(first) = mlock::lock(&buffer) else {
924            // Environment refused the lock; the bookkeeping must say so.
925            assert!(memory_lock_status().failed_buffers > 0);
926            return;
927        };
928        assert_eq!(mlock::page_refs(middle), Some(1));
929        let second = mlock::lock(&buffer);
930        assert!(second.is_some());
931        assert_eq!(mlock::page_refs(middle), Some(2));
932        // Dropping one holder must not unlock the page for the other.
933        drop(first);
934        assert_eq!(mlock::page_refs(middle), Some(1));
935        drop(second);
936        assert_eq!(mlock::page_refs(middle), None);
