1//! Shared AES-256-GCM envelope plumbing for the hybrid and PQ-only paths.
2//!
3//! Both `encrypt_hybrid`/`decrypt_hybrid` and `encrypt_pq_only_with_aad`/
4//! `decrypt_pq_only_with_aad` derive a 32-byte key through their own,
5//! deliberately domain-separated KDF constructions, then run the identical
6//! AES-256-GCM seal/open sequence. That sequence lives here once so a future
7//! change (cipher init, nonce policy, AAD pass-through) cannot silently drift
8//! between the two paths.
9//!
10//! Error handling contract: the helpers return [`EnvelopeStage`] naming which
11//! step failed. Callers map every stage to their own opaque Pattern-6 error
12//! and log the stage message under their own operation label — the stage enum
13//! never reaches a public error, so per-stage information stays out of
14//! adversary-visible surfaces.
15
16use crate::primitives::aead::aes_gcm::AesGcm256;
17use crate::primitives::aead::{AeadCipher, NONCE_LEN, TAG_LEN};
18use zeroize::Zeroizing;
19
20/// Which envelope step failed. Internal only — callers log the message and
21/// collapse to their opaque error.
22#[derive(Debug, Clone, Copy)]
23pub(crate) enum EnvelopeStage {
24    /// AES-256 cipher construction rejected the key.
25    CipherInit,
26    /// AEAD seal failed.
27    Seal,
28    /// AEAD open (authentication) failed.
29    Open,
30}
31
32impl EnvelopeStage {
33    /// Stage description for the caller's internal trace line. Matches the
34    /// strings the per-path implementations logged before consolidation so
35    /// operator dashboards keep their signal.
36    pub(crate) fn msg(self) -> &'static str {
37        match self {
38            Self::CipherInit => "AES-256 init failed",
39            Self::Seal => "AES-GCM seal failed",
40            Self::Open => "AEAD authentication failed",
41        }
42    }
43}
44
45/// `(ciphertext, nonce, tag)` produced by a successful seal.
46pub(crate) type SealedParts = (Vec<u8>, [u8; NONCE_LEN], [u8; TAG_LEN]);
47
48/// Seal `plaintext` under `key` with a fresh CSPRNG nonce.
49///
50/// `Some(&[])` and `None` AAD produce byte-identical AES-GCM output, so the
51/// AAD is passed through unconditionally.
52pub(crate) fn seal_aes256_gcm(
53    key: &[u8],
54    plaintext: &[u8],
55    aad: &[u8],
56) -> Result<SealedParts, EnvelopeStage> {
57    let cipher = AesGcm256::new(key).map_err(|_e| EnvelopeStage::CipherInit)?;
58    let nonce = AesGcm256::generate_nonce();
59    let (ciphertext, tag) =
60        cipher.encrypt(&nonce, plaintext, Some(aad)).map_err(|_e| EnvelopeStage::Seal)?;
61    Ok((ciphertext, nonce, tag))
62}
63
64/// Open `ciphertext` under `key`. Returns the plaintext in `Zeroizing` so a
65/// caller cannot accidentally hold an unscrubbed copy.
66pub(crate) fn open_aes256_gcm(
67    key: &[u8],
68    nonce: &[u8; NONCE_LEN],
69    ciphertext: &[u8],
70    tag: &[u8; TAG_LEN],
71    aad: &[u8],
72) -> Result<Zeroizing<Vec<u8>>, EnvelopeStage> {
73    let cipher = AesGcm256::new(key).map_err(|_e| EnvelopeStage::CipherInit)?;
74    cipher.decrypt(nonce, ciphertext, tag, Some(aad)).map_err(|_e| EnvelopeStage::Open)
75}
