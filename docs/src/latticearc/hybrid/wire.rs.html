1//! # Compact binary wire format (v1)
2//!
3//! A fixed-layout, versioned binary encoding for the hybrid types that
4//! applications store or transmit: hybrid KEM public keys, hybrid KEM
5//! encapsulations, hybrid ciphertexts, hybrid signature public keys and
6//! hybrid signatures. Intended for fixed-size records on disk or untrusted
7//! storage where JSON and CBOR are unwanted. The normative specification,
8//! including the compatibility policy, is `docs/WIRE_FORMAT.md`.
9//!
10//! ## Layout
11//!
12//! ```text
13//! magic "LA" (2) | version = 1 (1) | type tag (1) | algorithm id (1) | body
14//! ```
15//!
16//! Every body field has a length fixed by the algorithm id, so there are no
17//! length prefixes except for the variable AES-256-GCM ciphertext inside a
18//! [`HybridCiphertext`] (one `u32` big-endian prefix, capped at
19//! [`MAX_SYMMETRIC_CIPHERTEXT_LEN`]). Trailing bytes are rejected, so each
20//! value has exactly one encoding.
21//!
22//! | Tag | Type | Algorithm id |
23//! |---|---|---|
24//! | `0x01` | [`HybridKemPublicKey`] | ML-KEM level (1, 3, 5) |
25//! | `0x02` | [`EncapsulatedKey`] | ML-KEM level (1, 3, 5) |
26//! | `0x03` | [`HybridCiphertext`] | ML-KEM level (1, 3, 5) |
27//! | `0x04` | [`HybridSigPublicKey`] | ML-DSA parameter set (2, 3, 5) |
28//! | `0x05` | [`HybridSignature`] | ML-DSA parameter set (2, 3, 5) |
29//! | `0x06` | reserved for `WrappedKey` | — |
30//!
31//! Algorithm ids are the discriminants of [`MlKemSecurityLevel`] and
32//! [`MlDsaParameterSet`] (the NIST security category).
33//!
34//! ## Decoding guarantees
35//!
36//! - Total: every input yields `Ok` or a [`WireError`]; no panics.
37//! - The input length is checked against the type's hard maximum
38//!   ([`WireEncode::MAX_WIRE_LEN`]) before anything is parsed or allocated.
39//! - Allocation is bounded by the input actually supplied: a length prefix
40//!   is only honoured when that many bytes are present.
41//! - Structural only: component bytes are not validated cryptographically
42//!   (point/modulus checks happen at first use, as with
43//!   [`HybridKemPublicKey::from_bytes`]).
44//!
45//! ## Error detail
46//!
47//! [`WireError`] variants distinguish structural failures (truncation,
48//! version, type tag, trailing bytes). Every check reads only the public,
49//! attacker-supplied bytes and runs before any key material is touched, so
50//! the distinction reveals nothing the attacker did not already choose.
51//! Decryption and verification of a decoded value keep their opaque errors.
52//!
53//! ## Relationship to `HybridKemPublicKey::to_bytes`
54//!
55//! `HybridKemPublicKey::to_bytes` predates this module and carries its own
56//! version byte, a 1/2/3 level tag and `u32` length prefixes. It is left
57//! unchanged for existing stored keys. This format uses the shared header so
58//! all hybrid types share one envelope and one version rule, and drops the
59//! length prefixes because the algorithm id fixes every length.
60//!
61//! ## Example
62//!
63//! ```rust,no_run
64//! # fn main() -> Result<(), Box<dyn std::error::Error>> {
65//! use latticearc::hybrid::kem_hybrid::generate_keypair;
66//! use latticearc::{HybridKemPublicKey, WireEncode};
67//!
68//! let (pk, _sk) = generate_keypair()?;
69//! let record = pk.to_wire()?;
70//! assert_eq!(record.len(), 5 + 1184 + 32);
71//! let restored = HybridKemPublicKey::from_wire(&record)?;
72//! assert_eq!(restored.ml_kem_pk(), pk.ml_kem_pk());
73//! # Ok(())
74//! # }
75//! ```

87/// Magic bytes at the start of every wire value: ASCII `"LA"`.
88pub const WIRE_MAGIC: [u8; 2] = *b"LA";

90/// The only wire-format version this build encodes and decodes.
91pub const WIRE_VERSION: u8 = 1;

93/// Header length: magic (2) + version (1) + type tag (1) + algorithm id (1).

97///
98/// Decoders in this module reject it with [`WireError::WrongType`]. The

100/// header's version rules still apply to the envelope.
101pub const WRAPPED_KEY_TYPE_TAG: u8 = 0x06;

112/// Errors from encoding or decoding the wire format.
113///
114/// Decoding errors describe structural problems with public input only;
115/// see the module documentation for why they are distinguishable.
116#[derive(Debug, Clone, Error)]
117#[non_exhaustive]
118pub enum WireError {
119    /// The input is longer than the maximum for the requested type, or a
120    /// length prefix exceeds its cap. Checked before any allocation.
121    #[error("wire input length {len} exceeds the maximum of {max} bytes")]
122    TooLarge {
123        /// Observed length (whole input, or the prefixed field length).
124        len: usize,
125        /// Applicable maximum.
126        max: usize,
127    },
128    /// The input ends before the last field.
129    #[error("wire input is truncated")]
130    Truncated,
131    /// The input does not start with [`WIRE_MAGIC`].
132    #[error("wire input does not start with the LatticeArc magic bytes")]
133    BadMagic,
134    /// The version byte is not [`WIRE_VERSION`].
135    #[error("unsupported wire-format version {0}")]
136    UnsupportedVersion(u8),
137    /// The type tag does not match the type being decoded.
138    #[error("wire type tag {found:#04x} does not match expected {expected:#04x}")]
139    WrongType {
140        /// Tag of the type being decoded.
141        expected: u8,
142        /// Tag found in the input.
143        found: u8,
144    },
145    /// The algorithm id is not defined for this type.
146    #[error("unknown algorithm id {0} for this wire type")]
147    UnknownAlgorithm(u8),
148    /// Bytes remain after the last field.
149    #[error("wire input has trailing bytes")]
150    TrailingBytes,
151    /// A component of the value being encoded has a length no algorithm id
152    /// describes (for example a hand-built key of the wrong size).
153    #[error("cannot encode: invalid {0} length")]
154    InvalidComponent(&'static str),
155}

162/// Encode to and decode from the LatticeArc binary wire format.
163///
164/// Sealed: implemented only for [`HybridKemPublicKey`], [`EncapsulatedKey`],
165/// [`HybridCiphertext`], [`HybridSigPublicKey`] and [`HybridSignature`].
166pub trait WireEncode: sealed::Sealed + Sized {
167    /// Type tag written after the version byte.
168    const TYPE_TAG: u8;
169    /// Largest valid encoding of this type, in bytes. Longer input is
170    /// rejected with [`WireError::TooLarge`] before parsing.
171    const MAX_WIRE_LEN: usize;
172
173    /// Encode `self`.
174    ///
175    /// # Errors
176    ///
177    /// [`WireError::InvalidComponent`] if a component length does not match
178    /// any algorithm id (only possible for values assembled by hand), or
179    /// [`WireError::TooLarge`] if a variable field exceeds its cap.
180    fn to_wire(&self) -> Result<Vec<u8>, WireError>;
181
182    /// Decode a value previously produced by [`Self::to_wire`].
183    ///
184    /// # Errors
185    ///
186    /// A [`WireError`] describing the first structural problem found.
187    fn from_wire(bytes: &[u8]) -> Result<Self, WireError>;
188}

194const fn kem_level_id(level: MlKemSecurityLevel) -> u8 {
195    match level {
196        MlKemSecurityLevel::MlKem512 => 1,
197        MlKemSecurityLevel::MlKem768 => 3,
198        MlKemSecurityLevel::MlKem1024 => 5,
199    }
200}
201
202fn kem_level_from_id(id: u8) -> Result<MlKemSecurityLevel, WireError> {
203    match id {
204        1 => Ok(MlKemSecurityLevel::MlKem512),
205        3 => Ok(MlKemSecurityLevel::MlKem768),
206        5 => Ok(MlKemSecurityLevel::MlKem1024),
207        other => Err(WireError::UnknownAlgorithm(other)),
208    }
209}
210
211/// ML-KEM ciphertext sizes (768 / 1088 / 1568) are pairwise distinct.
212fn kem_level_from_ciphertext_len(len: usize) -> Result<MlKemSecurityLevel, WireError> {
213    [MlKemSecurityLevel::MlKem512, MlKemSecurityLevel::MlKem768, MlKemSecurityLevel::MlKem1024]
214        .into_iter()
215        .find(|level| level.ciphertext_size() == len)
216        .ok_or(WireError::InvalidComponent("ML-KEM ciphertext"))
217}

219const fn dsa_set_id(set: MlDsaParameterSet) -> u8 {
220    match set {
221        MlDsaParameterSet::MlDsa44 => 2,
222        MlDsaParameterSet::MlDsa65 => 3,
223        MlDsaParameterSet::MlDsa87 => 5,
224    }
225}
226
227fn dsa_set_from_id(id: u8) -> Result<MlDsaParameterSet, WireError> {
228    match id {
229        2 => Ok(MlDsaParameterSet::MlDsa44),
230        3 => Ok(MlDsaParameterSet::MlDsa65),
231        5 => Ok(MlDsaParameterSet::MlDsa87),
232        other => Err(WireError::UnknownAlgorithm(other)),
233    }
234}

236/// ML-DSA signature sizes (2420 / 3309 / 4627) are pairwise distinct.
237fn dsa_set_from_signature_len(len: usize) -> Result<MlDsaParameterSet, WireError> {
238    [MlDsaParameterSet::MlDsa44, MlDsaParameterSet::MlDsa65, MlDsaParameterSet::MlDsa87]
239        .into_iter()
240        .find(|set| set.signature_size() == len)
241        .ok_or(WireError::InvalidComponent("ML-DSA signature"))
242}

248/// Start an encoding: header plus capacity for `body_len` bytes.
249fn begin(type_tag: u8, alg_id: u8, body_len: usize) -> Result<Vec<u8>, WireError> {
250    let total = WIRE_HEADER_LEN
251        .checked_add(body_len)
252        .ok_or(WireError::TooLarge { len: usize::MAX, max: usize::MAX })?;
253    let mut out = Vec::with_capacity(total);
254    out.extend_from_slice(&WIRE_MAGIC);
255    out.push(WIRE_VERSION);
256    out.push(type_tag);
257    out.push(alg_id);
258    Ok(out)
259}
260
261/// Check the size cap and header; return the algorithm id and body.
262fn parse_header(bytes: &[u8], type_tag: u8, max_len: usize) -> Result<(u8, &[u8]), WireError> {
263    if bytes.len() > max_len {
264        return Err(WireError::TooLarge { len: bytes.len(), max: max_len });
265    }
266    let (header, body) = bytes.split_at_checked(WIRE_HEADER_LEN).ok_or(WireError::Truncated)?;
267    let [m0, m1, version, found, alg_id]: [u8; WIRE_HEADER_LEN] =
268        header.try_into().map_err(|_e| WireError::Truncated)?;
269    if [m0, m1] != WIRE_MAGIC {
270        return Err(WireError::BadMagic);
271    }
272    if version != WIRE_VERSION {
273        return Err(WireError::UnsupportedVersion(version));
274    }
275    if found != type_tag {
276        return Err(WireError::WrongType { expected: type_tag, found });
277    }
278    Ok((alg_id, body))
279}

281/// Forward-only reader over a body slice.
282struct Reader<'a> {
283    rest: &'a [u8],
284}
285
286impl<'a> Reader<'a> {
287    fn new(body: &'a [u8]) -> Self {
288        Self { rest: body }
289    }
290
291    fn take(&mut self, len: usize) -> Result<&'a [u8], WireError> {
292        let (head, tail) = self.rest.split_at_checked(len).ok_or(WireError::Truncated)?;
293        self.rest = tail;
294        Ok(head)
295    }
296
297    fn take_u32_be(&mut self) -> Result<u32, WireError> {
298        let raw: [u8; SYMMETRIC_LEN_PREFIX] =
299            self.take(SYMMETRIC_LEN_PREFIX)?.try_into().map_err(|_e| WireError::Truncated)?;
300        Ok(u32::from_be_bytes(raw))
301    }
302
303    fn finish(self) -> Result<(), WireError> {
304        if self.rest.is_empty() { Ok(()) } else { Err(WireError::TrailingBytes) }
305    }
306}
307
308fn require_len(field: &[u8], expected: usize, name: &'static str) -> Result<(), WireError> {
309    if field.len() == expected { Ok(()) } else { Err(WireError::InvalidComponent(name)) }
310}
311
312/// Sum of field lengths; overflow is impossible for the fixed sizes used here
313/// and is reported as `TooLarge` rather than wrapping.
314fn sum_lens(lens: &[usize]) -> Result<usize, WireError> {
315    lens.iter().try_fold(0usize, |acc, &len| {
316        acc.checked_add(len).ok_or(WireError::TooLarge { len: usize::MAX, max: usize::MAX })
317    })
318}

320// ----------------------------------------------------------------------------
321// HybridKemPublicKey (0x01)
322// ----------------------------------------------------------------------------
323
324impl sealed::Sealed for HybridKemPublicKey {}
325
326impl WireEncode for HybridKemPublicKey {
327    const TYPE_TAG: u8 = 0x01;
328    const MAX_WIRE_LEN: usize =
329        WIRE_HEADER_LEN + MlKemSecurityLevel::MlKem1024.public_key_size() + X25519_KEY_SIZE;
330
331    fn to_wire(&self) -> Result<Vec<u8>, WireError> {
332        let level = self.security_level();
333        require_len(self.ml_kem_pk(), level.public_key_size(), "ML-KEM public key")?;
334        require_len(self.ecdh_pk(), X25519_KEY_SIZE, "X25519 public key")?;
335        let body_len = sum_lens(&[self.ml_kem_pk().len(), self.ecdh_pk().len()])?;
336        let mut out = begin(Self::TYPE_TAG, kem_level_id(level), body_len)?;
337        out.extend_from_slice(self.ml_kem_pk());
338        out.extend_from_slice(self.ecdh_pk());
339        Ok(out)
340    }
341
342    fn from_wire(bytes: &[u8]) -> Result<Self, WireError> {
343        let (alg_id, body) = parse_header(bytes, Self::TYPE_TAG, Self::MAX_WIRE_LEN)?;
344        let level = kem_level_from_id(alg_id)?;
345        let mut reader = Reader::new(body);
346        let ml_kem_pk = reader.take(level.public_key_size())?;
347        let ecdh_pk = reader.take(X25519_KEY_SIZE)?;
348        reader.finish()?;
349        Ok(Self::new(ml_kem_pk.to_vec(), ecdh_pk.to_vec(), level))
350    }
351}

359/// Encodes only the public encapsulation (ML-KEM ciphertext and ephemeral
360/// X25519 public key). The shared secret is never written; decoding yields
361/// a value built with [`EncapsulatedKey::for_decapsulate`].
362impl WireEncode for EncapsulatedKey {
363    const TYPE_TAG: u8 = 0x02;
364    const MAX_WIRE_LEN: usize =
365        WIRE_HEADER_LEN + MlKemSecurityLevel::MlKem1024.ciphertext_size() + X25519_KEY_SIZE;
366
367    fn to_wire(&self) -> Result<Vec<u8>, WireError> {
368        let level = kem_level_from_ciphertext_len(self.ml_kem_ct().len())?;
369        require_len(self.ecdh_pk(), X25519_KEY_SIZE, "X25519 ephemeral public key")?;
370        let body_len = sum_lens(&[self.ml_kem_ct().len(), self.ecdh_pk().len()])?;
371        let mut out = begin(Self::TYPE_TAG, kem_level_id(level), body_len)?;
372        out.extend_from_slice(self.ml_kem_ct());
373        out.extend_from_slice(self.ecdh_pk());
374        Ok(out)
375    }
376
377    fn from_wire(bytes: &[u8]) -> Result<Self, WireError> {
378        let (alg_id, body) = parse_header(bytes, Self::TYPE_TAG, Self::MAX_WIRE_LEN)?;
379        let level = kem_level_from_id(alg_id)?;
380        let mut reader = Reader::new(body);
381        let ml_kem_ct = reader.take(level.ciphertext_size())?;
382        let ecdh_pk = reader.take(X25519_KEY_SIZE)?;
383        reader.finish()?;
384        Ok(Self::for_decapsulate(ml_kem_ct.to_vec(), ecdh_pk.to_vec()))
385    }
386}

388// ----------------------------------------------------------------------------
389// HybridCiphertext (0x03)
390// ----------------------------------------------------------------------------
391
392impl sealed::Sealed for HybridCiphertext {}
393
394impl WireEncode for HybridCiphertext {
395    const TYPE_TAG: u8 = 0x03;
396    const MAX_WIRE_LEN: usize = WIRE_HEADER_LEN
397        + MlKemSecurityLevel::MlKem1024.ciphertext_size()
398        + X25519_KEY_SIZE
399        + NONCE_LEN
400        + TAG_LEN
401        + SYMMETRIC_LEN_PREFIX
402        + MAX_SYMMETRIC_CIPHERTEXT_LEN;
403
404    fn to_wire(&self) -> Result<Vec<u8>, WireError> {
405        let level = kem_level_from_ciphertext_len(self.kem_ciphertext().len())?;
406        require_len(self.ecdh_ephemeral_pk(), X25519_KEY_SIZE, "X25519 ephemeral public key")?;
407        require_len(self.nonce(), NONCE_LEN, "AES-GCM nonce")?;
408        require_len(self.tag(), TAG_LEN, "AES-GCM tag")?;
409        let sym_len = self.symmetric_ciphertext().len();
410        if sym_len > MAX_SYMMETRIC_CIPHERTEXT_LEN {
411            return Err(WireError::TooLarge { len: sym_len, max: MAX_SYMMETRIC_CIPHERTEXT_LEN });
412        }
413        let prefix = u32::try_from(sym_len).map_err(|_e| WireError::TooLarge {
414            len: sym_len,
415            max: MAX_SYMMETRIC_CIPHERTEXT_LEN,
416        })?;
417        let body_len = sum_lens(&[
418            self.kem_ciphertext().len(),
419            X25519_KEY_SIZE,
420            NONCE_LEN,
421            TAG_LEN,
422            SYMMETRIC_LEN_PREFIX,
423            sym_len,
424        ])?;
425        let mut out = begin(Self::TYPE_TAG, kem_level_id(level), body_len)?;
426        out.extend_from_slice(self.kem_ciphertext());
427        out.extend_from_slice(self.ecdh_ephemeral_pk());
428        out.extend_from_slice(self.nonce());
429        out.extend_from_slice(self.tag());
430        out.extend_from_slice(&prefix.to_be_bytes());
431        out.extend_from_slice(self.symmetric_ciphertext());
432        Ok(out)
433    }
434
435    fn from_wire(bytes: &[u8]) -> Result<Self, WireError> {
436        let (alg_id, body) = parse_header(bytes, Self::TYPE_TAG, Self::MAX_WIRE_LEN)?;
437        let level = kem_level_from_id(alg_id)?;
438        let mut reader = Reader::new(body);
439        let kem_ct = reader.take(level.ciphertext_size())?;
440        let ecdh_pk = reader.take(X25519_KEY_SIZE)?;
441        let nonce = reader.take(NONCE_LEN)?;
442        let tag = reader.take(TAG_LEN)?;
443        let sym_len = usize::try_from(reader.take_u32_be()?).map_err(|_e| WireError::TooLarge {
444            len: usize::MAX,
445            max: MAX_SYMMETRIC_CIPHERTEXT_LEN,
446        })?;
447        if sym_len > MAX_SYMMETRIC_CIPHERTEXT_LEN {
448            return Err(WireError::TooLarge { len: sym_len, max: MAX_SYMMETRIC_CIPHERTEXT_LEN });
449        }
450        // `take` fails before any copy if fewer than `sym_len` bytes remain,
451        // so the allocation below is bounded by the supplied input.
452        let symmetric = reader.take(sym_len)?;
453        reader.finish()?;
454        Ok(Self::new(
455            kem_ct.to_vec(),
456            ecdh_pk.to_vec(),
457            symmetric.to_vec(),
458            nonce.to_vec(),
459            tag.to_vec(),
460        ))
461    }
462}

465// HybridSigPublicKey (0x04)
466// ----------------------------------------------------------------------------
467
468impl sealed::Sealed for HybridSigPublicKey {}
469
470impl WireEncode for HybridSigPublicKey {
471    const TYPE_TAG: u8 = 0x04;
472    const MAX_WIRE_LEN: usize =
473        WIRE_HEADER_LEN + MlDsaParameterSet::MlDsa87.public_key_size() + ED25519_PUBLIC_KEY_LEN;
474
475    fn to_wire(&self) -> Result<Vec<u8>, WireError> {
476        let set = self.parameter_set();
477        require_len(self.ml_dsa_pk(), set.public_key_size(), "ML-DSA public key")?;
478        require_len(self.ed25519_pk(), ED25519_PUBLIC_KEY_LEN, "Ed25519 public key")?;
479        let body_len = sum_lens(&[self.ml_dsa_pk().len(), self.ed25519_pk().len()])?;
480        let mut out = begin(Self::TYPE_TAG, dsa_set_id(set), body_len)?;
481        out.extend_from_slice(self.ml_dsa_pk());
482        out.extend_from_slice(self.ed25519_pk());
483        Ok(out)
484    }
485
486    fn from_wire(bytes: &[u8]) -> Result<Self, WireError> {
487        let (alg_id, body) = parse_header(bytes, Self::TYPE_TAG, Self::MAX_WIRE_LEN)?;
488        let set = dsa_set_from_id(alg_id)?;
489        let mut reader = Reader::new(body);
490        let ml_dsa_pk = reader.take(set.public_key_size())?;
491        let ed25519_pk = reader.take(ED25519_PUBLIC_KEY_LEN)?;
492        reader.finish()?;
493        // Lengths were fixed by `set` above, so `new`'s length validation
494        // cannot fail; map defensively instead of unwrapping.
495        Self::new(set, ml_dsa_pk.to_vec(), ed25519_pk.to_vec())
496            .map_err(|_e| WireError::InvalidComponent("ML-DSA or Ed25519 public key"))
497    }
498}

506/// `HybridSignature` does not store its parameter set; the encoder derives
507/// the algorithm id from the ML-DSA signature length.
508impl WireEncode for HybridSignature {
509    const TYPE_TAG: u8 = 0x05;
510    const MAX_WIRE_LEN: usize =
511        WIRE_HEADER_LEN + MlDsaParameterSet::MlDsa87.signature_size() + ED25519_SIGNATURE_LEN;
512
513    fn to_wire(&self) -> Result<Vec<u8>, WireError> {
514        let set = dsa_set_from_signature_len(self.ml_dsa_sig().len())?;
515        require_len(self.ed25519_sig(), ED25519_SIGNATURE_LEN, "Ed25519 signature")?;
516        let body_len = sum_lens(&[self.ml_dsa_sig().len(), self.ed25519_sig().len()])?;
517        let mut out = begin(Self::TYPE_TAG, dsa_set_id(set), body_len)?;
518        out.extend_from_slice(self.ml_dsa_sig());
519        out.extend_from_slice(self.ed25519_sig());
520        Ok(out)
521    }
522
523    fn from_wire(bytes: &[u8]) -> Result<Self, WireError> {
524        let (alg_id, body) = parse_header(bytes, Self::TYPE_TAG, Self::MAX_WIRE_LEN)?;
525        let set = dsa_set_from_id(alg_id)?;
526        let mut reader = Reader::new(body);
527        let ml_dsa_sig = reader.take(set.signature_size())?;
528        let ed25519_sig = reader.take(ED25519_SIGNATURE_LEN)?;
529        reader.finish()?;
530        Ok(Self::new(ml_dsa_sig.to_vec(), ed25519_sig.to_vec()))
531    }

551    /// Deterministic filler: `b = b * 29 + 7 (mod 256)`, emitting each `b`.
552    fn filled(len: usize, seed: u8) -> Vec<u8> {
553        let mut b = seed;
554        (0..len)
555            .map(|_| {
556                b = b.wrapping_mul(29).wrapping_add(7);
557                b
558            })
559            .collect()
560    }

596    /// Every structural mutation of a valid encoding must be rejected with
597    /// the matching error.
598    fn assert_rejections<T: WireEncode>(enc: &[u8]) {
599        assert!(T::from_wire(enc).is_ok());
600
601        for cut in 0..enc.len() {
602            assert!(
603                matches!(T::from_wire(&enc[..cut]), Err(WireError::Truncated)),
604                "prefix of {cut}/{} bytes must be Truncated",
605                enc.len()
606            );
607        }
608
609        let mut extended = enc.to_vec();
610        extended.push(0);
611        let result = T::from_wire(&extended);
612        if enc.len() == T::MAX_WIRE_LEN {
613            assert!(matches!(result, Err(WireError::TooLarge { .. })));
614        } else {
615            assert!(matches!(result, Err(WireError::TrailingBytes)));
616        }
617
618        for magic_index in 0..2 {
619            let mut bad = enc.to_vec();
620            bad[magic_index] ^= 0x20;
621            assert!(matches!(T::from_wire(&bad), Err(WireError::BadMagic)));
622        }
623
624        for version in [0u8, 2, 0x7F, 0xFF] {
625            let mut bad = enc.to_vec();
626            bad[2] = version;
627            assert!(matches!(
628                T::from_wire(&bad),
629                Err(WireError::UnsupportedVersion(v)) if v == version
630            ));
631        }
632
633        for tag in 0..=u8::MAX {
634            if tag == T::TYPE_TAG {
635                continue;
636            }
637            let mut bad = enc.to_vec();
638            bad[3] = tag;
639            assert!(matches!(
640                T::from_wire(&bad),
641                Err(WireError::WrongType { expected, found })
642                    if expected == T::TYPE_TAG && found == tag
643            ));
644        }
645
646        for alg in 0..=u8::MAX {
647            if valid_alg_ids::<T>().contains(&alg) {
648                continue;
649            }
650            let mut bad = enc.to_vec();
651            bad[4] = alg;
652            assert!(matches!(T::from_wire(&bad), Err(WireError::UnknownAlgorithm(a)) if a == alg));
653        }
654    }
