12/// `pub(super)`: called from `zero_trust::auth::ZeroTrustAuth::generate_challenge`.
13pub(super) fn generate_challenge_data(complexity: &ProofComplexity) -> Result<Vec<u8>> {
14    let size = match complexity {
15        ProofComplexity::Low => 32,
16        ProofComplexity::Medium => 64,
17        ProofComplexity::High => 128,
18    };
19
20    let data = crate::primitives::security::generate_secure_random_bytes(size).map_err(|e| {
21        CoreError::EntropyDepleted {
22            message: format!("Failed to generate challenge: {e}"),
23            action: "Check system entropy source".to_string(),
24        }
25    })?;
26    // Challenge data is sent over the wire and is not secret material; the
27    // ephemeral `Zeroizing` wrapper is dropped (and the source bytes wiped)
28    // when this function returns.
29    Ok(data.to_vec())
30}

32impl ZeroTrustAuth {
33    /// Compute proof data using Ed25519 signature-based challenge-response.
34    ///
35    /// This is a proper zero-knowledge proof: the signature proves knowledge
36    /// of the private key without revealing any information about it.
37    ///
38    /// All three complexity levels sign the same transcript shape:
39    ///
40    /// ```text
41    /// domain_tag || challenge || timestamp || public_key
42    /// ```
43    ///
44    /// `proof_complexity` selects only the 1-byte leading domain tag —
45    /// `0x01` for `Low`, `0x02` for `Medium`, `0x03` for `High` — which
46    /// separates the levels so a proof produced for one cannot satisfy a
47    /// verifier expecting another. The timestamp and public-key bindings are
48    /// unconditional; see the comment on the `match` below for why they must
49    /// not be complexity-dependent.
50    pub(super) fn compute_proof_data(&self, challenge: &[u8]) -> Result<Vec<u8>> {
51        let timestamp = Utc::now().timestamp_millis().to_le_bytes();
52
53        // All complexity levels bind the timestamp into the signed
54        // message so the verifier can enforce a freshness window. The
55        // previous Low variant signed only `challenge`, making
56        // `(challenge, signature)` pairs replayable indefinitely
57        // until session expiry. Replay
58        // protection is no longer opt-in.
59        // Every variant binds the public key into the signed transcript
60        // so a verifier holding pk can be sure the proof was produced
61        // *for that pk* and not relayed from a session under a different
62        // identity. Without PK binding, an attacker who captured a
63        // valid (challenge, signature) pair under pk_A could replay it
64        // against a verifier expecting pk_B if the signature happened
65        // to verify under pk_B too (catastrophic if a verifier holds
66        // multiple registered keys). The 1-byte domain tag separates
67        // the three complexity levels so a Low proof cannot satisfy a
68        // Medium or High verifier and vice versa.
69        let message_to_sign = match self.config.proof_complexity {
70            ProofComplexity::Low => {
71                let mut msg = vec![0x01];
72                msg.extend_from_slice(challenge);
73                msg.extend_from_slice(&timestamp);
74                msg.extend_from_slice(self.public_key.as_slice());
75                msg
76            }
77            ProofComplexity::Medium => {
78                let mut msg = vec![0x02];
79                msg.extend_from_slice(challenge);
80                msg.extend_from_slice(&timestamp);
81                msg.extend_from_slice(self.public_key.as_slice());
82                msg
83            }
84            ProofComplexity::High => {
85                let mut msg = vec![0x03];
86                msg.extend_from_slice(challenge);
87                msg.extend_from_slice(&timestamp);
88                msg.extend_from_slice(self.public_key.as_slice());
89                msg
90            }
91        };
92
93        // Sign the message - this IS zero-knowledge
94        // The signature proves knowledge of private key without revealing it
95        //
96        // M-A: bind ZK-proof signatures to a ZK-proof-specific Ed25519
97        // context. The internal 0x01/0x02/0x03 complexity tags inside
98        // `message_to_sign` keep Low/Medium/High distinguishable within
99        // this protocol; the SHA-512 prefix-padded context here keeps a
100        // captured ZK proof from replaying as a SignedData or PoP signed
101        // by the same Ed25519 key. Mirrors the post-M-A pure-Ed25519
102        // construction and the hybrid leg.
103        use crate::types::domains::{hash_with_context, zk_proof_sig_context};
104        let zk_digest = hash_with_context(zk_proof_sig_context(), &message_to_sign);
105
106        let signature = crate::unified_api::convenience::ed25519::sign_ed25519_internal(
107            &zk_digest,
108            self.private_key.expose_secret(),
109        )?;
110
111        // Append the timestamp to the proof bytes so the verifier can
112        // recover it. All three complexity levels now carry a timestamp
113        // suffix; older proofs without the
114        // suffix will fail length check at verify time.
115        let mut proof = signature;
116        proof.extend_from_slice(&timestamp);
117        Ok(proof)
118    }

120    /// Verify proof using PUBLIC KEY only.
121    ///
122    /// This is the correct way to verify a zero-knowledge proof:
123    /// only the public key is needed, not the private key.
124    pub(super) fn verify_proof_data(&self, proof: &[u8], challenge: &[u8]) -> Result<bool> {
125        // Ed25519 signatures are 64 bytes
126        if proof.len() < 64 {
127            return Ok(false);
128        }
129
130        match self.config.proof_complexity {
131            ProofComplexity::Low => {
132                // Low now requires the
133                // 8-byte timestamp suffix (mandatory replay protection).
134                if proof.len() < 72 {
135                    return Ok(false);
136                }
137                let signature = proof.get(..64).ok_or_else(|| {
138                    CoreError::AuthenticationFailed("Invalid proof format".to_string())
139                })?;
140                let timestamp_slice = proof.get(64..72).ok_or_else(|| {
141                    CoreError::AuthenticationFailed("Invalid proof format".to_string())
142                })?;
143                let timestamp_bytes: [u8; 8] = timestamp_slice.try_into().map_err(|_e| {
144                    CoreError::AuthenticationFailed("Invalid proof format".to_string())
145                })?;
146
147                // verify signature BEFORE applying the
148                // freshness check. The original ordering parsed the
149                // timestamp out of `proof[64..72]` and ran the 30-s/
150                // 5-min checks on raw adversary-supplied bytes; even
151                // though signature verification would still catch a
152                // tampered timestamp downstream, the principle
153                // "authenticate before acting on adversary content"
154                // wants verify first. Mirrored across Medium / High
155                // branches below.
156                let mut message = vec![0x01];
157                message.extend_from_slice(challenge);
158                message.extend_from_slice(&timestamp_bytes);
159                message.extend_from_slice(self.public_key.as_slice());
160                // M-A: mirror generate_proof_data's
161                // SHA-512(zk_ctx || 0x00 || message) construction.
162                use crate::types::domains::{hash_with_context, zk_proof_sig_context};
163                let zk_digest = hash_with_context(zk_proof_sig_context(), &message);
164                let sig_ok = crate::unified_api::convenience::ed25519::verify_ed25519_internal(
165                    &zk_digest,
166                    signature,
167                    self.public_key.as_slice(),
168                )?;
169                if !sig_ok {
170                    return Ok(false);
171                }
172
173                // Timestamp is now authenticated — freshness check is
174                // safe to run on `timestamp_bytes`.
175                let proof_ts_ms = i64::from_le_bytes(timestamp_bytes);
176                let now_ms = Utc::now().timestamp_millis();
177                // tighten the future-skew cap
178                // to match the sibling `verify_pop` path. `abs_diff(...)
179                // > 300_000` accepted proofs up to 5 min in the future,
180                // which gives an attacker with a forward-skewed clock a
181                // 10-min replay window. Reject anything more than 30 s
182                // ahead of "now"; the 5-min window only applies to the
183                // past direction.
184                if proof_ts_ms > now_ms.saturating_add(30_000) {
185                    tracing::warn!(
186                        proof_ts_ms,
187                        now_ms,
188                        "proof timestamp more than 30 s in the future"
189                    );
190                    return Ok(false);
191                }
192                let drift_ms = now_ms.abs_diff(proof_ts_ms);
193                if drift_ms > 300_000 {
194                    tracing::warn!(drift_ms, "proof timestamp outside 5-min freshness window");
195                    return Ok(false);
196                }
197
198                Ok(true)
199            }
200            ProofComplexity::Medium => {
201                // Extract signature and timestamp
202                if proof.len() < 72 {
203                    return Ok(false);
204                }
205                let signature = proof.get(..64).ok_or_else(|| {
206                    CoreError::AuthenticationFailed("Invalid proof format".to_string())
207                })?;
208                let timestamp_slice = proof.get(64..72).ok_or_else(|| {
209                    CoreError::AuthenticationFailed("Invalid proof format".to_string())
210                })?;
211                let timestamp_bytes: [u8; 8] = timestamp_slice.try_into().map_err(|_e| {
212                    CoreError::AuthenticationFailed("Invalid proof format".to_string())
213                })?;
214
215                // verify before freshness (mirrors Low).
216                // Reconstruct signed message. The 0x02 domain tag
217                // distinguishes Medium from Low/High; the public-key
218                // suffix binds the signature to this specific
219                // verifier identity (HPKE-style channel binding).
220                let mut message = vec![0x02];
221                message.extend_from_slice(challenge);
222                message.extend_from_slice(&timestamp_bytes);
223                message.extend_from_slice(self.public_key.as_slice());
224                // M-A: mirror generate_proof_data's
225                // SHA-512(zk_ctx || 0x00 || message) construction.
226                use crate::types::domains::{hash_with_context, zk_proof_sig_context};
227                let zk_digest = hash_with_context(zk_proof_sig_context(), &message);
228                let sig_ok = crate::unified_api::convenience::ed25519::verify_ed25519_internal(
229                    &zk_digest,
230                    signature,
231                    self.public_key.as_slice(),
232                )?;
233                if !sig_ok {
234                    return Ok(false);
235                }
236
237                // Reject stale proofs (>5 min drift).
238                // Timestamp is encoded as chrono milliseconds in little-endian.
239                let proof_ts_ms = i64::from_le_bytes(timestamp_bytes);
240                let now_ms = Utc::now().timestamp_millis();
241                // tighten the future-skew cap
242                // to match the sibling `verify_pop` path. `abs_diff(...)
243                // > 300_000` accepted proofs up to 5 min in the future,
244                // which gives an attacker with a forward-skewed clock a
245                // 10-min replay window. Reject anything more than 30 s
246                // ahead of "now"; the 5-min window only applies to the
247                // past direction.
248                if proof_ts_ms > now_ms.saturating_add(30_000) {
249                    tracing::warn!(
250                        proof_ts_ms,
251                        now_ms,
252                        "proof timestamp more than 30 s in the future"
253                    );
254                    return Ok(false);
255                }
256                let drift_ms = now_ms.abs_diff(proof_ts_ms);
257                if drift_ms > 300_000 {
258                    tracing::warn!(drift_ms, "proof timestamp outside 5-min freshness window");
259                    return Ok(false);
260                }
261
262                Ok(true)
263            }
264            ProofComplexity::High => {
265                // Extract signature and timestamp
266                if proof.len() < 72 {
267                    return Ok(false);
268                }
269                let signature = proof.get(..64).ok_or_else(|| {
270                    CoreError::AuthenticationFailed("Invalid proof format".to_string())
271                })?;
272                let timestamp_slice = proof.get(64..72).ok_or_else(|| {
273                    CoreError::AuthenticationFailed("Invalid proof format".to_string())
274                })?;
275                let timestamp_bytes: [u8; 8] = timestamp_slice.try_into().map_err(|_e| {
276                    CoreError::AuthenticationFailed("Invalid proof format".to_string())
277                })?;
278
279                // verify before freshness (mirrors Low).
280                // Reconstruct signed message with public key binding.
281                // High prepends a 0x03 domain
282                // tag so it is byte-distinguishable from both Low and
283                // Medium.
284                let mut message = vec![0x03];
285                message.extend_from_slice(challenge);
286                message.extend_from_slice(&timestamp_bytes);
287                message.extend_from_slice(self.public_key.as_slice());
288                // M-A: mirror generate_proof_data's
289                // SHA-512(zk_ctx || 0x00 || message) construction.
290                use crate::types::domains::{hash_with_context, zk_proof_sig_context};
291                let zk_digest = hash_with_context(zk_proof_sig_context(), &message);
292                let sig_ok = crate::unified_api::convenience::ed25519::verify_ed25519_internal(
293                    &zk_digest,
294                    signature,
295                    self.public_key.as_slice(),
296                )?;
297                if !sig_ok {
298                    return Ok(false);
299                }
300
301                // Reject stale proofs (>5 min drift).
302                // Timestamp is encoded as chrono milliseconds in little-endian.
303                let proof_ts_ms = i64::from_le_bytes(timestamp_bytes);
304                let now_ms = Utc::now().timestamp_millis();
305                // tighten the future-skew cap
306                // to match the sibling `verify_pop` path. `abs_diff(...)
307                // > 300_000` accepted proofs up to 5 min in the future,
308                // which gives an attacker with a forward-skewed clock a
309                // 10-min replay window. Reject anything more than 30 s
310                // ahead of "now"; the 5-min window only applies to the
311                // past direction.
312                if proof_ts_ms > now_ms.saturating_add(30_000) {
313                    tracing::warn!(
314                        proof_ts_ms,
315                        now_ms,
316                        "proof timestamp more than 30 s in the future"
317                    );
318                    return Ok(false);
319                }
320                let drift_ms = now_ms.abs_diff(proof_ts_ms);
321                if drift_ms > 300_000 {
322                    tracing::warn!(drift_ms, "proof timestamp outside 5-min freshness window");
323                    return Ok(false);
324                }
325
326                Ok(true)
327            }
328        }
329    }
