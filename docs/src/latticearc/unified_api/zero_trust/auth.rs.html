26/// Zero-trust authentication handler.
27///
28/// Manages challenge-response authentication, proof generation and verification,
29/// and continuous session monitoring.
30pub struct ZeroTrustAuth {
31    /// Public key for verification.
32    pub(crate) public_key: PublicKey,
33    /// Private key for proof generation.
34    ///
35    /// `pub(super)`: read by the `compute_proof_data` /
36    /// `verify_proof_data` helpers in `zero_trust::proof_data`.
37    pub(super) private_key: PrivateKey,
38    /// Authentication configuration.
39    ///
40    /// `pub(super)`: read by `zero_trust::proof_data` (proof-complexity
41    /// dispatch) in addition to this module's own trait impls.
42    pub(super) config: ZeroTrustConfig,
43    /// Session start timestamp.
44    session_start: DateTime<Utc>,
45    /// Timestamp of the most recent successful proof verification, or
46    /// `None` if no proof has yet been verified for this auth instance.
47    ///
48    /// The `None` ↔ `Some` boundary is the precondition gate for
49    /// [`Self::start_continuous_verification`]: starting a continuous
50    /// session before any successful challenge-response would let a
51    /// fresh `ZeroTrustAuth` skip the ZKP handshake entirely.
52    ///
53    /// `Mutex` (not `RefCell`) so the type stays `Sync` for use across
54    /// async tasks and multi-threaded callers; an `Arc<ZeroTrustAuth>`
55    /// shared between Tokio workers must compile.
56    last_verification: Mutex<Option<DateTime<Utc>>>,
57    /// Replay cache for proofs-of-possession.
58    ///
59    /// Keyed on `pk || sig || ts_micros_be` (microsecond precision —
60    /// see the M16 follow-up note in `generate_pop` for why second
61    /// precision was insufficient under Ed25519's deterministic
62    /// signatures). Values are the wall-clock seconds-since-epoch at
63    /// which the entry was inserted (used to bound cache lifetime to
64    /// the 5-minute PoP freshness window).
65    ///
66    /// Carries a parallel `per_pk_counts` map so the per-PK quota check
67    /// is O(1). Without it, the global cap alone would let a single
68    /// noisy public key fill the cache and lock out every other PK
69    /// (DoS against legitimate verifiers); with it, each PK is bounded
70    /// to `POP_CACHE_PER_PK_MAX` entries and other PKs proceed normally.
71    ///
72    /// `Mutex` (not `RefCell`) so the type stays `Sync` for use across
73    /// async/multi-threaded callers.
74    pop_replay_cache: Mutex<PopReplayCache>,
75}

77/// Internal helper: a replay cache that tracks both the full
78/// `(pk || sig || ts_micros)` entries and a per-PK count alongside.
79/// The two maps are mutated together under the surrounding `Mutex` so
80/// the per-PK count is always exact, never a stale approximation.
81///
82/// `pk_len` is fixed at construction (currently
83/// [`ED25519_PUBLIC_KEY_LEN`]) and is the authoritative length of the
84/// public-key prefix in every entry key. Mixing entries with different
85/// PK lengths would break the inverse-of-insert assumption in
86/// `expire_older_than` (the prefix slice would misidentify which entry
87/// belongs to which PK), silently leak per-PK quota state, and degrade
88/// the M2 fail-closed replay invariant. The `pk_len` field plus the
89/// debug-assert in [`Self::insert`] enforce uniformity at the
90/// API boundary so a future PoP type with a different PK length cannot
91/// silently coexist with the current one — adding such a type must
92/// either create a separate cache instance or extend the entry-key
93/// encoding to disambiguate.
94struct PopReplayCache {
95    /// Authoritative public-key prefix length. Set once at
96    /// construction; every `insert` debug-asserts that the supplied
97    /// `pk_bytes` length matches this value.
98    pk_len: usize,
99    /// Full keys (PK || sig || ts_micros) → insertion timestamp (seconds).
100    entries: std::collections::HashMap<Vec<u8>, i64>,
101    /// Public-key bytes → count of `entries` belonging to that PK.
102    /// Maintained in lockstep with `entries`; never read independently.
103    per_pk_counts: std::collections::HashMap<Vec<u8>, usize>,
104}
105
106impl PopReplayCache {
107    /// Construct an empty cache for entries whose PK prefix is exactly
108    /// `pk_len` bytes. Today the only caller passes
109    /// [`ED25519_PUBLIC_KEY_LEN`]; if a future PoP type uses a
110    /// different PK length, give it its own `PopReplayCache` rather
111    /// than mixing prefixes here.
112    fn new(pk_len: usize) -> Self {
113        Self {
114            pk_len,
115            entries: std::collections::HashMap::new(),
116            per_pk_counts: std::collections::HashMap::new(),
117        }
118    }
119
120    /// Remove entries older than `max_age_secs` and update per-PK counts
121    /// to match. The leading `self.pk_len` bytes of every entry key are
122    /// the public-key bytes; this is the inverse of how `insert` builds
123    /// the key, and is only well-defined because `insert` debug-asserts
124    /// every inserted `pk_bytes` matches `self.pk_len`.
125    fn expire_older_than(&mut self, now_secs: i64, max_age_secs: i64) {
126        let pk_len = self.pk_len;
127        let entries = &mut self.entries;
128        let per_pk = &mut self.per_pk_counts;
129        entries.retain(|key, ts| {
130            let fresh = now_secs.saturating_sub(*ts) <= max_age_secs;
131            if !fresh
132                && let Some(pk_bytes) = key.get(..pk_len)
133                && let Some(count) = per_pk.get_mut(pk_bytes)
134            {
135                *count = count.saturating_sub(1);
136                if *count == 0 {
137                    per_pk.remove(pk_bytes);
138                }
139            }
140            fresh
141        });
142    }
143
144    fn contains(&self, key: &[u8]) -> bool {
145        self.entries.contains_key(key)
146    }
147
148    fn count_for_pk(&self, pk_bytes: &[u8]) -> usize {
149        self.per_pk_counts.get(pk_bytes).copied().unwrap_or(0)
150    }
151
152    fn total(&self) -> usize {
153        self.entries.len()
154    }
155
156    fn insert(&mut self, pk_bytes: &[u8], full_key: Vec<u8>, ts_secs: i64) {
157        // Guards the `pk_len` uniformity invariant the rest of this
158        // type relies on. A mismatch here means the caller mixed PoP
159        // types into one cache; the per-PK count would key off a
160        // wrong-length prefix and `expire_older_than` would never
161        // decrement it, silently leaking quota. Dev/CI surface only —
162        // production stays branch-free in the hot path.
163        debug_assert_eq!(
164            pk_bytes.len(),
165            self.pk_len,
166            "PopReplayCache: pk_bytes.len()={} does not match cache pk_len={}; \
167             mixing PoP key types in one cache silently breaks per-PK quota tracking",
168            pk_bytes.len(),
169            self.pk_len,
170        );
171        if self.entries.insert(full_key, ts_secs).is_none() {
172            let slot = self.per_pk_counts.entry(pk_bytes.to_vec()).or_insert(0);
173            *slot = slot.saturating_add(1);
174        }
175    }
176}

230    /// Generate a new challenge for authentication.
231    ///
232    /// # Errors
233    /// Returns `CoreError::EntropyDepleted` if the system cannot generate random bytes.
234    pub fn generate_challenge(&self) -> Result<Challenge> {
235        let challenge_data = generate_challenge_data(&self.config.proof_complexity)?;
236        log_zero_trust_challenge_generated!(self.config.proof_complexity);
237        Ok(Challenge {
238            data: challenge_data,
239            timestamp: Utc::now(),
240            complexity: self.config.proof_complexity.clone(),
241            timeout_ms: self.config.challenge_timeout_ms,
242        })
243    }
244
245    /// Verifies whether a challenge is still within its timeout period.
246    ///
247    /// # Errors
248    ///
249    /// This function does not currently return errors, but returns `Result` for
250    /// API consistency and future extensibility.
251    pub fn verify_challenge_age(&self, challenge: &Challenge) -> Result<bool> {
252        let elapsed = Utc::now().signed_duration_since(challenge.timestamp());
253        let elapsed_ms = elapsed.num_milliseconds();
254
255        // Negative elapsed time means the challenge is from the future, which is invalid
256        // Convert safely: negative values are treated as invalid (elapsed > timeout)
257        let elapsed_u64 = u64::try_from(elapsed_ms).unwrap_or(u64::MAX);
258        Ok(elapsed_u64 <= challenge.timeout_ms())
259    }
260
261    /// Starts a new continuous verification session.
262    ///
263    /// # Errors
264    ///
265    /// Returns `CoreError::AuthenticationFailed` if no successful
266    /// challenge-response has occurred for this `ZeroTrustAuth`
267    /// instance. Continuous verification is supposed to *extend* a
268    /// trust relationship that was already established cryptographically;
269    /// without this gate, a freshly-constructed `ZeroTrustAuth` could
270    /// hand out a `ContinuousSession` whose `is_valid()` returns true,
271    /// bypassing the ZKP handshake entirely.
272    pub fn start_continuous_verification(&self) -> Result<ContinuousSession> {
273        let last_verified = match self.last_verification.lock() {
274            Ok(g) => *g,
275            Err(poisoned) => {
276                tracing::warn!(
277                    target: "latticearc::unified_api::zero_trust",
278                    "ZeroTrustAuth.last_verification mutex was poisoned in \
279                     start_continuous_verification; a previous writer panicked. \
280                     Recovering inner guard."
281                );
282                *poisoned.into_inner()
283            }
284        };
285        let last_verified = last_verified.ok_or_else(|| {
286            CoreError::AuthenticationFailed(
287                "start_continuous_verification requires a prior successful proof verification \
288                 (call verify_proof() against a fresh challenge first)"
289                    .to_string(),
290            )
291        })?;
292        Ok(ContinuousSession {
293            auth_public_key: self.public_key.clone(),
294            start_time: Utc::now(),
295            verification_interval_ms: self.config.verification_interval_ms,
296            last_verification: last_verified,
297        })
298    }

301impl ZeroTrustAuthenticable for ZeroTrustAuth {
302    type Proof = ZeroKnowledgeProof;
303    type Error = CoreError;
304
305    /// Generates a zero-knowledge proof for the given challenge.
306    ///
307    /// # Errors
308    ///
309    /// Returns `CoreError::AuthenticationFailed` if the challenge is empty.
310    ///
311    /// Returns `CoreError::InvalidKeyLength` if the private key has incorrect length
312    /// for Ed25519 signing.
313    ///
314    /// Returns `CoreError::InvalidInput` if the private key format is invalid.
315    fn generate_proof(&self, challenge: &[u8]) -> Result<Self::Proof> {
316        if challenge.is_empty() {
317            return Err(CoreError::AuthenticationFailed("Empty challenge".to_string()));
318        }
319
320        let proof_data = self.compute_proof_data(challenge)?;
321        let timestamp = Utc::now();
322
323        Ok(ZeroKnowledgeProof {
324            challenge: challenge.to_vec(),
325            proof: proof_data,
326            timestamp,
327            complexity: self.config.proof_complexity.clone(),
328        })
329    }
330
331    /// Verifies a zero-knowledge proof against the given challenge.
332    ///
333    /// # Errors
334    ///
335    /// Returns `CoreError::AuthenticationFailed` if the proof format is invalid
336    /// (e.g., timestamp bytes cannot be extracted for Medium/High complexity proofs).
337    ///
338    /// Returns `CoreError::InvalidInput` if the public key or signature format is invalid
339    /// during Ed25519 verification.
340    ///
341    /// Returns `CoreError::InvalidKeyLength` if the public key has incorrect length.
342    fn verify_proof(
343        &self,
344        proof: &Self::Proof,
345        challenge: &[u8],
346    ) -> std::result::Result<bool, Self::Error> {
347        // The prover writes `proof.complexity` from its own config but
348        // the verifier previously only consulted
349        // `self.config.proof_complexity` for dispatch — so a prover
350        // claiming Low complexity could ship a proof verified under
351        // the verifier's Medium policy (or vice versa). Pin the field
352        // by requiring it to match the verifier's expected complexity;
353        // mismatch collapses to `Ok(false)` (Pattern 6 — no
354        // distinguishable error per audit posture).
355        if proof.complexity() != &self.config.proof_complexity {
356            tracing::debug!(
357                expected = ?self.config.proof_complexity,
358                got = ?proof.complexity(),
359                "ZK proof rejected: complexity field mismatch"
360            );
361            log_zero_trust_proof_verified!(false);
362            return Ok(false);
363        }
364
365        // SECURITY: Use constant-time comparison to prevent timing attacks
366        // An attacker should not be able to determine which bytes of the challenge matched
367        let len_eq = proof.challenge().len().ct_eq(&challenge.len());
368        let content_eq = proof.challenge().ct_eq(challenge);
369        let challenge_matches: bool = (len_eq & content_eq).into();
370
371        if !challenge_matches {
372            log_zero_trust_proof_verified!(false);
373            return Ok(false);
374        }
375
376        let result = self.verify_proof_data(proof.proof_data(), challenge)?;
377        if result {
378            // Bump the last-verified timestamp so subsequent calls to
379            // `start_continuous_verification` and the elapsed-time
380            // gate in `maybe_continuous_verification` observe a real
381            // handshake rather than a fresh `None`. Recover from
382            // poison: a panicked writer would only have been writing
383            // a `Utc::now()` value; the cached timestamp is still
384            // valid.
385            let mut guard = match self.last_verification.lock() {
386                Ok(g) => g,
387                Err(poisoned) => {
388                    tracing::warn!(
389                        target: "latticearc::unified_api::zero_trust",
390                        "ZeroTrustAuth.last_verification mutex was poisoned in \
391                         verify_proof success path; a previous writer panicked. \
392                         Recovering inner guard."
393                    );
394                    poisoned.into_inner()
395                }
396            };
397            *guard = Some(Utc::now());
398        }
399        log_zero_trust_proof_verified!(result);
400        Ok(result)
401    }
402}

404/// Build the PoP transcript digest used by `generate_pop` / `verify_pop`.
405///
406/// Encodes
407/// `SHA-512(pop_ctx || 0x00 || verifier_pk_len_be4 || verifier_pk
408/// || 0x00 || ts_micros_be8 || 0x00 || challenge_len_be4 || challenge)`.
409///
410/// `verifier_pk` and `challenge` are each length-prefixed with a
411/// big-endian u32 so that concatenation cannot ambiguously parse as a
412/// different `(verifier_pk, challenge)` split. The 0x00 separators add
413/// a NUL-safety belt because `SIG_CONTEXT_POP_ED25519` is asserted
414/// NUL-free by its module-level test.
415///
416/// Sized [u8; 64] output. Returned by value so the digest lives in
417/// a register / stack slot the compiler can scrub.
418fn pop_transcript_digest(verifier_pk: &[u8], ts_micros: i64, challenge: &[u8]) -> [u8; 64] {
419    use sha2::{Digest, Sha512};
420    let mut hasher = Sha512::new();
421    hasher.update(crate::types::domains::pop_sig_context());
422    hasher.update([0x00]);
423    // Pre-encode the length prefixes as fixed-size arrays so the
424    // `.update(&[..])` calls stay branch-free and cannot be tricked by
425    // a future signature change that exposes a `usize`-vs-`u32` cast
426    // mistake. `.len() as u32` is bounded by the underlying slice cap
427    // (Ed25519 PK = 32 B, server-issued challenge ≤ 64 KiB at most for
428    // any sane caller).
429    let pk_len: u32 = verifier_pk.len().try_into().unwrap_or(u32::MAX);
430    hasher.update(pk_len.to_be_bytes());
431    hasher.update(verifier_pk);
432    hasher.update([0x00]);
433    hasher.update(ts_micros.to_be_bytes());
434    hasher.update([0x00]);
435    let ch_len: u32 = challenge.len().try_into().unwrap_or(u32::MAX);
436    hasher.update(ch_len.to_be_bytes());
437    hasher.update(challenge);
438    hasher.finalize().into()
439}

441impl ProofOfPossession for ZeroTrustAuth {
442    type Pop = ProofOfPossessionData;
443    type Error = CoreError;
444
445    /// Generates a proof of possession bound to a caller-supplied
446    /// challenge.
447    ///
448    /// The signed transcript is
449    /// `SHA-512(pop_ctx || 0x00 || verifier_pk_len_be4 || verifier_pk
450    /// || 0x00 || ts_micros_be8 || 0x00 || challenge_len_be4 ||
451    /// challenge)`. Binding `self.public_key` (the verifier identity this
452    /// auth represents) closes cross-identity replay; binding `challenge`
453    /// closes cross-verifier-instance replay within the freshness window.
454    /// Both are length-prefixed to prevent prefix-free ambiguity between
455    /// the two adjacent variable-length values.
456    ///
457    /// # Errors
458    ///
459    /// Returns `CoreError::InvalidKeyLength` if the private key has
460    /// incorrect length for Ed25519 signing. Returns
461    /// `CoreError::InvalidInput` if the private key format is invalid.
462    fn generate_pop(&self, challenge: &[u8]) -> Result<Self::Pop> {
463        // PoP-M1 follow-up: empty challenge silently disables cross-
464        // verifier-instance replay protection. The wire format encodes
465        // challenge_len so the signature still differs from an
466        // unbound-challenge legacy PoP, but a verifier that calls
467        // generate_pop(b"") and verify_pop(_, b"") accepts cross-
468        // verifier replays for the freshness window. Warn loudly at
469        // sign-side so operators don't ship the misuse silently;
470        // verify-side rejects with Err so silent fall-through is
471        // impossible.
472        if challenge.is_empty() {
473            tracing::warn!(
474                "PoP generate_pop called with empty challenge — cross-verifier-\
475                 instance replay protection requires a per-request nonce. \
476                 verify_pop will refuse to accept an empty challenge."
477            );
478        }
479        let timestamp = Utc::now();
480        // Microsecond precision: Ed25519 sigs are deterministic, so two
481        // PoPs in the same second from the same key produce identical
482        // wire bytes and trip the replay cache against a legitimate
483        // client regenerating in a tight loop. Microseconds also fit
484        // comfortably inside the 5-minute freshness window.
485        let ts_micros = timestamp.timestamp_micros();
486        let pop_digest = pop_transcript_digest(self.public_key.as_slice(), ts_micros, challenge);
487
488        let signature = crate::unified_api::convenience::ed25519::sign_ed25519_internal(
489            &pop_digest,
490            self.private_key.expose_secret(),
491        )?;
492
493        Ok(ProofOfPossessionData { public_key: self.public_key.clone(), signature, timestamp })
494    }
495
496    /// Verifies a proof of possession against `self.public_key` and the
497    /// caller-supplied expected challenge.
498    ///
499    /// # Returns
500    ///
501    /// `Ok(true)` if the proof is fresh AND identity-bound AND
502    /// challenge-bound AND cryptographically valid. `Ok(false)` for
503    /// every rejection cause — stale, future-dated, identity mismatch
504    /// (PoP-H1), challenge mismatch (PoP-M1), crypto reject, replay
505    /// within the freshness window. **All adversary-reachable rejection
506    /// causes collapse to `Ok(false)`** so callers cannot distinguish
507    /// "stale, refresh and retry" from "wrong PoP" / "replay" by
508    /// branching on the Result. Rejection cause is logged at
509    /// `tracing::debug!` for operator visibility.
510    ///
511    /// # Errors
512    ///
513    /// Returns `CoreError::InvalidInput` only for unrecoverable
514    /// structural failures (poisoned mutex on the replay cache). The
515    /// signature-length, public-key-length, and stale / future / replay
516    /// paths are all Pattern-6-collapsed to `Ok(false)`.
517    fn verify_pop(
518        &self,
519        pop: &Self::Pop,
520        expected_challenge: &[u8],
521    ) -> std::result::Result<bool, Self::Error> {
522        // PoP-M1 follow-up: refuse empty challenge. An operator that
523        // generates PoPs with empty challenge and verifies them with
524        // empty challenge structurally loses the cross-verifier-
525        // instance replay protection the parameter exists to provide
526        // — the wrapper bytes still differ from pre-fix raw-message
527        // PoPs, but a verifier that consistently passes b"" would
528        // accept cross-verifier replays for the freshness window. Hard
529        // reject is operator-misuse class, not adversary-attainable,
530        // so Err (not Ok(false)) is appropriate — distinguishing
531        // "missing challenge" from "wrong challenge" doesn't leak
532        // adversary-controllable state.
533        if expected_challenge.is_empty() {
534            return Err(CoreError::InvalidInput(
535                "PoP verify_pop requires a non-empty expected_challenge — the \
536                 per-request nonce is the only structural defence against \
537                 cross-verifier-instance replay. Pass the same bytes the \
538                 generator used."
539                    .to_string(),
540            ));
541        }
542        // Freshness check: reject proofs older than PROOF_OF_POSSESSION_MAX_AGE.
543        // This prevents replay of stale PoPs captured from prior sessions (P5.2 C4).
544        // Using chrono::Duration so the bound is expressed in seconds regardless
545        // of timezone or DST quirks.
546        const PROOF_OF_POSSESSION_MAX_AGE_SECS: i64 = 5 * 60; // 5 minutes
547        let elapsed = Utc::now().signed_duration_since(pop.timestamp());
548        if elapsed.num_seconds() > PROOF_OF_POSSESSION_MAX_AGE_SECS {
549            // Pattern-6: do not leak elapsed seconds or the configured
550            // max age via the error string. An adversary with a PoP-
551            // generation oracle could otherwise narrow server clock
552            // skew to seconds. Cause logged at tracing::debug! for
553            // operators.
554            tracing::debug!(
555                elapsed_secs = elapsed.num_seconds(),
556                max_age_secs = PROOF_OF_POSSESSION_MAX_AGE_SECS,
557                "verify_pop rejected: proof-of-possession too old"
558            );
559            return Ok(false);
560        }
561        // Also reject proofs dated more than 30 seconds in the future (clock skew tolerance).
562        if elapsed.num_seconds() < -30 {
563            tracing::debug!(
564                elapsed_secs = elapsed.num_seconds(),
565                "verify_pop rejected: proof-of-possession timestamp is in the future"
566            );
567            return Ok(false);
568        }
569
570        // PoP-H1: identity binding. The pre-fix verify dispatched against
571        // `pop.public_key()` — the public key the (potentially attacker-
572        // controlled) PoP carries. Any party with any Ed25519 keypair
573        // could then produce a self-signed PoP that returned Ok(true),
574        // proving possession of A KEY, not THIS IDENTITY's key. Compare
575        // the embedded pk against the verifier identity in constant
576        // time; mismatch is Pattern-6-collapsed to Ok(false) so the
577        // adversary cannot distinguish "wrong identity" from "wrong
578        // signature" by branching on the Result.
579        use subtle::ConstantTimeEq;
580        let embedded_pk = pop.public_key().as_slice();
581        let verifier_pk = self.public_key.as_slice();
582        if embedded_pk.len() != verifier_pk.len() {
583            tracing::debug!(
584                embedded_len = embedded_pk.len(),
585                verifier_len = verifier_pk.len(),
586                "verify_pop rejected: embedded public key length differs from \
587                 verifier identity (PoP-H1)"
588            );
589            return Ok(false);
590        }
591        if embedded_pk.ct_eq(verifier_pk).unwrap_u8() != 1u8 {
592            tracing::debug!(
593                "verify_pop rejected: embedded public key does not match verifier \
594                 identity (PoP-H1)"
595            );
596            return Ok(false);
597        }
598
599        // PoP-M1: reconstruct the post-fix transcript exactly as
600        // `generate_pop` produced it — verifier_pk + ts + challenge —
601        // and dispatch against the verifier identity. A captured PoP
602        // for a different verifier-id OR a different challenge round
603        // produces a different digest and the crypto verify rejects.
604        let ts_micros = pop.timestamp().timestamp_micros();
605        let pop_digest = pop_transcript_digest(verifier_pk, ts_micros, expected_challenge);
606
607        let valid = crate::unified_api::convenience::ed25519::verify_ed25519_internal(
608            &pop_digest,
609            pop.signature(),
610            verifier_pk,
611        )?;
612
613        // reject re-presentation of a
614        // verified PoP within the 5-minute acceptance window. Without
615        // a cache, an attacker who captured a single valid PoP at
616        // time T could replay it any number of times within
617        // (T, T + 5 min). The cache is keyed on (pk, sig, ts_micros);
618        // entries older than the freshness window above are
619        // self-evicting since the freshness check would already have
620        // rejected them on a re-replay attempt. The signature alone
621        // is byte-unique per timestamp under Ed25519 determinism (see
622        // `generate_pop` for why microsecond precision is necessary),
623        // so the cache key only needs to disambiguate within a single
624        // microsecond.
625        if valid {
626            let mut cache = self.pop_replay_cache.lock().map_err(|_poison| {
627                CoreError::InvalidInput("PoP replay cache poisoned".to_string())
628            })?;
629            // PoP-H1 enforced verifier_pk above, so the cache key uses
630            // verifier_pk (the canonical identity bytes) instead of the
631            // adversary-supplied `pop.public_key()`. Identical bytes by
632            // construction once H1 passes, but anchoring to the verifier
633            // makes the cache-key invariant resilient to future code
634            // motion that might forget to dereference `pop.public_key()`
635            // first.
636            let pk_bytes = verifier_pk;
637            let pk_len = pk_bytes.len();
638            // Evict expired entries opportunistically; per-PK counts are
639            // decremented in lockstep so they stay exact. `pk_len` is
640            // owned by the cache (locked at construction) so this call
641            // no longer carries it as a parameter.
642            let now_secs = Utc::now().timestamp();
643            cache.expire_older_than(now_secs, PROOF_OF_POSSESSION_MAX_AGE_SECS);
644            // Cache key combines PK + signature + microsecond timestamp;
645            // collisions require either a cryptographic break or a
646            // deliberate replay (the latter is what we're rejecting
647            // here). The challenge bytes already factor into the
648            // signature (PoP-M1), so adding them to the key would be
649            // redundant.
650            let key_cap = pk_len.saturating_add(pop.signature().len()).saturating_add(8);
651            let mut key = Vec::with_capacity(key_cap);
652            key.extend_from_slice(pk_bytes);
653            key.extend_from_slice(pop.signature());
654            key.extend_from_slice(&ts_micros.to_be_bytes());
655            if cache.contains(&key) {
656                // PoP-L1: collapse replay to Ok(false). Pre-fix this was
657                // Err(CoreError::InvalidInput("replay detected")), letting
658                // the caller distinguish "I've seen this before" from the
659                // sibling "stale, refresh and retry" path (which already
660                // returned Ok(false)). Pattern-6 contract requires every
661                // adversary-attainable rejection to look identical.
662                tracing::debug!(ts_micros, "PoP rejected: replay within 5-min window");
663                return Ok(false);
664            }
665            // Per-PK quota. Bounds the damage a single noisy public key can
666            // do: an attacker submitting many legitimate PoPs from one PK
667            // fills only their own slice of the cache, leaving every other
668            // PK's quota untouched. A 64-flight legitimate client (long
669            // poll, batched signing) still fits; ≥ 256 distinct PKs are
670            // required before the global cap can be saturated.
671            const POP_CACHE_PER_PK_MAX: usize = 64;
672            if cache.count_for_pk(pk_bytes) >= POP_CACHE_PER_PK_MAX {
673                // PoP-L1: collapse to Ok(false) — same Pattern-6
674                // requirement as the replay-detected branch. An attacker
675                // who floods their own PK to the quota cap could
676                // otherwise distinguish "valid-but-quota-exhausted" (Err)
677                // from "valid replay" (Ok(false)) or "invalid sig"
678                // (Ok(false)) by branching on the Result variant. The
679                // leaked bit is DoS-cache state, not key material, but
680                // the contract documented on verify_pop says "all
681                // adversary-reachable rejection causes collapse to
682                // Ok(false)". Operator cause stays in tracing::warn!.
683                tracing::warn!(
684                    per_pk_cap = POP_CACHE_PER_PK_MAX,
685                    "PoP per-key quota exhausted; rejecting new PoP from this PK \
686                     (other PKs unaffected)"
687                );
688                return Ok(false);
689            }
690            // Global cap. The opportunistic eviction above already removed
691            // entries older than the freshness window, so reaching this
692            // cap means ≥ `POP_CACHE_MAX` distinct legitimate PoPs from
693            // ≥ `POP_CACHE_MAX / POP_CACHE_PER_PK_MAX` distinct PKs were
694            // verified inside a single 5-minute window — that is DoS-class
695            // load. Silently skipping the insert would lift the replay
696            // window for subsequent presentations of the same PoP. Fail
697            // closed instead.
698            const POP_CACHE_MAX: usize = 16 * 1024;
699            if cache.total() >= POP_CACHE_MAX {
700                // PoP-L1: same Pattern-6 rationale as the per-PK quota
701                // branch above — adversary-distinguishable Err vs
702                // Ok(false) was the exact oracle the L1 fix was supposed
703                // to close. Operator cause stays in tracing::warn!.
704                tracing::warn!(
705                    cap = POP_CACHE_MAX,
706                    "PoP global replay cache full; rejecting new PoP rather than \
707                     skipping insert (silently skipping would re-open the 5-minute \
708                     replay window)"
709                );
710                return Ok(false);
711            }
712            cache.insert(pk_bytes, key, now_secs);
713        }
714
715        Ok(valid)
716    }
717}
718
719impl ContinuousVerifiable for ZeroTrustAuth {
720    type Error = CoreError;
721
722    /// Checks the current verification status of the session.
723    ///
724    /// # Errors
725    ///
726    /// This function does not currently return errors, but returns `Result` for
727    /// API consistency and future extensibility.
728    fn verify_continuously(&self) -> Result<VerificationStatus> {
729        let session_elapsed = Utc::now().signed_duration_since(self.session_start);
730
731        let max_session_time: u64 = 30 * 60 * 1000;
732
733        // Convert safely: negative elapsed times treated as 0 (just started)
734        let session_elapsed_u64 = u64::try_from(session_elapsed.num_milliseconds()).unwrap_or(0);
735        if session_elapsed_u64 > max_session_time {
736            return Ok(VerificationStatus::Expired);
737        }
738
739        if !self.config.continuous_verification {
740            return Ok(VerificationStatus::Verified);
741        }
742
743        // Recover from poison: a panicked write would only have been a
744        // failed `Utc::now()` assignment; the cached timestamp is still
745        // valid. Treating poison as fatal would lock callers out of an
746        // otherwise healthy auth session.
747        let last_verification_at = match self.last_verification.lock() {
748            Ok(g) => *g,
749            Err(poisoned) => {
750                tracing::warn!(
751                    target: "latticearc::unified_api::zero_trust",
752                    "ZeroTrustAuth.last_verification mutex was poisoned in \
753                     maybe_continuous_verification; a previous writer panicked. \
754                     Recovering inner guard."
755                );
756                *poisoned.into_inner()
757            }
758        };
759        // No prior verification ⇒ continuous-verify mode reports
760        // `Pending`, mirroring the "must reauthenticate" path. We never
761        // return `Verified` for an instance that has never completed a
762        // proof, even if the configured interval hasn't expired.
763        let Some(last_verification_at) = last_verification_at else {
764            return Ok(VerificationStatus::Pending);
765        };
766        let verification_elapsed = Utc::now().signed_duration_since(last_verification_at);
767
768        // Convert safely: negative elapsed times treated as 0 (just verified)
769        let verification_elapsed_u64 =
770            u64::try_from(verification_elapsed.num_milliseconds()).unwrap_or(0);
771        if verification_elapsed_u64 > self.config.verification_interval_ms {
772            return Ok(VerificationStatus::Pending);
773        }
774
775        Ok(VerificationStatus::Verified)
776    }
777
778    /// Performs reauthentication by generating and verifying a new challenge-proof pair.
779    ///
780    /// # Errors
781    ///
782    /// Returns `CoreError::EntropyDepleted` if the system cannot generate random bytes
783    /// for the challenge.
784    ///
785    /// Returns `CoreError::AuthenticationFailed` if proof generation fails due to an
786    /// empty challenge.
787    ///
788    /// Returns `CoreError::InvalidKeyLength` or `CoreError::InvalidInput` if the private
789    /// key format is invalid for Ed25519 signing.
790    fn reauthenticate(&self) -> Result<()> {
791        // The previous implementation generated a proof, discarded it
792        // via `let _proof = ...`, and bumped `last_verification`
793        // regardless. That made `last_verification` a "now" stamp on
794        // every call — an attacker who tampered with the private-key
795        // bytes (use-after-free, fuzz target, etc.) kept the session
796        // perpetually `Verified`. We now actually verify the proof
797        // against the challenge before bumping `last_verification`, and
798        // surface a hard error on mismatch.
799        let challenge = self.generate_challenge()?;
800        let proof = self.generate_proof(challenge.data())?;
801
802        let proof_valid = self.verify_proof(&proof, challenge.data())?;
803        if !proof_valid {
804            return Err(CoreError::AuthenticationFailed(
805                "Reauthentication proof verification failed".to_string(),
806            ));
807        }
808
809        let mut guard = match self.last_verification.lock() {
810            Ok(g) => g,
811            Err(poisoned) => {
812                tracing::warn!(
813                    target: "latticearc::unified_api::zero_trust",
814                    "ZeroTrustAuth.last_verification mutex was poisoned in \
815                     reauthenticate; a previous writer panicked. Recovering inner guard."
816                );
817                poisoned.into_inner()
818            }
819        };
820        *guard = Some(Utc::now());
821        Ok(())
822    }
823}
