22/// Returns the default compliance mode for a given use case.
23///
24/// Regulated use cases automatically get FIPS 140-3 compliance,
25/// which will be validated at operation time (failing gracefully
26/// if the `fips` feature is not enabled).
27fn default_compliance_for_use_case(use_case: UseCase) -> ComplianceMode {
28    match use_case {
29        // Regulated industries require FIPS 140-3 by default
30        UseCase::GovernmentClassified
31        | UseCase::HealthcareRecords
32        | UseCase::PaymentCard
33        | UseCase::FinancialTransactions => ComplianceMode::Fips140_3,
34        // Non-regulated use cases — explicitly listed so new variants trigger
35        // a compile error, forcing a conscious compliance decision.
36        UseCase::SecureMessaging
37        | UseCase::EmailEncryption
38        | UseCase::VpnTunnel
39        | UseCase::ApiSecurity
40        | UseCase::FileStorage
41        | UseCase::DatabaseEncryption
42        | UseCase::CloudStorage
43        | UseCase::BackupArchive
44        | UseCase::ConfigSecrets
45        | UseCase::Authentication
46        | UseCase::SessionToken
47        | UseCase::DigitalCertificate
48        | UseCase::KeyExchange
49        | UseCase::LegalDocuments
50        | UseCase::BlockchainTransaction
51        | UseCase::IoTDevice
52        | UseCase::FirmwareSigning
53        | UseCase::AuditLog => ComplianceMode::Default,
54    }
55}

61/// Unified configuration for cryptographic operations.
62///
63/// Provides a single, consistent way to configure encrypt, decrypt, sign, and verify
64/// operations. Uses a builder pattern for ergonomic configuration.
65///
66/// # Examples
67///
68/// ```rust,no_run
69/// # use latticearc::unified_api::{encrypt, CryptoConfig, UseCase, SecurityLevel, VerifiedSession, generate_keypair, EncryptKey};
70/// # fn main() -> Result<(), latticearc::unified_api::error::CoreError> {
71/// # let data = b"secret";
72///
73/// // Hybrid encryption (recommended - ML-KEM-768 + X25519 + AES-256-GCM)
74/// let (pk, _sk) = latticearc::generate_hybrid_keypair()?;
75/// encrypt(data, EncryptKey::Hybrid(&pk), CryptoConfig::new())?;
76///
77/// // With Zero Trust session
78/// # let (ed_pk, ed_sk) = generate_keypair()?;
79/// let session = VerifiedSession::establish(ed_pk.as_slice(), ed_sk.expose_secret())?;
80/// encrypt(data, EncryptKey::Hybrid(&pk),
81///     CryptoConfig::new().session(&session))?;
82///
83/// // With use case (recommended - library picks optimal algorithm)
84/// encrypt(data, EncryptKey::Hybrid(&pk), CryptoConfig::new()
85///     .session(&session)
86///     .use_case(UseCase::FileStorage))?;
87///
88/// // Symmetric encryption (AES-256-GCM). Generate a fresh key from the OS
89/// // CSPRNG — `[0u8; 32]` is rejected as a weak key.
90/// let key = latticearc::primitives::rand::random_bytes(32);
91/// encrypt(data, EncryptKey::Symmetric(&key), CryptoConfig::new()
92///     .force_scheme(latticearc::CryptoScheme::Symmetric))?;
93///
94/// // With FIPS compliance (requires `fips` feature)
95/// use latticearc::types::types::ComplianceMode;
96/// encrypt(data, EncryptKey::Hybrid(&pk), CryptoConfig::new()
97///     .compliance(ComplianceMode::Fips140_3))?;
98/// # Ok(())
99/// # }
100/// ```
101#[derive(Debug, Clone)]
102pub struct CryptoConfig<'a> {
103    /// Optional Zero Trust verified session.
104    /// If None, operates in unverified mode.
105    ///
106    /// Consumer: `CryptoConfig::validate()` (session expiry gate, called at
107    /// the top of every unified-API operation).
108    session: Option<&'a VerifiedSession>,
109    /// Algorithm selection mode (use case or security level).
110    ///
111    /// Consumer: `convenience::api::{select_encryption_scheme, select_signature_scheme}`
112    /// via `get_selection()`.
113    selection: AlgorithmSelection,
114    /// Compliance mode for regulatory requirements.
115    ///
116    /// Consumer: `CryptoConfig::validate_scheme_compliance()` via `get_compliance()`.
117    compliance: ComplianceMode,
118    /// Whether the user explicitly set compliance (vs. auto-set by use_case).
119    ///
120    /// Consumer: `CryptoConfig::validate_scheme_compliance()` (explicit
121    /// compliance is enforced; use-case-derived compliance is advisory).
122    compliance_explicit: bool,
123    /// Whether the user explicitly pinned a `SecurityLevel` (vs. left it
124    /// at the constructor default). Only when this is true does
125    /// `validate_scheme_compliance` enforce a minimum-strength gate
126    /// against the configured level — otherwise the existing scheme-
127    /// dispatch path (which routes by string name) decides what's
128    /// acceptable. Without this discrimination a defaulted-to-High
129    /// config would reject every ml-dsa-44 / ml-kem-512 caller.
130    security_level_explicit: bool,
131    /// Cryptographic mode: hybrid (default) or PQ-only.
132    ///
133    /// Consumer: `convenience::api` dispatch arms via `get_crypto_mode()`
134    /// (routes to the PQ-only scheme table when `CryptoMode::PqOnly`).
135    crypto_mode: CryptoMode,
136    /// Optional maximum age (seconds) for `EncryptedOutput.timestamp` on
137    /// decrypt. `None` (default) skips the check; `Some(n)` rejects any
138    /// ciphertext whose stamped timestamp is more than `n` seconds older
139    /// than the current wall-clock. See [`max_age`](Self::max_age).
140    ///
141    /// Consumer: `convenience::api::decrypt_with_aad()` replay-window guard
142    /// (surfaces as `CoreError::Replay`).
143    max_age_seconds: Option<u64>,
144}
145
146impl<'a> Default for CryptoConfig<'a> {
147    fn default() -> Self {
148        Self::new()
149    }
150}

156        Self {
157            session: None,

162            crypto_mode: CryptoMode::Hybrid,

167    /// Reject ciphertexts whose stamped `timestamp` is older than `seconds`.
168    ///
169    /// Defence-in-depth replay protection at the convenience-API layer.
170    /// `EncryptedOutput.timestamp` is set on encrypt to the current
171    /// `Utc::now().timestamp()` (seconds since epoch); when this option
172    /// is set, [`decrypt`](crate::unified_api::convenience::api::decrypt)
173    /// rejects ciphertexts whose timestamp is more than `seconds` behind
174    /// the receiver's wall-clock as
175    /// `CoreError::Replay { age_seconds, max_age_seconds }`.
176    ///
177    /// The dedicated `Replay` variant — not `ResourceExceeded` — is
178    /// deliberate: a caller can pattern-match a suspected replay and raise an
179    /// alert, distinct from a genuine size-quota rejection.
180    ///
181    /// # Caveats
182    ///
183    /// - **Wall-clock based.** Receivers with skewed clocks will reject
184    ///   freshly-encrypted ciphertexts; senders racing the receiver's
185    ///   clock will see false rejections. Pair with NTP discipline on
186    ///   both ends, and pick `seconds` large enough to swallow expected
187    ///   skew (typically 60s — 5min).
188    /// - **NOT a substitute for monotonic counter / nonce-cache replay
189    ///   protection** at the protocol layer. An attacker replaying a
190    ///   ciphertext WITHIN the `max_age` window still succeeds. This
191    ///   guard bounds the replay window; it does not eliminate it.
192    ///   Applications that need exactly-once delivery must layer their
193    ///   own replay-cache on top.
194    /// - **No-op when unset.** Default is `None` to preserve backward
195    ///   compatibility with callers who use the timestamp purely for
196    ///   audit / display.
197    #[must_use]
198    pub fn max_age(mut self, seconds: u64) -> Self {
199        self.max_age_seconds = Some(seconds);
200        self
201    }
202
203    /// Returns the configured replay-protection window, in seconds.
204    #[must_use]
205    pub fn max_age_seconds(&self) -> Option<u64> {
206        self.max_age_seconds
207    }

351    /// Check if a scheme string is acceptable under the current compliance mode.
352    ///
353    /// This enforces compliance policies against the algorithm encoded in
354    /// ciphertext or signature metadata. Following the OpenSSL model, compliance
355    /// is enforced on both encrypt/sign and decrypt/verify sides.
356    ///
357    /// # Scheme classification
358    ///
359    /// | Compliance | Allowed schemes | Rejected schemes |
360    /// |------------|-----------------|------------------|
361    /// | Default    | All             | None             |
362    /// | FIPS 140-3 | AES-256-GCM, ML-KEM-*, ML-DSA-*, SLH-DSA-*, hybrid-* | ChaCha20-Poly1305 |
363    /// | CNSA 2.0   | ML-KEM-*, ML-DSA-*, SLH-DSA-*, FN-DSA-*, hybrid-* | Ed25519, AES-256-GCM (pure classical) |
364    ///
365    /// # Errors
366    ///
367    /// Returns `CoreError::ComplianceViolation` if the scheme is not permitted
368    /// under the active compliance mode.
369    pub fn validate_scheme_compliance(
370        &self,
371        scheme: &str,
372    ) -> crate::unified_api::error::Result<()> {
373        use crate::unified_api::error::CoreError;
374
375        /// Schemes explicitly banned under FIPS 140-3.
376        const FIPS_BANNED: &[&str] = &["chacha20-poly1305"];
377
378        /// Pure classical schemes banned under CNSA 2.0.
379        /// Hybrid schemes containing these as a component are allowed
380        /// (transitional per NIST SP 800-227).
381        const CNSA_BANNED_CLASSICAL: &[&str] = &["ed25519", "aes-256-gcm"];
382
383        match self.compliance {
384            ComplianceMode::Default => {}
385            ComplianceMode::Fips140_3 => {
386                if FIPS_BANNED.contains(&scheme) {
387                    return Err(CoreError::ComplianceViolation(format!(
388                        "Scheme '{scheme}' is not FIPS 140-3 approved. \
389                         Use AES-256-GCM or a FIPS-validated algorithm.",
390                    )));
391                }
392            }
393            ComplianceMode::Cnsa2_0 => {
394                // Only reject exact matches — hybrid schemes that embed a
395                // classical component (e.g. "hybrid-ml-dsa-65-ed25519") are
396                // allowed as transitional per NIST SP 800-227.
397                if CNSA_BANNED_CLASSICAL.contains(&scheme) {
398                    return Err(CoreError::ComplianceViolation(format!(
399                        "Scheme '{scheme}' is not permitted under CNSA 2.0 (post-quantum required). \
400                         Use ML-KEM, ML-DSA, SLH-DSA, FN-DSA, or a hybrid scheme.",
401                    )));
402                }
403            }
404        }
405
406        // SECURITY: also gate on the configured `SecurityLevel`. Without
407        // this check, a server with `SecurityLevel::Maximum` would
408        // silently accept a wire-supplied `hybrid-ml-kem-512-…` scheme
409        // (NIST Level 1) if an attacker also supplied a valid 512-bit
410        // ciphertext. The compliance allowlist alone is not sufficient
411        // — `hybrid-ml-kem-512-aes-256-gcm` passes both FIPS and CNSA
412        // bans yet is below `Maximum`.
413        //
414        // Only enforced when the caller EXPLICITLY pinned a level via
415        // `.security_level(...)` — not when the constructor default
416        // `SecurityLevel::High` is in play. Without this distinction every
417        // ml-dsa-44 / ml-kem-512 caller using `CryptoConfig::new()` would
418        // be rejected, which contradicts the Standard-tier APIs being a
419        // first-class entry point. UseCase-driven and forced-scheme
420        // selections already constrain the scheme via dispatch.
421        //
422        // `None` from `scheme_min_security_level` means we don't recognise
423        // the scheme — let the caller's dispatch path return the canonical
424        // "unknown scheme" error.
425        if self.security_level_explicit
426            && let AlgorithmSelection::SecurityLevel(configured_level) = self.selection
427            && let Some(scheme_level) = scheme_min_security_level(scheme)
428            && (scheme_level as u8) < (configured_level as u8)
429        {
430            return Err(CoreError::ComplianceViolation(format!(
431                "Scheme '{scheme}' provides {scheme_level:?} security but \
432                 CryptoConfig requires at least {configured_level:?}. \
433                 Either downgrade the SecurityLevel via .security_level() \
434                 or provide a stronger scheme."
435            )));
436        }
437        Ok(())
438    }
439}

441/// Map a scheme tag string to the minimum `SecurityLevel` it provides.
442///
443/// Returns `None` for schemes this build doesn't recognise; callers
444/// should treat that as "let the dispatch layer reject as unknown"
445/// rather than as "any level is fine."
446fn scheme_min_security_level(scheme: &str) -> Option<SecurityLevel> {
447    // NIST PQC categories map to SecurityLevel as:
448    //   Category 1 (128-bit) → Standard (ML-KEM-512, ML-DSA-44, SLH-DSA-128*)
449    //   Category 3 (192-bit) → High     (ML-KEM-768, ML-DSA-65, SLH-DSA-192*)
450    //   Category 5 (256-bit) → Maximum  (ML-KEM-1024, ML-DSA-87, SLH-DSA-256*)
451    // Hybrid schemes inherit the level of their PQ component (the
452    // classical sidecar contributes zero PQ security).
453    //
454    // the previous shape used
455    // `s.contains("128")`/`"192"`/`"256"` substring matching, which
456    // accepted any future scheme name containing those literals
457    // (e.g. `sha-256`, `slh-dsa-shake-256s`, hypothetical
458    // `prefix-256xxx`) as a security-level claim regardless of actual
459    // algorithm strength. Substring matching across security-tier
460    // dispatch is a correctness footgun.
461    //
462    // The needles below are therefore full algorithm names
463    // (`ml-kem-512`), never bare size literals (`512`). A composite
464    // scheme tag embeds its PQ algorithm name verbatim in every
465    // spelling the library emits or accepts —
466    // `pq-ml-kem-512-aes-256-gcm`, `hybrid-ml-kem-512-aes-256-gcm`,
467    // `hybrid-ml-dsa-44-ed25519`, `ml-dsa-44-hybrid-ed25519` — so
468    // substring containment on the *algorithm name* classifies all of
469    // them without reintroducing the size-literal footgun.
470    //
471    // Matching must NOT be restricted to a prefix (`starts_with
472    // ("hybrid-")`) or to `-`-delimited whole tokens: the PQ algorithm
473    // name is itself hyphenated, so token equality never matches, and
474    // the `pq-` prefixed and `-hybrid-ed25519` suffixed forms carry no
475    // `hybrid-` prefix. Either restriction silently returns `None` for
476    // real wire schemes, which disables the minimum-strength gate in
477    // `validate_scheme_compliance` for exactly those spellings.
478    let s = scheme.to_ascii_lowercase();
479
480    // Only the SLH-DSA `-shake-*s` (small) variants are exposed by
481    // `SlhDsaSecurityLevel`; `-shake-*f` (fast) and `-sha2-*` are not
482    // wired through and would never reach here from a real keygen path.
483
484    // Standard (Category 1 / 128-bit PQ)
485    if matches!(
486        s.as_str(),
487        "ml-kem-512"
488            | "ml-dsa-44"
489            | "slh-dsa-shake-128s"
490            | "fn-dsa-512"
491            | "ed25519"
492            | "aes-256-gcm"
493            | "chacha20-poly1305"
494    ) || s.contains("ml-kem-512")
495        || s.contains("ml-dsa-44")
496    {
497        Some(SecurityLevel::Standard)
498    } else if matches!(s.as_str(), "ml-kem-768" | "ml-dsa-65" | "slh-dsa-shake-192s")
499        || s.contains("ml-kem-768")
500        || s.contains("ml-dsa-65")
501    {
502        Some(SecurityLevel::High)
503    } else if matches!(
504        s.as_str(),
505        "ml-kem-1024" | "ml-dsa-87" | "slh-dsa-shake-256s" | "fn-dsa-1024"
506    ) || s.contains("ml-kem-1024")
507        || s.contains("ml-dsa-87")
508    {
509        Some(SecurityLevel::Maximum)
510    } else {
511        None
512    }
513}

515impl<'a> CryptoConfig<'a> {
516    /// Validates the configuration.
517    ///
518    /// Checks:
519    /// 1. Session expiry (if present)
520    /// 2. FIPS feature availability (if compliance mode requires it)
521    /// 3. CNSA 2.0 requires `CryptoMode::PqOnly`
522    ///
523    /// # Errors
524    ///
525    /// Returns `CoreError::SessionExpired` if the session has expired.
526    /// Returns `CoreError::FeatureNotAvailable` if compliance mode requires FIPS
527    /// but the `fips` feature is not enabled.
528    /// Returns `CoreError::ConfigurationError` if CNSA 2.0 is set but the crypto
529    /// mode is not `PqOnly`.
530    pub fn validate(&self) -> crate::unified_api::error::Result<()> {
531        use crate::unified_api::error::CoreError;
532
533        // 1. Session expiry check
534        if let Some(session) = self.session {
535            session.verify_valid()?;
536        }
537
538        // 2. FIPS availability check
539        if self.compliance.requires_fips() && !fips_available() {
540            return Err(CoreError::FeatureNotAvailable(format!(
541                "{:?} compliance requires the `fips` feature. \
542                 Rebuild with: latticearc = {{ features = [\"fips\"] }}",
543                self.compliance
544            )));
545        }
546
547        // 3. CNSA 2.0 requires CryptoMode::PqOnly
548        if matches!(self.compliance, ComplianceMode::Cnsa2_0)
549            && !matches!(self.crypto_mode, CryptoMode::PqOnly)
550        {
551            return Err(CoreError::ConfigurationError(
552                "CNSA 2.0 compliance requires CryptoMode::PqOnly. \
553                 Use .crypto_mode(CryptoMode::PqOnly) on your CryptoConfig."
554                    .to_string(),
555            ));
556        }
557
558        Ok(())
559    }
560}

752    /// Every scheme spelling the library can emit or accept on the wire must
753    /// be classifiable by `scheme_min_security_level`. A `None` here silently
754    /// disables the minimum-strength gate for that spelling, which is exactly
755    /// how the `pq-`-prefixed and `-hybrid-ed25519`-suffixed forms escaped it.
756    #[test]
757    fn test_every_wire_scheme_spelling_is_level_classified_succeeds() {
758        // `pq-` prefixed encryption tags (crypto_types::EncryptionScheme).
759        // Signature tags, canonical hybrid and legacy alias spellings
760        // (types::domains + api.rs dispatch).
761        let level_1 = [
762            "pq-ml-kem-512-aes-256-gcm",
763            "hybrid-ml-kem-512-aes-256-gcm",
764            "pq-ml-dsa-44",
765            "hybrid-ml-dsa-44-ed25519",
766            "ml-dsa-44-hybrid-ed25519",
767        ];
768        let level_3 = [
769            "pq-ml-kem-768-aes-256-gcm",
770            "hybrid-ml-kem-768-aes-256-gcm",
771            "pq-ml-dsa-65",
772            "hybrid-ml-dsa-65-ed25519",
773            "ml-dsa-65-hybrid-ed25519",
774        ];
775        let level_5 = [
776            "pq-ml-kem-1024-aes-256-gcm",
777            "hybrid-ml-kem-1024-aes-256-gcm",
778            "pq-ml-dsa-87",
779            "hybrid-ml-dsa-87-ed25519",
780            "ml-dsa-87-hybrid-ed25519",
781        ];
782
783        for scheme in level_1 {
784            assert_eq!(
785                scheme_min_security_level(scheme),
786                Some(SecurityLevel::Standard),
787                "{scheme} must classify as Standard"
788            );
789        }
790        for scheme in level_3 {
791            assert_eq!(
792                scheme_min_security_level(scheme),
793                Some(SecurityLevel::High),
794                "{scheme} must classify as High"
795            );
796        }
797        for scheme in level_5 {
798            assert_eq!(
799                scheme_min_security_level(scheme),
800                Some(SecurityLevel::Maximum),
801                "{scheme} must classify as Maximum"
802            );
803        }
804    }

806    /// The gate must fire on the `pq-`/alias spellings, not just on the
807    /// `hybrid-` prefixed ones. Wire-supplied schemes reach this check from
808    /// `decrypt` and `verify`, so a spelling that skips it lets a Category-1
809    /// scheme through a `SecurityLevel::Maximum` configuration.
810    #[test]
811    fn test_below_minimum_level_rejected_for_all_spellings_fails() {
812        for scheme in [
813            "pq-ml-kem-512-aes-256-gcm",
814            "hybrid-ml-kem-512-aes-256-gcm",
815            "pq-ml-dsa-44",
816            "ml-dsa-44-hybrid-ed25519",
817            "hybrid-ml-dsa-44-ed25519",
818        ] {
819            let config = CryptoConfig::new().security_level(SecurityLevel::Maximum);
820            let result = config.validate_scheme_compliance(scheme);
821            assert!(result.is_err(), "{scheme} must be rejected under Maximum");
822            let err_msg = format!("{}", result.unwrap_err());
823            assert!(err_msg.contains("Standard"), "{scheme}: {err_msg}");
824            assert!(err_msg.contains("Maximum"), "{scheme}: {err_msg}");
825        }
826    }
