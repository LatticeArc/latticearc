12    /// Serialize to JSON string (human-readable format).
13    ///
14    /// # Errors
15    /// Returns an error if JSON serialization fails.
16    pub fn to_json(&self) -> Result<String> {
17        serde_json::to_string(self)
18            .map_err(|e| CoreError::SerializationError(format!("JSON serialization failed: {e}")))
19    }

21    /// Serialize to pretty-printed JSON.
22    ///
23    /// # Errors
24    /// Returns an error if JSON serialization fails.
25    pub fn to_json_pretty(&self) -> Result<String> {
26        serde_json::to_string_pretty(self)
27            .map_err(|e| CoreError::SerializationError(format!("JSON serialization failed: {e}")))
28    }

30    /// Maximum accepted size for a JSON-serialized key (1 MiB).
31    ///
32    /// Inputs larger than this are rejected before parsing to prevent
33    /// memory exhaustion from maliciously crafted payloads.
34    pub const MAX_KEY_JSON_SIZE: usize = 1024 * 1024; // 1 MiB
35
36    /// Maximum accepted size for a CBOR-serialized key (1 MiB).
37    ///
38    /// Inputs larger than this are rejected before parsing to prevent
39    /// memory exhaustion from maliciously crafted payloads.
40    pub const MAX_KEY_CBOR_SIZE: usize = 1024 * 1024; // 1 MiB
41
42    /// Maximum number of entries the open `metadata` map may carry.
43    /// `serde_json::Value` is recursive, so a 1 MiB JSON document can
44    /// expand into a much larger heap tree; the entry cap and the
45    /// per-value byte cap below bound that amplification.
46    pub const MAX_METADATA_ENTRIES: usize = 64;
47    /// Maximum byte length of a single metadata key.
48    pub const MAX_METADATA_KEY_LEN: usize = 256;
49    /// Maximum byte length of any single metadata value's serialized
50    /// JSON representation. AAD includes serialized metadata on every
51    /// encrypted-key decrypt; without this cap one oversized value
52    /// would amplify hashing cost across every load of the same key.
53    pub const MAX_METADATA_VALUE_JSON_LEN: usize = 4 * 1024;
54
55    /// Deserialize from JSON string.
56    ///
57    /// # Errors
58    /// Returns an error if the input exceeds [`MAX_KEY_JSON_SIZE`](Self::MAX_KEY_JSON_SIZE),
59    /// JSON parsing fails, or validation fails.
60    pub fn from_json(json: &str) -> Result<Self> {
61        if json.len() > Self::MAX_KEY_JSON_SIZE {
62            return Err(CoreError::ResourceExceeded(format!(
63                "Key JSON size {} exceeds limit {}",
64                json.len(),
65                Self::MAX_KEY_JSON_SIZE
66            )));
67        }
68        let key: Self = decode_json_opaque(json, "key.json")?;
69        key.validate()?;
70        key.check_use_case_or_security_level_invariant()?;
71        Ok(key)
72    }

74    // --- CBOR serialization ---
75
76    /// Serialize to CBOR bytes (compact binary format, RFC 8949).
77    ///
78    /// CBOR is the primary wire/storage format. Key material is stored as
79    /// native CBOR byte strings — no base64 encoding overhead.
80    ///
81    /// # Errors
82    /// Returns an error if CBOR serialization fails.
83    pub fn to_cbor(&self) -> Result<Vec<u8>> {
84        let mut buf = Vec::new();
85        ciborium::into_writer(self, &mut buf).map_err(|e| {
86            CoreError::SerializationError(format!("CBOR serialization failed: {e}"))
87        })?;
88        Ok(buf)
89    }
90
91    /// Deserialize from CBOR bytes.
92    ///
93    /// # Errors
94    /// Returns an error if the input exceeds [`MAX_KEY_CBOR_SIZE`](Self::MAX_KEY_CBOR_SIZE),
95    /// CBOR parsing fails, or validation fails.
96    pub fn from_cbor(data: &[u8]) -> Result<Self> {
97        if data.len() > Self::MAX_KEY_CBOR_SIZE {
98            return Err(CoreError::ResourceExceeded(format!(
99                "Key CBOR size {} exceeds limit {}",
100                data.len(),
101                Self::MAX_KEY_CBOR_SIZE
102            )));
103        }
104        let key: Self = decode_cbor_opaque(data, "key.cbor")?;
105        key.validate()?;
106        key.check_use_case_or_security_level_invariant()?;
107        Ok(key)
108    }
109
110    /// Enforce the documented invariant that at least one of `use_case`
111    /// or `security_level` is present. Applied only at the
112    /// deserialization boundary; locally-constructed keys always have
113    /// `security_level` set by the constructors, so the invariant cannot
114    /// be violated internally — only by hand-crafted wire payloads.
115    fn check_use_case_or_security_level_invariant(&self) -> Result<()> {
116        if self.use_case.is_none() && self.security_level.is_none() {
117            return Err(CoreError::InvalidKey(
118                "PortableKey requires at least one of `use_case` or `security_level`".to_string(),
119            ));
120        }
121        Ok(())
122    }

126    /// Write to a file as pretty JSON. Creates the file with 0600 permissions atomically on Unix
127    /// for secret/symmetric keys, preventing a window where the file is world-readable.
128    ///
129    /// # Errors
130    /// Returns an error if file writing or permission setting fails.
131    pub fn write_to_file(&self, path: &std::path::Path) -> Result<()> {
132        self.write_to_file_with_overwrite(path, false)
133    }
134
135    /// Like [`Self::write_to_file`] but with explicit overwrite control.
136    ///
137    /// `overwrite = false` (the recommended default) refuses to clobber
138    /// an existing file at `path` and returns
139    /// `CoreError::ConfigurationError` — caller should map this to a
140    /// `--force`-equivalent prompt or abort.
141    ///
142    /// `overwrite = true` replaces any existing file via atomic rename
143    /// (no truncate-then-write window where a crash leaves zero bytes
144    /// on disk + the prior key destroyed).
145    ///
146    /// On Unix, secret/symmetric files are written with mode `0o600`
147    /// applied BEFORE the rename. On Windows the tempfile inherits the
148    /// parent dir's ACL via `tempfile`'s NTFS path; further hardening
149    /// requires `windows-sys` and is left to the consumer.
150    ///
151    /// # Errors
152    ///
153    /// Returns `CoreError::ConfigurationError` on overwrite-refused or
154    /// tempfile creation failure, or `CoreError::Internal` on I/O.
155    pub fn write_to_file_with_overwrite(
156        &self,
157        path: &std::path::Path,
158        overwrite: bool,
159    ) -> Result<()> {
160        let json = self.to_json_pretty()?;
161        self.make_atomic_writer(json.as_bytes(), overwrite).write(path)
162    }
163
164    /// Build an `AtomicWrite` with the project's mode policy for this
165    /// key type pre-applied.
166    ///
167    /// Mode policy:
168    ///   Secret / Symmetric → 0o600 (owner read+write only)
169    ///   Public             → 0o644 (owner rw, others r) — public
170    ///                        keys are MEANT to be readable; without
171    ///                        this an explicit 0o644, tempfile's
172    ///                        default 0o600 would lock pub keys to
173    ///                        the creator and break key-distribution
174    ///                        flows.
175    ///
176    /// lifted out of `write_to_file_with_overwrite`
177    /// so the JSON and CBOR write paths share one source of truth for
178    /// mode selection. Drift between the two would silently regress the
179    /// secret-key threat model.
180    fn make_atomic_writer<'a>(
181        &self,
182        bytes: &'a [u8],
183        overwrite: bool,
184    ) -> crate::unified_api::atomic_write::AtomicWrite<'a> {
185        let writer =
186            crate::unified_api::atomic_write::AtomicWrite::new(bytes).overwrite_existing(overwrite);
187        if self.key_type == KeyType::Secret || self.key_type == KeyType::Symmetric {
188            writer.secret_mode()
189        } else {
190            writer.unix_mode(0o644)
191        }
192    }
193
194    /// Write to a file as CBOR. Creates the file with 0600 permissions atomically on Unix
195    /// for secret/symmetric keys, preventing a window where the file is world-readable.
196    ///
197    /// # Errors
198    /// Returns an error if CBOR serialization or file writing fails.
199    pub fn write_cbor_to_file(&self, path: &std::path::Path) -> Result<()> {
200        self.write_cbor_to_file_with_overwrite(path, false)
201    }
202
203    /// Like [`Self::write_cbor_to_file`] but with explicit overwrite control.
204    ///
205    /// Same contract as [`Self::write_to_file_with_overwrite`] — the JSON
206    /// and CBOR paths now both go through `AtomicWrite` instead of
207    /// the original truncate-then-write pattern that would destroy
208    /// prior key material on crash mid-write.
209    ///
210    /// # Errors
211    ///
212    /// Returns `CoreError::ConfigurationError` on overwrite-refused or
213    /// tempfile creation failure, or `CoreError::Internal` on I/O.
214    pub fn write_cbor_to_file_with_overwrite(
215        &self,
216        path: &std::path::Path,
217        overwrite: bool,
218    ) -> Result<()> {
219        let cbor = self.to_cbor()?;
220        self.make_atomic_writer(&cbor, overwrite).write(path)
221    }

223    /// Read from a JSON file with a bounded read.
224    ///
225    /// Reads at most [`Self::MAX_KEY_JSON_SIZE`] + 1 bytes via a
226    /// `Read::take` adapter so a symlink to `/dev/zero`, a pathological
227    /// attacker file, or any path whose `metadata().len()` is unreliable
228    /// (FIFOs, `/proc/*`) cannot OOM the process before the size cap
229    /// fires inside [`Self::from_json`].
230    ///
231    /// # Errors
232    /// Returns an error if file reading, the size cap, or JSON parsing fails.
233    pub fn read_from_file(path: &std::path::Path) -> Result<Self> {
234        use std::io::Read;
235        // +1 so anything longer than the cap reads as MAX+1 and is
236        // rejected by `from_json`'s length check rather than silently
237        // truncating to MAX.
238        const READ_CAP: u64 = (PortableKey::MAX_KEY_JSON_SIZE as u64).saturating_add(1);
239        let file = std::fs::File::open(path)?;
240        let mut buf = String::new();
241        file.take(READ_CAP).read_to_string(&mut buf).map_err(|e| {
242            tracing::debug!(error = %e, "key JSON file read rejected");
243            CoreError::SerializationError("key JSON file read failed".to_string())
244        })?;
245        Self::from_json(&buf)
246    }
247
248    /// Read from a CBOR file with a bounded read.
249    ///
250    /// Reads at most [`Self::MAX_KEY_CBOR_SIZE`] + 1 bytes — see
251    /// [`Self::read_from_file`] for the OOM-resistance rationale.
252    ///
253    /// # Errors
254    /// Returns an error if file reading, the size cap, or CBOR parsing fails.
255    pub fn read_cbor_from_file(path: &std::path::Path) -> Result<Self> {
256        use std::io::Read;
257        const READ_CAP: u64 = (PortableKey::MAX_KEY_CBOR_SIZE as u64).saturating_add(1);
258        let file = std::fs::File::open(path)?;
259        let mut buf = Vec::new();
260        file.take(READ_CAP).read_to_end(&mut buf).map_err(|e| {
261            tracing::debug!(error = %e, "key CBOR file read rejected");
262            CoreError::SerializationError("key CBOR file read failed".to_string())
263        })?;
264        Self::from_cbor(&buf)
265    }

269    /// Parse a legacy CLI v1 key file format.
270    ///
271    /// The CLI v1 format uses `"key"` for raw key bytes and `"algorithm"` as a string.
272    ///
273    /// ```json
274    /// {
275    ///   "algorithm": "ML-DSA-65",
276    ///   "key_type": "public",
277    ///   "key": "Base64..."
278    /// }
279    /// ```
280    ///
281    /// # Errors
282    /// Returns an error if the JSON is invalid or the algorithm is unrecognized.
283    pub fn from_legacy_json(json: &str) -> Result<Self> {
284        // parallel `from_json` enforces a 1 MiB size
285        // guard before parsing to prevent memory exhaustion from
286        // maliciously crafted payloads. The legacy path was missing this
287        // check, leaving a DoS vector when migrating older keyfiles.
288        if json.len() > Self::MAX_KEY_JSON_SIZE {
289            return Err(CoreError::ResourceExceeded(format!(
290                "Legacy key JSON size {} exceeds limit {}",
291                json.len(),
292                Self::MAX_KEY_JSON_SIZE
293            )));
294        }
295
296        #[derive(Deserialize)]
297        struct LegacyKeyFile {
298            algorithm: String,
299            key_type: String,
300            key: String,
301            #[serde(default)]
302            label: Option<String>,
303        }
304
305        let legacy: LegacyKeyFile = decode_json_opaque(json, "key.legacy_json")?;
306
307        let algorithm = parse_legacy_algorithm(&legacy.algorithm)?;
308        // The legacy schema carries a single flat `key` field, so it
309        // physically cannot represent a hybrid key's separate post-quantum
310        // and classical components. Reject hybrid algorithms here with the
311        // real reason: otherwise the key is built as `KeyData::Single` and
312        // `validate()` rejects it with the misleading "must use composite
313        // key data" message, which reads as a caller error rather than a
314        // format limitation.
315        if algorithm.is_hybrid() {
316            return Err(CoreError::InvalidKey(format!(
317                "Legacy key format cannot represent hybrid algorithm {algorithm:?}: \
318                 the legacy schema has a single `key` field and no way to encode \
319                 the separate post-quantum and classical components a hybrid key \
320                 requires. Re-export this key in the current (non-legacy) format."
321            )));
322        }
323        let key_type = match legacy.key_type.to_lowercase().as_str() {
324            "public" | "pub" => KeyType::Public,
325            "secret" | "private" | "sk" => KeyType::Secret,
326            "symmetric" | "sym" => KeyType::Symmetric,
327            other => {
328                return Err(CoreError::InvalidKey(format!(
329                    "Unrecognized legacy key_type: '{other}'"
330                )));
331            }
332        };
333
334        let key_data = KeyData::Single { raw: legacy.key };
335
336        let mut key = Self::new(algorithm, key_type, key_data);
337        if let Some(label) = legacy.label {
338            key.set_label(label)?;
339        }
340        key.validate()?;
341        Ok(key)
342    }
343}

345/// Parse legacy algorithm strings to [`KeyAlgorithm`].
346///
347/// Thin wrapper over [`KeyAlgorithm::from_canonical_name`] that maps
348/// `None` to a [`CoreError::InvalidKey`] for the `from_legacy_json`
349/// path.
350fn parse_legacy_algorithm(s: &str) -> Result<KeyAlgorithm> {
351    KeyAlgorithm::from_canonical_name(s)
352        .ok_or_else(|| CoreError::InvalidKey(format!("Unrecognized algorithm: '{s}'")))
