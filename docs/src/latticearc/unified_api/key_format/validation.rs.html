18    /// Validate internal consistency.
19    ///
20    /// Checks:
21    /// - Format version matches [`CURRENT_VERSION`](Self::CURRENT_VERSION)
22    /// - Symmetric algorithms (`aes-256`, `chacha20`) require `KeyType::Symmetric`
23    /// - Hybrid algorithms require composite `KeyData`
24    /// - Non-hybrid algorithms require single `KeyData`
25    /// - Base64 key data decodes successfully
26    ///
27    /// # Errors
28    /// Returns `CoreError::InvalidKey` on validation failure.
29    pub fn validate(&self) -> Result<()> {
30        // Version check — reject keys serialized by future or incompatible versions
31        if self.version != Self::CURRENT_VERSION {
32            return Err(CoreError::InvalidKey(format!(
33                "Unsupported key format version {}, expected {}",
34                self.version,
35                Self::CURRENT_VERSION
36            )));
37        }
38
39        // Metadata caps: enforce on every deserialization path so an
40        // attacker-supplied document cannot smuggle a 100k-entry map
41        // past the 1 MiB whole-document gate. `serde_json::Value` is
42        // recursive — even a small JSON string can build a much larger
43        // heap tree — and the metadata is hashed into the AAD on
44        // every encrypted-key load.
45        if self.metadata.len() > Self::MAX_METADATA_ENTRIES {
46            return Err(CoreError::InvalidKey(format!(
47                "metadata entry count {} exceeds maximum {}",
48                self.metadata.len(),
49                Self::MAX_METADATA_ENTRIES
50            )));
51        }
52        for (key, value) in &self.metadata {
53            if key.len() > Self::MAX_METADATA_KEY_LEN {
54                return Err(CoreError::InvalidKey(format!(
55                    "metadata key length {} exceeds maximum {}",
56                    key.len(),
57                    Self::MAX_METADATA_KEY_LEN
58                )));
59            }
60            let value_len = serde_json::to_string(value).map(|s| s.len()).map_err(|e| {
61                CoreError::SerializationError(format!(
62                    "metadata value re-serialization failed: {e}"
63                ))
64            })?;
65            if value_len > Self::MAX_METADATA_VALUE_JSON_LEN {
66                // `key` is an attacker-controlled `String` from the
67                // deserialized BTreeMap. Surface only its length (not the
68                // value) so a tampered key file can't be used to fingerprint
69                // which key-name tripped the cap.
70                return Err(CoreError::InvalidKey(format!(
71                    "metadata value JSON length {} for key (len={}) exceeds maximum {}",
72                    value_len,
73                    key.len(),
74                    Self::MAX_METADATA_VALUE_JSON_LEN
75                )));
76            }
77        }
78
79        // Symmetric algorithm ↔ key type
80        if self.algorithm.is_symmetric() && self.key_type != KeyType::Symmetric {
81            return Err(CoreError::InvalidKey(format!(
82                "Algorithm {:?} requires KeyType::Symmetric, got {:?}",
83                self.algorithm, self.key_type
84            )));
85        }
86        if !self.algorithm.is_symmetric() && self.key_type == KeyType::Symmetric {
87            return Err(CoreError::InvalidKey(format!(
88                "KeyType::Symmetric is not valid for algorithm {:?}",
89                self.algorithm
90            )));
91        }
92
93        // Hybrid ↔ composite key data (encrypted variant is opaque — skip this check).
94        match (&self.key_data, self.algorithm.is_hybrid()) {
95            (KeyData::Composite { .. }, false) => {
96                return Err(CoreError::InvalidKey(format!(
97                    "Non-hybrid algorithm {:?} must use single key data",
98                    self.algorithm
99                )));
100            }
101            (KeyData::Single { .. }, true) => {
102                return Err(CoreError::InvalidKey(format!(
103                    "Hybrid algorithm {:?} must use composite key data",
104                    self.algorithm
105                )));
106            }
107            _ => {}
108        }
109
110        // Verify base64 decodes / envelope shape
111        match &self.key_data {
112            KeyData::Single { raw } => {
113                let _ = decode_b64_opaque(raw, "validate.single")?;
114            }
115            KeyData::Composite { pq, classical } => {
116                let pq_bytes = decode_b64_opaque(pq, "validate.composite.pq")?;
117                let classical_bytes = decode_b64_opaque(classical, "validate.composite.classical")?;
118                // validate component sizes for hybrid keys.
119                // The composite arm previously only verified that
120                // base64 decoded — a truncated PQ component or a
121                // wrong-level keyfile slipped past `validate()` and
122                // failed downstream with less-informative errors. We
123                // bound the components against the expected sizes for
124                // the (algorithm, key_type) pair so the failure point
125                // is the load site, not three layers deep.
126                //
127                // Sizes are upper bounds: secret keys hold a seed +
128                // pk; we accept anything from `min_seed` (32) through
129                // the full secret-key size. Public-key composite
130                // values must match the level's pk size exactly.
131                Self::validate_composite_lengths(
132                    self.algorithm,
133                    pq_bytes.len(),
134                    classical_bytes.len(),
135                )?;
136            }
137            KeyData::Encrypted { enc, kdf, kdf_iterations, kdf_salt, aead, nonce, ciphertext } => {
138                Self::validate_encrypted_envelope_fields(
139                    *enc,
140                    kdf,
141                    *kdf_iterations,
142                    kdf_salt,
143                    aead,
144                    nonce,
145                    ciphertext,
146                )?;
147            }
148        }
149
150        Ok(())
151    }

153    /// Validate this key AND reject it if it has expired at `now`.
154    ///
155    /// # M4 fix
156    ///
157    /// [`validate`](Self::validate) intentionally **does not** consult
158    /// `not_after` — that's a documented design choice ("informational
159    /// lifecycle field") so that consumers loading a possibly-expired key
160    /// for inspection or migration don't get failed at parse time. The
161    /// auditor flagged the absence of any "is this key safe to USE right
162    /// now?" gate as a footgun: a captured-but-expired key file is
163    /// indistinguishable from a fresh one to anyone calling `validate()`.
164    ///
165    /// This helper is the explicit gate. Callers about to perform a crypto
166    /// operation with the key should route through `validate_with_expiry`
167    /// instead of `validate`. Callers loading a key for reporting,
168    /// migration, or expiry-aware lifecycle handling continue to use
169    /// `validate` and check [`is_expired_at`](Self::is_expired_at)
170    /// themselves.
171    ///
172    /// `now` is passed explicitly so the call is testable without freezing
173    /// the system clock and so determinism is preserved on the verify
174    /// path. Pass `chrono::Utc::now()` from production callers.
175    ///
176    /// # Errors
177    ///
178    /// Returns the existing `validate()` errors on structural failures, or
179    /// `CoreError::InvalidKey("key has expired")` if `now >= not_after`.
180    /// The "expired" error is intentionally opaque: an attacker who can
181    /// see the error variant could otherwise binary-search the `not_after`
182    /// timestamp off-line via repeated calls with different `now` values.
183    pub fn validate_with_expiry(&self, now: DateTime<Utc>) -> Result<()> {
184        self.validate()?;
185        if self.is_expired_at(now) {
186            return Err(CoreError::InvalidKey("key has expired".to_string()));
187        }
188        Ok(())
189    }

191    /// Wall-clock-anchored variant of [`Self::validate_with_expiry`]. Routes every
192    /// key-extraction method on this type (`to_hybrid_public_key`,
193    /// `to_hybrid_secret_key`, `to_hybrid_sig_public_key`,
194    /// `to_hybrid_sig_secret_key`) through the expiry gate before the
195    /// returned typed key escapes into a sign / encrypt / agree path. Reads
196    /// `Utc::now()` once at call time so the gate can't be bypassed by a
197    /// caller that holds a stale `DateTime<Utc>`.
198    ///
199    /// The expiry-aware gate is intentionally NOT folded into
200    /// [`validate`](Self::validate): a tool inspecting an expired key for
201    /// migration or reporting purposes must still be able to load it. The
202    /// boundary is "loaded ≠ usable" — `validate()` accepts expired keys,
203    /// `validate_with_expiry_now()` rejects them, and only the latter sits
204    /// in the critical path before extracted key material reaches crypto.
205    ///
206    /// # Errors
207    ///
208    /// Forwards [`Self::validate_with_expiry`](Self::validate_with_expiry)'s
209    /// errors: structural failures from [`validate`](Self::validate), or
210    /// `CoreError::InvalidKey("key has expired")` if `not_after` is at or
211    /// before the current wall clock.
212    pub fn validate_with_expiry_now(&self) -> Result<()> {
213        self.validate_with_expiry(Utc::now())
214    }

216    /// Validate an encrypted-envelope's metadata and base64-decodable sizes.
217    ///
218    /// Shared between [`Self::validate`] (which rejects malformed keys at
219    /// load time) and [`Self::decrypt_with_passphrase`] (which uses this to
220    /// defend against direct construction of an unvalidated encrypted
221    /// variant). Does not verify the AEAD tag — that happens in
222    /// `decrypt_with_passphrase` after key derivation.
223    ///
224    /// `pub(super)`: called cross-module from
225    /// `key_format::encryption::PortableKey::decrypt_with_passphrase`.
226    pub(super) fn validate_encrypted_envelope_fields(
227        enc: u32,
228        kdf: &str,
229        kdf_iterations: u32,
230        kdf_salt: &str,
231        aead: &str,
232        nonce: &str,
233        ciphertext: &str,
234    ) -> Result<()> {
235        // Defense-in-depth: reject pathologically large field strings
236        // BEFORE handing them to `base64::decode`. The 1 MiB
237        // [`MAX_KEY_JSON_SIZE`] / [`MAX_KEY_CBOR_SIZE`] gate fires
238        // earlier on the whole document, but post-decode allocation
239        // amplifies (a 1 MiB base64 expands to ~750 KiB raw which is
240        // then often re-allocated), and these per-field caps mirror
241        // the pattern in `serialization.rs` for `EncryptedOutput`.
242        // Generous slack vs. the cryptographic minimums lets future
243        // schemes (e.g. larger PBKDF2 salts, XChaCha nonces) ship
244        // without revisiting these constants.
245        const MAX_KDF_SALT_B64: usize = 256; // raw cap is 64 B (PBKDF2)
246        const MAX_NONCE_B64: usize = 64; // 12 B raw for AES-GCM
247        const MAX_CIPHERTEXT_B64: usize = 1024 * 1024; // 1 MiB raw cap
248        if kdf_salt.len() > MAX_KDF_SALT_B64 {
249            return Err(CoreError::InvalidKey(format!(
250                "kdf_salt base64 length {} exceeds maximum {MAX_KDF_SALT_B64}",
251                kdf_salt.len()
252            )));
253        }
254        if nonce.len() > MAX_NONCE_B64 {
255            return Err(CoreError::InvalidKey(format!(
256                "nonce base64 length {} exceeds maximum {MAX_NONCE_B64}",
257                nonce.len()
258            )));
259        }
260        if ciphertext.len() > MAX_CIPHERTEXT_B64 {
261            return Err(CoreError::InvalidKey(format!(
262                "ciphertext base64 length {} exceeds maximum {MAX_CIPHERTEXT_B64}",
263                ciphertext.len()
264            )));
265        }
266        // An envelope from an older schema version has a valid wire shape
267        // but an older AEAD AAD layout, so authentication would fail
268        // opaquely as "wrong passphrase". Detect it by version and emit a
269        // distinct, actionable error instead.
270        if enc < ENCRYPTED_ENVELOPE_VERSION {
271            return Err(CoreError::InvalidKey(format!(
272                "v{enc} encrypted-key envelope: the AEAD AAD layout changed in \
273                 v{ENCRYPTED_ENVELOPE_VERSION}, so a v{enc} envelope cannot be \
274                 decrypted by current code even with the correct passphrase. \
275                 Re-protect the key: decrypt it with the release that wrote it, \
276                 then re-encrypt with the current release."
277            )));
278        }
279        if enc != ENCRYPTED_ENVELOPE_VERSION {
280            return Err(CoreError::InvalidKey(format!(
281                "Unsupported encrypted key envelope version {enc}, expected {ENCRYPTED_ENVELOPE_VERSION}",
282            )));
283        }
284        // `kdf` and `aead` are attacker-controlled fields validated
285        // before AEAD authentication. Echoing them in the typed error
286        // would give an attacker a per-field fingerprint of which
287        // check tripped; surface a fixed string and route the raw
288        // value to `tracing::debug!` for operator visibility.
289        if kdf != PBKDF2_KDF_ID {
290            tracing::debug!(received = %kdf, expected = %PBKDF2_KDF_ID, "encrypted key envelope: unsupported KDF identifier");
291            return Err(CoreError::InvalidKey("Unsupported KDF identifier".to_string()));
292        }
293        if aead != AES_GCM_AEAD_ID {
294            tracing::debug!(received = %aead, expected = %AES_GCM_AEAD_ID, "encrypted key envelope: unsupported AEAD identifier");
295            return Err(CoreError::InvalidKey("Unsupported AEAD identifier".to_string()));
296        }
297        if kdf_iterations < PBKDF2_MIN_ITERATIONS {
298            return Err(CoreError::InvalidKey(format!(
299                "PBKDF2 iteration count {kdf_iterations} below minimum {PBKDF2_MIN_ITERATIONS}",
300            )));
301        }
302        if kdf_iterations > PBKDF2_MAX_ITERATIONS {
303            // reject adversary-supplied envelopes with
304            // pathological `kdf_iterations` before any HMAC-SHA256 rounds run.
305            return Err(CoreError::InvalidKey(format!(
306                "PBKDF2 iteration count {kdf_iterations} exceeds maximum {PBKDF2_MAX_ITERATIONS}",
307            )));
308        }
309        // Warn when a loaded key uses fewer iterations than the current
310        // OWASP-recommended default. The minimum (100k) is retained as a
311        // hard floor for backwards compatibility with keys generated under
312        // OWASP 2018 guidance; the default (600k) is OWASP 2023 for
313        // HMAC-SHA256. Callers should re-protect keys below the default.
314        //
315        // Deduped per *distinct iteration count* per process: a mixed-fleet
316        // operator loading key-A at 10k and key-B at 1k must see two
317        // warnings (one per cohort), not one — otherwise the audit trail
318        // hides the lower-iteration cohort behind the first cohort that
319        // happened to trigger. Using `Mutex<HashSet<u32>>` here is fine
320        // because key load is cold-path; the mutex is never contended on
321        // a hot loop.
322        if kdf_iterations < PBKDF2_DEFAULT_ITERATIONS {
323            // Capacity cap prevents unbounded memory growth from an adversary
324            // feeding distinct iteration-count keys. 256 distinct values is
325            // far more than any realistic operator fleet (cohorts cluster
326            // around OWASP guidance dates); beyond the cap we silence
327            // further warnings — a single previously-emitted warning is
328            // already enough audit signal that the cohort exists.
329            const MAX_DISTINCT_ITERATION_COUNTS: usize = 256;
330            static LOW_ITER_WARNED: std::sync::OnceLock<
331                std::sync::Mutex<std::collections::HashSet<u32>>,
332            > = std::sync::OnceLock::new();
333            let table = LOW_ITER_WARNED.get_or_init(|| std::sync::Mutex::new(Default::default()));
334            // Lock-poisoning here is non-fatal — emit the warning anyway,
335            // since not warning is strictly worse than a duplicate warning.
336            let mut seen = table.lock().unwrap_or_else(std::sync::PoisonError::into_inner);
337            let should_warn = if seen.contains(&kdf_iterations) {
338                false
339            } else if seen.len() < MAX_DISTINCT_ITERATION_COUNTS {
340                seen.insert(kdf_iterations);
341                true
342            } else {
343                // Capacity reached — drop the new value silently. The
344                // operator already has 256 prior warnings to act on; a
345                // 257th cohort would not change the response.
346                false
347            };
348            if should_warn {
349                tracing::warn!(
350                    kdf_iterations,
351                    recommended = PBKDF2_DEFAULT_ITERATIONS,
352                    "PBKDF2 iteration count is below the current OWASP recommendation; \
353                     re-protect this key (decrypt + re-encrypt with passphrase) at the \
354                     next opportunity."
355                );
356            }
357        }
358        let salt = decode_b64_opaque(kdf_salt, "validate.encrypted.kdf_salt")?;
359        if salt.len() < PBKDF2_MIN_SALT_LEN {
360            return Err(CoreError::InvalidKey(format!(
361                "PBKDF2 salt length {} below minimum {PBKDF2_MIN_SALT_LEN}",
362                salt.len(),
363            )));
364        }
365        let nonce_bytes = decode_b64_opaque(nonce, "validate.encrypted.nonce")?;
366        if nonce_bytes.len() != AES_GCM_NONCE_LEN {
367            return Err(CoreError::InvalidKey(format!(
368                "AES-GCM nonce length {} != {AES_GCM_NONCE_LEN}",
369                nonce_bytes.len(),
370            )));
371        }
372        let ct_bytes = decode_b64_opaque(ciphertext, "validate.encrypted.ciphertext")?;
373        if ct_bytes.len() < AES_GCM_TAG_LEN {
374            return Err(CoreError::InvalidKey(
375                "Encrypted key ciphertext shorter than AES-GCM tag".to_string(),
376            ));
377        }
378        Ok(())
379    }

381    /// Expected classical-leg length for a hybrid algorithm.
382    ///
383    /// L6 fix: dispatch the classical-leg length per algorithm so a future
384    /// hybrid using a different classical curve (e.g. `Hybrid…+secp256k1`,
385    /// 65-byte uncompressed SEC1 PK) does not have to relax the global
386    /// `classical_len != 32` guard and re-open the loose-validation hole
387    /// the guard exists to close. Every current hybrid uses a 32-byte
388    /// classical leg (X25519 KEM, Ed25519 sig); non-hybrid algorithms are
389    /// not legitimate inputs to this dispatch and surface as `None`.
390    const fn expected_hybrid_classical_len(algorithm: KeyAlgorithm) -> Option<usize> {
391        match algorithm {
392            KeyAlgorithm::HybridMlKem512X25519
393            | KeyAlgorithm::HybridMlKem768X25519
394            | KeyAlgorithm::HybridMlKem1024X25519
395            | KeyAlgorithm::HybridMlDsa44Ed25519
396            | KeyAlgorithm::HybridMlDsa65Ed25519
397            | KeyAlgorithm::HybridMlDsa87Ed25519 => Some(32),
398            _ => None,
399        }
400    }

402    /// bound the composite key component lengths against
403    /// the expected sizes for the (algorithm, key_type) pair. The
404    /// classical leg length is dispatched per algorithm via
405    /// [`expected_hybrid_classical_len`](Self::expected_hybrid_classical_len);
406    /// the PQ leg has algorithm- and key-type-dependent sizes.
407    ///
408    /// We use ranges rather than exact equality because the `Composite`
409    /// secret-key encoding holds the seed + embedded public key (the
410    /// FIPS 203 §6.1 layout for ML-KEM SKs), which varies in length
411    /// across the encoding choices used historically. Lower bound is
412    /// 32 (any-curve seed); upper bound is the full SK size for the
413    /// level. Public-key composites must match the level's PK size
414    /// exactly because the SEC1/raw-bytes encoding is fixed.
415    fn validate_composite_lengths(
416        algorithm: KeyAlgorithm,
417        pq_len: usize,
418        classical_len: usize,
419    ) -> Result<()> {
420        let Some(expected_classical) = Self::expected_hybrid_classical_len(algorithm) else {
421            return Err(CoreError::InvalidKey(format!(
422                "Composite key data is not legitimate for non-hybrid algorithm {algorithm:?}"
423            )));
424        };
425        if classical_len != expected_classical {
426            return Err(CoreError::InvalidKey(format!(
427                "Hybrid key classical component for {algorithm:?} must be {expected_classical} bytes, got {classical_len}",
428            )));
429        }
430        // Conservative PQ bounds — any future algorithm with larger
431        // sizes can extend this without loosening the per-(level,
432        // type) check below for current algorithms.
433        const PQ_LEN_FLOOR: usize = 32;
434        const PQ_LEN_CEILING: usize = 16 * 1024; // SLH-DSA-256 sigs aren't here, but keep headroom
435        if !(PQ_LEN_FLOOR..=PQ_LEN_CEILING).contains(&pq_len) {
436            return Err(CoreError::InvalidKey(format!(
437                "Hybrid key PQ component length {pq_len} outside acceptable range [{PQ_LEN_FLOOR}, {PQ_LEN_CEILING}]",
438            )));
439        }
440        Ok(())
441    }
