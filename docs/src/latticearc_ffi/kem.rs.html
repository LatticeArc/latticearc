23/// ML-KEM parameter set of the post-quantum half of a hybrid key.
24#[derive(Debug, Clone, Copy, PartialEq, Eq, uniffi::Enum)]
25pub enum MlKemSecurityLevel {
26    /// ML-KEM-512 (NIST category 1).
27    MlKem512,
28    /// ML-KEM-768 (NIST category 3), the default.
29    MlKem768,
30    /// ML-KEM-1024 (NIST category 5).
31    MlKem1024,
32}
33
34impl From<MlKemSecurityLevel> for CoreLevel {
35    fn from(l: MlKemSecurityLevel) -> Self {
36        match l {
37            MlKemSecurityLevel::MlKem512 => Self::MlKem512,
38            MlKemSecurityLevel::MlKem768 => Self::MlKem768,
39            MlKemSecurityLevel::MlKem1024 => Self::MlKem1024,
40        }
41    }
42}
43
44impl TryFrom<CoreLevel> for MlKemSecurityLevel {
45    type Error = LatticeArcError;
46
47    fn try_from(l: CoreLevel) -> Result<Self, Self::Error> {
48        match l {
49            CoreLevel::MlKem512 => Ok(Self::MlKem512),
50            CoreLevel::MlKem768 => Ok(Self::MlKem768),
51            CoreLevel::MlKem1024 => Ok(Self::MlKem1024),
52            _ => Err(LatticeArcError::internal(format!("unsupported ML-KEM level {l:?}"))),
53        }
54    }
55}

57/// A hybrid KEM public key (ML-KEM encapsulation key + X25519 public key).
58#[derive(uniffi::Object)]
59pub struct HybridKemPublicKey {
60    inner: CoreKemPublicKey,
61    level: MlKemSecurityLevel,
62}
63
64impl HybridKemPublicKey {
65    fn wrap(inner: CoreKemPublicKey) -> Result<Arc<Self>, LatticeArcError> {
66        let level = inner.security_level().try_into()?;
67        Ok(Arc::new(Self { inner, level }))
68    }
69}

71#[uniffi::export]
72impl HybridKemPublicKey {
73    /// Parses the LatticeArc wire encoding produced by `to_wire`.
74    ///
75    /// # Errors
76    ///
77    /// `WireFormat` for anything that is not a well-formed public key.
78    #[uniffi::constructor]
79    pub fn from_wire(bytes: Vec<u8>) -> Result<Arc<Self>, LatticeArcError> {
80        Self::wrap(CoreKemPublicKey::from_wire(&bytes)?)
81    }
82
83    /// The versioned wire encoding (see `docs/WIRE_FORMAT.md`).
84    ///
85    /// # Errors
86    ///
87    /// `WireFormat` if the key's components have inconsistent lengths.
88    pub fn to_wire(&self) -> Result<Vec<u8>, LatticeArcError> {
89        Ok(self.inner.to_wire()?)
90    }
91
92    /// The ML-KEM parameter set.
93    #[must_use]
94    pub fn security_level(&self) -> MlKemSecurityLevel {
95        self.level
96    }
97}

99/// The components of a hybrid KEM secret key, returned only by
100/// [`HybridKemSecretKey::export_parts`].
101///
102/// This is an explicit export: the fields are plain byte strings the caller
103/// now owns and must clear after persisting them. The record is not
104/// comparable, cloneable or printable, so secrets cannot leak through
105/// equality checks or logs.
106#[derive(uniffi::Record)]
107pub struct HybridKemSecretKeyParts {
108    /// ML-KEM parameter set.
109    pub security_level: MlKemSecurityLevel,
110    /// ML-KEM decapsulation key bytes.
111    pub ml_kem_secret_key: Vec<u8>,
112    /// ML-KEM encapsulation key bytes.
113    pub ml_kem_public_key: Vec<u8>,
114    /// 32-byte X25519 seed.
115    pub ecdh_seed: Vec<u8>,
116}
117
118/// A hybrid KEM secret key. Generate one with
119/// [`HybridKemSecretKey::generate`]; it never leaves Rust except through
120/// `export_parts`.
121#[derive(uniffi::Object)]
122pub struct HybridKemSecretKey {
123    inner: CoreKemSecretKey,
124    level: MlKemSecurityLevel,
125}
126
127impl HybridKemSecretKey {
128    fn wrap(inner: CoreKemSecretKey) -> Result<Arc<Self>, LatticeArcError> {
129        let level = inner.security_level().try_into()?;
130        Ok(Arc::new(Self { inner, level }))
131    }
132}
133
134#[uniffi::export]
135impl HybridKemSecretKey {
136    /// Generates a fresh keypair at `level` from the OS CSPRNG.
137    ///
138    /// # Errors
139    ///
140    /// `Internal` if the underlying key generation fails.
141    #[uniffi::constructor]
142    pub fn generate(level: MlKemSecurityLevel) -> Result<Arc<Self>, LatticeArcError> {
143        let (_pk, sk) = generate_keypair_with_level(level.into())?;
144        Self::wrap(sk)
145    }
146
147    /// Rebuilds a key from parts previously returned by `export_parts`.
148    ///
149    /// # Errors
150    ///
151    /// `InvalidInput` when a component has the wrong length for `level`.
152    #[uniffi::constructor]
153    pub fn from_parts(mut parts: HybridKemSecretKeyParts) -> Result<Arc<Self>, LatticeArcError> {
154        let seed_len = parts.ecdh_seed.len();
155        let seed: Zeroizing<[u8; 32]> =
156            parts.ecdh_seed.as_slice().try_into().map(Zeroizing::new).map_err(|_e| {
157                LatticeArcError::invalid_input(format!(
158                    "ecdh seed must be 32 bytes, got {seed_len}"
159                ))
160            })?;
161        let result = CoreKemSecretKey::from_serialized(
162            parts.security_level.into(),
163            &parts.ml_kem_secret_key,
164            &parts.ml_kem_public_key,
165            &seed,
166        );
167        // Nothing secret stays behind in the caller's record once the key
168        // object exists (or the parts were rejected).
169        parts.ml_kem_secret_key.zeroize();
170        parts.ecdh_seed.zeroize();
171        Self::wrap(result?)
172    }
173
174    /// The matching public key.
175    #[must_use]
176    pub fn public_key(&self) -> Arc<HybridKemPublicKey> {
177        Arc::new(HybridKemPublicKey {
178            inner: CoreKemPublicKey::new(
179                self.inner.ml_kem_pk_bytes(),
180                self.inner.ecdh_public_key_bytes(),
181                self.inner.security_level(),
182            ),
183            level: self.level,
184        })
185    }
186
187    /// The ML-KEM parameter set.
188    #[must_use]
189    pub fn security_level(&self) -> MlKemSecurityLevel {
190        self.level
191    }
192
193    /// Explicitly exports the secret components, for storage under a
194    /// wrapped key. Clear the result as soon as it is persisted.
195    ///
196    /// # Errors
197    ///
198    /// `Internal` if the ML-KEM key cannot be serialized.
199    pub fn export_parts(&self) -> Result<HybridKemSecretKeyParts, LatticeArcError> {
200        Ok(HybridKemSecretKeyParts {
201            security_level: self.level,
202            ml_kem_secret_key: take_zeroizing(self.inner.ml_kem_sk_bytes()?),
203            ml_kem_public_key: self.inner.ml_kem_pk_bytes(),
204            ecdh_seed: self.inner.ecdh_seed_bytes()?.to_vec(),
205        })
206    }
207}

209/// Result of [`encapsulate`]: the public encapsulated key to send, and the
210/// shared secret to keep.
211#[derive(uniffi::Record)]
212pub struct Encapsulation {
213    /// Wire-encoded encapsulated key (ML-KEM ciphertext + ephemeral X25519
214    /// public key); pass it to [`decapsulate`].
215    pub encapsulated_key: Vec<u8>,
216    /// The 64-byte shared secret.
217    pub shared_secret: Arc<SecretBytes>,
218}
219
220/// Encapsulates a fresh shared secret to `public_key`. `context` is
221/// required and non-empty; the same context must be used to decapsulate.
222///
223/// # Errors
224///
225/// `InvalidContext` for an empty or over-long context.
226#[uniffi::export]
227pub fn encapsulate(
228    public_key: &HybridKemPublicKey,
229    context: Vec<u8>,
230) -> Result<Encapsulation, LatticeArcError> {
231    let enc = encapsulate_with_context(&public_key.inner, &context)?;
232    let shared_secret = SecretBytes::from_vec(enc.expose_secret().to_vec());
233    Ok(Encapsulation { encapsulated_key: enc.to_wire()?, shared_secret })
234}
235
236/// Recovers the shared secret from a wire-encoded encapsulated key.
237///
238/// ML-KEM uses implicit rejection: a well-formed encapsulated key that was
239/// made for another secret key, or decapsulated under another `context`,
240/// yields a *different* 64-byte secret with no error. Key confirmation must
241/// come from the authenticated step that uses the secret (for example
242/// [`decrypt_hybrid`] or an AEAD), never from this call succeeding.
243///
244/// # Errors
245///
246/// `WireFormat` for a malformed encapsulated key, `InvalidContext` for an
247/// empty context.
248#[uniffi::export]
249pub fn decapsulate(
250    secret_key: &HybridKemSecretKey,
251    encapsulated_key: Vec<u8>,
252    context: Vec<u8>,
253) -> Result<Arc<SecretBytes>, LatticeArcError> {
254    let enc = EncapsulatedKey::from_wire(&encapsulated_key)?;
255    let ss = decapsulate_with_context(&secret_key.inner, &enc, &context)?;
256    Ok(SecretBytes::from_vec(ss.expose_secret().to_vec()))
257}
258
259/// Encrypts `plaintext` to `public_key` (hybrid KEM + AES-256-GCM) and
260/// returns the wire-encoded ciphertext. `aad` is authenticated but not
261/// encrypted.
262///
263/// # Errors
264///
265/// `InvalidInput` for plaintext above the library's size cap.
266#[uniffi::export]
267pub fn encrypt_hybrid(
268    public_key: &HybridKemPublicKey,
269    plaintext: Vec<u8>,
270    aad: Option<Vec<u8>>,
271) -> Result<Vec<u8>, LatticeArcError> {
272    let ctx = aad.map(HybridEncryptionContext::with_aad);
273    let ct = core_encrypt_hybrid(&public_key.inner, &plaintext, ctx.as_ref())?;
274    Ok(ct.to_wire()?)
275}
276
277/// Decrypts a ciphertext from [`encrypt_hybrid`].
278///
279/// # Errors
280///
281/// `WireFormat` for a malformed ciphertext, `AuthenticationFailed` for a
282/// wrong key, wrong `aad` or tampered bytes.
283#[uniffi::export]
284pub fn decrypt_hybrid(
285    secret_key: &HybridKemSecretKey,
286    ciphertext: Vec<u8>,
287    aad: Option<Vec<u8>>,
288) -> Result<Vec<u8>, LatticeArcError> {
289    let ct = HybridCiphertext::from_wire(&ciphertext)?;
290    let ctx = aad.map(HybridEncryptionContext::with_aad);
291    Ok(take_zeroizing(core_decrypt_hybrid(&secret_key.inner, &ct, ctx.as_ref())?))
292}

294#[cfg(test)]
295#[expect(clippy::unwrap_used, reason = "Tests use unwrap for simplicity")]
296#[expect(clippy::indexing_slicing, reason = "Tests index into buffers of known length")]
297mod tests {
298    use super::*;
299
300    #[test]
301    fn kem_roundtrip_all_levels() {
302        for level in [
303            MlKemSecurityLevel::MlKem512,
304            MlKemSecurityLevel::MlKem768,
305            MlKemSecurityLevel::MlKem1024,
306        ] {
307            let sk = HybridKemSecretKey::generate(level).unwrap();
308            assert_eq!(sk.security_level(), level);
309            let pk = HybridKemPublicKey::from_wire(sk.public_key().to_wire().unwrap()).unwrap();
310            assert_eq!(pk.security_level(), level);
311            let enc = encapsulate(&pk, b"app/v1/transfer".to_vec()).unwrap();
312            assert_eq!(enc.shared_secret.len(), 64);
313            let ss = decapsulate(&sk, enc.encapsulated_key, b"app/v1/transfer".to_vec()).unwrap();
314            assert_eq!(ss.export_bytes(), enc.shared_secret.export_bytes());
315        }
316    }
317
318    #[test]
319    fn kem_context_is_required_and_bound() {
320        let sk = HybridKemSecretKey::generate(MlKemSecurityLevel::MlKem768).unwrap();
321        let pk = sk.public_key();
322        assert!(matches!(encapsulate(&pk, vec![]), Err(LatticeArcError::InvalidContext { .. })));
323        let enc = encapsulate(&pk, b"a".to_vec()).unwrap();
324        let other = decapsulate(&sk, enc.encapsulated_key.clone(), b"b".to_vec()).unwrap();
325        assert_ne!(other.export_bytes(), enc.shared_secret.export_bytes());
326        // Implicit rejection: a wrong secret key also yields a different
327        // secret without an error.
328        let wrong = HybridKemSecretKey::generate(MlKemSecurityLevel::MlKem768).unwrap();
329        let implicit = decapsulate(&wrong, enc.encapsulated_key, b"a".to_vec()).unwrap();
330        assert_ne!(implicit.export_bytes(), enc.shared_secret.export_bytes());
331    }
332
333    #[test]
334    fn secret_key_parts_roundtrip() {
335        let sk = HybridKemSecretKey::generate(MlKemSecurityLevel::MlKem768).unwrap();
336        let parts = sk.export_parts().unwrap();
337        assert_eq!(parts.ecdh_seed.len(), 32);
338        let back = HybridKemSecretKey::from_parts(parts).unwrap();
339        let again = back.export_parts().unwrap();
340        let original = sk.export_parts().unwrap();
341        assert_eq!(again.ml_kem_secret_key, original.ml_kem_secret_key);
342        assert_eq!(again.ecdh_seed, original.ecdh_seed);
343        assert_eq!(back.public_key().to_wire().unwrap(), sk.public_key().to_wire().unwrap());
344
345        let mut bad = sk.export_parts().unwrap();
346        bad.ecdh_seed.pop();
347        assert!(matches!(
348            HybridKemSecretKey::from_parts(bad),
349            Err(LatticeArcError::InvalidInput { .. })
350        ));
351    }
352
353    #[test]
354    fn hybrid_encrypt_decrypt_roundtrip_and_failures() {
355        let sk = HybridKemSecretKey::generate(MlKemSecurityLevel::MlKem768).unwrap();
356        let pk = sk.public_key();
357        let ct = encrypt_hybrid(&pk, b"hello".to_vec(), Some(b"hdr".to_vec())).unwrap();
358        assert_eq!(decrypt_hybrid(&sk, ct.clone(), Some(b"hdr".to_vec())).unwrap(), b"hello");
359        assert_eq!(
360            decrypt_hybrid(&sk, ct.clone(), Some(b"other".to_vec())).unwrap_err(),
361            LatticeArcError::AuthenticationFailed
362        );
363        let other = HybridKemSecretKey::generate(MlKemSecurityLevel::MlKem768).unwrap();
364        assert_eq!(
365            decrypt_hybrid(&other, ct.clone(), Some(b"hdr".to_vec())).unwrap_err(),
366            LatticeArcError::AuthenticationFailed
367        );
368        let mut tampered = ct.clone();
369        let last = tampered.len() - 1;
370        tampered[last] ^= 1;
371        assert_eq!(
372            decrypt_hybrid(&sk, tampered, Some(b"hdr".to_vec())).unwrap_err(),
373            LatticeArcError::AuthenticationFailed
374        );
375        assert!(matches!(
376            decrypt_hybrid(&sk, ct[..7].to_vec(), None),
377            Err(LatticeArcError::WireFormat { .. })
378        ));
379        let no_aad = encrypt_hybrid(&pk, vec![], None).unwrap();
380        assert_eq!(decrypt_hybrid(&sk, no_aad, None).unwrap(), Vec::<u8>::new());
381    }

383    #[test]
384    fn wire_rejects_garbage() {
385        assert!(matches!(
386            HybridKemPublicKey::from_wire(vec![0; 3]),
387            Err(LatticeArcError::WireFormat { .. })
388        ));
389        let sk = HybridKemSecretKey::generate(MlKemSecurityLevel::MlKem512).unwrap();
390        assert!(matches!(
391            decapsulate(&sk, b"LA\x01junk".to_vec(), b"c".to_vec()),
392            Err(LatticeArcError::WireFormat { .. })
393        ));
394    }
