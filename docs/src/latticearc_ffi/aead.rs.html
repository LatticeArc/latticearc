24/// Wraps a 16-, 32- or 64-byte `key` under a 32-byte `kek` with AES-256-GCM
25/// plus an HMAC-SHA256 commitment to the KEK and the context (format 2).
26/// `context` is required, non-empty, committed and bound as associated data;
27/// the same context must be given to [`unwrap_key`].
28///
29/// # Errors
30///
31/// `InvalidInput` for a KEK that is not 32 bytes or a key of unsupported
32/// length; `InvalidContext` for an empty context.
33#[uniffi::export]
34pub fn wrap_key(
35    kek: &SecretBytes,
36    key: &SecretBytes,
37    context: Vec<u8>,
38) -> Result<Vec<u8>, LatticeArcError> {
39    let kek = kek.key32("key-encryption key")?;
40    Ok(core_wrap_key(&kek, key.expose(), &context)?.to_bytes())
41}
42
43/// Like [`wrap_key`] but pads under the AEAD so the output is exactly
44/// `total_len` bytes, whatever the wrapped key's length.
45///
46/// # Errors
47///
48/// As [`wrap_key`], plus `InvalidInput` when `total_len` is too small for
49/// the key or above the 1 MiB cap.
50#[uniffi::export]
51pub fn wrap_key_padded(
52    kek: &SecretBytes,
53    key: &SecretBytes,
54    context: Vec<u8>,
55    total_len: u32,
56) -> Result<Vec<u8>, LatticeArcError> {
57    let kek = kek.key32("key-encryption key")?;
58    Ok(core_wrap_key_padded(&kek, key.expose(), &context, total_len as usize)?.to_bytes())
59}
60
61/// Recovers a key wrapped by [`wrap_key`] or [`wrap_key_padded`] (format 2
62/// only).
63///
64/// # Errors
65///
66/// `AuthenticationFailed` for a wrong KEK, wrong context or tampered bytes;
67/// `InvalidInput` for a KEK that is not 32 bytes, an encoding that is not
68/// even the right shape, or a legacy format-1 record (read those with
69/// [`unwrap_key_allowing_v1`] and re-wrap them).
70#[uniffi::export]
71pub fn unwrap_key(
72    kek: &SecretBytes,
73    wrapped: Vec<u8>,
74    context: Vec<u8>,
75) -> Result<Arc<SecretBytes>, LatticeArcError> {
76    unwrap_with_policy(kek, &wrapped, &context, FormatPolicy::RequireV2)
77}
78
79/// Like [`unwrap_key`], but also accepts a legacy format-1 record written by
80/// LatticeArc 0.13–0.15, whose commitment covers the KEK but not the
81/// context. Use only to migrate old records: unwrap, then re-wrap with
82/// [`wrap_key`].
83///
84/// # Errors
85///
86/// As [`unwrap_key`], except that a format-1 record is accepted.
87#[uniffi::export]
88pub fn unwrap_key_allowing_v1(
89    kek: &SecretBytes,
90    wrapped: Vec<u8>,
91    context: Vec<u8>,
92) -> Result<Arc<SecretBytes>, LatticeArcError> {
93    unwrap_with_policy(kek, &wrapped, &context, FormatPolicy::AllowV1)
94}
95
96fn unwrap_with_policy(
97    kek: &SecretBytes,
98    wrapped: &[u8],
99    context: &[u8],
100    policy: FormatPolicy,
101) -> Result<Arc<SecretBytes>, LatticeArcError> {
102    let kek = kek.key32("key-encryption key")?;
103    let wrapped = WrappedKey::from_bytes(wrapped)?;
104    let key = core_unwrap_key_with_policy(&kek, &wrapped, context, policy)?;
105    Ok(SecretBytes::from_vec(take_zeroizing(key)))
106}

112/// Encrypts a stream as AES-256-GCM segments under a committing header.
113///
114/// Call `seal_segment` once per full plaintext segment and `finish` with the
115/// (possibly empty, possibly short) tail. Every segment except the first
116/// holds `plaintext_segment_size` bytes; the first holds
117/// `first_plaintext_segment_size`.
118#[derive(uniffi::Object)]
119pub struct StreamEncryptor {
120    inner: Mutex<Option<CoreStreamEncryptor>>,
121    header: Vec<u8>,
122    segment_size: u32,
123    plaintext_segment_size: u32,
124    first_plaintext_segment_size: u32,
125}
126
127#[uniffi::export]
128impl StreamEncryptor {
129    /// Starts a stream under a 32-byte `key` with the required, non-empty
130    /// `aad` bound to every segment and ciphertext segments of
131    /// `segment_size` bytes (64 bytes to 16 MiB).
132    ///
133    /// # Errors
134    ///
135    /// `InvalidInput` for a key that is not 32 bytes or a segment size out of
136    /// range; `InvalidContext` for an empty `aad`.
137    #[uniffi::constructor]
138    pub fn new(
139        key: &SecretBytes,
140        aad: Vec<u8>,
141        segment_size: u32,
142    ) -> Result<Arc<Self>, LatticeArcError> {
143        let key = key.key32("stream key")?;
144        let enc = CoreStreamEncryptor::new(&key, &aad, segment_size)?;
145        Ok(Arc::new(Self {
146            header: enc.header().to_vec(),
147            segment_size: enc.segment_size(),
148            plaintext_segment_size: usize_to_u32(enc.plaintext_segment_size()),
149            first_plaintext_segment_size: usize_to_u32(enc.first_plaintext_segment_size()),
150            inner: Mutex::new(Some(enc)),
151        }))
152    }
153
154    /// The stream header; store it with the ciphertext.
155    #[must_use]
156    pub fn header(&self) -> Vec<u8> {
157        self.header.clone()
158    }
159
160    /// Ciphertext segment size in bytes.
161    #[must_use]
162    pub fn segment_size(&self) -> u32 {
163        self.segment_size
164    }
165
166    /// Plaintext bytes carried by every segment after the first.
167    #[must_use]
168    pub fn plaintext_segment_size(&self) -> u32 {
169        self.plaintext_segment_size
170    }
171
172    /// Plaintext bytes carried by the first segment.
173    #[must_use]
174    pub fn first_plaintext_segment_size(&self) -> u32 {
175        self.first_plaintext_segment_size
176    }
177
178    /// Index of the next segment `seal_segment` will produce.
179    ///
180    /// # Errors
181    ///
182    /// `StreamExhausted` after `finish`.
183    pub fn next_index(&self) -> Result<u32, LatticeArcError> {
184        let guard = self.inner.lock().unwrap_or_else(PoisonError::into_inner);
185        guard.as_ref().map(CoreStreamEncryptor::next_index).ok_or(LatticeArcError::StreamExhausted)
186    }
187
188    /// Seals one full segment of plaintext and returns the ciphertext
189    /// segment.
190    ///
191    /// # Errors
192    ///
193    /// `InvalidInput` when `plaintext` is not exactly the segment's
194    /// capacity; `StreamExhausted` after `finish` or when the segment
195    /// counter would wrap.
196    pub fn seal_segment(&self, plaintext: Vec<u8>) -> Result<Vec<u8>, LatticeArcError> {
197        let mut guard = self.inner.lock().unwrap_or_else(PoisonError::into_inner);
198        let enc = guard.as_mut().ok_or(LatticeArcError::StreamExhausted)?;
199        Ok(enc.seal_segment(&plaintext)?)
200    }
201
202    /// Seals the final segment (any length up to the segment's capacity,
203    /// including empty) and closes the stream.
204    ///
205    /// # Errors
206    ///
207    /// `InvalidInput` when the tail is longer than a segment;
208    /// `StreamExhausted` when the stream was already finished.
209    pub fn finish(&self, last_plaintext: Vec<u8>) -> Result<Vec<u8>, LatticeArcError> {
210        let mut guard = self.inner.lock().unwrap_or_else(PoisonError::into_inner);
211        let live = guard.as_ref().ok_or(LatticeArcError::StreamExhausted)?;
212        // Check the only input error before consuming the encryptor, so a
213        // caller can fix the tail and retry instead of losing the stream.
214        let capacity = if live.next_index() == 0 {
215            live.first_plaintext_segment_size()
216        } else {
217            live.plaintext_segment_size()
218        };
219        if last_plaintext.len() > capacity {
220            return Err(LatticeArcError::invalid_input(format!(
221                "last segment holds at most {capacity} bytes, got {}",
222                last_plaintext.len()
223            )));
224        }
225        let enc = guard.take().ok_or(LatticeArcError::StreamExhausted)?;
226        Ok(enc.finish(&last_plaintext)?)
227    }
228}

230/// Decrypts segments produced by [`StreamEncryptor`], in order or by index.
231///
232/// A wrong key or a wrong `aad` fails in the constructor, before any segment
233/// is touched, thanks to the key- and context-commitment in the format-2
234/// header. (A legacy format-1 header opened with
235/// [`StreamDecryptor::new_allowing_v1`] commits to the key only; there a
236/// wrong `aad` fails on the first segment opened.)
237#[derive(uniffi::Object)]
238pub struct StreamDecryptor {
239    inner: Mutex<CoreStreamDecryptor>,
240    segment_size: u32,
241    plaintext_segment_size: u32,
242    first_plaintext_segment_size: u32,
243}
244
245#[uniffi::export]
246impl StreamDecryptor {
247    /// Opens a format-2 stream from its `header`.
248    ///
249    /// # Errors
250    ///
251    /// `AuthenticationFailed` for a wrong key, wrong `aad` or tampered
252    /// header; `InvalidInput` for a key that is not 32 bytes or a legacy
253    /// format-1 header (open those with [`Self::new_allowing_v1`] and
254    /// re-encrypt); `InvalidContext` for an empty `aad`.
255    #[uniffi::constructor]
256    pub fn new(
257        key: &SecretBytes,
258        aad: Vec<u8>,
259        header: Vec<u8>,
260    ) -> Result<Arc<Self>, LatticeArcError> {
261        Self::open_with_policy(key, &aad, &header, FormatPolicy::RequireV2)
262    }
263
264    /// Like [`Self::new`], but also opens a legacy format-1 header written by
265    /// LatticeArc 0.13–0.15, which commits to the key but not to `aad`. Use
266    /// only to migrate old streams: decrypt, then re-encrypt with
267    /// [`StreamEncryptor`].
268    ///
269    /// # Errors
270    ///
271    /// As [`Self::new`], except that a format-1 header is accepted.
272    #[uniffi::constructor]
273    pub fn new_allowing_v1(
274        key: &SecretBytes,
275        aad: Vec<u8>,
276        header: Vec<u8>,
277    ) -> Result<Arc<Self>, LatticeArcError> {
278        Self::open_with_policy(key, &aad, &header, FormatPolicy::AllowV1)
279    }
280
281    /// Format version of the opened header: 2, or 1 for a legacy header
282    /// opened with [`Self::new_allowing_v1`].
283    #[must_use]
284    pub fn format_version(&self) -> u8 {
285        self.inner.lock().unwrap_or_else(PoisonError::into_inner).format_version()
286    }
287
288    /// Ciphertext segment size in bytes.
289    #[must_use]
290    pub fn segment_size(&self) -> u32 {
291        self.segment_size
292    }
293
294    /// Plaintext bytes carried by every segment after the first.
295    #[must_use]
296    pub fn plaintext_segment_size(&self) -> u32 {
297        self.plaintext_segment_size
298    }
299
300    /// Plaintext bytes carried by the first segment.
301    #[must_use]
302    pub fn first_plaintext_segment_size(&self) -> u32 {
303        self.first_plaintext_segment_size
304    }
305
306    /// Index `open_next` will read next.
307    #[must_use]
308    pub fn next_index(&self) -> u32 {
309        self.inner.lock().unwrap_or_else(PoisonError::into_inner).next_index()
310    }
311
312    /// `true` once `open_next` has consumed the last segment.
313    #[must_use]
314    pub fn is_finished(&self) -> bool {
315        self.inner.lock().unwrap_or_else(PoisonError::into_inner).is_finished()
316    }
317
318    /// Opens the segment at `index` (random access). `is_last` must say
319    /// whether this is the stream's final segment.
320    ///
321    /// # Errors
322    ///
323    /// `AuthenticationFailed` for a tampered, reordered or misdescribed
324    /// segment.
325    pub fn open_segment(
326        &self,
327        index: u32,
328        segment: Vec<u8>,
329        is_last: bool,
330    ) -> Result<Vec<u8>, LatticeArcError> {
331        let guard = self.inner.lock().unwrap_or_else(PoisonError::into_inner);
332        Ok(take_zeroizing(guard.open_segment(index, &segment, is_last)?))
333    }
334
335    /// Opens the next segment in order.
336    ///
337    /// # Errors
338    ///
339    /// `AuthenticationFailed` for a tampered or misdescribed segment;
340    /// `StreamExhausted` after the last segment.
341    pub fn open_next(&self, segment: Vec<u8>, is_last: bool) -> Result<Vec<u8>, LatticeArcError> {
342        let mut guard = self.inner.lock().unwrap_or_else(PoisonError::into_inner);
343        Ok(take_zeroizing(guard.open_next(&segment, is_last)?))
344    }
345}
346
347impl StreamDecryptor {
348    fn open_with_policy(
349        key: &SecretBytes,
350        aad: &[u8],
351        header: &[u8],
352        policy: FormatPolicy,
353    ) -> Result<Arc<Self>, LatticeArcError> {
354        let key = key.key32("stream key")?;
355        let dec = CoreStreamDecryptor::new_with_policy(&key, aad, header, policy)?;
356        Ok(Arc::new(Self {
357            segment_size: dec.segment_size(),
358            plaintext_segment_size: usize_to_u32(dec.plaintext_segment_size()),
359            first_plaintext_segment_size: usize_to_u32(dec.first_plaintext_segment_size()),
360            inner: Mutex::new(dec),
361        }))
362    }
363}

431    #[test]
432    fn stream_roundtrip_sequential_and_random_access() {
433        let key = SecretBytes::import(vec![0x44; 32]);
434        let enc = StreamEncryptor::new(&key, b"file".to_vec(), 256).unwrap();
435        let first = vec![1u8; enc.first_plaintext_segment_size() as usize];
436        let middle = vec![2u8; enc.plaintext_segment_size() as usize];
437        let tail = vec![3u8; 10];
438        assert_eq!(enc.next_index().unwrap(), 0);
439        let s0 = enc.seal_segment(first.clone()).unwrap();
440        let s1 = enc.seal_segment(middle.clone()).unwrap();
441        assert_eq!(enc.next_index().unwrap(), 2);
442        let s2 = enc.finish(tail.clone()).unwrap();
443        assert_eq!(enc.seal_segment(vec![]).unwrap_err(), LatticeArcError::StreamExhausted);
444        assert_eq!(enc.next_index().unwrap_err(), LatticeArcError::StreamExhausted);
445
446        let dec = StreamDecryptor::new(&key, b"file".to_vec(), enc.header()).unwrap();
447        assert_eq!(dec.segment_size(), 256);
448        assert_eq!(dec.open_segment(1, s1.clone(), false).unwrap(), middle);
449        assert_eq!(dec.open_next(s0, false).unwrap(), first);
450        assert_eq!(dec.open_next(s1.clone(), false).unwrap(), middle);
451        assert!(!dec.is_finished());
452        assert_eq!(dec.open_next(s2.clone(), true).unwrap(), tail);
453        assert!(dec.is_finished());
454        assert_eq!(dec.open_next(s2, true).unwrap_err(), LatticeArcError::StreamExhausted);
455
456        // Misdescribed, reordered or tampered segments fail opaquely.
457        assert_eq!(
458            dec.open_segment(1, s1.clone(), true).unwrap_err(),
459            LatticeArcError::AuthenticationFailed
460        );
461        assert_eq!(
462            dec.open_segment(0, s1.clone(), false).unwrap_err(),
463            LatticeArcError::AuthenticationFailed
464        );
465        let mut bad = s1;
466        bad[5] ^= 0x80;
467        assert_eq!(
468            dec.open_segment(1, bad, false).unwrap_err(),
469            LatticeArcError::AuthenticationFailed
470        );
471    }

494        (0..s.len()).step_by(2).map(|i| u8::from_str_radix(&s[i..i + 2], 16).unwrap()).collect()
495    }
