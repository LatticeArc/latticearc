1//! UniFFI bindings for LatticeArc.
2//!
3//! This crate exposes the LatticeArc 0.13 mobile-core APIs to Swift and
4//! Kotlin through [UniFFI](https://mozilla.github.io/uniffi-rs/): Argon2id,
5//! HKDF and SP 800-108 with required contexts, committing key wrap,
6//! streaming AEAD, hybrid ML-KEM + X25519 encapsulation and encryption,
7//! hybrid ML-DSA + Ed25519 and plain ML-DSA signatures (context-bound,
8//! pre-hashed and seed-derived), the binary wire format, and the memory-lock
9//! status. See `docs/FFI.md` for the language-level API map and the build
10//! scripts under `scripts/ffi/`.
11//!
12//! # Rules the surface follows
13//!
14//! - **Secrets are handles.** Derived keys, unwrapped keys, shared secrets
15//!   and signing keys are opaque objects ([`SecretBytes`],
16//!   [`HybridKemSecretKey`], [`HybridSigningKey`], [`MlDsaSigningKey`]).
17//!   Secret bytes cross the boundary only through methods named `export*`.
18//! - **Errors are typed.** Every failure is a [`LatticeArcError`] variant;
19//!   nothing panics across the boundary. Verification mismatches return
20//!   `false`; forged ciphertexts and wrong keys in the AEAD, key-wrap,
21//!   stream and hybrid-encryption paths return the opaque
22//!   `AuthenticationFailed`. ML-KEM decapsulation uses implicit rejection
23//!   and never fails for a wrong key; the authenticated step that follows
24//!   does.
25//! - **Contexts are mandatory** wherever the Rust API requires them: hybrid
26//!   KEM and signatures, HKDF, SP 800-108, key wrap, stream `aad` and
27//!   HashML-DSA. Plain ML-DSA `sign` / `verify` accept an empty context, as
28//!   FIPS 204 does.
29//! - **Public values use the wire format** (`docs/WIRE_FORMAT.md`), so a
30//!   key or ciphertext written on one platform parses on every other.
31//!
32//! The hand-written source of this crate contains no `unsafe`; the C ABI is
33//! generated by UniFFI's macros.

57/// Process-wide report on memory locking of secret buffers.
58///
59/// On Android the per-process `RLIMIT_MEMLOCK` is often 64 KiB, so
60/// `failed_buffers > 0` is expected there once enough secrets are alive;
61/// show it rather than hide it.
62#[derive(Debug, Clone, PartialEq, Eq, uniffi::Record)]
63pub struct MemoryLockStatus {
64    /// `true` when this build attempts to lock secret buffers.
65    pub supported: bool,
66    /// `true` when the library was built with the `secret-mlock` feature.
67    pub feature_enabled: bool,
68    /// Live secret buffers whose pages are locked.
69    pub locked_buffers: u64,
70    /// Lock attempts that failed since process start.
71    pub failed_buffers: u64,
72    /// OS error text of the most recent failed attempt, if any.
73    pub last_error: Option<String>,
74}
75
76/// Returns the process-wide [`MemoryLockStatus`].
77#[uniffi::export]
78#[must_use]
79pub fn memory_lock_status() -> MemoryLockStatus {
80    let s = latticearc::memory_lock_status();
81    MemoryLockStatus {
82        supported: s.supported,
83        feature_enabled: s.feature_enabled,
84        locked_buffers: u64::try_from(s.locked_buffers).unwrap_or(u64::MAX),
85        failed_buffers: u64::try_from(s.failed_buffers).unwrap_or(u64::MAX),
86        last_error: s.last_error,
87    }
88}

97#[cfg(test)]
98mod tests {
99    use super::*;
100
101    #[test]
102    fn version_is_the_workspace_version() {
103        assert_eq!(version(), env!("CARGO_PKG_VERSION"));
104    }
105
106    #[test]
107    fn memory_lock_status_reports_feature() {
108        let s = memory_lock_status();
109        assert_eq!(s.feature_enabled, cfg!(feature = "secret-mlock"));
110        if !s.feature_enabled {
111            assert!(!s.supported);
112            assert_eq!(s.locked_buffers, 0);
113        }
114    }
115
116    #[cfg(all(feature = "secret-mlock", not(target_os = "windows")))]
117    #[test]
118    fn locked_buffer_count_tracks_live_secrets() {
119        let before = memory_lock_status().locked_buffers;
120        let s = SecretBytes::import(vec![1u8; 64]);
121        let during = memory_lock_status();
122        assert!(during.supported);
123        if s.is_locked() {
124            assert!(during.locked_buffers > before);
125        }
126        drop(s);
127    }
